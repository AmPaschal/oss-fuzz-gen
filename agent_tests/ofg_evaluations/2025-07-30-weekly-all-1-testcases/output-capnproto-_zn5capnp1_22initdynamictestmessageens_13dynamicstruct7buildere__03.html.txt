// The function under test is declared in test-util.h, which is included above.
void capnp::_::initDynamicTestMessage(capnp::DynamicStruct::Builder);

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) {
  // Cap'n Proto requires word-aligned buffers. If we don't receive one,
  // we can't do anything. libFuzzer should notice and try again with an
  // aligned buffer.
  if (Size > 0 && reinterpret_cast<uintptr_t>(Data) % sizeof(capnp::word) != 0) {
    return 0;
  }

  kj::ArrayPtr<const capnp::word> wordArray(
      reinterpret_cast<const capnp::word*>(Data), Size / sizeof(capnp::word));

  if (kj::none != kj::runCatchingExceptions([&]() {
        // Create a message builder. We will initialize it with the fuzzed data,
        // and then pass a builder for its root to the function-under-test.
        // This tests whether the function can correctly overwrite potentially
        // garbage data.
        capnp::MallocMessageBuilder builder;

        // Create a reader for the fuzzed data. This treats the input as a
        // single-segment message. This may throw if the data is too small
        // or malformed, which is caught below.
        capnp::SegmentArrayMessageReader reader(
            kj::ArrayPtr<const kj::ArrayPtr<const capnp::word>>(&wordArray, 1));

        // Copy the data from the reader to the builder.
        builder.setRoot(reader.getRoot<capnp::AnyPointer>());

        // Get a typed builder for the root. This may also throw if the pointer
        // is invalid for the schema.
        auto schema = capnp::Schema::from<capnp::_::TestAllTypes>();
        auto root = builder.getRoot<capnp::DynamicStruct>(schema);

        // Call the function under test.
        capnp::_::initDynamicTestMessage(root);
      })) {
    KJ_LOG(ERROR, "threw");
  }
  return 0;
}
