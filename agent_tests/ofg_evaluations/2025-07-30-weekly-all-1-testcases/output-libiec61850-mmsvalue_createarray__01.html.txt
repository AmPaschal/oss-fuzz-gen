MmsValue * MmsValue_createArray(const MmsVariableSpecification *, int);

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size < sizeof(int) + 1) {
        return 0;
    }

    int arraySize;
    memcpy(&arraySize, data, sizeof(int));
    arraySize = abs(arraySize) % 256;

    MmsVariableSpecification elementSpec;
    memset(&elementSpec, 0, sizeof(MmsVariableSpecification));

    elementSpec.type = (MmsType)data[sizeof(int)] % (MMS_DATA_ACCESS_ERROR + 1);
    elementSpec.name = NULL;

    const uint8_t* typeSpecData = data + sizeof(int) + 1;
    size_t typeSpecSize = size - (sizeof(int) + 1);

    switch (elementSpec.type) {
        case MMS_INTEGER:
        case MMS_UNSIGNED:
            if (typeSpecSize < sizeof(int)) return 0;
            int intSize;
            memcpy(&intSize, typeSpecData, sizeof(int));
            if (elementSpec.type == MMS_INTEGER)
                elementSpec.typeSpec.integer = abs(intSize) % 65;
            else
                elementSpec.typeSpec.unsignedInteger = abs(intSize) % 65;
            break;
        case MMS_FLOAT:
            if (typeSpecSize < 2) return 0;
            elementSpec.typeSpec.floatingpoint.exponentWidth = typeSpecData[0];
            elementSpec.typeSpec.floatingpoint.formatWidth = typeSpecData[1];
            break;
        case MMS_BIT_STRING:
        case MMS_OCTET_STRING:
        case MMS_VISIBLE_STRING:
        case MMS_STRING:
            if (typeSpecSize < sizeof(int)) return 0;
            int strSize;
            memcpy(&strSize, typeSpecData, sizeof(int));
            strSize = abs(strSize) % 1024;
            if (elementSpec.type == MMS_BIT_STRING) elementSpec.typeSpec.bitString = strSize;
            else if (elementSpec.type == MMS_OCTET_STRING) elementSpec.typeSpec.octetString = strSize;
            else if (elementSpec.type == MMS_VISIBLE_STRING) elementSpec.typeSpec.visibleString = strSize;
            else elementSpec.typeSpec.mmsString = strSize;
            break;
        case MMS_BINARY_TIME:
            if (typeSpecSize < 1) return 0;
            elementSpec.typeSpec.binaryTime = (typeSpecData[0] % 2 == 0) ? 4 : 6;
            break;
        case MMS_ARRAY:
        {
            if (typeSpecSize < sizeof(int)) return 0;

            static MmsVariableSpecification innerElementSpec;
            memset(&innerElementSpec, 0, sizeof(MmsVariableSpecification));
            innerElementSpec.type = MMS_INTEGER;
            innerElementSpec.typeSpec.integer = 32;

            elementSpec.typeSpec.array.elementTypeSpec = &innerElementSpec;
            
            int elementCount;
            memcpy(&elementCount, typeSpecData, sizeof(int));
            elementSpec.typeSpec.array.elementCount = abs(elementCount) % 64;
            break;
        }
        case MMS_STRUCTURE:
            return 0;
        default:
            break;
    }

    MmsValue* mmsArray = MmsValue_createArray(&elementSpec, arraySize);

    if (mmsArray != NULL) {
        MmsValue_delete(mmsArray);
    }

    return 0;
}
