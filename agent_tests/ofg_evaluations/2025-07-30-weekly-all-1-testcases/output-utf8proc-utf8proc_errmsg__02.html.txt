
	
<description>
The `utf8proc_errmsg` function translates an error code into a human-readable string. It takes an `utf8proc_ssize_t` error code as input and returns a pointer to a constant character string describing the error. The function uses a `switch` statement to map the error code to the appropriate message.
</description>
<requirements>
<requirement>
The `errcode` argument should be one of the defined error codes: `UTF8PROC_ERROR_NOMEM` (-1), `UTF8PROC_ERROR_OVERFLOW` (-2), `UTF8PROC_ERROR_INVALIDUTF8` (-3), `UTF8PROC_ERROR_NOTASSIGNED` (-4), or `UTF8PROC_ERROR_INVALIDOPTS` (-5). This is because the function is designed to return specific messages for these codes, and providing one of these will result in a more descriptive output.
</requirement>
<requirement>
The `errcode` input can be fuzzed with arbitrary values. The function has a default case in its switch statement that handles any unspecified error code, preventing crashes and ensuring robust behavior even with unexpected inputs.
</requirement>

#include <string.h>
#include "/src/utf8proc/utf8proc.h"

const char * utf8proc_errmsg(utf8proc_ssize_t);

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  /* Direct fuzzing of utf8proc_errmsg */
  if (size >= sizeof(utf8proc_ssize_t)) {
    utf8proc_ssize_t errcode;
    memcpy(&errcode, data, sizeof(errcode));
    utf8proc_errmsg(errcode);
  }

  if (size < 1)
    return 0;

  /* Avoid timeout with long inputs */
  if (size > (64 * 1024))
    return 0;

  if (data[size - 1] != '\0')
    return 0;

  const uint8_t *ptr = data;
  utf8proc_int32_t c = 0, c_prev = 0, state = 0;
  utf8proc_option_t options;
  utf8proc_ssize_t ret, bytes = 0;
  utf8proc_uint8_t *str = NULL;
  size_t len = strlen((const char *)data);

  while (bytes != len) {
    ret = utf8proc_iterate(ptr, -1, &c);

    if (ret < 0) {
      utf8proc_errmsg(ret);
      break;
    }
    if (ret == 0) {
      break;
    }

    bytes += ret;
    ptr += ret;

    utf8proc_tolower(c);
    utf8proc_toupper(c);
    utf8proc_totitle(c);
    utf8proc_islower(c);
    utf8proc_isupper(c);
    utf8proc_charwidth(c);
    utf8proc_category(c);
    utf8proc_category_string(c);
    utf8proc_codepoint_valid(c);

    utf8proc_grapheme_break(c_prev, c);
    utf8proc_grapheme_break_stateful(c_prev, c, &state);

    c_prev = c;
  }

  /*
   * The original fuzzer has a bug here. It allocates `size` bytes,
   * then divides `size` by 4, and then uses the smaller `size` for memcpy.
   * This should use a temporary variable for the length.
   */
  size_t original_size = size;
  utf8proc_int32_t *copy = original_size >= 4 ? malloc(original_size) : NULL;

  if (copy) {
    size_t len_utf32 = original_size / 4;

    options = UTF8PROC_STRIPCC | UTF8PROC_NLF2LS | UTF8PROC_NLF2PS;
    memcpy(copy, data, len_utf32 * 4);
    ret = utf8proc_normalize_utf32(copy, len_utf32, options);
    if (ret < 0) utf8proc_errmsg(ret);

    options = UTF8PROC_STRIPCC | UTF8PROC_NLF2LS;
    memcpy(copy, data, len_utf32 * 4);
    ret = utf8proc_normalize_utf32(copy, len_utf32, options);
    if (ret < 0) utf8proc_errmsg(ret);

    options = UTF8PROC_STRIPCC | UTF8PROC_NLF2PS;
    memcpy(copy, data, len_utf32 * 4);
    ret = utf8proc_normalize_utf32(copy, len_utf32, options);
    if (ret < 0) utf8proc_errmsg(ret);

    options = UTF8PROC_STRIPCC;
    memcpy(copy, data, len_utf32 * 4);
    ret = utf8proc_normalize_utf32(copy, len_utf32, options);
    if (ret < 0) utf8proc_errmsg(ret);

    options = UTF8PROC_LUMP;
    memcpy(copy, data, len_utf32 * 4);
    ret = utf8proc_normalize_utf32(copy, len_utf32, options);
    if (ret < 0) utf8proc_errmsg(ret);

    options = 0;
    memcpy(copy, data, len_utf32 * 4);
    ret = utf8proc_normalize_utf32(copy, len_utf32, options);
    if (ret < 0) utf8proc_errmsg(ret);

    free(copy);
  }

  free(utf8proc_NFD(data));
  free(utf8proc_NFC(data));
  free(utf8proc_NFKD(data));
  free(utf8proc_NFKC(data));
  free(utf8proc_NFKC_Casefold(data));

  ret = utf8proc_map(data, len, &str, UTF8PROC_CHARBOUND | UTF8PROC_STRIPNA);
  if (ret < 0) utf8proc_errmsg(ret);
  free(str);

  ret = utf8proc_map(data, len, &str, UTF8PROC_LUMP | UTF8PROC_NLF2LS | UTF8PROC_NLF2PS);
  if (ret < 0) utf8proc_errmsg(ret);
  free(str);

  ret = utf8proc_map(data, len, &str, UTF8PROC_COMPOSE | UTF8PROC_STRIPMARK);
  if (ret < 0) utf8proc_errmsg(ret);
  free(str);

  return 0;
}
