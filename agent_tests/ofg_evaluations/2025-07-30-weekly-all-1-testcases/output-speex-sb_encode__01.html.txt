
	
<description>
The `sb_encode` function is the main encoding function for the Speex wideband codec. It takes an encoder state, an input audio frame, and a `SpeexBits` struct as input, and writes the encoded bitstream to the `SpeexBits` struct.
</description>
<requirements>

<requirement>
The first argument, `state`, must be a valid pointer to an `SBEncState` struct initialized by `sb_encoder_init` with `&speex_wb_mode`. This is because `sb_encode` is the encoder function for the wideband mode, and the state must be initialized accordingly.
</requirement>
<requirement>
The second argument, `vin`, must be a pointer to a buffer of `spx_word16_t` of size `st->full_frame_size`. The `full_frame_size` is a member of the `SBEncState` struct and is initialized in `sb_encoder_init`.
</requirement>
<requirement>
The third argument, `bits`, must be a valid pointer to a `SpeexBits` struct initialized by `speex_bits_init`. This is because the encoded bitstream is written to this struct.
</requirement>
<requirement>
Before calling `sb_encode`, `speex_bits_reset` should be called on the `SpeexBits` struct. This is to ensure that the struct is empty and ready to receive the new encoded frame.
</requirement>

</requirements>

#define FIXED_POINT
#include <fuzzer/FuzzedDataProvider.h>
#include <stdint.h>
#include <stdlib.h>
#include <vector>

#include "speex/speex.h"

/*
 * Wrap the C header include in extern "C" to prevent C++ name mangling,
 * which resolves the "undefined reference" linker error.
 */
extern "C" {
#include "/src/speex/libspeex/sb_celp.h"
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider fdp(data, size);

    /* sb_encode is part of the sub-band codec, so we must use wideband or ultra-wideband mode.
       We use the public API to create a valid state object. */
    void *st = speex_encoder_init(&speex_wb_mode);
    if (!st) {
        return 0;
    }

    /* Set various encoder parameters using fuzzed data to explore more code paths.
       This logic is borrowed from the existing speexenc_fuzzer.cc */
    spx_int32_t quality = fdp.ConsumeIntegralInRange(0, 10);
    spx_int32_t complexity = fdp.ConsumeIntegralInRange(0, 10);
    spx_int32_t vbr_enabled = fdp.ConsumeBool();
    spx_int32_t vad_enabled = fdp.ConsumeBool();
    spx_int32_t dtx_enabled = fdp.ConsumeBool();

    speex_encoder_ctl(st, SPEEX_SET_QUALITY, &quality);
    speex_encoder_ctl(st, SPEEX_SET_COMPLEXITY, &complexity);
    speex_encoder_ctl(st, SPEEX_SET_VBR, &vbr_enabled);
    speex_encoder_ctl(st, SPEEX_SET_VAD, &vad_enabled);
    speex_encoder_ctl(st, SPEEX_SET_DTX, &dtx_enabled);

    /* Get frame size. For wideband, the full input frame for sb_encode is 2x this size. */
    spx_int32_t frame_size;
    speex_encoder_ctl(st, SPEEX_GET_FRAME_SIZE, &frame_size);
    if (frame_size <= 0) {
        speex_encoder_destroy(st);
        return 0;
    }
    int full_frame_size = 2 * frame_size;

    /* Prepare input buffer for sb_encode. It expects a buffer of spx_int16_t. */
    std::vector<spx_int16_t> input_frame(full_frame_size);
    size_t consumed_bytes = fdp.ConsumeData(input_frame.data(), input_frame.size() * sizeof(spx_int16_t));

    /* We need a full frame to proceed. */
    if (consumed_bytes < input_frame.size() * sizeof(spx_int16_t)) {
        speex_encoder_destroy(st);
        return 0;
    }

    /* Initialize SpeexBits for the encoded output. */
    SpeexBits bits;
    speex_bits_init(&bits);

    /* Call the target function. */
    sb_encode(st, input_frame.data(), &bits);

    /* Clean up resources. */
    speex_bits_destroy(&bits);
    speex_encoder_destroy(st);

    return 0;
}
