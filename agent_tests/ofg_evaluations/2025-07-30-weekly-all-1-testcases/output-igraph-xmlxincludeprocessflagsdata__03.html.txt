</requirement>
<requirement>
The doc argument should be created by functions like xmlReadFile or xmlReadMemory. This is to ensure that the document is well-formed and can be processed correctly.
</requirement>
<requirement>
The third argument, void *data, is passed as the private data to the XInclude context. Based on the calling context, this can be set to NULL.
</requirement>

#include <fuzzer/FuzzedDataProvider.h>
#include <cstddef>
#include <cstdint>
#include <vector>

extern "C" {
#include "libxml/parser.h"
#include "libxml/xinclude.h"
#include "libxml/xmlerror.h"
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    const int flags = provider.ConsumeIntegral<int>();
    std::vector<char> xml_data = provider.ConsumeRemainingBytes<char>();

    // Suppress error messages
    xmlSetGenericErrorFunc(NULL, NULL);

    xmlDocPtr doc = xmlReadMemory(xml_data.data(), xml_data.size(),
                                "fuzz.xml", NULL, 0);

    if (doc != NULL) {
        xmlXIncludeProcessFlagsData(doc, flags, NULL);
        xmlFreeDoc(doc);
    }

    return 0;
}
