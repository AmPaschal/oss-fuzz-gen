
	
<description>
The function processes a FLAC stream metadata block. It determines the type of the metadata block and creates a corresponding object to handle it. For certain metadata types (SEEKTABLE, PICTURE, CUESHEET), it performs additional validation by calling the is_legal() method. The function is designed to be used as a callback during the FLAC stream decoding process.
</description>
<requirements>
<requirement>
The `metadata` argument must be a pointer to a valid `FLAC__StreamMetadata` structure. This is because the function dereferences this pointer to access its members.
</requirement>
<requirement>
The `metadata->type` field should be initialized with one of the following values: `FLAC__METADATA_TYPE_STREAMINFO`, `FLAC__METADATA_TYPE_PADDING`, `FLAC__METADATA_TYPE_APPLICATION`, `FLAC__METADATA_TYPE_SEEKTABLE`, `FLAC__METADATA_TYPE_VORBIS_COMMENT`, `FLAC__METADATA_TYPE_CUESHEET`, or `FLAC__METADATA_TYPE_PICTURE`. The function only processes these specific metadata types and will return early for any other type.
</requirement>
<requirement>
The `metadata->data` field must contain valid data corresponding to the `metadata->type`. This is crucial because the function uses this data to construct different metadata objects, and invalid data can lead to crashes or undefined behavior during object construction or in subsequent method calls like `is_valid()` and `is_legal()`.
</requirement>
<requirement>
The `metadata->length` field must accurately represent the size of the data in `metadata->data`. An incorrect length can cause buffer over-reads or other memory-related errors within the object constructors or validation methods.
</requirement>
</requirements>

namespace FLAC {
	namespace Decoder {
        class FuzzerStream : public Stream {
            public:
                FuzzerStream() : Stream() { }

                ::FLAC__StreamDecoderReadStatus read_callback(FLAC__byte buffer[], size_t *bytes)  override {
                    (void)buffer;
                    (void)bytes;
	                return FLAC__STREAM_DECODER_READ_STATUS_ABORT;
                }

                ::FLAC__StreamDecoderWriteStatus write_callback(const ::FLAC__Frame *frame, const FLAC__int32 * const buffer[])  override {
                    (void)frame;
                    (void)buffer;
                    return FLAC__STREAM_DECODER_WRITE_STATUS_CONTINUE;
                }

                void error_callback(::FLAC__StreamDecoderErrorStatus status)  override {
                    (void)status;
                }

                void metadata_callback(const ::FLAC__StreamMetadata *metadata) override {
                    Metadata::Prototype * cloned_object = nullptr;
                    if (metadata->type == FLAC__METADATA_TYPE_STREAMINFO)
                        cloned_object = new Metadata::StreamInfo(metadata);
                    else if (metadata->type == FLAC__METADATA_TYPE_PADDING)
                        cloned_object = new Metadata::Padding(metadata);
                    else if (metadata->type == FLAC__METADATA_TYPE_APPLICATION)
                        cloned_object = new Metadata::Application(metadata);
                    else if (metadata->type == FLAC__METADATA_TYPE_SEEKTABLE)
                        cloned_object = new Metadata::SeekTable(metadata);
                    else if (metadata->type == FLAC__METADATA_TYPE_VORBIS_COMMENT)
                        cloned_object = new Metadata::VorbisComment(metadata);
                    else if (metadata->type == FLAC__METADATA_TYPE_CUESHEET)
                        cloned_object = new Metadata::CueSheet(metadata);
                    else if (metadata->type == FLAC__METADATA_TYPE_PICTURE)
                        cloned_object = new Metadata::Picture(metadata);
                    else
                        return;
                    if (0 != cloned_object && *cloned_object == *metadata && cloned_object->is_valid()) {
                        if (cloned_object->get_type() == FLAC__METADATA_TYPE_SEEKTABLE)
                            dynamic_cast<Metadata::SeekTable *>(cloned_object)->is_legal();
                        if (cloned_object->get_type() == FLAC__METADATA_TYPE_PICTURE)
                            dynamic_cast<Metadata::Picture *>(cloned_object)->is_legal(NULL);
                        if (cloned_object->get_type() == FLAC__METADATA_TYPE_CUESHEET)
                            dynamic_cast<Metadata::CueSheet *>(cloned_object)->is_legal(true,NULL);
                    }
                    delete cloned_object;
                }
        };
    }
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    FLAC::Decoder::FuzzerStream decoder;
    
    FLAC__StreamMetadata_StreamInfo stream_info;
    if (provider.remaining_bytes() < sizeof(stream_info)) {
        return 0;
    }
    provider.ConsumeData(&stream_info, sizeof(stream_info));

    FLAC__StreamMetadata metadata;
    metadata.type = FLAC__METADATA_TYPE_STREAMINFO;
    metadata.is_last = provider.ConsumeBool();
    metadata.length = sizeof(stream_info);
    memcpy(&metadata.data.stream_info, &stream_info, sizeof(stream_info));
    
    decoder.metadata_callback(&metadata);

    return 0;
}
