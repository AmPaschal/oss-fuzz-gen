
	
<description>
The function `netsnmp_query_getnext` is a wrapper around the `_query` function, which sends an SNMP GETNEXT request to the specified agent. It takes a variable list and a session as input, and returns the result of the query. The variable list contains the OIDs to be queried, and the session contains the information about the agent to be queried.
</description>
<requirements>

<requirement>
The first argument, a pointer to a `netsnmp_variable_list` struct, must not be NULL. This is to prevent a null-pointer dereference in the `_query` function.
</requirement>
<requirement>
The `netsnmp_variable_list` should be initialized by calling `snmp_varlist_add_variable`. This function correctly allocates and initializes the list, and adds a variable to it.
</requirement>
<requirement>
The second argument, a pointer to a `netsnmp_session` struct, must not be NULL unless a default session is set. If this argument is NULL and no default session is set, the function will return an error.
</requirement>
<requirement>
The `netsnmp_session` should be initialized by calling `snmp_open`. This ensures that the session is valid and connected.
</requirement>
<requirement>
Before calling `snmp_open`, the `snmp_parse_args` function must be called to initialize the `netsnmp_session` structure. This is because `snmp_open` relies on the fields of this structure, which are populated by `snmp_parse_args`.
</requirement>
<requirement>
Before calling `snmp_parse_args`, the `init_snmp` function must be called. This is because `snmp_parse_args` relies on the MIBs loaded by `init_snmp`.
</requirement>

</requirements>

#include <net-snmp/net-snmp-config.h>
#include <net-snmp/net-snmp-includes.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

/* Function signature to be fuzzed */
int netsnmp_query_getnext(netsnmp_variable_list *, netsnmp_session *);

int LLVMFuzzerInitialize(int *argc, char ***argv) {
    if (getenv("NETSNMP_DEBUGGING") != NULL) {
        snmp_enable_stderrlog();
        snmp_set_do_debugging(1);
        debug_register_tokens("");
    }
    /* Initializes the SNMP library */
    init_snmp("netsnmp_query_getnext_fuzzer");
    return 0;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size == 0 || size > 256) {
        return 0;
    }

    /* 1. Initialize session */
    netsnmp_session session, *ss;
    snmp_sess_init(&session);
    session.peername = "udp:127.0.0.1:9161";
    session.version = SNMP_VERSION_2c;

    /* Use fuzzer data for community string. Must be NULL-terminated. */
    char community[257];
    memcpy(community, data, size);
    community[size] = '\0';
    session.community = (u_char *)community;
    session.community_len = strlen(community);

    /* Avoid blocking */
    session.retries = 0;
    session.timeout = 50000; /* 50 ms */

    ss = snmp_open(&session);
    if (ss == NULL) {
        /* Expected to fail in network-less sandbox */
        return 0;
    }

    /* 2. Create variable list */
    netsnmp_variable_list *vars =
        (netsnmp_variable_list *)calloc(1, sizeof(netsnmp_variable_list));
    if (vars == NULL) {
        snmp_close(ss);
        return 0;
    }

    /* Use a fixed valid OID for the GETNEXT query */
    oid an_oid[] = {1, 3, 6, 1, 2, 1, 1, 1, 0};
    size_t an_oid_len = OID_LENGTH(an_oid);
    snmp_set_var_objid(vars, an_oid, an_oid_len);
    vars->type = ASN_NULL;

    /* 3. Call function under test */
    netsnmp_query_getnext(vars, ss);

    /* 4. Cleanup */
    snmp_free_varbind(vars);
    snmp_close(ss);

    return 0;
}
