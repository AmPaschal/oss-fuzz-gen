extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  std::string input(reinterpret_cast<const char *>(data), size);
  SQLParserResult result;
  SQLParser::parse(input, &result);

  if (!result.isValid()) {
    return 0;
  }

  // Define a recursive lambda to traverse the expression tree.
  std::function<void(const Expr*)> FuzzExprRecursive =
      [&](const Expr* expr) {
    if (!expr) {
      return;
    }
    // Call the function under test.
    expr->isLiteral();

    // Recurse into sub-expressions.
    FuzzExprRecursive(expr->expr);
    FuzzExprRecursive(expr->expr2);
    if (expr->exprList) {
      for (const Expr* e : *expr->exprList) {
        FuzzExprRecursive(e);
      }
    }
  };

  // Iterate through all statements and start the recursive traversal.
  for (const SQLStatement* stmt : result.getStatements()) {
    if (stmt->isType(kStmtSelect)) {
      const auto* select_stmt = static_cast<const SelectStatement*>(stmt);
      if (select_stmt->selectList) {
        for (const Expr* expr : *select_stmt->selectList) {
          FuzzExprRecursive(expr);
        }
      }
      FuzzExprRecursive(select_stmt->whereClause);
    } else if (stmt->isType(kStmtInsert)) {
        const auto* insert_stmt = static_cast<const InsertStatement*>(stmt);
        if (insert_stmt->values) {
            for (const Expr* expr : *insert_stmt->values) {
                FuzzExprRecursive(expr);
            }
        }
    } else if (stmt->isType(kStmtUpdate)) {
        const auto* update_stmt = static_cast<const UpdateStatement*>(stmt);
        if (update_stmt->updates) {
            for (const UpdateClause* uc : *update_stmt->updates) {
                FuzzExprRecursive(uc->value);
            }
        }
        FuzzExprRecursive(update_stmt->where);
    }
  }

  return 0;
}
