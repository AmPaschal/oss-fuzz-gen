The function takes a pointer to a struct evhttp and a file descriptor as input. It returns 0 on success and -1 on failure. The function is used in libevent's test suite to test the server's ability to accept connections from pre-existing sockets.
</description>

#include <fuzzer/FuzzedDataProvider.h>
#include <sys/socket.h>
#include <unistd.h>

#include "/src/libevent/include/event2/http.h"
#include "/src/libevent/include/event2/event.h"
#include "/src/libevent/include/event2/util.h"

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    struct event_base *base = event_base_new();
    if (!base) {
        return 0;
    }

    struct evhttp *http = evhttp_new(base);
    if (!http) {
        event_base_free(base);
        return 0;
    }

    evutil_socket_t fd[2];
    if (evutil_socketpair(AF_UNIX, SOCK_STREAM, 0, fd) == -1) {
        evhttp_free(http);
        event_base_free(base);
        return 0;
    }
    
    FuzzedDataProvider provider(data, size);
    std::string client_data = provider.ConsumeRemainingBytesAsString();
    if (!client_data.empty()) {
        // Simulate a client sending data.
        write(fd[1], client_data.c_str(), client_data.size());
    }

    // Call the target function.
    evhttp_accept_socket(http, fd[0]);

    // The accepted socket is managed by the evhttp connection object,
    // which is freed by evhttp_free. We just need to close our ends.
    evutil_closesocket(fd[0]);
    evutil_closesocket(fd[1]);
    evhttp_free(http);
    event_base_free(base);

    return 0;
}
