
	
<description>
The function `modbus_write_and_read_registers` writes a block of contiguous registers and reads another block of contiguous registers in a single atomic transaction. It takes a pointer to a `modbus_t` context, the write address and number of registers to write, a pointer to the data to write, the read address and number of registers to read, and a pointer to store the read data.
</description>
<requirements>

<requirement>
The first argument, `ctx`, must be a valid pointer to a `modbus_t` struct initialized by `modbus_new_tcp`, `modbus_new_tcp_pi`, or `modbus_new_rtu`. This is because the function relies on a properly initialized context to manage the modbus connection and state.
</requirement>
<requirement>
The `modbus_connect` function must be called on the `ctx` object before calling `modbus_write_and_read_registers`. This is necessary to establish a connection to the modbus server.
</requirement>
<requirement>
The third argument, `write_nb`, must be in the range [1, 121]. This is to prevent a buffer overflow, as the function checks this value against `MODBUS_MAX_WR_WRITE_REGISTERS`.
</requirement>
<requirement>
The sixth argument, `read_nb`, must be in the range [1, 125]. This is to prevent a buffer overflow, as the function checks this value against `MODBUS_MAX_WR_READ_REGISTERS`.
</requirement>
<requirement>
The fourth argument, `src`, must be a valid pointer to an array of `uint16_t` of at least `write_nb` elements. This is because the function reads `write_nb` elements from this array to send to the modbus server.
</requirement>
<requirement>
The seventh argument, `dest`, must be a valid pointer to an array of `uint16_t` of at least `read_nb` elements. This is because the function writes `read_nb` elements to this array from the data read from the modbus server.
</requirement>
<requirement>
The `write_addr` and `read_addr` arguments should be fuzzed. These arguments represent the starting addresses for writing and reading registers, and their values can be arbitrary.
</requirement>

</requirements>

#define PORT 8080
#define kMinInputLength 9
#define kMaxInputLength MODBUS_RTU_MAX_ADU_LENGTH

struct Fuzzer {
  uint16_t port;
  char *file;

  FILE *inFile;
  uint64_t size;
  const uint8_t *buffer;

  pthread_t thread;
  int socket;
};
typedef struct Fuzzer Fuzzer;

int client(Fuzzer *fuzzer);

void fuzzinit(Fuzzer *fuzzer) {
  struct sockaddr_in server_addr;
  fuzzer->socket = socket(AF_INET, SOCK_STREAM, 0);

  server_addr.sin_family = AF_INET;
  server_addr.sin_port = htons(fuzzer->port);
  server_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

  setsockopt(fuzzer->socket, SOL_SOCKET, SO_REUSEADDR, &(int){1}, sizeof(int));

  bind(fuzzer->socket, (struct sockaddr *)&server_addr, sizeof(server_addr));
  listen(fuzzer->socket, 1);
}

void *Server(void *args) {

  Fuzzer *fuzzer = (Fuzzer *)args;
  {
    int client;
    char clientData[10240];
    struct sockaddr_in clientAddr;
    uint32_t clientSZ = sizeof(clientAddr);

    client = accept(fuzzer->socket, (struct sockaddr *)&clientAddr, &clientSZ);

    /* The client sends a request, but we don't use it. */
    recv(client, clientData, sizeof(clientData), 0);
    /* Instead we send the fuzzer data as a response. */
    send(client, fuzzer->buffer, fuzzer->size, 0);

    shutdown(client, SHUT_RDWR);
    close(client);
  }
  pthread_exit(NULL);
}

void clean(Fuzzer *fuzzer) {
  { // Server
    shutdown(fuzzer->socket, SHUT_RDWR);
    close(fuzzer->socket);
  }
  free(fuzzer);
}

extern int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

  if (size < kMinInputLength || size > kMaxInputLength) {
    return 0;
  }

  Fuzzer *fuzzer = (Fuzzer *)malloc(sizeof(Fuzzer));
  fuzzer->port = PORT;
  fuzzer->size = size;
  fuzzer->buffer = data;

  fuzzinit(fuzzer);

  pthread_create(&fuzzer->thread, NULL, Server, fuzzer);
  client(fuzzer);
  pthread_join(fuzzer->thread, NULL); /* To Avoid UAF*/

  clean(fuzzer);
  return 0;
}

int client(Fuzzer *fuzzer) {
  uint16_t *tab_rp_registers = NULL;
  modbus_t *ctx = NULL;
  int rc;
  int nb_points;

  ctx = modbus_new_tcp("127.0.0.1", fuzzer->port);

  if (ctx == NULL) {
    fprintf(stderr, "Unable to allocate libmodbus context\n");
    return -1;
  }

  if (modbus_connect(ctx) == -1) {
    fprintf(stderr, "Connection failed: %s\n", modbus_strerror(errno));
    modbus_free(ctx);
    return -1;
  }

  /* Allocate and initialize the memory to store the registers */
  nb_points = (UT_REGISTERS_NB > UT_INPUT_REGISTERS_NB) ? UT_REGISTERS_NB : UT_INPUT_REGISTERS_NB;
  tab_rp_registers = (uint16_t *)malloc(nb_points * sizeof(uint16_t));
  memset(tab_rp_registers, 0, nb_points * sizeof(uint16_t));

  /*
   * Call the target function.
   * It will send a request and then receive the response from our mock server,
   * which is controlled by the fuzzer input.
   */
  rc = modbus_write_and_read_registers(ctx, UT_REGISTERS_ADDRESS, UT_REGISTERS_NB, UT_REGISTERS_TAB, UT_REGISTERS_ADDRESS, UT_REGISTERS_NB, tab_rp_registers);

  /* Free the memory */
  free(tab_rp_registers);

  /* Close the connection */
  modbus_close(ctx);
  modbus_free(ctx);

  return rc;
}
