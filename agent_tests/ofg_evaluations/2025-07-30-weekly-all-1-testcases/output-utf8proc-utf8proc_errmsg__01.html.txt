#include <string.h>
#include <stdint.h>
#include "/src/utf8proc/utf8proc.h"

const char * utf8proc_errmsg(utf8proc_ssize_t);

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size < 1)
    return 0;

  /* Avoid timeout with long inputs */
  if (size > (64 * 1024))
    return 0;

  /* Test with raw data to hit the default case and other values */
  if (size >= sizeof(utf8proc_ssize_t)) {
    utf8proc_ssize_t err_code;
    memcpy(&err_code, data, sizeof(utf8proc_ssize_t));
    utf8proc_errmsg(err_code);
  }

  /* The rest of the fuzzer is based on the original fuzzer.c */
  /* to generate valid error codes from the library's functions. */

  if (data[size - 1] != '\0')
    return 0;

  const uint8_t *ptr = data;
  utf8proc_int32_t c = 0, c_prev = 0, state = 0;
  utf8proc_option_t options;
  utf8proc_ssize_t ret, bytes = 0;
  size_t len = strlen((const char *)data);

  while (bytes < len) {
    ret = utf8proc_iterate(ptr, -1, &c);

    if (ret < 0) {
      utf8proc_errmsg(ret);
      break;
    }
    if (ret == 0) {
      break;
    }

    bytes += ret;
    ptr += ret;

    utf8proc_tolower(c);
    utf8proc_toupper(c);
    utf8proc_totitle(c);
    utf8proc_islower(c);
    utf8proc_isupper(c);
    utf8proc_charwidth(c);
    utf8proc_category(c);
    utf8proc_category_string(c);
    utf8proc_codepoint_valid(c);

    utf8proc_grapheme_break(c_prev, c);
    utf8proc_grapheme_break_stateful(c_prev, c, &state);

    c_prev = c;
  }

  if (size >= 4) {
    /* size must be a multiple of 4 for utf8proc_int32_t */
    size_t utf32_len = size / 4;
    utf8proc_int32_t *copy = malloc(utf32_len * sizeof(utf8proc_int32_t));
    if (copy) {
      memcpy(copy, data, utf32_len * sizeof(utf8proc_int32_t));

      options = UTF8PROC_STRIPCC | UTF8PROC_NLF2LS | UTF8PROC_NLF2PS;
      ret = utf8proc_normalize_utf32(copy, utf32_len, options);
      if (ret < 0) utf8proc_errmsg(ret);

      options = UTF8PROC_LUMP;
      ret = utf8proc_normalize_utf32(copy, utf32_len, options);
      if (ret < 0) utf8proc_errmsg(ret);

      free(copy);
    }
  }

  free(utf8proc_NFD(data));
  free(utf8proc_NFC(data));
  free(utf8proc_NFKD(data));
  free(utf8proc_NFKC(data));
  free(utf8proc_NFKC_Casefold(data));

  return 0;
}
