
	
<description>
The function avahi_string_list_new creates a new AvahiStringList and populates it with strings. It takes a variable number of const char * arguments, with the list terminated by a NULL pointer. The first argument is mandatory and if it is NULL the function returns NULL. Subsequent arguments are added to the list until a NULL argument is encountered.
</description>
<requirements>
<requirement>
The first argument, txt, must not be NULL. This is because the function will not create the string list if the first argument is NULL.
</requirement>
<requirement>
All subsequent string arguments passed to the function must not be NULL. This is because avahi_string_list_add asserts that its text argument is not NULL.
</requirement>
<requirement>
The variable argument list must be terminated by a NULL pointer. This is to signal the end of the arguments to avahi_string_list_add_many_va.
</requirement>
</requirements>

#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

#include "/src/avahi/avahi-common/strlst.h"

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    /* avahi_string_list_new() expects a NUL-terminated string.
     * The fuzzer input is not guaranteed to be NUL-terminated, so we must create a copy. */
    char *nul_terminated_str = (char *) malloc(size + 1);
    if (!nul_terminated_str) {
        return 0; /* Out of memory */
    }
    memcpy(nul_terminated_str, data, size);
    nul_terminated_str[size] = '\0';

    /* Call the function under test.
     * The function is variadic and the argument list must be terminated by a NULL pointer.
     * We pass one string from the fuzzer and the terminating NULL. */
    AvahiStringList *s = avahi_string_list_new(nul_terminated_str, NULL);

    if (s) {
        avahi_string_list_free(s);
    }

    free(nul_terminated_str);

    return 0;
}
