
	
<description>
The function `firebase::firestore::core::Query::MatchesBounds(const Document &)` checks if a given `Document` falls within the start and end bounds of a `Query`. The function retrieves the `start_at_` and `end_at_` bounds from the `Query` object and uses the `SortsBeforeDocument` and `SortsAfterDocument` methods of the `Bound` class to perform the check. The `Document` object is a wrapper around a `MutableDocument`, which can be created using various factory methods like `FoundDocument`. Creating a `FoundDocument` involves creating a `DocumentKey` and an `ObjectValue`. The `Bound` objects are created from `google_firestore_v1_ArrayValue` protobuf messages.
</description>

// Helper to create a Bound with a single fuzzed integer value. This simplifies
// the complex task of creating a nanopb ArrayValue message from scratch.
absl::optional<Bound> CreateBound(FuzzedDataProvider& provider) {
  try {
    Message<google_firestore_v1_ArrayValue> pos_proto;

    pos_proto->values_count = 1;
    pos_proto->values[0].which_value_type =
        google_firestore_v1_Value_integer_value_tag;
    pos_proto->values[0].integer_value = provider.ConsumeIntegral<int64_t>();

    return Bound::FromValue(std::move(pos_proto), provider.ConsumeBool());
  } catch (...) {
    // Errors can happen during nanopb message creation.
    return absl::nullopt;
  }
}

  FuzzedDataProvider provider(data, size);

  // Create a Document. A valid document path must have an even number of

  std::string doc_path_str = "c/" + provider.ConsumeRandomLengthString(128);
  Document doc;
  try {
    DocumentKey doc_key = DocumentKey::FromPathString(doc_path_str);
    // Use InvalidDocument to avoid Timestamp/SnapshotVersion issues.
    MutableDocument mutable_doc = MutableDocument::InvalidDocument(doc_key);
    doc = Document(mutable_doc);
  } catch (...) {
    // Invalid path or other creation error.
    return 0;
  }

  // Create a Query. A query path can have any number of segments.
  std::string query_path_str = provider.ConsumeRandomLengthString(128);
  Query query;
  try {
    query = Query(ResourcePath::FromString(query_path_str));
  } catch (...) {
    // Invalid path or other creation error.
    return 0;
  }

  // Randomly add start and end bounds to the query.
  if (provider.ConsumeBool()) {
    if (absl::optional<Bound> bound = CreateBound(provider)) {
      query = query.StartingAt(*bound);
    }
  }
  if (provider.ConsumeBool()) {
    if (absl::optional<Bound> bound = CreateBound(provider)) {
      query = query.EndingAt(*bound);
    }
  }

  // Call the function under test.
  query.Matches(doc);

  return 0;
