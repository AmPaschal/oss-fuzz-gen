// To use FLAC::Decoder::File, we must subclass it and implement the
// pure virtual callbacks from the FLAC::Decoder::Stream base class.
class FuzzDecoder : public FLAC::Decoder::File {
public:
	FuzzDecoder() = default;
	~FuzzDecoder() override = default;

protected:
	// This is called when the decoder has a chunk of decoded audio.
	// We don't need to do anything with it for fuzzing.
	::FLAC__StreamDecoderWriteStatus write_callback(const ::FLAC__Frame *frame, const FLAC__int32 * const buffer[]) override {
		(void)frame;
		(void)buffer;
		return FLAC__STREAM_DECODER_WRITE_STATUS_CONTINUE;
	}

	// This is called when the decoder encounters an error.
	void error_callback(::FLAC__StreamDecoderErrorStatus status) override {
		(void)status;
		// Do nothing.
	}
};

	}

	// The decoder's destructor will call finish() and clean up resources.
	return 0;
