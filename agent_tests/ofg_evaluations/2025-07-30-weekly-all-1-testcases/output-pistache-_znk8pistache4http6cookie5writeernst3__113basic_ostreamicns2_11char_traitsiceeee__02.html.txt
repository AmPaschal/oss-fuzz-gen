extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    std::string name = provider.ConsumeRandomLengthString(50);
    std::string value = provider.ConsumeRandomLengthString(100);
    
    // The Cookie constructor does not allow an empty name.
    if (name.empty()) {
        return 0;
    }

    Pistache::Http::Cookie cookie(name, value);

    if (provider.ConsumeBool()) {
        cookie.path = provider.ConsumeRandomLengthString(50);
    }
    if (provider.ConsumeBool()) {
        cookie.domain = provider.ConsumeRandomLengthString(50);
    }
    if (provider.ConsumeBool()) {
        cookie.maxAge = provider.ConsumeIntegral<int>();
    }
    cookie.secure = provider.ConsumeBool();
    cookie.httpOnly = provider.ConsumeBool();

    if (provider.ConsumeBool()) {
        auto time_t_val = provider.ConsumeIntegral<time_t>();
        std::chrono::system_clock::time_point tp = std::chrono::system_clock::from_time_t(time_t_val);
        Pistache::Http::FullDate full_date(tp);
        cookie.expires = full_date;
    }

    while (provider.remaining_bytes() > 10 && provider.ConsumeBool()) {
        std::string ext_key = provider.ConsumeRandomLengthString(20);
        std::string ext_val = provider.ConsumeRandomLengthString(50);
        if (!ext_key.empty()) {
            cookie.ext.insert({ext_key, ext_val});
        }
    }

    std::stringstream ss;
    ss << cookie;

    return 0;
