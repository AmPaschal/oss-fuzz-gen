enum class ParseableTypes {
  kInt,
  kString,
  kVectorString,
  kFloat,
  kDouble,

  // Marker for fuzzer.
  kMaxValue,

template <typename T> void add_fuzzed_option(cxxopts::Options *options, FuzzedDataProvider *provider) {
  assert(options);
  assert(provider);

  options->add_options()(provider->ConsumeRandomLengthString(kMaxArgSize), provider->ConsumeRandomLengthString(kMaxArgSize), cxxopts::value<T>());
}
