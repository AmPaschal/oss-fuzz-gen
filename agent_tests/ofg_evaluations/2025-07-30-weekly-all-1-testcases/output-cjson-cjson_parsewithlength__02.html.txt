  size_t offset = 4;
  unsigned char *copied;
  char *printed_json = NULL;
  int minify, formatted, buffered;

  if (size <= offset)
    return 0;

  /* We use the first few bytes of the input as options. */
  if (data[0] != '1' && data[0] != '0')
    return 0;
  /* byte 1 is unused */
  if (data[2] != '1' && data[2] != '0')
    return 0;
  if (data[3] != '1' && data[3] != '0')
    return 0;

  minify = data[0] == '1' ? 1 : 0;
  formatted = data[2] == '1' ? 1 : 0;
  buffered = data[3] == '1' ? 1 : 0;

  /* Call the function under test */
  json = cJSON_ParseWithLength((const char *)data + offset, size - offset);

  if (json == NULL)
    return 0;

  /* The following code is kept from the original fuzzer to exercise the
   * library with the parsed json object. */
  if (buffered) {
    printed_json = cJSON_PrintBuffered(json, 1, formatted);
  } else {
    /* unbuffered printing */
    if (formatted) {
      printed_json = cJSON_Print(json);
    } else {
      printed_json = cJSON_PrintUnformatted(json);
    }
  }

  if (printed_json != NULL)
    free(printed_json);

  if (minify) {
    copied = (unsigned char *)malloc(size);
    if (copied == NULL) {
      cJSON_Delete(json);
      return 0;
    }

    memcpy(copied, data, size);

    cJSON_Minify((char *)copied + offset);

    free(copied);
  }

  cJSON_Delete(json);

  return 0;
}
