The `Pistache::Rest::Private::RouterHandler::onRequest` function is a key component in the Pistache routing mechanism. It receives an HTTP request and a response writer, and its primary responsibility is to delegate the routing of the request to the underlying `Pistache::Rest::Router` object. The `RouterHandler` itself doesn't contain complex logic; it acts as a bridge between the HTTP handling layer and the routing engine. To function correctly, the `RouterHandler` must be initialized with a valid `Router` object that has at least one route configured. The `onRequest` method then calls the `route` method of this `Router` object, passing the request and response writer. The `route` method, in turn, is responsible for matching the request's resource path against the configured routes and invoking the appropriate handler.
</description>

<requirement>

void fuzz_request_parser(const std::string &input) {
  constexpr size_t maxDataSize = 4096;
  Pistache::Http::RequestParser rparser(maxDataSize);

  if (rparser.feed(input.data(), input.size())) {
    auto state = Pistache::Http::Private::State::Done;
    ignoreExceptions([&] { state = rparser.parse(); });

    if (state == Pistache::Http::Private::State::Again)
      ignoreExceptions([&] { rparser.parse(); });
  }
}

  auto router = std::make_shared<Pistache::Rest::Router>();
  router->get("/fuzz", [](const Pistache::Rest::Request &, Pistache::Http::ResponseWriter response) {
    response.send(Pistache::Http::Code::Ok);
    return Pistache::Rest::Route::Result::Ok;
  });
  Pistache::Rest::Private::RouterHandler handler(router);

  Pistache::Http::RequestParser parser(input.size());
  parser.feed(input.c_str(), input.size());

  try {
    if (parser.parse() == Pistache::Http::Private::State::Done) {
      Pistache::Http::ResponseWriter response(Pistache::Http::Version::Http11, nullptr, &handler, std::weak_ptr<Pistache::Tcp::Peer>());
      handler.onRequest(parser.request, std::move(response));
    }
  } catch (...) {
  }
}
