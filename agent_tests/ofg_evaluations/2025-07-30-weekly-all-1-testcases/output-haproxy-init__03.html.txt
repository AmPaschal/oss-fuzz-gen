/*
 * The build script renames the main function in haproxy.c to main2
 * to avoid a name clash with the fuzzer's main function. We declare
 * it here so we can call it.
 */
int main2(int argc, char **argv);

/*

 */
int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    /* Don't waste time on empty inputs. */
    if (size == 0) {
        return 0;
    }

    /*
     * We need a null-terminated string for argv. Malloc a buffer that is
     * one byte larger, copy the fuzzing data, and null-terminate it.
     */
    char *arg_buffer = (char *)malloc(size + 1);
    if (arg_buffer == NULL) {
        perror("malloc");
        return 0;
    }
    memcpy(arg_buffer, data, size);
    arg_buffer[size] = '\0';

    /*
     * Create argument vector. argv[0] is the program name.
     * We'll pass the entire fuzzer input as the first argument, argv[1].
     * Some command line options like '-d' cause haproxy to fork, which
     * is undesirable when fuzzing. We add '-dM' to disable this.
     * The haproxy main function may call exit(), which will terminate the
     * fuzzer. This is a known limitation when fuzzing main entry points.
     */
    char *argv[] = {"haproxy", "-dM", arg_buffer, NULL};
    int argc = 3;

    /* Call the target function. */
    main2(argc, argv);

    /* Clean up allocated memory. */
    free(arg_buffer);

    return 0;
}
