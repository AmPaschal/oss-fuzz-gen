// (TShader::parse) which will call this method internally when parsing HLSL.
// bool glslang::HlslGrammar::acceptPostDecls(TQualifier &);

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  glslang::InitializeProcess();

  const EShLanguage stage = provider.PickValueInArray<EShLanguage>({
      EShLangVertex, EShLangTessControl, EShLangTessEvaluation,
      EShLangGeometry, EShLangFragment,  EShLangCompute,
  });

  glslang::TShader shader(stage);
  const std::string shader_source = provider.ConsumeRemainingBytesAsString();
  const char *shader_source_ptr = shader_source.c_str();
  const int shader_source_len = static_cast<int>(shader_source.length());

  shader.setStringsWithLengths(&shader_source_ptr, &shader_source_len, 1);
  shader.setEntryPoint("main");

  // Enable HLSL parsing, which is the path that calls the target function.
  EShMessages messages =
      static_cast<EShMessages>(EShMsgReadHlsl | EShMsgVulkanRules | EShMsgSpvRules);

  // Fuzz other potentially interesting message flags.
  if (provider.ConsumeBool()) {
    messages = static_cast<EShMessages>(messages | EShMsgRelaxedErrors);
  }
  if (provider.ConsumeBool()) {
    messages = static_cast<EShMessages>(messages | EShMsgSuppressWarnings);
  }
  if (provider.ConsumeBool()) {
    messages = static_cast<EShMessages>(messages | EShMsgHlslOffsets);
  }
  if (provider.ConsumeBool()) {
    messages = static_cast<EShMessages>(messages | EShMsgHlslEnable16BitTypes);
  }

  // This call will trigger the HLSL parser and, with the right input,
  // the acceptPostDecls function.
  shader.parse(GetDefaultResources(), 100, false, messages);

  glslang::FinalizeProcess();
  return 0;
}
