The caller `js_json_stringify` passes its arguments directly from the JavaScript environment, which means the inputs can be of any type.

#include <stdlib.h>

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size == 0 || size > 4096) { // Limit input size
        return 0;
    }

    JSRuntime *rt = JS_NewRuntime();
    if (!rt) {
        return 0;
    }
    JSContext *ctx = JS_NewContext(rt);
    if (!ctx) {
        JS_FreeRuntime(rt);
        return 0;
    }

    test_one_input_init(rt, ctx);

    // JS_ParseJSON requires a null-terminated string.
    char *null_terminated_data = (char*)malloc(size + 1);
    if (!null_terminated_data) {
        js_std_free_handlers(rt);
        JS_FreeContext(ctx);
        JS_FreeRuntime(rt);
        return 0;
    }
    memcpy(null_terminated_data, data, size);
    null_terminated_data[size] = '\0';

    JSValue obj = JS_ParseJSON(ctx, (const char *)null_terminated_data, size, "<fuzz>");
    free(null_terminated_data);

    if (!JS_IsException(obj)) {
        JSValue res;

        // Case 1: No replacer or space
        res = JS_JSONStringify(ctx, obj, JS_UNDEFINED, JS_UNDEFINED);
        JS_FreeValue(ctx, res);

        // Case 2: With a simple string replacer
        JSValue replacer = JS_NewString(ctx, "a");
        res = JS_JSONStringify(ctx, obj, replacer, JS_UNDEFINED);
        JS_FreeValue(ctx, res);
        JS_FreeValue(ctx, replacer);

        // Case 3: With a number for space
        JSValue space = JS_NewInt32(ctx, 4);
        res = JS_JSONStringify(ctx, obj, JS_UNDEFINED, space);
        JS_FreeValue(ctx, res);
        JS_FreeValue(ctx, space);
        
        // Case 4: With a string for space
        space = JS_NewString(ctx, "\t");
        res = JS_JSONStringify(ctx, obj, JS_UNDEFINED, space);
        JS_FreeValue(ctx, res);
        JS_FreeValue(ctx, space);
    }

    JS_FreeValue(ctx, obj);

    js_std_free_handlers(rt);
    JS_FreeContext(ctx);
    JS_FreeRuntime(rt);

    return 0;
}
