
	
<description>
The `mg_hello` function initializes a simple HTTP server that listens on a given URL. It takes a URL as input, creates a listening socket, and then enters a loop to poll for incoming connections.
</description>
<requirements>
<requirement>
The input `url` must be a valid URL string that can be parsed by `mg_url_host` and `mg_url_port`. This is because the host and port are extracted from the URL to create a listening socket.
</requirement>
<requirement>
The `url` must contain a valid host and port to which a socket can be bound. An invalid host or port will cause the `socket` or `bind` calls to fail.
</requirement>
<requirement>
The `url` can optionally start with "udp://" to indicate a UDP connection. This is checked to set the connection type.
</requirement>
<requirement>
The `url` can optionally be a TLS URL. The `mg_url_is_ssl` function is used to determine if the URL is a TLS URL, and if so, the connection is configured for TLS.
</requirement>
</requirements>

#define MG_ENABLE_SOCKET 1
#define MG_ENABLE_MBEDTLS 1
#define MG_ENABLE_LOG 0
#include "mongoose.c"

#ifdef __cplusplus
extern "C" int LLVMFuzzerTestOneInput(const uint8_t *, size_t);
#else
int LLVMFuzzerTestOneInput(const uint8_t *, size_t);
#endif

static int s_done;
static void fn(struct mg_connection *c, int ev, void *ev_data) {
  if (ev != MG_EV_POLL && ev != MG_EV_WAKEUP) {
    s_done = 1;
  }
  (void)c;
  (void)ev_data;
}

void mg_hello(const char *url) {
  struct mg_mgr mgr;
  mg_mgr_init(&mgr);
  s_done = 0;
  mg_http_connect(&mgr, url, fn, NULL);
  // Poll for a limited number of times to avoid hangs in the fuzzer
  for (int i = 0; i < 10 && s_done == 0; i++) {
    mg_mgr_poll(&mgr, 0);
  }
  mg_mgr_free(&mgr);
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  // We need a null-terminated string for the URL.
  // Limit size to avoid excessive memory allocation.
  if (size == 0 || size > 2048) {
    return 0;
  }
  char *url = (char *) malloc(size + 1);
  if (url == NULL) {
    return 0;
  }
  memcpy(url, data, size);
  url[size] = '\0';

  mg_hello(url);

  free(url);
  return 0;
}
