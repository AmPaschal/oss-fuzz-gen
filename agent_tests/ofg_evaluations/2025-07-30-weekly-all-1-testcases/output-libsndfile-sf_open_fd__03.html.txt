extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    const int mode = provider.PickValueInArray({SFM_READ, SFM_WRITE, SFM_RDWR});
    const int close_desc = provider.ConsumeBool();

    SF_INFO sfinfo;
    memset(&sfinfo, 0, sizeof(sfinfo));

    if (mode != SFM_READ) {
        sfinfo.samplerate = provider.ConsumeIntegralInRange<int>(1, 192000);
        sfinfo.channels = provider.ConsumeIntegralInRange<int>(1, 16);
        const int major_format = provider.PickValueInArray({
            SF_FORMAT_WAV, SF_FORMAT_AIFF, SF_FORMAT_FLAC, SF_FORMAT_OGG,
            SF_FORMAT_RAW, SF_FORMAT_W64, SF_FORMAT_RF64
        });
        const int sub_format = provider.PickValueInArray({
            SF_FORMAT_PCM_16, SF_FORMAT_PCM_24, SF_FORMAT_PCM_32,
            SF_FORMAT_FLOAT, SF_FORMAT_DOUBLE, SF_FORMAT_VORBIS,
            SF_FORMAT_ULAW, SF_FORMAT_ALAW, SF_FORMAT_IMA_ADPCM
        });
        sfinfo.format = major_format | sub_format;
        if (sfinfo.format == 0) {
            sfinfo.format = SF_FORMAT_WAV | SF_FORMAT_PCM_16;
        }
    }

    std::vector<uint8_t> file_data = provider.ConsumeRemainingBytes<uint8_t>();

    int fd = memfd_create("fuzz_sndfile", 0);
    if (fd < 0) {
        return 0;
    }

    if (!file_data.empty()) {
        ssize_t written = write(fd, file_data.data(), file_data.size());
        if (written < 0 || (size_t)written != file_data.size()) {
            close(fd);
            return 0;
        }
    }

    if (lseek(fd, 0, SEEK_SET) < 0) {
        close(fd);
        return 0;
    }

    SNDFILE *sndfile = sf_open_fd(fd, mode, &sfinfo, close_desc);

    if (sndfile) {
        sf_close(sndfile);
    }

    if (!close_desc) {
        close(fd);
    }

    return 0;
}
