
	
<description>
The function `zip_file_replace` replaces the content of a file in a zip archive. The file to be replaced is identified by its index. The new content is provided as a `zip_source_t`.
</description>
<requirements>
<requirement>
The first argument, `za`, must be a valid pointer to a `zip_t` struct. This is because the function dereferences this pointer to access archive information.
</requirement>
<requirement>
The `zip_t` archive must be opened in write mode. This is because `_zip_file_replace` checks if the archive is read-only and fails if it is.
</requirement>
<requirement>
The second argument, `idx`, must be a valid index of a file in the archive, i.e., `idx < za->nentry`. This is because the function checks this condition and returns an error if it's not met.
</requirement>
<requirement>
The third argument, `source`, must be a non-NULL pointer to a `zip_source_t` struct. The function checks for `NULL` and returns an error. The `zip_source_t` can be created using functions like `zip_source_buffer` or `zip_source_file`.
</requirement>
<requirement>
The `flags` argument can be 0 or a bitmask of `ZIP_FL_*` constants. Callers in the codebase often use 0.
</requirement>
</requirements>

#include <stdint.h>
#include <stdlib.h>
#include <string.h>
#include <zip.h>

int
LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    const char *path = "fuzz_replace.zip";
    zip_t *archive;
    int error;

    /* We need at least enough data for the flags */
    if (size < sizeof(zip_flags_t)) {
        return 0;
    }

    (void)remove(path);
    if ((archive = zip_open(path, ZIP_CREATE, &error)) == NULL) {
        return 0;
    }

    /* Add an initial file that we can then replace. */
    zip_source_t *initial_source = zip_source_buffer(archive, "initial", 7, 0);
    if (initial_source == NULL) {
        zip_discard(archive);
        (void)remove(path);
        return 0;
    }

    zip_int64_t idx = zip_file_add(archive, "file_to_replace", initial_source, ZIP_FL_OVERWRITE);
    if (idx < 0) {
        /* zip_file_add frees the source on both success and error */
        zip_discard(archive);
        (void)remove(path);
        return 0;
    }

    /* Prepare arguments for zip_file_replace from fuzzer data */
    zip_flags_t flags;
    memcpy(&flags, data, sizeof(zip_flags_t));
    const uint8_t *replace_content = data + sizeof(zip_flags_t);
    size_t replace_size = size - sizeof(zip_flags_t);

    zip_source_t *replace_source = zip_source_buffer(archive, replace_content, replace_size, 0);
    if (replace_source == NULL) {
        zip_discard(archive);
        (void)remove(path);
        return 0;
    }

    /* Call the function under test */
    zip_file_replace(archive, (zip_uint64_t)idx, replace_source, flags);
    /* On success, zip_file_replace takes ownership of and frees the source.
       On most errors, it also frees the source. We don't need to free it. */

    if (zip_close(archive) < 0) {
        /* zip_close frees the archive regardless of return value. */
    }

    (void)remove(path);

    return 0;
}
