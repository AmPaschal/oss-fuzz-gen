
	
<description>
The `mbedtls_ssl_close_notify` function is responsible for sending a close_notify alert to the peer, signaling the intent to close the TLS connection. This is the proper way to gracefully shut down an SSL/TLS connection. The function first checks if the SSL context and its configuration are valid. It then checks if the TLS handshake has been completed. If the handshake is over, it proceeds to call `mbedtls_ssl_send_alert_message` to transmit the warning-level alert `MBEDTLS_SSL_ALERT_MSG_CLOSE_NOTIFY`. If the handshake is not yet complete, the function does nothing and returns successfully.
</description>
<requirements>
<requirement>
The `mbedtls_ssl_context` pointer `ssl` must not be NULL. This is a basic sanity check to prevent null pointer dereferences within the function.
</requirement>
<requirement>
The `mbedtls_ssl_context` must be initialized by calling `mbedtls_ssl_init()`. This ensures the context's internal fields are set to a consistent initial state.
</requirement>
<requirement>
An `mbedtls_ssl_config` must be initialized and configured, at a minimum with `mbedtls_ssl_config_defaults()`. This configuration is necessary for the subsequent setup of the SSL context.
</requirement>
<requirement>
`mbedtls_ssl_setup()` must be called to associate the configured `mbedtls_ssl_config` with the `mbedtls_ssl_context`. This step is crucial as it populates the `ssl->conf` field, which is checked for non-NULL by `mbedtls_ssl_close_notify`.
</requirement>
<requirement>
A successful TLS handshake must be performed using `mbedtls_ssl_handshake()` before calling this function. `mbedtls_ssl_close_notify` will only send an alert if the handshake state is `MBEDTLS_SSL_HANDSHAKE_OVER`.
</requirement>
</requirements>

static int dummy_send(void *ctx, const unsigned char *buf, size_t len)
{
    (void) ctx;
    (void) buf;
    if (len > INT_MAX) {
        return -1;
    }
    return (int) len;

static int fuzz_recv(void *ctx, unsigned char *buf, size_t len)
{
    fuzzBufferOffset_t *biomemfuzz = (fuzzBufferOffset_t *) ctx;

    if (biomemfuzz->Offset == biomemfuzz->Size) {
        return 0;
    }
    if (len > INT_MAX) {
        return -1;
    }
    if (len + biomemfuzz->Offset > biomemfuzz->Size) {
        len = biomemfuzz->Size - biomemfuzz->Offset;
    }
    memcpy(buf, biomemfuzz->Data + biomemfuzz->Offset, len);
    biomemfuzz->Offset += len;
    return (int) len;
}

static int dummy_random(void *p_rng, unsigned char *output, size_t output_len)
{
    int ret;
    size_t i;

#if defined(MBEDTLS_CTR_DRBG_C)
    if (p_rng != NULL) {
        ret = mbedtls_ctr_drbg_random(p_rng, output, output_len);
    } else {
        ret = 0;
    }
#else
    (void) p_rng;
    ret = 0;
#endif
    for (i = 0; i < output_len; i++) {
        output[i] = (unsigned char) rand();
    }
    return ret;
}

#if defined(MBEDTLS_SSL_SRV_C) && defined(MBEDTLS_ENTROPY_C) && defined(MBEDTLS_CTR_DRBG_C)
  int ret;
  size_t len;
  mbedtls_ssl_context ssl;
  mbedtls_ssl_config conf;
  mbedtls_ctr_drbg_context ctr_drbg;
  mbedtls_entropy_context entropy;
#if defined(MBEDTLS_SSL_SESSION_TICKETS) && defined(MBEDTLS_SSL_TICKET_C)
  mbedtls_ssl_ticket_context ticket_ctx;
#endif
  unsigned char buf[4096];
  fuzzBufferOffset_t biomemfuzz;
  uint8_t options;

  if (Size < 1) {
    return 0;
  }
  options = Data[Size - 1];

  mbedtls_ctr_drbg_init(&ctr_drbg);
  mbedtls_entropy_init(&entropy);
#if defined(MBEDTLS_X509_CRT_PARSE_C) && defined(MBEDTLS_PEM_PARSE_C)
  mbedtls_x509_crt_init(&srvcert);
  mbedtls_pk_init(&pkey);
#endif
  mbedtls_ssl_init(&ssl);
  mbedtls_ssl_config_init(&conf);
#if defined(MBEDTLS_SSL_SESSION_TICKETS) && defined(MBEDTLS_SSL_TICKET_C)
  mbedtls_ssl_ticket_init(&ticket_ctx);
#endif
#if defined(MBEDTLS_USE_PSA_CRYPTO)
  psa_status_t status = psa_crypto_init();
  if (status != PSA_SUCCESS) {
    goto exit;
  }
#endif /* MBEDTLS_USE_PSA_CRYPTO */

  if (mbedtls_ctr_drbg_seed(&ctr_drbg, dummy_entropy, &entropy, (const unsigned char *)pers, strlen(pers)) != 0) {
    goto exit;
  }

  if (initialized == 0) {

#if defined(MBEDTLS_X509_CRT_PARSE_C) && defined(MBEDTLS_PEM_PARSE_C)
    if (mbedtls_x509_crt_parse(&srvcert, (const unsigned char *)mbedtls_test_srv_crt, mbedtls_test_srv_crt_len) != 0) {
      goto exit;
    }
    if (mbedtls_x509_crt_parse(&srvcert, (const unsigned char *)mbedtls_test_cas_pem, mbedtls_test_cas_pem_len) != 0) {
      goto exit;
    }
    if (mbedtls_pk_parse_key(&pkey, (const unsigned char *)mbedtls_test_srv_key, mbedtls_test_srv_key_len, NULL, 0, dummy_random, &ctr_drbg) != 0) {
      goto exit;
    }
#endif

    alpn_list[0] = "HTTP";
    alpn_list[1] = "fuzzalpn";
    alpn_list[2] = NULL;

    dummy_init();

    initialized = 1;
  }

  if (mbedtls_ssl_config_defaults(&conf, MBEDTLS_SSL_IS_SERVER, MBEDTLS_SSL_TRANSPORT_STREAM, MBEDTLS_SSL_PRESET_DEFAULT) != 0) {
    goto exit;
  }

  srand(1);
  mbedtls_ssl_conf_rng(&conf, dummy_random, &ctr_drbg);

#if defined(MBEDTLS_X509_CRT_PARSE_C) && defined(MBEDTLS_PEM_PARSE_C)
  mbedtls_ssl_conf_ca_chain(&conf, srvcert.next, NULL);
  if (mbedtls_ssl_conf_own_cert(&conf, &srvcert, &pkey) != 0) {
    goto exit;
  }
#endif

  mbedtls_ssl_conf_cert_req_ca_list(&conf, (options & 0x1) ? MBEDTLS_SSL_CERT_REQ_CA_LIST_ENABLED : MBEDTLS_SSL_CERT_REQ_CA_LIST_DISABLED);
#if defined(MBEDTLS_SSL_ALPN)
  if (options & 0x2) {
    mbedtls_ssl_conf_alpn_protocols(&conf, alpn_list);
  }
#endif
#if defined(MBEDTLS_SSL_SESSION_TICKETS) && defined(MBEDTLS_SSL_TICKET_C)
  if (options & 0x4) {
    if (mbedtls_ssl_ticket_setup(&ticket_ctx, dummy_random, &ctr_drbg, MBEDTLS_CIPHER_AES_256_GCM, 86400) != 0) {
      goto exit;
    }

    mbedtls_ssl_conf_session_tickets_cb(&conf, mbedtls_ssl_ticket_write, mbedtls_ssl_ticket_parse, &ticket_ctx);
  }
#endif
#if defined(MBEDTLS_SSL_EXTENDED_MASTER_SECRET)
  mbedtls_ssl_conf_extended_master_secret(&conf, (options & 0x10) ? MBEDTLS_SSL_EXTENDED_MS_DISABLED : MBEDTLS_SSL_EXTENDED_MS_ENABLED);
#endif
#if defined(MBEDTLS_SSL_ENCRYPT_THEN_MAC)
  mbedtls_ssl_conf_encrypt_then_mac(&conf, (options & 0x20) ? MBEDTLS_SSL_ETM_ENABLED : MBEDTLS_SSL_ETM_DISABLED);
#endif
#if defined(MBEDTLS_KEY_EXCHANGE_SOME_PSK_ENABLED)
  if (options & 0x40) {
    mbedtls_ssl_conf_psk(&conf, psk, sizeof(psk), (const unsigned char *)psk_id, sizeof(psk_id) - 1);
  }
#endif
#if defined(MBEDTLS_SSL_RENEGOTIATION)
  mbedtls_ssl_conf_renegotiation(&conf, (options & 0x80) ? MBEDTLS_SSL_RENEGOTIATION_ENABLED : MBEDTLS_SSL_RENEGOTIATION_DISABLED);

  if (ret == 0) {
    do {
      len = sizeof(buf) - 1;
      ret = mbedtls_ssl_read(&ssl, buf, len);

        continue;
      } else if (ret <= 0) {
        break;
      }
    } while (1);
  }

  mbedtls_ssl_close_notify(&ssl);

exit:
#if defined(MBEDTLS_SSL_SESSION_TICKETS) && defined(MBEDTLS_SSL_TICKET_C)

#endif

  mbedtls_ctr_drbg_free(&ctr_drbg);
  mbedtls_ssl_config_free(&conf);
#if defined(MBEDTLS_X509_CRT_PARSE_C) && defined(MBEDTLS_PEM_PARSE_C)
  mbedtls_x509_crt_free(&srvcert);
  mbedtls_pk_free(&pkey);
#endif
  mbedtls_ssl_free(&ssl);
#if defined(MBEDTLS_USE_PSA_CRYPTO)
  mbedtls_psa_crypto_free();
#endif
#else
  (void)Data;
  (void)Size;
#endif /* MBEDTLS_SSL_SRV_C && MBEDTLS_ENTROPY_C && MBEDTLS_CTR_DRBG_C */

  return 0;
}
