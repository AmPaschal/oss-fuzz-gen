extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size == 0) {
    return 0;
  }

  // Create a connected pair of sockets.
  int sv[2];
  if (socketpair(AF_UNIX, SOCK_STREAM, 0, sv) == -1) {
    return 0;
  }

  socket_t server_sock = sv[0];
  int client_sock = sv[1];

  // Write the fuzzer data to the client end of the socket.
  ssize_t written_bytes = write(client_sock, data, size);
  (void)written_bytes; // Prevent unused variable warning

  // Close the client end. This signals EOF to the server's read.
  close(client_sock);

  // Call the function under test via the public wrapper.
  g_server.FuzzProcessAndCloseSocket(server_sock);

  return 0;
}
