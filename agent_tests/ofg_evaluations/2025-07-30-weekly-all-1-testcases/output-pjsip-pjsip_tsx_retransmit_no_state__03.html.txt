
	
<description>
The function `pjsip_tsx_retransmit_no_state` retransmits a SIP message. It acquires a group lock from the transaction, sends the message, and then releases the lock. If the transmission is successful, it decrements the reference counter of the transmitted data. The function can either be provided with the transmission data directly, or it can use the last transmitted data from the transaction object.
</description>
<requirements>
<requirement>
The `tsx` argument must be a valid pointer to a `pjsip_transaction` object. The function dereferences this pointer to access the group lock and, if necessary, the last transmitted data, so it cannot be NULL.
</requirement>
<requirement>
The `grp_lock` member of the `tsx` object must be a valid and initialized group lock. This is because the function calls `pj_grp_lock_acquire` and `pj_grp_lock_release` on this lock.
</requirement>
<requirement>
If the `tdata` argument is NULL, the `last_tx` member of the `tsx` object must be a valid `pjsip_tx_data` object. In this case, the function will use `tsx->last_tx` as the data to be transmitted.
</requirement>
<requirement>
If the `tdata` argument is not NULL, it must be a valid pointer to a `pjsip_tx_data` object. The function will attempt to send the data pointed to by `tdata`.
</requirement>
</requirements>

#define kMinInputLength 1

// Dummy callback for the transaction state.
static void on_tsx_state_dummy(pjsip_transaction *tsx, pjsip_event *event) {
    (void)tsx;
    (void)event;
}

// Dummy module to act as transaction user.
static pjsip_module mod_fuzz = {
    NULL, NULL,
    {"mod-fuzz", 8},
    -1, 0,
    NULL, NULL, NULL, NULL,
    NULL, NULL, NULL, NULL,
    &on_tsx_state_dummy,
};

extern int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) {
    if (Size < kMinInputLength) {
        return 0;
    }
    // Input data is not used to ensure complex objects can be created
    // to reach the target function.
    (void)Data;

    pj_caching_pool cp;
    pj_pool_t *pool = NULL;
    pjsip_endpoint *endpt = NULL;
    pjsip_tx_data *tdata = NULL;
    pjsip_transaction *tsx = NULL;
    pj_status_t status;

    pj_init();
    pj_caching_pool_init(&cp, &pj_pool_factory_default_policy, 0);
    pj_log_set_level(0);
    
    pool = pj_pool_create(&cp.factory, "fuzz-pool", 4000, 4000, NULL);
    if (!pool) {
        pj_caching_pool_destroy(&cp);
        return 0;
    }

    if (pjsip_endpt_create(&cp.factory, "fuzzer", &endpt) != PJ_SUCCESS) {
        pj_pool_release(pool);
        pj_caching_pool_destroy(&cp);
        return 0;
    }

    if (pjsip_tsx_layer_init_module(endpt) != PJ_SUCCESS) {
        pjsip_endpt_destroy(endpt);
        pj_pool_release(pool);
        pj_caching_pool_destroy(&cp);
        return 0;
    }

    if (pjsip_endpt_register_module(endpt, &mod_fuzz) != PJ_SUCCESS) {
        pjsip_endpt_destroy(endpt);
        pj_pool_release(pool);
        pj_caching_pool_destroy(&cp);
        return 0;
    }

    if (pjsip_endpt_create_tdata(endpt, &tdata) != PJ_SUCCESS) {
        pjsip_endpt_destroy(endpt);
        pj_pool_release(pool);
        pj_caching_pool_destroy(&cp);
        return 0;
    }
    
    pjsip_msg *msg = pjsip_msg_create(pool, PJSIP_REQUEST_MSG);
    pj_str_t method = pj_str("INVITE");
    pjsip_method_init(&msg->line.req.method, pool, &method);
    char uri_str[] = "sip:fuzz@fuzz.com";
    msg->line.req.uri = pjsip_parse_uri(pool, uri_str, strlen(uri_str), 0);

    pjsip_cseq_hdr *cseq = pjsip_cseq_hdr_create(pool);
    cseq->cseq = 1;
    cseq->method = msg->line.req.method;
    pjsip_msg_add_hdr(msg, (pjsip_hdr*)cseq);

    pjsip_cid_hdr *cid = pjsip_cid_hdr_create(pool);
    cid->id = pj_str("fuzz-call-id");
    pjsip_msg_add_hdr(msg, (pjsip_hdr*)cid);

    pjsip_from_hdr *from = pjsip_from_hdr_create(pool);
    from->uri = pjsip_parse_uri(pool, "sip:from@fuzz", 13, 0);
    from->tag = pj_str("from-tag");
    pjsip_msg_add_hdr(msg, (pjsip_hdr*)from);

    pjsip_to_hdr *to = pjsip_to_hdr_create(pool);
    to->uri = pjsip_parse_uri(pool, "sip:to@fuzz", 11, 0);
    pjsip_msg_add_hdr(msg, (pjsip_hdr*)to);
    
    pjsip_via_hdr *via = pjsip_via_hdr_create(pool);
    pjsip_msg_add_hdr(msg, (pjsip_hdr*)via);

    tdata->msg = msg;

    // pjsip_tsx_create_uac will add a reference to tdata on success.
    status = pjsip_tsx_create_uac(&mod_fuzz, tdata, &tsx);
    if (status == PJ_SUCCESS) {
        // Path 1: tdata is not NULL.
        pjsip_tsx_retransmit_no_state(tsx, tdata);
        // Path 2: tdata is NULL. This will use tsx->last_tx.
        pjsip_tsx_retransmit_no_state(tsx, NULL);
        
        // Terminate the transaction to release its reference to tdata.
        pjsip_tsx_terminate(tsx, PJSIP_SC_REQUEST_TERMINATED);
    }
    
    // Release the initial reference from pjsip_endpt_create_tdata.
    pjsip_tx_data_dec_ref(tdata);

    pjsip_endpt_destroy(endpt);
    pj_pool_release(pool);
    pj_caching_pool_destroy(&cp);
    
    return 0;
}
