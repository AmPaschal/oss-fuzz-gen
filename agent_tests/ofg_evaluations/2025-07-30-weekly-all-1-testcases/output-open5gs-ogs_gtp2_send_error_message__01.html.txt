
	
<description>
The function `ogs_gtp2_send_error_message` sends a GTPv2 error message. It takes a transaction, TEID, message type, and cause value as input. It constructs an error message based on the message type and sends it. The function will cause a fatal error if an invalid message type is provided.
</description>
<requirements>
<requirement>
The `xact` parameter must be a valid pointer to an `ogs_gtp_xact_t` struct. This is because the function dereferences this pointer to update and commit the transaction.
</requirement>
<requirement>
The `type` parameter must be one of the OGS_GTP2_*_RESPONSE_TYPE or OGS_GTP2_*_INDICATION_TYPE values defined in `lib/gtp/v2/message.h`. Supplying a value for `type` not handled by the switch statement will result in a call to `ogs_fatal`.
</requirement>
<requirement>
The `cause_value` can be any `uint8_t` value. This is because it is directly assigned to the `cause.value` field.
</requirement>
<requirement>
The `teid` parameter can be any `uint32_t` value. This is because it is directly assigned to the `errmsg.h.teid` field.
</requirement>
</requirements>

#include <stdio.h>
#include <stdint.h>
#include <string.h>

#include "fuzzing.h"
#include "ogs-gtp.h"

#define kMinInputLength 7
#define kMaxInputLength 1024

static const uint8_t valid_types[] = {
    OGS_GTP2_CREATE_SESSION_RESPONSE_TYPE,
    OGS_GTP2_MODIFY_BEARER_RESPONSE_TYPE,
    OGS_GTP2_DELETE_SESSION_RESPONSE_TYPE,
    OGS_GTP2_RELEASE_ACCESS_BEARERS_RESPONSE_TYPE,
    OGS_GTP2_DOWNLINK_DATA_NOTIFICATION_ACKNOWLEDGE_TYPE,
    OGS_GTP2_CREATE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_UPDATE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_DELETE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_CREATE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE,
    OGS_GTP2_DELETE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE,
    OGS_GTP2_MODIFY_BEARER_FAILURE_INDICATION_TYPE,
    OGS_GTP2_DELETE_BEARER_FAILURE_INDICATION_TYPE,
    OGS_GTP2_BEARER_RESOURCE_FAILURE_INDICATION_TYPE,
};

extern int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) 
{
    if (Size < kMinInputLength || Size > kMaxInputLength) {
        return 0;
    }

    if (!initialized) {
        initialize();
        ogs_log_install_domain(&__ogs_gtp_domain, "gtp", OGS_LOG_NONE);
        ogs_log_install_domain(&__ogs_tlv_domain, "tlv", OGS_LOG_NONE);
    }

    uint32_t teid = 0;
    uint8_t type_index = 0;
    uint8_t cause_value = 0;

    memcpy(&teid, Data, sizeof(teid));
    type_index = Data[sizeof(uint32_t)];
    cause_value = Data[sizeof(uint32_t) + sizeof(uint8_t)];
    const uint8_t *pkbuf_data = Data + sizeof(uint32_t) + sizeof(uint8_t) + sizeof(uint8_t);
    size_t pkbuf_size = Size - (sizeof(uint32_t) + sizeof(uint8_t) + sizeof(uint8_t));

    uint8_t type = valid_types[type_index % (sizeof(valid_types)/sizeof(valid_types[0]))];

    ogs_sockaddr_t *sa = NULL;
    if (ogs_addaddrinfo(&sa, AF_INET, "127.0.0.1", 2123, 0) != OGS_OK) {
        return 0;
    }

    ogs_gtp_node_t *gnode = ogs_gtp_node_new(sa);
    if (gnode == NULL) {
        ogs_freeaddrinfo(sa);
        return 0;
    }

    ogs_pkbuf_t *pkbuf = ogs_pkbuf_alloc(NULL, OGS_MAX_SDU_LEN);
    if (pkbuf == NULL) {
        ogs_gtp_node_free(gnode);
        return 0;
    }
    ogs_pkbuf_put_data(pkbuf, pkbuf_data, pkbuf_size);

    ogs_gtp2_header_t h;
    memset(&h, 0, sizeof(h));
    h.type = OGS_GTP2_ECHO_REQUEST_TYPE;

    ogs_gtp_xact_t *xact = ogs_gtp_xact_local_create(gnode, &h, pkbuf, NULL, NULL);
    if (xact == NULL) {
        ogs_pkbuf_free(pkbuf);
        ogs_gtp_node_free(gnode);
        return 0;
    }
    
    ogs_gtp2_send_error_message(xact, teid, type, cause_value);

    ogs_gtp_node_free(gnode);

    return 0;
}
