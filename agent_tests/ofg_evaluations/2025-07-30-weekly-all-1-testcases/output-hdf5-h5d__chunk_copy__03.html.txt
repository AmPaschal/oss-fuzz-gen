
	
<description>
The function H5D__chunk_copy is responsible for copying chunked dataset data from a source to a destination file. It handles different chunk indexing schemes and performs data type conversions if necessary. It is called by H5O__layout_copy_file when the layout type is chunked.
</description>
<requirements>
<requirement>
The f_src, storage_src, layout_src, f_dst, storage_dst, ds_extent_src, and dt_src pointers must not be NULL. This is to prevent assertion failures at the beginning of the function.
</requirement>
<requirement>
The storage_src->ops->copy_setup and storage_src->ops->iterate function pointers must be valid. The function calls these pointers and a null pointer would lead to a crash.
</requirement>
<requirement>
The idx_type and ops fields of storage_src and storage_dst must be a valid pair. This is enforced by the H5D_CHUNK_STORAGE_INDEX_CHK macro, and valid pairs are (H5D_CHUNK_IDX_EARRAY, H5D_COPS_EARRAY), (H5D_CHUNK_IDX_FARRAY, H5D_COPS_FARRAY), (H5D_CHUNK_IDX_BT2, H5D_COPS_BT2), or (H5D_CHUNK_IDX_BTREE, H5D_COPS_BTREE).
</requirement>
<requirement>
The cpy_info pointer must not be NULL. This is to prevent a null pointer dereference when accessing its members.
</requirement>
</requirements>

extern int
LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    char filename_src[256];
    sprintf(filename_src, "/tmp/libfuzzer_src.%d", getpid());
    char filename_dst[256];
    sprintf(filename_dst, "/tmp/libfuzzer_dst.%d", getpid());

    if (size < 1)
        return 0;

    /* Create source file */
    hid_t fapl_src = H5Pcreate(H5P_FILE_ACCESS);
    if (fapl_src < 0)
        return 0;
    H5Pset_fapl_core(fapl_src, 4096, 0);
    hid_t fcpl_src = H5Pcreate(H5P_FILE_CREATE);
    if (fcpl_src < 0) {
        H5Pclose(fapl_src);
        return 0;
    }
    H5Pset_file_space_strategy(fcpl_src, H5F_FSPACE_STRATEGY_FSM_AGGR, 0, (hsize_t)1);
    hid_t file_src_id = H5Fcreate(filename_src, H5F_ACC_TRUNC, fcpl_src, fapl_src);
    if (file_src_id < 0) {
        H5Pclose(fapl_src);
        H5Pclose(fcpl_src);
        return 0;
    }

    /* Create destination file */
    hid_t fapl_dst = H5Pcreate(H5P_FILE_ACCESS);
    if (fapl_dst < 0) {
        H5Fclose(file_src_id);
        H5Pclose(fapl_src);
        H5Pclose(fcpl_src);
        return 0;
    }
    H5Pset_fapl_core(fapl_dst, 4096, 0);
    hid_t fcpl_dst = H5Pcreate(H5P_FILE_CREATE);
    if (fcpl_dst < 0) {
        H5Fclose(file_src_id);
        H5Pclose(fapl_src);
        H5Pclose(fcpl_src);
        H5Pclose(fapl_dst);
        return 0;
    }
    H5Pset_file_space_strategy(fcpl_dst, H5F_FSPACE_STRATEGY_FSM_AGGR, 0, (hsize_t)1);
    hid_t file_dst_id = H5Fcreate(filename_dst, H5F_ACC_TRUNC, fcpl_dst, fapl_dst);
    if (file_dst_id < 0) {
        H5Fclose(file_src_id);
        H5Pclose(fapl_src);
        H5Pclose(fcpl_src);
        H5Pclose(fapl_dst);
        H5Pclose(fcpl_dst);
        return 0;
    }

    H5F_t *f_src = (H5F_t *)H5I_object(file_src_id);
    H5F_t *f_dst = (H5F_t *)H5I_object(file_dst_id);

    /* Create a dataset in the source file */
    hsize_t dims[1] = {size};
    hid_t   space_id = H5Screate_simple(1, dims, NULL);
    if (space_id < 0)
        goto cleanup;
    hid_t dcpl_id = H5Pcreate(H5P_DATASET_CREATE);
    if (dcpl_id < 0) {
        H5Sclose(space_id);
        goto cleanup;
    }
    hsize_t chunk_dims[1] = {1};
    H5Pset_chunk(dcpl_id, 1, chunk_dims);
    hid_t dset_id = H5Dcreate2(file_src_id, "dset", H5T_NATIVE_UINT8, space_id, H5P_DEFAULT, dcpl_id, H5P_DEFAULT);
    if (dset_id < 0) {
        H5Sclose(space_id);
        H5Pclose(dcpl_id);
        goto cleanup;
    }

    H5Dwrite(dset_id, H5T_NATIVE_UINT8, H5S_ALL, H5S_ALL, H5P_DEFAULT, data);

    H5D_t *dset = (H5D_t *)H5I_object(dset_id);
    if (dset) {
        H5O_layout_chunk_t layout_dst;
        memcpy(&layout_dst, &dset->shared->layout.u.chunk, sizeof(H5O_layout_chunk_t));

        H5O_storage_chunk_t storage_dst;
        memcpy(&storage_dst, &dset->shared->layout.storage.u.chunk, sizeof(H5O_storage_chunk_t));

        H5O_copy_t cpy_info;
        memset(&cpy_info, 0, sizeof(H5O_copy_t));

        H5D__chunk_copy(f_src, &dset->shared->layout.storage.u.chunk, &dset->shared->layout.u.chunk, f_dst,
                        &storage_dst, &dset->shared->space->extent, dset->shared->type,
                        &dset->shared->dcpl_cache.pline, &cpy_info);
    }

    H5Dclose(dset_id);
    H5Sclose(space_id);
    H5Pclose(dcpl_id);

cleanup:
    H5Fclose(file_src_id);
    H5Fclose(file_dst_id);
    H5Pclose(fapl_src);
    H5Pclose(fcpl_src);
    H5Pclose(fapl_dst);
    H5Pclose(fcpl_dst);

    unlink(filename_src);
    unlink(filename_dst);

    return 0;
}
