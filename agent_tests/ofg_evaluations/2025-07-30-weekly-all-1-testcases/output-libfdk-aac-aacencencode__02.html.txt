
	
<description>
The aacEncEncode function is the core encoding function of the libfdk-aac library. It takes an encoder handle, input and output buffer descriptors, and input/output argument structures to perform the encoding of a single audio frame. Before calling this function, the encoder must be properly initialized by calling aacEncOpen, which sets up the necessary memory and internal states. The input and output buffers must be correctly described by the AACENC_BufDesc structures, specifying buffer pointers, sizes, and identifiers. The AACENC_InArgs structure provides essential information like the number of input samples. The function then processes the input audio, encodes it into the AAC format (potentially with SBR/PS extensions), and writes the resulting bitstream to the provided output buffer. It updates the AACENC_OutArgs structure with the results of the encoding operation, such as the number of bytes written.
</description>
<requirements>

<requirement>
The `aacEncOpen` function must be called to initialize the encoder and obtain a valid `HANDLE_AACENCODER` before calling `aacEncEncode`. This is because `aacEncEncode` operates on an initialized encoder state and will fail if the handle is NULL or points to uninitialized memory.
</requirement>
<requirement>
The `hAacEncoder` argument must be a valid handle returned by `aacEncOpen`. The function performs a NULL check on this handle and will return `AACENC_INVALID_HANDLE` if it is invalid.
</requirement>
<requirement>
The `inargs` and `outargs` arguments must not be NULL. The function checks for NULL and returns `AACENC_UNSUPPORTED_PARAMETER` if they are, preventing null pointer dereferences later.
</requirement>
<requirement>
To encode audio, the `inBufDesc` argument must point to a valid `AACENC_BufDesc` structure with a buffer identifier of `IN_AUDIO_DATA`. This is required for the function to locate and process the input audio samples.
</requirement>
<requirement>
The `outBufDesc` argument must point to a valid `AACENC_BufDesc` structure with a buffer identifier of `OUT_BITSTREAM_DATA` and a sufficiently large buffer. This is necessary for the function to write the encoded bitstream; an undersized buffer will cause an `AACENC_ENCODE_ERROR`.
</requirement>
<requirement>
The `inargs->numInSamples` field must be set correctly to the number of samples in the input buffer, or -1 to signal the end of the stream. The function relies on this value to determine how much data to process and to manage flushing operations.
</requirement>

</requirements>

#include "/src/aac/libAACenc/include/aacenc_lib.h"
#include "FDK_audio.h"
#include <fuzzer/FuzzedDataProvider.h>
#include <vector>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  HANDLE_AACENCODER hAacEncoder;
  if (aacEncOpen(&hAacEncoder, 0, 0) != AACENC_OK) {
    return 0;
  }

  // Consume parameters for configuration
  const AUDIO_OBJECT_TYPE aot = provider.PickValueInArray(
      {AOT_AAC_LC, AOT_SBR, AOT_PS, AOT_ER_AAC_LD, AOT_ER_AAC_ELD});
  const int sample_rate = provider.PickValueInArray(
      {8000, 11025, 12000, 16000, 22050, 24000, 32000, 44100, 48000});
  const CHANNEL_MODE channel_mode = provider.PickValueInArray({MODE_1, MODE_2});
  const int bitrate = provider.ConsumeIntegralInRange<int>(8000, 256000);
  const TRANSPORT_TYPE transmux =
      provider.PickValueInArray({TT_MP4_RAW, TT_MP4_ADTS, TT_MP4_LOAS});
  const int afterburner = provider.ConsumeBool();

  // Set parameters
  if (aacEncoder_SetParam(hAacEncoder, AACENC_AOT, aot) != AACENC_OK) {
    aacEncClose(&hAacEncoder);
    return 0;
  }
  if (aacEncoder_SetParam(hAacEncoder, AACENC_SAMPLERATE, sample_rate) !=
      AACENC_OK) {
    aacEncClose(&hAacEncoder);
    return 0;
  }
  if (aacEncoder_SetParam(hAacEncoder, AACENC_CHANNELMODE, channel_mode) !=
      AACENC_OK) {
    aacEncClose(&hAacEncoder);
    return 0;
  }
  if (aacEncoder_SetParam(hAacEncoder, AACENC_BITRATE, bitrate) != AACENC_OK) {
    aacEncClose(&hAacEncoder);
    return 0;
  }
  if (aacEncoder_SetParam(hAacEncoder, AACENC_TRANSMUX, transmux) !=
      AACENC_OK) {
    aacEncClose(&hAacEncoder);
    return 0;
  }
  if (aacEncoder_SetParam(hAacEncoder, AACENC_AFTERBURNER, afterburner) !=
      AACENC_OK) {
    aacEncClose(&hAacEncoder);
    return 0;
  }

  // Finalize initialization by calling aacEncEncode with NULL parameters
  if (aacEncEncode(hAacEncoder, nullptr, nullptr, nullptr, nullptr) !=
      AACENC_OK) {
    aacEncClose(&hAacEncoder);
    return 0;
  }

  AACENC_InfoStruct info = {0};
  if (aacEncInfo(hAacEncoder, &info) != AACENC_OK) {
    aacEncClose(&hAacEncoder);
    return 0;
  }

  // Prepare input and output buffers
  std::vector<uint8_t> input_bytes =
      provider.ConsumeRemainingBytes<uint8_t>();
  std::vector<uint8_t> output_buf(info.maxOutBufBytes);

  AACENC_BufDesc in_buf_desc, out_buf_desc;
  AACENC_InArgs in_args;
  AACENC_OutArgs out_args;

  // Input buffer
  void *in_ptr = input_bytes.data();
  int in_size = input_bytes.size();
  int in_element_size = sizeof(int16_t);
  int in_identifier = IN_AUDIO_DATA;
  in_buf_desc.numBufs = 1;
  in_buf_desc.bufs = &in_ptr;
  in_buf_desc.bufferIdentifiers = &in_identifier;
  in_buf_desc.bufSizes = &in_size;
  in_buf_desc.bufElSizes = &in_element_size;

  // Output buffer
  void *out_ptr = output_buf.data();
  int out_size = output_buf.size();
  int out_element_size = sizeof(uint8_t);
  int out_identifier = OUT_BITSTREAM_DATA;
  out_buf_desc.numBufs = 1;
  out_buf_desc.bufs = &out_ptr;
  out_buf_desc.bufferIdentifiers = &out_identifier;
  out_buf_desc.bufSizes = &out_size;
  out_buf_desc.bufElSizes = &out_element_size;

  // Input arguments
  in_args.numInSamples = input_bytes.size() / sizeof(int16_t);
  in_args.numAncBytes = 0;

  // Encode the frame
  aacEncEncode(hAacEncoder, &in_buf_desc, &out_buf_desc, &in_args, &out_args);

  // Flush the encoder
  in_args.numInSamples = -1;
  aacEncEncode(hAacEncoder, nullptr, &out_buf_desc, &in_args, &out_args);

  aacEncClose(&hAacEncoder);

  return 0;
}
