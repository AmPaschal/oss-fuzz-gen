
	
<description>
The gnutls_session_ticket_send function is used by a TLS 1.3 server to send one or more session tickets to a client after the main handshake has been completed. This allows the client to use these tickets to perform an abbreviated \"resumption\" handshake in subsequent connections, improving performance. The function requires a fully established TLS 1.3 server session and will fail if called on a client session, on a session that is not TLS 1.3, before the handshake is complete, or if the number of requested tickets is zero.
</description>
<requirements>
<requirement>
The session must be initialized as a server session using `gnutls_init()` with the `GNUTLS_SERVER` flag. This is because the function checks if the session entity is a client and fails if it is.
</requirement>
<requirement>
The session must be configured to use the TLS 1.3 protocol by setting a priority string such as \"NORMAL:-VERS-ALL:+VERS-TLS1.3\" using `gnutls_priority_set_direct()`. This is because the function checks the `tls13_sem` flag in the session's version information.
</requirement>
<requirement>
A successful TLS handshake must be performed on the session before calling this function. This is because the function is designed to send session tickets after the initial handshake is complete.
</requirement>
<requirement>
The `nr` parameter, which specifies the number of tickets to send, must be greater than zero. The function will return an error for a value of zero.
</requirement>
<requirement>
The `flags` parameter should be set to 0. This parameter is currently unused in the function's implementation.
</requirement>
</requirements>

#include <assert.h>
#include <stdint.h>
#include <unistd.h>
#include <string.h>
#include <stdlib.h>

#include <gnutls/crypto.h>
#include <gnutls/gnutls.h>

#include "certs.h"
#include "handshake.h"
#include "fuzzer.h"

/* A helper struct to pass both the data and the read offset to the pull function. */
typedef struct {
	struct mem_st *mem;
	size_t read_offset;
} pull_data_st;

static ssize_t fuzz_pull(gnutls_transport_ptr_t ptr, void *buffer, size_t len)
{
	pull_data_st *pull_data = ptr;
	struct mem_st *mem = pull_data->mem;

	if (pull_data->read_offset >= mem->size) {
		return GNUTLS_E_AGAIN;
	}
	size_t to_read = len;
	if (to_read > mem->size - pull_data->read_offset) {
		to_read = mem->size - pull_data->read_offset;
	}
	memcpy(buffer, mem->data + pull_data->read_offset, to_read);
	pull_data->read_offset += to_read;
	return to_read;
}

static ssize_t fuzz_push(gnutls_transport_ptr_t ptr, const void *buffer, size_t len)
{
	/* Do nothing */
	(void)ptr;
	(void)buffer;
	return len;
}

int __attribute__((visibility("protected")))
gnutls_rnd(gnutls_rnd_level_t level, void *data, size_t len)
{
	memset(data, 0xff, len);

	/* Flip the first byte to avoid infinite loop in the RSA
	 * blinding code of Nettle */
	if (len > 0)
		memset(data, 0x0, 1);
	return 0;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
	if (size < 1) {
		return 0;
	}

	/* Use the first byte to determine the number of tickets to send */
	unsigned int num_tickets = (data[0] % 4) + 1;
	const uint8_t *handshake_data = data + 1;
	size_t handshake_size = size - 1;

	int res;
	gnutls_session_t session;
	gnutls_certificate_credentials_t xcred;
	struct mem_st memdata;
	pull_data_st pull_data;
	gnutls_datum_t rsa_cert, rsa_key;
	gnutls_datum_t ecdsa_cert, ecdsa_key;
	gnutls_datum_t ed25519_cert, ed25519_key;
	gnutls_datum_t ed448_cert, ed448_key;

	res = gnutls_init(&session, GNUTLS_SERVER);
	assert(res >= 0);

	res = gnutls_certificate_allocate_credentials(&xcred);
	assert(res >= 0);

	rsa_cert.data = (unsigned char *)kRSACertificateDER;
	rsa_cert.size = sizeof(kRSACertificateDER);
	rsa_key.data = (unsigned char *)kRSAPrivateKeyDER;
	rsa_key.size = sizeof(kRSAPrivateKeyDER);

	ecdsa_cert.data = (unsigned char *)kECDSACertificateDER;
	ecdsa_cert.size = sizeof(kECDSACertificateDER);
	ecdsa_key.data = (unsigned char *)kECDSAPrivateKeyDER;
	ecdsa_key.size = sizeof(kECDSAPrivateKeyDER);

	ed25519_cert.data = (unsigned char *)kEd25519CertificateDER;
	ed25519_cert.size = sizeof(kEd25519CertificateDER);
	ed25519_key.data = (unsigned char *)kEd25519PrivateKeyDER;
	ed25519_key.size = sizeof(kEd25519PrivateKeyDER);

	ed448_cert.data = (unsigned char *)kEd448CertificateDER;
	ed448_cert.size = sizeof(kEd448CertificateDER);
	ed448_key.data = (unsigned char *)kEd448PrivateKeyDER;
	ed448_key.size = sizeof(kEd448PrivateKeyDER);

	res = gnutls_certificate_set_x509_key_mem(xcred, &rsa_cert, &rsa_key,
						  GNUTLS_X509_FMT_DER);
	assert(res >= 0);

	res = gnutls_certificate_set_x509_key_mem(
		xcred, &ecdsa_cert, &ecdsa_key, GNUTLS_X509_FMT_DER);
	assert(res >= 0);

	res = gnutls_certificate_set_x509_key_mem(
		xcred, &ed25519_cert, &ed25519_key, GNUTLS_X509_FMT_DER);
	assert(res >= 0);

	res = gnutls_certificate_set_x509_key_mem(
		xcred, &ed448_cert, &ed448_key, GNUTLS_X509_FMT_DER);
	assert(res >= 0);

	gnutls_certificate_set_known_dh_params(xcred, GNUTLS_SEC_PARAM_MEDIUM);

	res = gnutls_credentials_set(session, GNUTLS_CRD_CERTIFICATE, xcred);
	assert(res >= 0);

	res = gnutls_priority_set_direct(
		session, "NORMAL:-VERS-TLS-ALL:+VERS-TLS1.3", NULL);
	assert(res >= 0);

	memdata.data = handshake_data;
	memdata.size = handshake_size;

	pull_data.mem = &memdata;
	pull_data.read_offset = 0;

	gnutls_transport_set_push_function(session, fuzz_push);
	gnutls_transport_set_pull_function(session, fuzz_pull);
	gnutls_transport_set_ptr(session, &pull_data);

	do {
		res = gnutls_handshake(session);
	} while (res == GNUTLS_E_AGAIN || res == GNUTLS_E_INTERRUPTED);

	if (res >= 0) {
		/* Handshake successful, now we can send a ticket.
		 * The function checks for TLS1.3 and server mode internally.
		 * The flags parameter must be 0.
		 */
		gnutls_session_ticket_send(session, num_tickets, 0);
	}

	gnutls_deinit(session);
	gnutls_certificate_free_credentials(xcred);
	return 0;
}
