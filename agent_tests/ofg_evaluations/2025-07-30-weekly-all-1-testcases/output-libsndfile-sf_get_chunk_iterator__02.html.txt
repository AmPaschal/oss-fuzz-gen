
	
<description>
The sf_get_chunk_iterator function is used to get an iterator for chunks in a sound file. It takes a SNDFILE pointer and a pointer to a SF_CHUNK_INFO struct. If the chunk_info pointer is NULL, it returns an iterator for all chunks. Otherwise, it returns an iterator for chunks matching the ID in the chunk_info struct.
</description>
<requirements>
<requirement>
The `sndfile` argument must be a valid pointer to an `SNDFILE` structure returned by `sf_open`. This is because the function validates the `SNDFILE` pointer and its internal file pointer, and a valid `SNDFILE` is required to avoid errors.
</requirement>
<requirement>
The `chunk_info` argument can be NULL. If it is not NULL, it must point to a valid `SF_CHUNK_INFO` structure, with the `id` member being a valid null-terminated string, as it is used to identify the chunk to iterate.
</requirement>
</requirements>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size < 1) {
    return 0;
  }

  VIO_DATA vio_data;
  SF_VIRTUAL_IO vio;
  SF_INFO sndfile_info;
  SNDFILE *sndfile = NULL;

  // Use a portion of the data for file initialization
  FuzzedDataProvider provider(data, size);
  size_t file_size = provider.ConsumeIntegralInRange<size_t>(0, provider.remaining_bytes());
  std::vector<uint8_t> file_data = provider.ConsumeBytes<uint8_t>(file_size);

  int err = sf_init_file(file_data.data(), file_data.size(), &sndfile, &vio_data, &vio, &sndfile_info);
  if (err) {
    if (sndfile)
      sf_close(sndfile);
    return 0;
  }

  SF_CHUNK_ITERATOR *iterator = NULL;
  if (provider.ConsumeBool()) {
    // Test with a specific chunk info
    SF_CHUNK_INFO chunk_info;
    memset(&chunk_info, 0, sizeof(chunk_info));
    std::string id = provider.ConsumeRandomLengthString(sizeof(chunk_info.id) - 1);
    strncpy(chunk_info.id, id.c_str(), sizeof(chunk_info.id) - 1);
    chunk_info.id[sizeof(chunk_info.id) - 1] = '\0'; // Ensure null termination
    chunk_info.id_size = strlen(chunk_info.id);

    iterator = sf_get_chunk_iterator(sndfile, &chunk_info);
  } else {
    // Test with a NULL chunk info to iterate all chunks
    iterator = sf_get_chunk_iterator(sndfile, NULL);
  }

  // The iterator itself doesn't need to be freed, sf_close handles it.
  // We can optionally use the iterator to exercise more code.
  if (iterator) {
    SF_CHUNK_INFO chunk_info_out;
    memset(&chunk_info_out, 0, sizeof(chunk_info_out));
    // Loop through chunks using sf_get_chunk_iterator and sf_next_chunk_iterator
    while (sf_get_chunk_size(iterator, &chunk_info_out) == SF_ERR_NO_ERROR) {
      // We don't need to do anything with the chunk data here.
      iterator = sf_next_chunk_iterator(iterator);
      if (!iterator)
        break;
    }
  }

  if (sndfile)
    sf_close(sndfile);

  return 0;
}
