  uint8_t *data;
  size_t size;

static ssize_t push_func(gnutls_transport_ptr_t ptr, const void *data,
                         size_t len) {
  struct mem_buffer *buf = (struct mem_buffer *)ptr;
  buf->data = (uint8_t *)gnutls_realloc(buf->data, buf->size + len);
  assert(buf->data != NULL);
  memcpy(buf->data + buf->size, data, len);
  buf->size += len;
  return len;
}

static ssize_t pull_func(gnutls_transport_ptr_t ptr, void *data, size_t len) {
  struct mem_buffer *buf = (struct mem_buffer *)ptr;
  size_t available = buf->size - buf->read_pos;

  if (available == 0) {
    return GNUTLS_E_AGAIN;
  }

  size_t to_read = (available < len) ? available : len;
  memcpy(data, buf->data + buf->read_pos, to_read);
  buf->read_pos += to_read;

  return to_read;
}

      &(gnutls_datum_t){(unsigned char *)kRSACertificateDER,
                        sizeof(kRSACertificateDER)},
      &(gnutls_datum_t){(unsigned char *)kRSAPrivateKeyDER,
                        sizeof(kRSAPrivateKeyDER)},
      GNUTLS_X509_FMT_DER);
  gnutls_credentials_set(server_session, GNUTLS_CRD_CERTIFICATE, scred);

  /* Client setup */
  gnutls_certificate_allocate_credentials(&ccred);
  gnutls_certificate_set_x509_trust_mem(
      ccred, &(gnutls_datum_t){_ca_der, _ca_der_len}, GNUTLS_X509_FMT_DER);
  gnutls_init(&client_session, GNUTLS_CLIENT);
  gnutls_server_name_set(client_session, GNUTLS_NAME_DNS, "localhost",
                         strlen("localhost"));
  gnutls_priority_set_direct(client_session, "NORMAL", NULL);
  gnutls_credentials_set(client_session, GNUTLS_CRD_CERTIFICATE, ccred);

  /* Transport setup */
  gnutls_transport_set_push_function(client_session, push_func);
  gnutls_transport_set_pull_function(client_session, pull_func);
  gnutls_transport_set_push_function(server_session, push_func);
  gnutls_transport_set_pull_function(server_session, pull_func);
  gnutls_transport_set_ptr(client_session, &server_buf);
  gnutls_transport_set_ptr(server_session, &client_buf);

  /* Handshake */
  int client_ret = 0, server_ret = 0;
  do {
    client_ret = gnutls_handshake(client_session);
    server_ret = gnutls_handshake(server_session);
  } while ((client_ret < 0 && gnutls_error_is_fatal(client_ret) == 0) ||
           (server_ret < 0 && gnutls_error_is_fatal(server_ret) == 0));

  if (client_ret >= 0 && server_ret >= 0) {
    /* Fuzz the hostname verification */
    hostname = (char *)malloc(size + 1);
    memcpy(hostname, data, size);
    hostname[size] = '\0';
    gnutls_certificate_verify_peers3(client_session, hostname, &status);
    free(hostname);
  }

  /* Cleanup */
  gnutls_deinit(client_session);
  gnutls_deinit(server_session);
  gnutls_certificate_free_credentials(ccred);
  gnutls_certificate_free_credentials(scred);
  gnutls_free(client_buf.data);
  gnutls_free(server_buf.data);
  gnutls_global_deinit();

  return 0;
}
