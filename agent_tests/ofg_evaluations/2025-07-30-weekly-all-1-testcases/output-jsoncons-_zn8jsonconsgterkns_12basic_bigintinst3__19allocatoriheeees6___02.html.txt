// Header for the function-under-test's class
#include "jsoncons/utility/bigint.hpp"

// Define the type alias for convenience
using my_bigint = jsoncons::basic_bigint<std::allocator<unsigned char>>;

// A global volatile variable to act as a sink, preventing dead code elimination.
volatile bool g_sink;

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    // Need enough data for two numbers.
    if (provider.remaining_bytes() < 2) {
        return 0;
    }

    // Create two complex bigints from the fuzzer data.
    size_t len1 = provider.ConsumeIntegralInRange<size_t>(1, 32);
    std::vector<uint8_t> bytes1 = provider.ConsumeBytes<uint8_t>(len1);
    if (bytes1.empty()) {
        return 0;
    }

    if (provider.remaining_bytes() < 1) {
        return 0;
    }
    size_t len2 = provider.ConsumeIntegralInRange<size_t>(1, 32);
    std::vector<uint8_t> bytes2 = provider.ConsumeBytes<uint8_t>(len2);
    if (bytes2.empty()) {
        return 0;
    }

    my_bigint b1 = my_bigint::from_bytes_be(1, bytes1.data(), bytes1.size());
    my_bigint b2 = my_bigint::from_bytes_be(1, bytes2.data(), bytes2.size());

    // With the -fno-inline flag added to the build script, a direct call
    // to the operator should now result in a non-inlined function call
    // that is visible to the coverage tooling.
    bool result = b1 > b2;

    // Store the result in a volatile sink to be absolutely sure the
    // compiler does not optimize the call away as dead code.
    g_sink = result;

    return 0;
}
