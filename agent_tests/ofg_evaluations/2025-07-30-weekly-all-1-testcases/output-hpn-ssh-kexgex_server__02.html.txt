// Simplified key preparation from kex_fuzz.cc
static struct sshkey *
prepare_key(void)
{
    const char *privstr = PRIV_RSA;
    struct sshkey *privkey = NULL;
    struct sshbuf *b = NULL;
    int r;

    if ((b = sshbuf_from(privstr, strlen(privstr))) == NULL)
        abort();
    if ((r = sshkey_parse_private_fileblob(b, "", &privkey, NULL)) != 0) {
        error_fr(r, "priv");
        abort();
    }
    sshbuf_free(b);
    return privkey;
}

// Function under test
int kexgex_server(struct ssh *);

extern "C" int
LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    static struct sshkey *privkey = NULL;
    struct ssh *server = NULL;
    struct kex_params kex_params;
    const char *proposal[PROPOSAL_MAX] = { KEX_SERVER };
    char *myproposal[PROPOSAL_MAX] = {0};
    int r;

    if (privkey == NULL) {
        log_init("kexgex_fuzz", SYSLOG_LEVEL_ERROR, SYSLOG_FACILITY_AUTH, 1);
        privkey = prepare_key();
    }

    // Set up a server instance
    for (int i = 0; i < PROPOSAL_MAX; i++) {
        if ((myproposal[i] = strdup(proposal[i])) == NULL)
            abort();
    }
    // Use our prepared key type
    free(myproposal[PROPOSAL_SERVER_HOST_KEY_ALGS]);
    if ((myproposal[PROPOSAL_SERVER_HOST_KEY_ALGS] =
        strdup(sshkey_ssh_name(privkey))) == NULL)
        abort();

    memcpy(kex_params.proposal, myproposal, sizeof(myproposal));

    if ((r = ssh_init(&server, 1, &kex_params)) != 0) {
        error_fr(r, "ssh_init server");
        goto out;
    }
    if ((r = ssh_add_hostkey(server, privkey)) != 0) {
        error_fr(r, "ssh_add_hostkey");
        goto out;
    }

    // Call the function to be fuzzed
    if (kexgex_server(server) != 0) {
        goto out;
    }

    // Feed the fuzzer input to the server
    if (ssh_input_append(server, data, size) != 0) {
        error_f("ssh_input_append failed");
        goto out;
    }

    // Run the dispatch loop to process the input
    ssh_dispatch_run(server, DISPATCH_NONBLOCK, NULL);

out:
    for (int i = 0; i < PROPOSAL_MAX; i++)
        free(myproposal[i]);
    if (server != NULL)
        ssh_free(server);

    return 0;
}
