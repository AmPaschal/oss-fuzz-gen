
	
<description>
The gnutls_record_recv_packet function receives a GnuTLS packet. It's used after a successful handshake to receive data from the peer. The function will block until a packet is received, or a timeout occurs.
</description>
<requirements>
<requirement>
The gnutls_handshake function must be called on the session before calling gnutls_record_recv_packet. This is because gnutls_record_recv_packet expects the session to be in an active state, which is achieved after a successful handshake.
</requirement>
<requirement>
The second argument, a pointer to gnutls_packet_t, must be a valid pointer. The pointed-to structure will be populated by the function, and the caller is responsible for deallocating it using gnutls_packet_deinit.
</requirement>
<requirement>
The first argument, gnutls_session_t, should be initialized by calling gnutls_init and a client or server-specific initialization function. This is to ensure that the session object is in a valid state before being used.
</requirement>
</requirements>

#include <assert.h>
#include <stdint.h>
#include <string.h>
#include <stdlib.h>
#include <stdbool.h>

#include <gnutls/gnutls.h>
#include <gnutls/x509.h>

#include "fuzzer.h"

/* In-memory transport functions */
typedef struct mem_st {
	const uint8_t *data;
	size_t size;
} mem_st;

#define MIN(x, y) ((x) < (y) ? (x) : (y))

static ssize_t mem_push(gnutls_transport_ptr_t tr, const void *data, size_t len)
{
	(void)tr;
	(void)data;
	return len;
}

static ssize_t mem_pull(gnutls_transport_ptr_t tr, void *data, size_t len)
{
	struct mem_st *p = (struct mem_st *)tr;

	if (p->size == 0) {
		return 0;
	}

	len = MIN(len, p->size);
	memcpy(data, p->data, len);

	p->size -= len;
	p->data += len;

	return len;
}

static int mem_pull_timeout(gnutls_transport_ptr_t tr, unsigned int ms)
{
	(void)ms;
	struct mem_st *p = (struct mem_st *)tr;

	if (p->size > 0)
		return 1; /* available data */
	else
		return 0; /* timeout */
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
	int res;
	gnutls_session_t session;
	gnutls_certificate_credentials_t xcred;
	struct mem_st memdata;

	res = gnutls_init(&session, GNUTLS_CLIENT);
	assert(res >= 0);

	res = gnutls_certificate_allocate_credentials(&xcred);
	assert(res >= 0);
	res = gnutls_credentials_set(session, GNUTLS_CRD_CERTIFICATE, xcred);
	assert(res >= 0);

	res = gnutls_priority_set_direct(session, "NORMAL", NULL);
	assert(res >= 0);

	memdata.data = data;
	memdata.size = size;

	gnutls_transport_set_push_function(session, mem_push);
	gnutls_transport_set_pull_function(session, mem_pull);
	gnutls_transport_set_pull_timeout_function(session, mem_pull_timeout);
	gnutls_transport_set_ptr(session, &memdata);

	do {
		res = gnutls_handshake(session);
	} while (res < 0 && gnutls_error_is_fatal(res) == 0);

	if (res >= 0) {
		while (true) {
			gnutls_packet_t packet = NULL;
			res = gnutls_record_recv_packet(session, &packet);
			if (res > 0) {
				gnutls_packet_deinit(packet);
			} else {
				if (packet) {
					gnutls_packet_deinit(packet);
				}
				break;
			}
		}
	}

	gnutls_deinit(session);
	gnutls_certificate_free_credentials(xcred);
	return 0;
}
