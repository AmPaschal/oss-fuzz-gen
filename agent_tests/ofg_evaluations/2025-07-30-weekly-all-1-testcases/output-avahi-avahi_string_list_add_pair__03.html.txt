
	
<description>
The function `avahi_string_list_add_pair` adds a key-value pair to an `AvahiStringList`. It takes a pointer to a string list `l`, a key, and a value as input. If `l` is `NULL`, a new list is created. The `key` cannot be `NULL`. The `value` can be `NULL`. If the value is not `NULL`, the pair is added in the format "key=value". Otherwise, only the key is added. The function returns the modified string list.
</description>
<requirements>
<requirement>
The first argument `l` can be `NULL`. If `l` is `NULL`, a new `AvahiStringList` will be created, so it is not necessary to initialize it.
</requirement>
<requirement>
The `key` argument must not be `NULL`. The function has an `assert(key)` which will cause a crash if the key is `NULL`.
</requirement>
<requirement>
The `key` and `value` arguments should be null-terminated strings. These are passed to `avahi_string_list_add_printf` which expects C-style strings, and incorrect formatting can lead to memory errors.
</requirement>
<requirement>
The `value` argument can be `NULL`. The function handles the `NULL` case for the value, so no specific checks are needed before calling.
</requirement>
</requirements>

#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

#include "/src/avahi/avahi-common/strlst.h"
#include "avahi-core/log.h"

void log_function(AvahiLogLevel level, const char *txt) {}

AvahiStringList * avahi_string_list_add_pair(AvahiStringList *, const char *, const char *);

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    avahi_set_log_function(log_function);

    if (size == 0) {
        return 0;
    }

    const char *key_data = (const char *)data;
    size_t key_len = size;
    const char *value_data = NULL;
    size_t value_len = 0;

    const void *nul_pos = memchr(data, '\0', size);
    if (nul_pos) {
        key_len = (const uint8_t *)nul_pos - data;
        if (key_len < size - 1) {
            value_data = (const char *)nul_pos + 1;
            value_len = size - key_len - 1;
        }
    }

    char *key = (char *)malloc(key_len + 1);
    if (!key) {
        return 0;
    }
    memcpy(key, key_data, key_len);
    key[key_len] = '\0';

    char *value = NULL;
    if (value_data) {
        value = (char *)malloc(value_len + 1);
        if (!value) {
            free(key);
            return 0;
        }
        memcpy(value, value_data, value_len);
        value[value_len] = '\0';
    }

    AvahiStringList *slist = NULL;
    slist = avahi_string_list_add_pair(slist, key, value);

    avahi_string_list_free(slist);
    free(key);
    if (value) {
        free(value);
    }

    return 0;
}
