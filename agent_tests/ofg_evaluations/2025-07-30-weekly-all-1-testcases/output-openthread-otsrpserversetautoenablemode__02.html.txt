
	
<description>
The `otSrpServerSetAutoEnableMode` function is used to enable or disable the auto-enable mode of the SRP server. When auto-enable mode is enabled, the SRP server is automatically started when the Border Routing Manager detects that the device is acting as a border router. When disabled, the SRP server must be manually started and stopped. The first argument is a pointer to an `otInstance` object, which represents the OpenThread instance. The second argument is a boolean value that indicates whether to enable or disable the auto-enable mode.
</description>
<requirements>

<requirement>
The `aInstance` pointer must be a valid pointer to an `otInstance` object. This is because the function dereferences this pointer to access the SRP server instance.
</requirement>
<requirement>
The `aEnabled` argument can be fuzzed with either true or false values. The function is designed to handle both enabling and disabling the auto-enable mode.
</requirement>

// This custom FuzzDataProvider mimics the one in the original radio-one-node.cpp
// to ensure structural compatibility for compilation.
class FuzzDataProvider
{
public:
    FuzzDataProvider(const uint8_t *aData, size_t aSize)
        : mData(aData)
        , mSize(aSize)
    {
    }

    void ConsumeData(void *aBuf, size_t aLength)
    {
        if (aLength > mSize)
        {
            aLength = mSize;
        }
        memcpy(aBuf, mData, aLength);
        mData += aLength;
        mSize -= aLength;
    }

    uint8_t ConsumeIntegralInRange(uint8_t aMin, uint8_t aMax)
    {
        assert(aMin <= aMax);
        uint16_t range = aMax - aMin;
        uint8_t  result = 0;

        if (mSize > 0)
        {
            ConsumeData(&result, sizeof(result));
        }

        if (range < 255)
        {
            result = aMin + (result % (range + 1));
        }
        else
        {
            result = aMin + result;
        }
        return result;
    }

    size_t RemainingBytes(void) { return mSize; }

private:
    const uint8_t *mData;
    size_t         mSize;
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    FuzzDataProvider fdp(data, size);

    Core nexus;
    Node &node = nexus.CreateNode();
    otInstance *instance = &node.GetInstance();

    bool enable = false;
    if (fdp.RemainingBytes() > 0)
    {
        enable = fdp.ConsumeIntegralInRange(0, 1);
    }

    otSrpServerSetAutoEnableMode(instance, enable);

    nexus.AdvanceTime(1000);

    return 0;
}
