// Port for the test server
constexpr const int SERVER_PORT = 18080;

// This function will run in a separate thread.
// It sets up a simple Crow application and waits for connections.
void start_server() {
    crow::SimpleApp app;

    // A catch-all route is sufficient. When a request comes in,
    // the server will parse the URL, which triggers `handle_url`.
    CROW_ROUTE(app, "/<path>")
    ([](const std::string& /*path*/) {
        return "OK";
    });

    // Suppress verbose logging
    crow::logger::setLogLevel(crow::LogLevel::CRITICAL);

    // Run the server
    app.port(SERVER_PORT).bindaddr("127.0.0.1").run();
}

// This function is called once to initialize the server in a background thread.
void initialize_server() {
    static std::thread server_thread(start_server);
    // Give the server a moment to start up.
    std::this_thread::sleep_for(std::chrono::milliseconds(100));
    server_thread.detach();
}

// The main fuzzing entry point
extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    // Initialize the server on the first run
    static bool server_initialized = false;
    if (!server_initialized) {
        initialize_server();
        server_initialized = true;
    }

    FuzzedDataProvider fdp(data, size);
    std::string url_path = fdp.ConsumeRemainingBytesAsString();

    // Create a raw HTTP request with the fuzzed URL path
    std::string http_request = "GET /" + url_path + " HTTP/1.1\r\nHost: localhost\r\nConnection: close\r\n\r\n";

    // Create a socket
    int sock = socket(AF_INET, SOCK_STREAM, 0);
    if (sock < 0) {
        return 0;
    }

    // Set up server address
    struct sockaddr_in serv_addr;
    serv_addr.sin_family = AF_INET;
    serv_addr.sin_port = htons(SERVER_PORT);
    if (inet_pton(AF_INET, "127.0.0.1", &serv_addr.sin_addr) <= 0) {
        close(sock);
        return 0;
    }

    // Connect to the server
    if (connect(sock, (struct sockaddr *)&serv_addr, sizeof(serv_addr)) < 0) {
        close(sock);
        return 0;
    }

    // Send the fuzzed HTTP request
    send(sock, http_request.c_str(), http_request.length(), 0);

    // We don't need to wait for a response, as we are only interested
    // in the server-side processing which might lead to a crash.
    close(sock);

    return 0;
}
