
	
<description>
The function `JS_ParseJSON` parses a JSON string from a buffer and returns a `JSValue`. It is a wrapper for `JS_ParseJSON2` which performs the main parsing logic. `JS_ParseJSON2` initializes a `JSParseState` struct and then iteratively calls `json_next_token` to tokenize the input buffer and `json_parse_value` to construct the corresponding JavaScript objects. The function expects a valid `JSContext` to be provided, which is used for memory management and error reporting. The input buffer is treated as a JSON string, and its length must be provided. The `filename` argument is used for error reporting purposes.
</description>

/*
 * Fuzzing for JSON parsing.
 * This target creates a new runtime and context for each input to ensure
 * a clean state, which is more robust for fuzzing.
 */
int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size == 0)
        return 0;

    JSRuntime *rt = JS_NewRuntime();
    if (!rt) {
        return 0;
    }

    JSContext *ctx = JS_NewContext(rt);
    if (!ctx) {
        JS_FreeRuntime(rt);
        return 0;
    }

    /* A minimal context is needed for JS_ParseJSON to work correctly.
     * We add the base objects and the JSON intrinsic. */
    JS_AddIntrinsicBaseObjects(ctx);
    JS_AddIntrinsicJSON(ctx);

    const char *json_str = (const char *)data;
    JSValue val = JS_ParseJSON(ctx, json_str, size, "<fuzz>");

    /* Free the returned JSValue, regardless of whether it's a valid
     * object or an exception, to prevent memory leaks. */
    JS_FreeValue(ctx, val);

    JS_FreeContext(ctx);
    JS_FreeRuntime(rt);

    return 0;
}
