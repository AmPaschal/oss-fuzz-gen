
	
<description>
The `pr_data_xfer` function is the core data transfer engine for the proftpd server. It handles both uploading (reading from the network and writing to a buffer) and downloading (reading from a buffer and writing to the network). The direction of the transfer is determined by the global `session.xfer.direction` variable. The function supports both binary and ASCII transfer modes, with the mode being controlled by flags in the global `session.sf_flags` variable. The function relies heavily on a pre-configured global `session` object, which must be properly initialized before this function is called, including setting up the data connection socket (`session.d`).
</description>

</requirement>

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size < 2) {
        return 0;
    }

    pool *p = NULL;
    char *cl_buf = NULL;
    int in_fds[2] = {-1, -1};
    int out_fds[2] = {-1, -1};

    p = make_sub_pool(NULL);
    if (!p) {
        return 0;
    }

    // --- Setup global session state ---
    memset(&session, 0, sizeof(session));
    session.pool = p;

    // To bypass poll_ctrl(), set control connection to NULL
    session.c = NULL;

    // Setup data connection (session.d)
    session.d = pcalloc(p, sizeof(conn_t));
    if (!session.d) {
        goto cleanup;
    }
    session.d->pool = p;

    // Create pipes to simulate network connection
    if (pipe(in_fds) < 0 || pipe(out_fds) < 0) {
        goto cleanup;
    }

    // Make pipes non-blocking
    fcntl(in_fds[0], F_SETFL, O_NONBLOCK);
    fcntl(in_fds[1], F_SETFL, O_NONBLOCK);
    fcntl(out_fds[0], F_SETFL, O_NONBLOCK);
    fcntl(out_fds[1], F_SETFL, O_NONBLOCK);

    // Setup input stream (for reading from "network")
    session.d->instrm = pcalloc(p, sizeof(pr_netio_stream_t));
    if (!session.d->instrm) {
        goto cleanup;
    }
    session.d->instrm->strm_fd = in_fds[0];
    session.d->instrm->strm_pool = p;

    // Setup output stream (for writing to "network")
    session.d->outstrm = pcalloc(p, sizeof(pr_netio_stream_t));
    if (!session.d->outstrm) {
        goto cleanup;
    }
    session.d->outstrm->strm_fd = out_fds[1];
    session.d->outstrm->strm_pool = p;

    // --- Consume fuzzer input ---
    session.xfer.direction = (data[0] % 2 == 0) ? PR_NETIO_IO_RD : PR_NETIO_IO_WR;
    session.sf_flags = (data[1] & 0x01) ? SF_ASCII : 0;

    const uint8_t *payload = data + 2;
    size_t payload_size = size - 2;

    if (payload_size == 0) {
        goto cleanup;
    }

    // --- Setup transfer-specific state ---
    session.xfer.p = make_sub_pool(p);
    if (!session.xfer.p) {
        goto cleanup;
    }

    // If reading, write fuzzer data to the "network" pipe
    if (session.xfer.direction == PR_NETIO_IO_RD) {
        write(in_fds[1], payload, payload_size);

        // If ASCII mode, setup the transfer buffer
        if (session.sf_flags & SF_ASCII) {
            session.xfer.bufsize = 4096;
            session.xfer.buf = pcalloc(session.xfer.p, session.xfer.bufsize + 1);
            if (session.xfer.buf) {
                session.xfer.buf++; // As seen in data.c
                session.xfer.buflen = 0;
            }
        }
    }

    // --- Call the target function ---
    cl_buf = (char *) malloc(payload_size);
    if (!cl_buf) {
        goto cleanup;
    }

    if (session.xfer.direction == PR_NETIO_IO_WR) {
        // For writing, the fuzzer input is the client buffer
        memcpy(cl_buf, payload, payload_size);
    }
    
    pr_data_xfer(cl_buf, payload_size);

cleanup:
    if (cl_buf) free(cl_buf);
    if (in_fds[0] != -1) close(in_fds[0]);
    if (in_fds[1] != -1) close(in_fds[1]);
    if (out_fds[0] != -1) close(out_fds[0]);
    if (out_fds[1] != -1) close(out_fds[1]);
    if (p) destroy_pool(p);

    return 0;
}
