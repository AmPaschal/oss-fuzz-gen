#include <cstdint>
#include <string>
#include <functional>

#include "httplib.h"

class FuzzedStream : public httplib::Stream {
public:
  FuzzedStream(const uint8_t *data, size_t size)
      : data_(data), size_(size), read_pos_(0) {}

  bool is_readable() const override { return true; }
  bool wait_readable() const override { return true; }
  bool wait_writable() const override { return true; }

  ssize_t read(char *ptr, size_t size) override {
    if (read_pos_ >= size_) {
      return 0;
    }
    size_t available = size_ - read_pos_;
    size_t to_read = std::min(size, available);
    memcpy(ptr, data_ + read_pos_, to_read);
    read_pos_ += to_read;
    return static_cast<ssize_t>(to_read);
  }

  ssize_t write(const char * /*ptr*/, size_t size) override {
    return static_cast<ssize_t>(size);
  }

  void get_remote_ip_and_port(std::string &ip, int &port) const override {
    ip = "127.0.0.1";
    port = 8080;
  }

  void get_local_ip_and_port(std::string &ip, int &port) const override {
    ip = "127.0.0.1";
    port = 80;
  }

  socket_t socket() const override { return 0; }

  time_t duration() const override { return 0; }

private:
  const uint8_t *data_;
  size_t size_;
  size_t read_pos_;
};

class FuzzableServer : public httplib::Server {
public:
  void ProcessFuzzedRequest(FuzzedStream &stream) {
    std::string remote_addr;
    int remote_port = -1;
    stream.get_remote_ip_and_port(remote_addr, remote_port);

    std::string local_addr;
    int local_port = -1;
    stream.get_local_ip_and_port(local_addr, local_port);

    bool connection_close = false;
    process_request(stream, remote_addr, remote_port, local_addr, local_port,
                    /*close_connection=*/false, connection_close, nullptr);
  }
};
