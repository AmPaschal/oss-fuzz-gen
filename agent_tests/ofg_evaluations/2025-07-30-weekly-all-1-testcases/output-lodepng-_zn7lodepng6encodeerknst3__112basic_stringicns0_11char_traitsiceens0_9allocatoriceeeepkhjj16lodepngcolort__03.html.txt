
	
<description>
The lodepng::encode function encodes raw pixel data into a PNG image and saves it to a file. It is a wrapper around another encode function that takes a std::vector<unsigned char> as an output buffer. This inner function then calls the C-style lodepng_encode_memory function, which performs the actual PNG encoding. The final PNG data is then written to the specified file.
</description>
<requirements>

<requirement>
The `in` buffer must be large enough to hold the raw image data. Its size is determined by the formula `w * h * bytes_per_pixel`, where `bytes_per_pixel` depends on the `colortype` and `bitdepth`.
</requirement>
<requirement>
The `colortype` and `bitdepth` parameters must be a valid combination as specified in the `LodePNGColorType` enum. For example, `LCT_GREY` supports bit depths of 1, 2, 4, 8, and 16, while `LCT_RGB` only supports 8 and 16.
</requirement>
<requirement>
The `filename` parameter should be a valid, writable file path. The function will attempt to save the encoded PNG to this location.
</requirement>
<requirement>
The width `w` and height `h` parameters should be greater than zero. A width or height of zero would result in an image with no pixels, which may not be a valid PNG.
</requirement>

</requirements>

	Fuzz target source:
#include <fuzzer/FuzzedDataProvider.h>
#include "/src/lodepng/lodepng.h"

#include <string>
#include <vector>
#include <cstdint>

// Valid non-palette colortype/bitdepth combinations for raw input.
// Palette types are excluded as the simple encode API has no way to provide a palette.
const size_t num_combinations = 11;

const LodePNGColorType colortypes[num_combinations] = {
    LCT_GREY,       LCT_GREY,       LCT_GREY,    LCT_GREY,    LCT_GREY, // 1, 2, 4, 8 or 16 bits
    LCT_RGB,        LCT_RGB,                                            // 8 or 16 bits
    LCT_GREY_ALPHA, LCT_GREY_ALPHA,                                     // 8 or 16 bits
    LCT_RGBA,       LCT_RGBA,                                           // 8 or 16 bits
};

const unsigned bitdepths[num_combinations] = {
    1, 2,  4, 8, 16, // gray
    8, 16,           // rgb
    8, 16,           // gray+alpha
    8, 16,           // rgb+alpha
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size < 10) {
    return 0;
  }
  FuzzedDataProvider provider(data, size);

  // Consume dimensions for the image.
  // Keep them small to avoid excessive memory allocation.
  unsigned int w = provider.ConsumeIntegralInRange<unsigned int>(1, 256);
  unsigned int h = provider.ConsumeIntegralInRange<unsigned int>(1, 256);

  // Pick a color type and bit depth for the raw image.
  size_t combination_index = provider.ConsumeIntegralInRange<size_t>(0, num_combinations - 1);
  LodePNGColorType color_type = colortypes[combination_index];
  unsigned int bit_depth = bitdepths[combination_index];

  LodePNGColorMode mode;
  lodepng_color_mode_init(&mode);
  mode.colortype = color_type;
  mode.bitdepth = bit_depth;

  unsigned bpp = lodepng_get_bpp(&mode);
  if (bpp == 0) { // Should not happen with our valid combinations
      return 0;
  }
  
  // Calculate size of the raw image buffer.
  // The raw buffer does not have padding bits at the end of scanlines.
  size_t raw_size = (static_cast<size_t>(w) * h * bpp + 7) / 8;

  // Avoid large allocations that could lead to OOM.
  if (raw_size > 1024 * 1024) {
      return 0;
  }

  std::vector<uint8_t> raw_image = provider.ConsumeBytes<uint8_t>(raw_size);
  if (raw_image.size() != raw_size) {
    return 0;
  }

  // Call the target function.
  // Using "/dev/null" as the filename avoids filesystem side-effects.
  lodepng::encode("/dev/null", raw_image.data(), w, h, color_type, bit_depth);

  return 0;
}
