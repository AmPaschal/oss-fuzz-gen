It is a wrapper around ZSTD_compressStream2, which performs the actual compression. The function is designed to be called multiple times, consuming input from the input buffer and writing compressed data to the output buffer. The ZSTD_CStream object must be created and initialized before calling this function. The input and output buffers must be properly set up with valid memory regions and sizes. The function returns a hint for the next input size.
</description>

size_t ZSTD_compressStream(ZSTD_CStream *, ZSTD_outBuffer *, ZSTD_inBuffer *);

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    /* Create and initialize streaming context */
    ZSTD_CStream* const cstream = ZSTD_createCStream();
    if (cstream == NULL) {
        return 0;
    }
    /* The compression level is chosen arbitrarily. A more advanced fuzzer could
     * derive this from the input data.
     */
    int const compressionLevel = 3;
    size_t const initResult = ZSTD_initCStream(cstream, compressionLevel);
    if (ZSTD_isError(initResult)) {
        ZSTD_freeCStream(cstream);
        return 0;
    }

    /* Allocate output buffer */
    size_t const outBuffSize = ZSTD_CStreamOutSize();
    void*  const outBuff = malloc(outBuffSize);
    if (outBuff == NULL) {
        ZSTD_freeCStream(cstream);
        return 0;
    }

    /* Setup input and output buffers */
    ZSTD_inBuffer inBuffer = { data, size, 0 };
    ZSTD_outBuffer outBuffer = { outBuff, outBuffSize, 0 };

    /* Continuously call ZSTD_compressStream until the input is fully consumed */
    while (inBuffer.pos < inBuffer.size) {
        /* When the output buffer is full, we just reset its position.
         * For fuzzing, we only care about exercising the code paths,
         * not about the actual compressed data.
         */
        outBuffer.pos = 0;

        size_t const cStreamResult = ZSTD_compressStream(cstream, &outBuffer, &inBuffer);
        if (ZSTD_isError(cStreamResult)) {
            /* An error occurred, break the loop and clean up */
            break;
        }
    }

    /* After the input is consumed, end the stream to write the epilogue.
     * This may produce more output, so it needs its own loop.
     */
    size_t remaining;
    do {
        outBuffer.pos = 0;
        remaining = ZSTD_endStream(cstream, &outBuffer);
        if (ZSTD_isError(remaining)) {
            /* An error occurred, break the loop and clean up */
            break;
        }
    } while (remaining > 0);

    /* Clean up resources */
    ZSTD_freeCStream(cstream);
    free(outBuff);

    return 0;
}
