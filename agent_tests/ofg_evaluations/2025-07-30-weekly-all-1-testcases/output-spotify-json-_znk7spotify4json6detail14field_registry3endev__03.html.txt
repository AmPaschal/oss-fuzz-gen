extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);
  spotify::json::detail::field_registry registry;

  // Populate the registry with fuzzed data by calling save().
  // Leave at least one byte for the final choice.
  while (provider.remaining_bytes() > 1) {
    const std::string name = provider.ConsumeRandomLengthString(provider.remaining_bytes() / 2);
    const bool required = provider.ConsumeBool();
    std::shared_ptr<spotify::json::detail::field> f = std::make_shared<dummy_field>();
    registry.save(name, required, f);
  }

  // Define a pointer to the member function.
  using const_iterator_func =
      spotify::json::detail::field_registry::const_iterator (
          spotify::json::detail::field_registry::*)() const noexcept;

  // Use the fuzzer input to choose which function to call. This makes it
  // impossible for the compiler to know at compile time which function will be
  // called, thus preventing inlining and forcing a call instruction.
  const_iterator_func func_ptr = provider.ConsumeBool()
                                      ? &spotify::json::detail::field_registry::begin
                                      : &spotify::json::detail::field_registry::end;

  // Call the chosen function (begin or end) through the function pointer.
  // This will call the target function `end()` in roughly 50% of cases.
  auto it_end = (registry.*func_ptr)();

  // Use the resulting iterator to prevent the call from being optimized away.
  auto it_begin = registry.begin();
  volatile size_t count = 0;
  for (auto it = it_begin; it != it_end; ++it) {
    count++;
  }

  return 0;
}
