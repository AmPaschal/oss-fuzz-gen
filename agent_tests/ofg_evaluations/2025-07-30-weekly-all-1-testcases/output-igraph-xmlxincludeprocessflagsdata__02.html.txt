extern "C" {
#include <libxml/parser.h>

#include <libxml/xmlerror.h>
}

// xmlInitParser is not thread-safe and should be called once.
// We can use a static variable to ensure it's called only for the first input.
static void InitXmlParser() {
    static bool initialized = false;
    if (!initialized) {
        xmlInitParser();
        // Suppress default error output to keep logs clean.
        xmlSetGenericErrorFunc(NULL, NULL);
        initialized = true;
    }
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    InitXmlParser();

    FuzzedDataProvider provider(data, size);

    // Consume data for flags
    int flags = provider.ConsumeIntegral<int>();

    // Consume remaining data for XML content
    std::string xml_data = provider.ConsumeRemainingBytesAsString();
    if (xml_data.empty()) {
        return 0;
    }

    xmlDocPtr doc = xmlReadMemory(xml_data.c_str(), (int)xml_data.size(), "fuzz.xml", NULL, 0);
    if (doc == NULL) {
        return 0;
    }

    xmlXIncludeProcessFlagsData(doc, flags, NULL);

    xmlFreeDoc(doc);
    
    // xmlCleanupParser() is not called here because it cleans up global state,
    // which is expensive and unnecessary to do for every fuzz input.
    // The fuzzer process will be restarted often, handling the cleanup.
    // Resetting the last error is good practice between runs.
    xmlResetLastError();

    return 0;
}
