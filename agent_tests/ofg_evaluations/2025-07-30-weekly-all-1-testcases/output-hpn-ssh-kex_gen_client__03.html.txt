
	
<description>
The `kex_gen_client` function generates the client's ephemeral key pair for a key exchange and sends the public key to the server. It uses the `kex_type` from the `ssh->kex` struct to determine which key generation algorithm to use. After generating the keypair, it sends the public key to the server in a `SSH2_MSG_KEX_ECDH_INIT` message and sets up a handler for the server's reply.
</description>
<requirements>
<requirement>
The `ssh` argument must be a valid pointer to a `struct ssh` initialized by `ssh_alloc_session_state`. This ensures that the session state is properly set up for subsequent operations.
</requirement>
<requirement>
The `ssh->kex` member must be a valid pointer to a `struct kex` initialized by `kex_new`. This is because the function's logic is entirely dependent on the state contained within the `kex` structure.
</requirement>
<requirement>
The `ssh->kex->kex_type` field must be set to one of the supported key exchange types. Supported types are `KEX_DH_GRP1_SHA1`, `KEX_DH_GRP14_SHA1`, `KEX_DH_GRP14_SHA256`, `KEX_DH_GRP16_SHA512`, `KEX_DH_GRP18_SHA512`, `KEX_ECDH_SHA2`, `KEX_C25519_SHA256`, `KEX_KEM_SNTRUP761X25519_SHA512`, or `KEX_KEM_MLKEM768X25519_SHA256`. This is to ensure a valid code path is taken in the switch statement.
</requirement>
<requirement>
The `ssh` object must be associated with a connection using `ssh_packet_set_connection`. This is necessary for the packet sending functions (`sshpkt_start`, `sshpkt_put_stringb`, `sshpkt_send`) to operate correctly.
</requirement>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    static bool initialized = false;
    if (!initialized) {
        log_init("kex_gen_client_fuzz", SYSLOG_LEVEL_QUIET, SYSLOG_FACILITY_AUTH, 1);
        initialized = true;
    }

    FuzzedDataProvider provider(data, size);

    const char* kex_algs[] = {
#ifdef WITH_OPENSSL
        KEX_DH1,
        KEX_DH14_SHA1,
        KEX_DH14_SHA256,
        KEX_DH16_SHA512,
        KEX_DH18_SHA512,
        KEX_ECDH_SHA2_NISTP256,
        KEX_ECDH_SHA2_NISTP384,
        KEX_ECDH_SHA2_NISTP521,
#endif
        KEX_CURVE25519_SHA256,
        KEX_CURVE25519_SHA256_OLD,
        KEX_SNTRUP761X25519_SHA512,
        KEX_SNTRUP761X25519_SHA512_OLD,
	KEX_MLKEM768X25519_SHA256,
    };

    const char* chosen_kex = provider.PickValueInArray(kex_algs);

    char *myproposal[PROPOSAL_MAX];
    for (int i = 0; i < PROPOSAL_MAX; i++) {
        myproposal[i] = NULL;
    }

    const char *proposal_template[PROPOSAL_MAX] = {
	    "sntrup761x25519-sha512@openssh.com,mlkem768x25519-sha256,curve25519-sha256,curve25519-sha256@libssh.org,ecdh-sha2-nistp256,ecdh-sha2-nistp384,ecdh-sha2-nistp521,diffie-hellman-group-exchange-sha256,diffie-hellman-group16-sha512,diffie-hellman-group18-sha512,diffie-hellman-group14-sha256,diffie-hellman-group14-sha1,diffie-hellman-group-exchange-sha1,diffie-hellman-group1-sha1",
	    "rsa-sha2-512,rsa-sha2-256,ssh-ed25519",
	    "aes128-ctr,aes192-ctr,aes256-ctr",
	    "aes128-ctr,aes192-ctr,aes256-ctr",
	    "hmac-sha2-256,hmac-sha2-512",
	    "hmac-sha2-256,hmac-sha2-512",
	    "none,zlib@openssh.com",
	    "none,zlib@openssh.com",
	    "",
	    ""
    };

    bool proposal_ok = true;
    for (int i = 0; i < PROPOSAL_MAX; i++) {
        const char *ccp = proposal_template[i];
        if (i == PROPOSAL_KEX_ALGS) {
            ccp = chosen_kex;
        }
        myproposal[i] = xstrdup(ccp);
        if (myproposal[i] == NULL) {
            proposal_ok = false;
            break;
        }
    }

    if (proposal_ok) {
        struct kex_params kex_params;
        memcpy(kex_params.proposal, myproposal, sizeof(myproposal));

        struct ssh *client = NULL;
        if (ssh_init(&client, 0, &kex_params) == 0 && client != NULL && client->kex != NULL) {
            int kex_type_id = client->kex->kex_type;
            if (kex_type_id > 0 && kex_type_id < KEX_MAX && client->kex->kex[kex_type_id] != NULL) {
                (client->kex->kex[kex_type_id])(client);
            }
        }
        ssh_free(client);
    }

    for (int i = 0; i < PROPOSAL_MAX; i++) {
        free(myproposal[i]);
    }

    return 0;
}
