#include "hdf5.h"

#include <unistd.h>
#include <stdlib.h>
#include <stdio.h>

extern int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size == 0) {
        return 0;
    }

    char filename[256];
    sprintf(filename, "/tmp/libfuzzer.%d", getpid());

    // Create a new HDF5 file.
    hid_t file_id = H5Fcreate(filename, H5F_ACC_TRUNC, H5P_DEFAULT, H5P_DEFAULT);
    if (file_id < 0) {
        unlink(filename);
        return 0;
    }

    // Create a 1D dataspace for the dataset.
    hsize_t dims[1] = {size};
    hid_t space_id = H5Screate_simple(1, dims, NULL);
    if (space_id < 0) {
        H5Fclose(file_id);
        unlink(filename);
        return 0;
    }

    // Create the dataset.
    hid_t dset_id = H5Dcreate2(file_id, "fuzz_dset", H5T_NATIVE_UINT8, space_id,
                               H5P_DEFAULT, H5P_DEFAULT, H5P_DEFAULT);
    if (dset_id < 0) {
        H5Sclose(space_id);
        H5Fclose(file_id);
        unlink(filename);
        return 0;
    }

    // Write the fuzzer's data to the dataset. This public API call
    // will trigger the internal H5D__write function.
    H5Dwrite(dset_id, H5T_NATIVE_UINT8, H5S_ALL, H5S_ALL, H5P_DEFAULT, data);

    // Clean up HDF5 resources.
    H5Dclose(dset_id);
    H5Sclose(space_id);
    H5Fclose(file_id);

    // Remove the temporary file.
    unlink(filename);

    return 0;
}
