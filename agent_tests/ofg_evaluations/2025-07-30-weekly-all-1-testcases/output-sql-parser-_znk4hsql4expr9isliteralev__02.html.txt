// Helper function to traverse a list of expressions.
void FuzzExprList(const std::vector<Expr*>* exprList) {
  if (exprList) {
    for (const Expr* expr : *exprList) {
      FuzzExpr(expr);
    }
  }
}

// Recursive function to traverse the expression tree and call the target function.
void FuzzExpr(const Expr* expr) {
  if (!expr) {
    return;
  }

  // Call the function under test.
  expr->isLiteral();

  // Recursively call on sub-expressions.
  FuzzExpr(expr->expr);
  FuzzExpr(expr->expr2);
  FuzzExprList(expr->exprList);

  if (expr->select) {
    const SelectStatement* select = expr->select;
    FuzzExprList(select->selectList);
    FuzzExpr(select->whereClause);
    if (select->groupBy) {
      FuzzExprList(select->groupBy->columns);
      FuzzExpr(select->groupBy->having);
    }
  }
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t* data, size_t size) {
  std::string input(reinterpret_cast<const char*>(data), size);
  SQLParserResult res;
  SQLParser::parse(input, &res);

  if (!res.isValid()) {
    return 0;
  }

  for (const SQLStatement* stmt : res.getStatements()) {
    if (!stmt) {
      continue;
    }

    // Traverse expressions based on the statement type.
    switch (stmt->type()) {
      case kStmtSelect: {
        const auto* select_stmt = static_cast<const SelectStatement*>(stmt);
        FuzzExprList(select_stmt->selectList);
        FuzzExpr(select_stmt->whereClause);
        if (select_stmt->groupBy) {
          FuzzExprList(select_stmt->groupBy->columns);
          FuzzExpr(select_stmt->groupBy->having);
        }
        break;
      }
      case kStmtInsert: {
        const auto* insert_stmt = static_cast<const InsertStatement*>(stmt);
        FuzzExprList(insert_stmt->values);
        break;
      }
      case kStmtUpdate: {
        const auto* update_stmt = static_cast<const UpdateStatement*>(stmt);
        if (update_stmt->updates) {
          for (const UpdateClause* update : *update_stmt->updates) {
            FuzzExpr(update->value);
          }
        }
        FuzzExpr(update_stmt->where);
        break;
      }
      case kStmtDelete: {
        const auto* delete_stmt = static_cast<const DeleteStatement*>(stmt);
        FuzzExpr(delete_stmt->expr);
        break;
      }
      default:
        // Other statement types could be handled here if they have expressions.
        break;
    }
  }
  return 0;
}
