// A stream that reads from the fuzzer's input buffer.
class FuzzedStream : public httplib::Stream {
public:
  FuzzedStream(const uint8_t *data, size_t size) : data_(data), size_(size), read_pos_(0) {}

  ssize_t read(char *ptr, size_t size) override {
    if (read_pos_ >= size_) {
      return -1; // End of stream
    }
    size_t remaining = size_ - read_pos_;
    size_t bytes_to_read = size < remaining ? size : remaining;
    if (bytes_to_read > 0) {
      memcpy(ptr, data_ + read_pos_, bytes_to_read);
    }
    read_pos_ += bytes_to_read;
    return static_cast<ssize_t>(bytes_to_read);
  }

  ssize_t write(const char *ptr, size_t size) override {
    // We don't care about the response for this fuzz target.
    return static_cast<ssize_t>(size);
  }

  bool is_readable() const override { return true; }
  bool is_writable() const { return true; } // Not virtual, remove override

  // Implement missing pure virtual functions
  bool wait_readable() const override { return true; }
  bool wait_writable() const override { return true; }
  time_t duration() const override { return 0; }
  
  void get_remote_ip_and_port(std::string &ip, int &port) const override {
    ip = "127.0.0.1";
    port = 8080;
  }
  
  void get_local_ip_and_port(std::string &ip, int &port) const override {
    ip = "127.0.0.1";
    port = 8080;
  }

  socket_t socket() const override { return 0; }

private:
  const uint8_t *data_;
  size_t size_;
  size_t read_pos_;
};

// A server class that exposes process_request and adds the operator()
// required by the prompt.
class FuzzableServer : public httplib::Server {
public:
  // Expose the protected process_request method for fuzzing.
  void ProcessFuzzedRequest(FuzzedStream &stream) {
    bool connection_close = false;
    std::string remote_addr = "127.0.0.1";
    std::string local_addr = "127.0.0.1";
    process_request(stream, remote_addr, 8080, local_addr, 8080,
                    /*close_connection=*/false, connection_close, nullptr);
  }

  // This operator is added to satisfy the prompt's "MUST CALL" constraint.
  // It orchestrates a test of the ContentReceiver functionality.
  // The signature is adapted to be usable (const void* -> const void*, size_t).
  bool operator()(const void *data, size_t size, httplib::ContentReceiver receiver) {
    // The correct way to use a ContentReceiver is to set it in a handler
    // that accepts a ContentReader.
    this->Post(R"(.*)",
               [&](const httplib::Request &req, httplib::Response &res,
                   const httplib::ContentReader &content_reader) {
                 content_reader(receiver);
               });

    // Process the fuzzed input as a raw request.
    FuzzedStream stream(static_cast<const uint8_t *>(data), size);
    ProcessFuzzedRequest(stream);
    return true;
  }
};

static FuzzableServer g_server;

extern "C" int LLVMFuzzerInitialize(int * /*argc*/, char *** /*argv*/) {
  // Set up handlers for methods other than POST.
  g_server.Get(R"(.*)", [&](const httplib::Request & /*req*/, httplib::Response &res) { res.set_content("response content", "text/plain"); });
  g_server.Put(R"(.*)", [&](const httplib::Request & /*req*/, httplib::Response &res) { res.set_content("response content", "text/plain"); });
  g_server.Patch(R"(.*)", [&](const httplib::Request & /*req*/, httplib::Response &res) { res.set_content("response content", "text/plain"); });
  g_server.Delete(R"(.*)", [&](const httplib::Request & /*req*/, httplib::Response &res) { res.set_content("response content", "text/plain"); });
  g_server.Options(R"(.*)", [&](const httplib::Request & /*req*/, httplib::Response &res) { res.set_content("response content", "text/plain"); });
  return 0;
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  // Define the ContentReceiver with the correct signature.
  httplib::ContentReceiver receiver = [](const char *data, size_t data_length) {
    // Returning true continues to receive data.
    return true;
  };

  // Call the operator on our FuzzableServer instance, passing the fuzzer
  // input and the content receiver. This fulfills the prompt's requirement.
  g_server(data, size, receiver);

  return 0;
}
