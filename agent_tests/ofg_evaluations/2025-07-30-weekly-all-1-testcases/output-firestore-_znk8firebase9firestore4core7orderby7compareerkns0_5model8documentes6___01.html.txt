
	
<description>
The function `firebase::firestore::core::OrderBy::Compare` compares two `Document` objects based on a specified field and direction. The comparison logic depends on whether the `OrderBy` object is configured to sort by the document key or by a specific field within the document. If sorting by a field, the function retrieves the field's value from both documents and compares them. The function will fail with an assertion if the specified field is not present in both documents.
</description>
<requirements>
<requirement>
Both `Document` arguments, `lhs` and `rhs`, must be created from `MutableDocument` objects that have been initialized as `FoundDocument`. This ensures that the documents contain data and can be compared.
</requirement>
<requirement>
The `OrderBy` object must be initialized with a `FieldPath` that exists in both `Document` arguments. This is to prevent an assertion failure when the function tries to access a non-existent field.
</requirement>
<requirement>
If the `OrderBy` object is configured to sort by a specific field (i.e., `field_.IsKeyFieldPath()` is false), both `Document` arguments must contain the field specified by the `OrderBy` object. This is because the function asserts that both documents have the field before comparison, and will crash if the assertion fails.
</requirement>
</requirements>

// Creates a document for testing.
//
// The document will have a single field, specified by `field_path`, whose
// value is generated from the `FuzzedDataProvider`.
model::Document MakeDocument(
    FuzzedDataProvider* provider,
    const model::FieldPath& field_path) {
  std::string document_path = provider->ConsumeRandomLengthString(100);
  // DocumentKey::FromPathString requires an even number of segments.
  if (std::count(document_path.begin(), document_path.end(), '/') % 2 == 0) {
    document_path += "/a";
  }
  auto key = model::DocumentKey::FromPathString(document_path);

  auto version = model::SnapshotVersion{
      Timestamp{provider->ConsumeIntegral<int64_t>(),
                provider->ConsumeIntegralInRange<int32_t>(0, 999999999)}};

  model::ObjectValue data;
  switch (provider->ConsumeIntegralInRange<int>(0, 4)) {
    case 0:
      data.Set(field_path, Value(provider->ConsumeBool()));
      break;
    case 1:
      data.Set(field_path, Value(provider->ConsumeIntegral<int64_t>()));
      break;
    case 2:
      data.Set(field_path, Value(provider->ConsumeFloatingPoint<double>()));
      break;
    case 3:
      data.Set(field_path, Value(provider->ConsumeRandomLengthString(100)));
      break;
    case 4:
      data.Set(field_path, Value(Timestamp{
                                provider->ConsumeIntegral<int64_t>(),
                                provider->ConsumeIntegralInRange<int32_t>(
                                    0, 999999999)}));
      break;
  }

  return model::Document{
      model::MutableDocument::FoundDocument(key, version, std::move(data))};
}

  FuzzedDataProvider provider(data, size);

  try {
    const auto& direction = provider.ConsumeBool() ? Direction::Ascending
                                                  : Direction::Descending;
    std::string field_path_str = provider.ConsumeRandomLengthString(100);
    // `FieldPath::FromDotSeparatedString` has assertions against invalid
    // characters.
    if (field_path_str.empty() ||
        field_path_str.find_first_of(".[]*`") != std::string::npos) {
      return 0;
    }
    FieldPath field = FieldPath::FromDotSeparatedString(field_path_str);

    OrderBy order_by(field, direction);

    Document doc1 = firebase::firestore::MakeDocument(&provider, field);
    Document doc2 = firebase::firestore::MakeDocument(&provider, field);

    order_by.Compare(doc1, doc2);

  } catch (...) {
    // Ignore exceptions from invalid, fuzzer-generated input.
  }

  return 0;
