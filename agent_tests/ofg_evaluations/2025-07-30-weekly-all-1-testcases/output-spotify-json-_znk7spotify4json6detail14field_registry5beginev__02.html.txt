  std::string value;

// Define the type for the member function pointer to ensure it matches the signature exactly.
using begin_ptr_t = spotify::json::detail::field_registry::const_iterator (spotify::json::detail::field_registry::*)() const noexcept;

// This function is marked as noinline to prevent the compiler from seeing
// the call to begin() and the construction of the codec in the same optimization
// unit. This creates an optimization barrier.
__attribute__((noinline))
void invoke_begin_via_pointer(const spotify::json::detail::field_registry &registry, begin_ptr_t p_begin, const std::string &check_str) {
    // Call the function through the pointer.
    const auto it = (registry.*p_begin)();
    const auto it_end = registry.end();

    // Create an un-optimizable control flow path.
    // The compiler cannot know the contents of the strings from the fuzzer,
    // so it cannot prove that this condition is always false. Therefore, it
    // must generate the code for the check, which depends on the result of
    // the function call. The call to abort() is a non-returning call, making
    // the dependency even stronger.
    if (it != it_end) {
        if (it->first == check_str) {
            // This path is unlikely to be taken, but its existence forces the
            // compiler to preserve the code that leads to it.
            abort();
        }
    }
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

  // Create an instance of the codec.
  auto codec = spotify::json::codec::object<fuzz_object>();

  // We will add one field to the codec and then check if its name matches
  // another string from the fuzzer.
  const std::string field_to_add = provider.ConsumeRandomLengthString(32);
  const std::string field_to_check = provider.ConsumeRemainingBytesAsString();

  // Ensure we add at least one field so that begin() returns a valid iterator.
  if (field_to_add.empty()) {
      codec.required("a_default_key", &fuzz_object::value);
  } else {
      codec.required(field_to_add, &fuzz_object::value);
  }

  // Because we used the #define hack, _fields is now public in object_t's base
  // class, and since object_t inherits publicly, we can access it.
  const spotify::json::detail::field_registry &registry = codec._fields;

  // 1. Force a non-inlined instance of begin() to be emitted by taking its address.
  begin_ptr_t p_begin = &spotify::json::detail::field_registry::begin;

  // 2. Call the noinline helper function, passing the necessary components.
  invoke_begin_via_pointer(registry, p_begin, field_to_check);

  return 0;
