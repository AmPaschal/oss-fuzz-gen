
	
<description>
The function `cmsCreateProofingTransformTHR` creates a proofing transform. It takes input, output, and proofing profiles, their formats, intents, and a flags bitmask as arguments. It can also perform black point compensation. The context ID is derived from the input profile.
</description>
<requirements>
<requirement>
The `InputProfile`, `OutputProfile`, and `ProofingProfile` arguments must be valid `cmsHPROFILE` handles. These can be created by calling `cmsOpenProfileFromFile` on existing ICC profile files.
</requirement>
<requirement>
The `InputFormat` and `OutputFormat` arguments must be valid format specifiers. These are defined in `lcms2.h` with the `PT_` and `TYPE_` prefixes.
</requirement>
<requirement>
The `nIntent` and `ProofingIntent` arguments must be valid intent values. The valid values are defined in `lcms2.h` with the `INTENT_` prefix.
</requirement>
<requirement>
The `dwFlags` argument must be a bitmask of flags defined in `lcms2.h` with the `cmsFLAGS_` prefix. To ensure the proofing-specific logic is tested, the `dwFlags` argument should have either the `cmsFLAGS_SOFTPROOFING` or `cmsFLAGS_GAMUTCHECK` flag set.
</requirement>
<requirement>
The `ContextID` argument should be created by calling `cmsGetProfileContextID` on the `InputProfile`. This is how the only caller of the target function creates the context ID.
</requirement>
</requirements>

#include "/src/lcms/include/lcms2.h"
#include <stdint.h>

cmsHTRANSFORM cmsCreateProofingTransformTHR(cmsContext, cmsHPROFILE, cmsUInt32Number, cmsHPROFILE, cmsUInt32Number, cmsHPROFILE, cmsUInt32Number, cmsUInt32Number, cmsUInt32Number);

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size < 16) {
        return 0;
    }

    const uint32_t nIntent = *((const uint32_t *)data) % 16;
    const uint32_t ProofingIntent = *((const uint32_t *)data + 1) % 16;
    const uint32_t dwFlags = *((const uint32_t *)data + 2);
    data += 12;
    size -= 12;

    if (size == 0) {
        return 0;
    }

    cmsHPROFILE InputProfile = cmsOpenProfileFromMem(data, size);
    if (!InputProfile) {
        return 0;
    }

    cmsHPROFILE OutputProfile = cmsCreate_sRGBProfile();
    if (!OutputProfile) {
        cmsCloseProfile(InputProfile);
        return 0;
    }

    cmsHPROFILE ProofingProfile = cmsCreate_sRGBProfile();
    if (!ProofingProfile) {
        cmsCloseProfile(InputProfile);
        cmsCloseProfile(OutputProfile);
        return 0;
    }

    cmsColorSpaceSignature srcCS = cmsGetColorSpace(InputProfile);
    cmsUInt32Number nSrcComponents = cmsChannelsOf(srcCS);
    cmsUInt32Number InputFormat;
    if (srcCS == cmsSigLabData) {
        InputFormat = COLORSPACE_SH(PT_Lab) | CHANNELS_SH(nSrcComponents) | BYTES_SH(0);
    } else {
        InputFormat = COLORSPACE_SH(PT_ANY) | CHANNELS_SH(nSrcComponents) | BYTES_SH(1);
    }

    cmsUInt32Number OutputFormat = TYPE_BGR_8;

    cmsHTRANSFORM hTransform = cmsCreateProofingTransformTHR(
        NULL,
        InputProfile,
        InputFormat,
        OutputProfile,
        OutputFormat,
        ProofingProfile,
        nIntent,
        ProofingIntent,
        dwFlags);

    if (hTransform) {
        cmsDeleteTransform(hTransform);
    }
    cmsCloseProfile(InputProfile);
    cmsCloseProfile(OutputProfile);
    cmsCloseProfile(ProofingProfile);

    return 0;
}
