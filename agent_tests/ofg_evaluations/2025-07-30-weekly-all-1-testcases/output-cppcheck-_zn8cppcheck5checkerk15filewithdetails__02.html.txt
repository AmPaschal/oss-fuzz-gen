  std::ofstream out(filename);
  out << code;
  out.close();

  // Create Settings and Suppressions
  Settings settings;
  settings.addEnabled("all");
  settings.certainty.setEnabled(Certainty::inconclusive, true);
  Suppressions suppressions;

  // Create FileWithDetails for the temporary file
  const FileWithDetails file(filename, Standards::Language::CPP, code.size());

  // Create and configure CppCheck instance
  CppCheck cppcheck(settings, suppressions, s_errorLogger, false, nullptr);

  // Call the target function
  cppcheck.check(file);

  return 0;
}
