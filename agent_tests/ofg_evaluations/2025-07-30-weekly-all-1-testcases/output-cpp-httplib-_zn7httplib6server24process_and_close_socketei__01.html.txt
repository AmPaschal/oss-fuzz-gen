// A stream that reads from the fuzzer's input buffer.
class FuzzedStream : public httplib::Stream {
public:
  FuzzedStream(const uint8_t *data, size_t size)
      : data_(data), size_(size), read_pos_(0) {}

  ~FuzzedStream() override = default;

  bool is_readable() const override { return read_pos_ < size_; }

  bool wait_readable() const override { return true; }

  bool wait_writable() const override { return true; }

  ssize_t read(char *ptr, size_t size) override {
    if (read_pos_ >= size_) {
      return -1; // No more data
    }
    size_t remaining = size_ - read_pos_;
    size_t bytes_to_read = size > remaining ? remaining : size;
    memcpy(ptr, data_ + read_pos_, bytes_to_read);
    read_pos_ += bytes_to_read;
    return static_cast<ssize_t>(bytes_to_read);
  }

  ssize_t write(const char *ptr, size_t size) override {
    // For fuzzing, we can just discard the written data.
    return static_cast<ssize_t>(size);
  }

  void get_remote_ip_and_port(std::string &ip, int &port) const override {
    ip = "127.0.0.1";
    port = 12345;
  }

  void get_local_ip_and_port(std::string &ip, int &port) const override {
    ip = "127.0.0.1";
    port = 8080;
  }

  socket_t socket() const override {
    // Return a dummy socket descriptor.
    return 0;
  }

  time_t duration() const override { return 0; }

private:
  const uint8_t *data_;
  size_t size_;
  size_t read_pos_;
};

// A server class with a public method to fuzz the socket processing logic.
class FuzzableServer : public httplib::Server {
public:
  // This method simulates the core logic of `process_and_close_socket`
  // but uses our `FuzzedStream` instead of a real socket.
  bool ProcessAndCloseFuzzedSocket(httplib::Stream &strm) {
    std::string remote_addr;
    int remote_port = 0;
    strm.get_remote_ip_and_port(remote_addr, remote_port);

    std::string local_addr;
    int local_port = 0;
    strm.get_local_ip_and_port(local_addr, local_port);

    bool close_connection = false;
    bool connection_closed = false;

    // This is the core logic from `process_and_close_socket` that we want to fuzz.
    // It processes one request from the stream.
    auto ret = process_request(strm, remote_addr, remote_port, local_addr,
                               local_port, false, close_connection,
                               nullptr);

    return ret;
  }
};

static FuzzableServer g_server;

// Initialize the server with some routes.
extern "C" int LLVMFuzzerInitialize(int * /*argc*/, char *** /*argv*/) {
  g_server.Get(R"(.*)",
               [&](const httplib::Request &, httplib::Response &res) {
                 res.set_content("response", "text/plain");
               });
  g_server.Post(R"(.*)",
                [&](const httplib::Request &, httplib::Response &res) {
                  res.set_content("response", "text/plain");
                });
  g_server.Put(R"(.*)",
               [&](const httplib::Request &, httplib::Response &res) {
                 res.set_content("response", "text/plain");
               });
  return 0;
}

// The main fuzzing entry point.
extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size == 0) {
    return 0;
  }
  FuzzedStream stream(data, size);
  g_server.ProcessAndCloseFuzzedSocket(stream);
  return 0;
}
