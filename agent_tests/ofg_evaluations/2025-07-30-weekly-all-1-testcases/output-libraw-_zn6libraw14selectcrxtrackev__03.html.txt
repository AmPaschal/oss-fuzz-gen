
	
<description>
The function `selectCRXTrack` is responsible for selecting the primary RAW track from a CRX file. It inspects the available tracks, calculates which one contains the highest resolution RAW data, and then sets various fields in the LibRaw data structure to point to this track's data. It also identifies associated JPEG and metadata tracks. The caller `LibRaw::identify` first calls `parseCR3` to populate the CRX track metadata before calling this function.
</description>
<requirements>

<requirement>
The member variable `libraw_internal_data.unpacker_data.crx_track_count` must be initialized to a value between 0 and 15. This is because it determines the number of valid entries in the `crx_header` array, and the code iterates up to this value, capped at `LIBRAW_CRXTRACKS_MAXCOUNT` (16).
</requirement>
<requirement>
The `libraw_internal_data.unpacker_data.crx_header` array must be populated with valid data for the number of tracks specified by `crx_track_count`. The fields within each `crx_data_header_t` struct, such as `MediaType`, `nBits`, `f_width`, `f_height`, `sample_count`, `MediaSize` and `MediaOffset`, must be initialized to prevent crashes and ensure correct track selection.
</requirement>
<requirement>
The `shot_select` member variable must be initialized. This variable is used to select a specific track or frame when multiple are available, and an uninitialized value could lead to unpredictable behavior.
</requirement>
<requirement>
The function `parseCR3()` must be called before `selectCRXTrack()`. This is because `parseCR3()` is responsible for populating the `crx_track_count` and `crx_header` fields, which are essential for the logic of `selectCRXTrack()`.
</requirement>

</requirements>

#include <stddef.h>
#include <stdint.h>

#include <string>
#include <vector>
#include <fuzzer/FuzzedDataProvider.h>

// Define this macro to allow inclusion of internal libraw headers
#define LIBRAW_LIBRARY_BUILD
#include <libraw/libraw.h>
#include "internal/libraw_internal_funcs.h"

// Derived class to access the protected selectCRXTrack method
class FuzzLibRaw : public LibRaw {
public:
  using LibRaw::selectCRXTrack;
};


enum InterpolationOptions {
  Linear = 0,
  Vng = 1,
  Ppg = 2,
  Ahd = 3,
  Dcb = 4,
  Dht = 11,
  AhdModified = 12
};
static const InterpolationOptions options[] = {Linear, Vng, Ppg, Ahd, Dcb, Dht, AhdModified};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t* data, size_t size) {
  // Input less than 15mb
  if (size > 15000000) {
    return 0;
  }

  FuzzedDataProvider fdp(data, size);
  FuzzLibRaw lib_raw;

  for(int i =0; i < 4; i++)
    lib_raw.output_params_ptr()->aber[i] = fdp.ConsumeFloatingPoint<double>();
  for(int i =0; i < 4; i++)
    lib_raw.output_params_ptr()->user_mul[i] = fdp.ConsumeFloatingPoint<float>();
  for(int i =0; i < 2; i++)
    lib_raw.output_params_ptr()->gamm[i] = fdp.ConsumeFloatingPoint<double>();
  lib_raw.output_params_ptr()->bright = fdp.ConsumeFloatingPoint<float>();
  lib_raw.output_params_ptr()->threshold = fdp.ConsumeFloatingPoint<float>();
  lib_raw.output_params_ptr()->use_auto_wb = fdp.ConsumeIntegral<int>();
  lib_raw.output_params_ptr()->output_color = fdp.ConsumeIntegralInRange<int>(0, 6);
  lib_raw.output_params_ptr()->user_flip = fdp.ConsumeIntegralInRange<int>(0, 7);
  lib_raw.output_params_ptr()->user_black = fdp.ConsumeIntegral<int>();
  lib_raw.output_params_ptr()->user_sat = fdp.ConsumeIntegral<int>();
  lib_raw.output_params_ptr()->auto_bright_thr = fdp.ConsumeFloatingPoint<float>();
  lib_raw.output_params_ptr()->adjust_maximum_thr = fdp.ConsumeFloatingPointInRange<float>(0.f, 1.f);
  lib_raw.output_params_ptr()->fbdd_noiserd = fdp.ConsumeIntegralInRange<int>(0, 5);
  
  // The location of shot_select is unclear and causing compilation errors.
  // Removing it to ensure the fuzzer compiles and calls the target function.
  // The function will use the default value of shot_select.

  std::vector<char> payload = fdp.ConsumeRemainingBytes<char>();
  int result = lib_raw.open_buffer(payload.data(), payload.size());
  if (result != LIBRAW_SUCCESS) {
    return 0;
  }

  // Call the function under test.
  lib_raw.selectCRXTrack();

  result = lib_raw.unpack();
  if (result != LIBRAW_SUCCESS) {
    return 0;
  }

  result = lib_raw.unpack_thumb();
  if (result != LIBRAW_SUCCESS) {
    return 0;
  }

  result = lib_raw.raw2image();
  if (result != LIBRAW_SUCCESS) {
    lib_raw.free_image();
    return 0;
  }
  lib_raw.free_image();

  for (size_t i = 0; i < sizeof(options)/sizeof(*options); i++) {
    lib_raw.output_params_ptr()->user_qual = static_cast<int>(options[i]);

    result = lib_raw.dcraw_process();
    if (result != LIBRAW_SUCCESS) {
      // continue
    }
  }

  return 0;
}
