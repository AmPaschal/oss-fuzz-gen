
	
<description>
The `mbedtls_ssl_write_early_data` function is used by a TLS 1.3 client to send application data to the server before the handshake is complete. This is known as 0-RTT (zero round trip time) data. The function will only send data if early data is enabled on both the client and the server, and if the server has indicated that it is willing to accept early data.
</description>
<requirements>

<requirement>
The `ssl` context must be initialized and configured for early data. This means the endpoint must be a client, TLS 1.3 must be enabled, the transport must not be datagram-based, and early data must be enabled in the configuration.
</requirement>
<requirement>
The `ssl->tls_version` must be `MBEDTLS_SSL_VERSION_TLS1_3`. The function will return an error if the version is not TLS 1.3.
</requirement>
<requirement>
The `ssl->early_data_state` must be either `MBEDTLS_SSL_EARLY_DATA_STATE_CAN_WRITE` or `MBEDTLS_SSL_EARLY_DATA_STATE_ACCEPTED`. The function will not send any data if the state is not one of these values.
</requirement>
<requirement>
The `len` parameter must be less than or equal to the remaining early data size. The maximum early data size is determined by the server and communicated to the client in the session ticket.
</requirement>
<requirement>
The `buf` parameter must be a valid pointer to a buffer of size `len`. A null or invalid pointer will lead to a crash.
</requirement>

</requirements>

int fuzz_recv(void *ctx, unsigned char *buf, size_t len) {
  fuzzBufferOffset_t *fuzz_data = (fuzzBufferOffset_t *)ctx;
  if (fuzz_data->Offset + len > fuzz_data->Size) {
    len = fuzz_data->Size - fuzz_data->Offset;
  }
  if (len > 0) {
    memcpy(buf, fuzz_data->Data + fuzz_data->Offset, len);
    fuzz_data->Offset += len;
  }
  return (int)len;
}

int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) {
#if defined(MBEDTLS_SSL_CLI_C) && defined(MBEDTLS_ENTROPY_C) && defined(MBEDTLS_CTR_DRBG_C)
  int ret;
  size_t len;
  mbedtls_ssl_context ssl;
  mbedtls_ssl_config conf;
  mbedtls_ctr_drbg_context ctr_drbg;
  mbedtls_entropy_context entropy;
  unsigned char buf[4096];
  fuzzBufferOffset_t biomemfuzz;

  if (Size < 1) {
    return 0;
  }
  const uint8_t options = Data[Size - 1];
  const uint8_t *input_data = Data;
  size_t input_size = Size - 1;

  mbedtls_ssl_init(&ssl);
  mbedtls_ssl_config_init(&conf);
  mbedtls_ctr_drbg_init(&ctr_drbg);
  mbedtls_entropy_init(&entropy);
#if defined(MBEDTLS_USE_PSA_CRYPTO)
  psa_status_t status = psa_crypto_init();
  if (status != PSA_SUCCESS) {
    goto exit;
  }
#endif /* MBEDTLS_USE_PSA_CRYPTO */

  if (mbedtls_ctr_drbg_seed(&ctr_drbg, dummy_entropy, &entropy, (const unsigned char *)pers, strlen(pers)) != 0) {
    return 1;
  }

  if (initialized == 0) {
#if defined(MBEDTLS_X509_CRT_PARSE_C)
    mbedtls_x509_crt_init(&cacert);
    if (mbedtls_x509_crt_parse(&cacert, (const unsigned char *)mbedtls_test_cas_pem, mbedtls_test_cas_pem_len) != 0) {
      return 1;
    }
#endif
#if defined(MBEDTLS_SSL_SESSION_TICKETS)
    mbedtls_ssl_session_init(&saved_session);
#endif

    dummy_init();

    initialized = 1;
  }

  if (mbedtls_ssl_config_defaults(&conf, MBEDTLS_SSL_IS_CLIENT, MBEDTLS_SSL_TRANSPORT_STREAM, MBEDTLS_SSL_PRESET_DEFAULT) != 0) {
    goto exit;
  }

  mbedtls_ssl_conf_rng(&conf, dummy_random, &ctr_drbg);

#if defined(MBEDTLS_X509_CRT_PARSE_C)
  mbedtls_ssl_conf_authmode(&conf, (options & 1) ? MBEDTLS_SSL_VERIFY_REQUIRED : MBEDTLS_SSL_VERIFY_NONE);
  mbedtls_ssl_conf_ca_chain(&conf, &cacert, NULL);
#endif

#if defined(MBEDTLS_KEY_EXCHANGE_SOME_PSK_ENABLED)
  if (options & 2) {
    mbedtls_ssl_conf_psk(&conf, psk, sizeof(psk), (const unsigned char *)psk_id, sizeof(psk_id) - 1);
  }
#endif

#if defined(MBEDTLS_SSL_EARLY_DATA)
  if (options & 8) {
    mbedtls_ssl_conf_early_data(&conf, MBEDTLS_SSL_EARLY_DATA_ENABLED);
  }
#endif

  if (mbedtls_ssl_setup(&ssl, &conf) != 0) {
    goto exit;
  }

#if defined(MBEDTLS_X509_CRT_PARSE_C)
  if (mbedtls_ssl_set_hostname(&ssl, "localhost") != 0) {
    goto exit;
  }
#endif

#if defined(MBEDTLS_SSL_SESSION_TICKETS)
  if (options & 4) {
    mbedtls_ssl_set_session(&ssl, &saved_session);
  }
#endif

#if defined(MBEDTLS_SSL_EARLY_DATA)
  if ((options & 8) && (options & 4)) {
    size_t early_data_len = input_size > EARLY_DATA_MAX_LEN ? EARLY_DATA_MAX_LEN : input_size;
    mbedtls_ssl_write_early_data(&ssl, input_data, early_data_len);

    input_data += early_data_len;
    input_size -= early_data_len;
  }
#endif

  biomemfuzz.Data = input_data;
  biomemfuzz.Size = input_size;
  biomemfuzz.Offset = 0;
  mbedtls_ssl_set_bio(&ssl, &biomemfuzz, dummy_send, fuzz_recv, NULL);

  ret = mbedtls_ssl_handshake(&ssl);

  if (ret == 0) {
    do {
      len = sizeof(buf) - 1;
      ret = mbedtls_ssl_read(&ssl, buf, len);
      if (ret == MBEDTLS_ERR_SSL_WANT_READ)
        continue;
      else if (ret <= 0) {
        break;
      }
    } while (1);
#if defined(MBEDTLS_SSL_SESSION_TICKETS)
    if (options & 4) {
      if (mbedtls_ssl_get_session(&ssl, &saved_session) != 0) {
        mbedtls_ssl_session_free(&saved_session);
        mbedtls_ssl_session_init(&saved_session);
      }
    }
#endif
  }

exit:
  mbedtls_ssl_free(&ssl);
  mbedtls_ssl_config_free(&conf);
  mbedtls_ctr_drbg_free(&ctr_drbg);
  mbedtls_entropy_free(&entropy);
#if defined(MBEDTLS_USE_PSA_CRYPTO)
  mbedtls_psa_crypto_free();
#endif
#else
  (void)Data;
  (void)Size;
#endif /* MBEDTLS_SSL_CLI_C && MBEDTLS_ENTROPY_C && MBEDTLS_CTR_DRBG_C */

  return 0;
}
