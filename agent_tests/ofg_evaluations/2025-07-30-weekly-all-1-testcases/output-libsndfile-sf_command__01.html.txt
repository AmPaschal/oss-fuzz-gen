extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size < 10) {
    return 0;
  }

  SF_INFO sndfile_info;
  VIO_DATA vio_data;
  SF_VIRTUAL_IO vio;
  SNDFILE *sndfile = NULL;

  int err = sf_init_file(data, size, &sndfile, &vio_data, &vio, &sndfile_info);
  if (err || sndfile == NULL) {
    if (sndfile)
      sf_close(sndfile);
    return 0;
  }

  FuzzedDataProvider provider(data, size);
  int command = provider.PickValueInArray(kCommands);

  char buffer[1024];
  double dbl_val;
  SF_BROADCAST_INFO binfo;
  SF_DITHER_INFO dinfo;
  SF_INSTRUMENT inst;
  SF_CUES cue_info;
  sf_count_t count;

  switch (command) {
  case SFC_GET_LIB_VERSION:
  case SFC_GET_LOG_INFO:
    sf_command(sndfile, command, buffer, provider.ConsumeIntegralInRange<int>(0, sizeof(buffer)));
    break;
  case SFC_SET_VBR_ENCODING_QUALITY:
  case SFC_SET_COMPRESSION_LEVEL:
  case SFC_CALC_SIGNAL_MAX:
  case SFC_CALC_NORM_SIGNAL_MAX:
  case SFC_GET_SIGNAL_MAX:
    dbl_val = provider.ConsumeFloatingPoint<double>();
    sf_command(sndfile, command, &dbl_val, sizeof(dbl_val));
    break;
  case SFC_SET_BROADCAST_INFO:
    provider.ConsumeData(&binfo, sizeof(binfo));
    sf_command(sndfile, command, &binfo, sizeof(binfo));
    break;
  case SFC_SET_DITHER_ON_WRITE:
  case SFC_SET_DITHER_ON_READ:
    provider.ConsumeData(&dinfo, sizeof(dinfo));
    sf_command(sndfile, command, &dinfo, sizeof(dinfo));
    break;
  case SFC_SET_INSTRUMENT:
    provider.ConsumeData(&inst, sizeof(inst));
    sf_command(sndfile, command, &inst, sizeof(inst));
    break;
  case SFC_SET_CUE:
    provider.ConsumeData(&cue_info, sizeof(cue_info));
    sf_command(sndfile, command, &cue_info, sizeof(cue_info));
    break;
  case SFC_FILE_TRUNCATE:
  case SFC_SET_RAW_START_OFFSET:
    count = provider.ConsumeIntegral<sf_count_t>();
    sf_command(sndfile, command, &count, sizeof(count));
    break;
  default:
    sf_command(sndfile, command, NULL, provider.ConsumeIntegral<int>());
    break;
  }

  if (sndfile != NULL)
    sf_close(sndfile);

  return 0;
}
