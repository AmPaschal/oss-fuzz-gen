
	
<description>
The `JS_ParseJSON` function parses a string containing JSON text and returns a JavaScript value. It is a wrapper around the `JS_ParseJSON2` function, which performs the actual parsing. The caller `js_json_parse` obtains the string and its length from the arguments passed to it, and then invokes `JS_ParseJSON`. The filename argument is hardcoded to `<input>` in the caller.
</description>
<requirements>
<requirement>
The `JSContext *ctx` argument must be a valid context created by `JS_NewContext` or `JS_NewContextRaw`. This is because the function needs a valid context to create and manipulate JavaScript values.
</requirement>
<requirement>
The `buf` argument must be a valid pointer to a buffer of at least `buf_len` bytes. This is because the function reads `buf_len` bytes from this buffer to parse the JSON data.
</requirement>
<requirement>
The `filename` argument can be a null-terminated string. This argument is used for error reporting and can be set to a constant string like `<input>`, as is done in the caller `js_json_parse`.
</requirement>
</requirements>

#include "/src/quickjs/quickjs.h"
#include "quickjs-libc.h"
#include "fuzz/fuzz_common.h"

#include <stdint.h>
#include <stdlib.h>
#include <string.h>

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size == 0) {
        return 0;
    }

    // Although JS_ParseJSON takes a size, the underlying implementation might
    // rely on null-termination in some paths. Creating a null-terminated
    // copy is safer, following the pattern of other fuzzers.
    char *json_string = (char *)malloc(size + 1);
    if (!json_string) {
        return 0;
    }
    memcpy(json_string, data, size);
    json_string[size] = '\0';

    JSRuntime *rt = JS_NewRuntime();
    if (!rt) {
        free(json_string);
        return 0;
    }
    JSContext *ctx = JS_NewContext(rt);
    if (!ctx) {
        free(json_string);
        JS_FreeRuntime(rt);
        return 0;
    }

    // Initialize the context with standard libraries, interrupt handlers etc.
    // This is crucial for the context to be able to handle JSON parsing.
    test_one_input_init(rt, ctx);

    reset_nbinterrupts();

    // Call the function-under-test
    JSValue val = JS_ParseJSON(ctx, json_string, size, "<fuzz>");

    free(json_string);

    // If parsing is successful, free the resulting JSValue.
    // If it's an exception, the value is not a valid object to be freed.
    if (!JS_IsException(val)) {
        JS_FreeValue(ctx, val);
    }

    // Clean up QuickJS resources
    js_std_free_handlers(rt);
    JS_FreeContext(ctx);
    JS_FreeRuntime(rt);

    return 0;
}
