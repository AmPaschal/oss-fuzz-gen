
	
<description>
The function `xmlTextReaderRelaxNGValidateCtxt` validates an XML document contained in an `xmlTextReader` against a RelaxNG schema. It is a wrapper around the internal function `xmlTextReaderRelaxNGValidateInternal`. The function returns 0 if the document is valid, a positive error code if it is invalid, and -1 in case of an internal error.
</description>
<requirements>
<requirement>
The `reader` argument must be a valid `xmlTextReaderPtr` created from a valid XML file. This can be done by calling `xmlNewTextReaderFilename()` with a path to the XML file.
</requirement>
<requirement>
The `ctxt` argument must be a valid `xmlRelaxNGValidCtxtPtr`. This should be created by parsing a RelaxNG schema using `xmlRelaxNGNewParserCtxt`, `xmlRelaxNGParse`, and `xmlRelaxNGNewValidCtxt`.
</requirement>
<requirement>
The `reader` argument must not be NULL. The internal function `xmlTextReaderRelaxNGValidateInternal` checks for a NULL reader and returns -1.
</requirement>
<requirement>
The `ctxt` argument must not be NULL. The internal function `xmlTextReaderRelaxNGValidateInternal` checks for a NULL `ctxt` (when `rng` is also NULL) and returns -1.
</requirement>
<requirement>
The `reader` must be associated with a document. The function `xmlTextReaderCurrentDoc(reader)` is called, and if it returns NULL, the validation fails.
</requirement>
<requirement>
The `options` argument can be set to 0. This is based on its usage in the project's test files.
</requirement>
</requirements>

#include <fuzzer/FuzzedDataProvider.h>
#include <cstddef>

extern "C" {
#include <libxml/parser.h>
#include <libxml/relaxng.h>
#include <libxml/xmlreader.h>
#include <libxml/xmlerror.h>
}

// Suppress generic error messages.
static void ignore_error(void *ctx, const char *msg, ...) {
    // No-op.
}

// Suppress structured error messages.
static void ignore_structured_error(void *ctx, const xmlError *err) {
    // No-op.
}

// One-time initialization and cleanup.
class XmlFuzzerInitializer {
public:
    XmlFuzzerInitializer() {
        xmlInitParser();
        // Redirect error messages to avoid spamming the fuzzer output.
        xmlSetGenericErrorFunc(nullptr, ignore_error);
        xmlSetStructuredErrorFunc(nullptr, ignore_structured_error);
    }
    ~XmlFuzzerInitializer() {
        xmlCleanupParser();
    }
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    // Static initializer to ensure xmlInitParser is called only once.
    static XmlFuzzerInitializer initializer;

    FuzzedDataProvider provider(data, size);

    // Consume the options parameter first.
    int options = provider.ConsumeIntegral<int>();

    // Consume data for the schema and the XML document from the rest of the input.
    size_t schema_size = provider.ConsumeIntegralInRange<size_t>(0, provider.remaining_bytes() / 2);
    std::vector<uint8_t> schema_data = provider.ConsumeBytes<uint8_t>(schema_size);
    std::vector<uint8_t> xml_data = provider.ConsumeRemainingBytes<uint8_t>();

    if (schema_data.empty() || xml_data.empty()) {
        return 0;
    }

    // 1. Create a RelaxNG parser context from the schema data.
    xmlRelaxNGParserCtxtPtr pctxt = xmlRelaxNGNewMemParserCtxt(
        reinterpret_cast<const char *>(schema_data.data()), schema_data.size());
    if (pctxt == nullptr) {
        return 0;
    }

    // 2. Parse the schema.
    xmlRelaxNGPtr schema = xmlRelaxNGParse(pctxt);
    xmlRelaxNGFreeParserCtxt(pctxt);
    if (schema == nullptr) {
        return 0;
    }

    // 3. Create a RelaxNG validation context from the schema.
    xmlRelaxNGValidCtxtPtr vctxt = xmlRelaxNGNewValidCtxt(schema);
    if (vctxt == nullptr) {
        xmlRelaxNGFree(schema);
        return 0;
    }

    // 4. Create an XML text reader from the document data.
    xmlTextReaderPtr reader = xmlReaderForMemory(
        reinterpret_cast<const char *>(xml_data.data()), xml_data.size(),
        nullptr, nullptr, 0);
    if (reader == nullptr) {
        xmlRelaxNGFreeValidCtxt(vctxt);
        xmlRelaxNGFree(schema);
        return 0;
    }

    // 5. Call the function under test while iterating through the document.
    while (xmlTextReaderRead(reader) == 1) {
        xmlTextReaderRelaxNGValidateCtxt(reader, vctxt, options);
    }
    // Final call to validate after reading is complete.
    xmlTextReaderRelaxNGValidateCtxt(reader, vctxt, options);

    // 6. Clean up resources.
    xmlFreeTextReader(reader);
    xmlRelaxNGFreeValidCtxt(vctxt);
    xmlRelaxNGFree(schema);

    return 0;
}
