  std::vector<char> xml_data = provider.ConsumeRemainingBytes<char>();

  // 1. Create schema from the first part of the input.

    return 0;
  }

  xmlSchemaPtr schema = xmlSchemaParse(parser_ctxt);
  xmlSchemaFreeParserCtxt(parser_ctxt);

  if (schema == nullptr) {
    xmlCleanupParser();
    return 0;
  }

  // 2. Create a validation context from the schema.
  xmlSchemaValidCtxtPtr valid_ctxt = xmlSchemaNewValidCtxt(schema);
  if (valid_ctxt == nullptr) {
    xmlSchemaFree(schema);
    xmlCleanupParser();
    return 0;
  }

  // 3. Create a reader for the second part of the input.
  xmlTextReaderPtr reader = xmlReaderForMemory(
      (const char*)xml_data.data(), xml_data.size(), "fuzz.xml", nullptr, 0);
  if (reader == nullptr) {
    xmlSchemaFreeValidCtxt(valid_ctxt);
    xmlSchemaFree(schema);
    xmlCleanupParser();
    return 0;
  }

  // 4. Fuzz the target function.
  int options = 0; // Using default options.

  xmlTextReaderSchemaValidateCtxt(reader, valid_ctxt, options);

  // 5. Cleanup.

  xmlSchemaFreeValidCtxt(valid_ctxt);
  xmlSchemaFree(schema);

  xmlCleanupParser();

  return 0;
