
	
<description>
The function `sf_set_string` sets a string in a sound file. The first argument is a pointer to a `SNDFILE` object, which must have been opened in write or read/write mode. The second argument is an integer that specifies the type of string being set, and it must be one of the `SF_STR_*` values. The third argument is the string to be set. The function returns 0 on success and a non-zero error code on failure.
</description>
<requirements>

<requirement>
The `SNDFILE` pointer, `sndfile`, must be initialized by calling `sf_open` with the mode set to `SFM_WRITE` or `SFM_RDWR`. This ensures that the file is opened in a mode that allows writing.
</requirement>
<requirement>
The `str_type` argument must be one of the following values: `SF_STR_TITLE`, `SF_STR_COPYRIGHT`, `SF_STR_SOFTWARE`, `SF_STR_ARTIST`, `SF_STR_COMMENT`, `SF_STR_DATE`, `SF_STR_ALBUM`, `SF_STR_LICENSE`, `SF_STR_TRACKNUMBER`, or `SF_STR_GENRE`. Any other value will cause the function to return an error.
</requirement>
<requirement>
The `str` argument must be a non-NULL, null-terminated string. A NULL string will cause a crash.
</requirement>
<requirement>
If the `str_type` argument is not `SF_STR_SOFTWARE`, the `str` argument must not be an empty string. An empty string will cause the function to return an error in this case.
</requirement>

</requirements>

#include <fuzzer/FuzzedDataProvider.h>
#include <unistd.h>

extern "C" {
#include "/src/libsndfile/include/sndfile.h"

extern "C" int sf_set_string(SNDFILE *, int, const char *);

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    // To test sf_set_string, we need to open a file for writing.
    // The simplest way is to use a temporary file.
    char filename[] = "/tmp/sf_fuzz.XXXXXX";
    int fd = mkstemp(filename);
    if (fd < 0) {
        return 0;
    }
    close(fd);

    SF_INFO sfinfo;
    memset(&sfinfo, 0, sizeof(sfinfo));

    // Populate SF_INFO with fuzzed data for opening in write mode.
    sfinfo.samplerate = provider.ConsumeIntegralInRange<int>(8000, 96000);
    sfinfo.channels = provider.ConsumeIntegralInRange<int>(1, 8);

    // Pick a valid format combination to increase chances of sf_open succeeding.
    int major_formats[] = {
        SF_FORMAT_WAV,  SF_FORMAT_AIFF,    SF_FORMAT_AU,      SF_FORMAT_RAW,
        SF_FORMAT_PAF,  SF_FORMAT_SVX,     SF_FORMAT_NIST,    SF_FORMAT_VOC,
        SF_FORMAT_IRCAM,SF_FORMAT_W64,     SF_FORMAT_MAT4,    SF_FORMAT_MAT5,
        SF_FORMAT_PVF,  SF_FORMAT_XI,      SF_FORMAT_HTK,     SF_FORMAT_SDS,
        SF_FORMAT_AVR,  SF_FORMAT_WAVEX,   SF_FORMAT_SD2,     SF_FORMAT_FLAC,
        SF_FORMAT_CAF,  SF_FORMAT_WVE,     SF_FORMAT_OGG,     SF_FORMAT_MPC2K,
        SF_FORMAT_RF64};
    int sub_formats[] = {
        SF_FORMAT_PCM_S8,   SF_FORMAT_PCM_16,    SF_FORMAT_PCM_24,
        SF_FORMAT_PCM_32,   SF_FORMAT_PCM_U8,    SF_FORMAT_FLOAT,
        SF_FORMAT_DOUBLE,   SF_FORMAT_ULAW,      SF_FORMAT_ALAW,
        SF_FORMAT_IMA_ADPCM,SF_FORMAT_MS_ADPCM,  SF_FORMAT_GSM610,
        SF_FORMAT_VOX_ADPCM,SF_FORMAT_G721_32,   SF_FORMAT_G723_24,
        SF_FORMAT_G723_40,  SF_FORMAT_DWVW_12,   SF_FORMAT_DWVW_16,
        SF_FORMAT_DWVW_24,  SF_FORMAT_DWVW_N,    SF_FORMAT_DPCM_8,
        SF_FORMAT_DPCM_16,  SF_FORMAT_VORBIS};
    
    sfinfo.format = provider.PickValueInArray(major_formats) |
                    provider.PickValueInArray(sub_formats);

    SNDFILE *sndfile = sf_open(filename, SFM_WRITE, &sfinfo);
    if (sndfile == NULL) {
        unlink(filename);
        return 0;
    }

    // Fuzz the parameters for sf_set_string.
    const int str_type = provider.ConsumeIntegralInRange<int>(SF_STR_TITLE, SF_STR_GENRE);
    const std::string str = provider.ConsumeRemainingBytesAsString();

    sf_set_string(sndfile, str_type, str.c_str());

    sf_close(sndfile);
    unlink(filename);

    return 0;
}
