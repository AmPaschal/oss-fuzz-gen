
	
<description>
The function `GEOSMakeValidWithParams_r` takes a geometry and a set of parameters to make the geometry valid. The `params` argument controls the validation method. There are two methods: `GEOS_MAKE_VALID_LINEWORK` and `GEOS_MAKE_VALID_STRUCTURE`. The `keepCollapsed` parameter is used only with the `GEOS_MAKE_VALID_STRUCTURE` method. The function is called by `GEOSMakeValidWithParams` and `GEOSMakeValid_r`. The latter provides default parameters.
</description>
<requirements>

<requirement>
The first argument, `extHandle`, must be a valid `GEOSContextHandle_t` object. This is because it is used internally by the `execute` function and to report errors.
</requirement>
<requirement>
The second argument, `g`, must not be a null pointer. This is because it is dereferenced to get the SRID without a null check.
</requirement>
<requirement>
The `method` field of the `GEOSMakeValidParams` struct, pointed to by the third argument `params`, should be either `GEOS_MAKE_VALID_LINEWORK` (0) or `GEOS_MAKE_VALID_STRUCTURE` (1). Any other value will result in an error and a null return.
</requirement>
<requirement>
If the `method` field of the `GEOSMakeValidParams` struct is `GEOS_MAKE_VALID_STRUCTURE`, the `keepCollapsed` field is used. It is converted to a boolean, so it should be fuzzed with 0 and 1 for different behaviors.
</requirement>

</requirements>

static GEOSContextHandle_t handle = NULL;

void
log_notice(const char *fmt, ...)
{
    va_list ap;
    va_start(ap, fmt);
    vfprintf(stderr, fmt, ap);
    va_end(ap);
    fprintf(stderr, "\n");
}

void
log_error(const char *fmt, ...)
{
    va_list ap;
    va_start(ap, fmt);
    vfprintf(stderr, fmt, ap);
    va_end(ap);
    fprintf(stderr, "\n");
}

int
LLVMFuzzerInitialize(int *argc, char ***argv)
{
    handle = GEOS_init_r();
    GEOSContext_setNoticeHandler_r(handle, log_notice);
    GEOSContext_setErrorHandler_r(handle, log_error);
    return 0;
}

int
LLVMFuzzerTestOneInput(const uint8_t* data, size_t size)
{
    // We need at least 2 bytes for params and 1 for the geometry
    if (size < 3) {
        return 0;
    }

    // The rest of the data is used for the geometry
    const uint8_t* wkb_data = data + 2;
    size_t wkb_size = size - 2;
    GEOSGeometry* g1 = GEOSGeomFromWKB_buf_r(handle, wkb_data, wkb_size);
    if ( ! g1 ) {
        return 0;
    }

    // Create and setup params for the function-under-test
    GEOSMakeValidParams* params = GEOSMakeValidParams_create_r(handle);
    if ( ! params ) {
        GEOSGeom_destroy_r(handle, g1);
        return 0;
    }

    // Use first byte to select method: GEOS_MAKE_VALID_LINEWORK (0) or GEOS_MAKE_VALID_STRUCTURE (1)
    enum GEOSMakeValidMethods method = (data[0] % 2);
    GEOSMakeValidParams_setMethod_r(handle, params, method);

    // Use second byte to select keepCollapsed: 0 or 1
    int keepCollapsed = data[1] % 2;
    GEOSMakeValidParams_setKeepCollapsed_r(handle, params, keepCollapsed);

    // Call the function-under-test
    GEOSGeometry* g_valid = GEOSMakeValidWithParams_r(handle, g1, params);

    if (g_valid) {
        GEOSGeom_destroy_r(handle, g_valid);
    }

    GEOSMakeValidParams_destroy_r(handle, params);
    GEOSGeom_destroy_r(handle, g1);

    return 0;
}
