#include <cstdint>

#include "spirv-tools/libspirv.hpp"
#include "test/fuzzers/random_generator.h"

extern "C" int LLVMFuzzerTestOneInput(const uint8_t* data, size_t size) {
  if (size < 4 || size % 4 != 0) {
    // Input must be non-empty and a multiple of 4 bytes.
    return 0;
  }

  // Use the input data to seed the random generator, which will determine the
  // target environment.
  spvtools::fuzzers::RandomGenerator random_gen(data, size);
  spv_target_env target_env = random_gen.GetTargetEnv();

  spv_optimizer_t* optimizer = spvOptimizerCreate(target_env);
  if (!optimizer) {
    return 0;
  }

  spv_optimizer_options options = spvOptimizerOptionsCreate();
  if (!options) {
    spvOptimizerDestroy(optimizer);
    return 0;
  }

  std::vector<uint32_t> input_binary(size / 4);
  memcpy(input_binary.data(), data, size);

  spv_binary optimized_binary = nullptr;
  spvOptimizerRun(optimizer, input_binary.data(), input_binary.size(),
                  &optimized_binary, options);

  if (optimized_binary) {
    spvBinaryDestroy(optimized_binary);
  }

  spvOptimizerOptionsDestroy(options);
  spvOptimizerDestroy(optimizer);

  return 0;
}
