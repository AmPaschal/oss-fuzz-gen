
	
<description>
The function `modbus_write_and_read_registers` writes a block of contiguous registers and reads another block of contiguous registers in a single atomic transaction. It takes a pointer to a `modbus_t` context, the write address and number of registers to write, a source buffer for the write data, the read address and number of registers to read, and a destination buffer for the read data. The function first validates the input parameters, then builds a request, sends it to the server, receives the response, and finally parses the response to extract the read register values.
</description>
<requirements>

<requirement>
The first argument, `ctx`, must be a valid pointer to a `modbus_t` structure, initialized by `modbus_new_tcp` or `modbus_new_rtu` and connected using `modbus_connect`. This is because the function dereferences the `ctx` pointer to access backend-specific functions and data, and an invalid pointer will cause a crash.
</requirement>
<requirement>
The `write_nb` argument, representing the number of registers to write, must be less than or equal to `MODBUS_MAX_WR_WRITE_REGISTERS` (121). The function checks this condition and returns an error if it's not met, preventing a buffer overflow when constructing the request packet.
</requirement>
<requirement>
The `read_nb` argument, representing the number of registers to read, must be less than or equal to `MODBUS_MAX_WR_READ_REGISTERS` (125). The function checks this condition and returns an error if it's not met, preventing a buffer overflow when constructing the request packet.
</requirement>
<requirement>
The `src` argument must point to a buffer of `uint16_t` with a size of at least `write_nb` elements. The function reads up to `write_nb` elements from this buffer, and an undersized buffer will lead to a heap buffer overflow.
</requirement>
<requirement>
The `dest` argument must point to a buffer of `uint16_t` with a size of at least `read_nb` elements. The function writes the read register values into this buffer, and an undersized buffer will lead to a heap buffer overflow.
</requirement>

</requirements>

#define PORT 8080
#define kMinInputLength 9
#define kMaxInputLength MODBUS_RTU_MAX_ADU_LENGTH

struct Fuzzer {
  uint16_t port;
  char *file;

  FILE *inFile;
  uint64_t size;
  const uint8_t *buffer;

  pthread_t thread;
  int socket;
};
typedef struct Fuzzer Fuzzer;

int client(Fuzzer *fuzzer);

void fuzzinit(Fuzzer *fuzzer) {
  struct sockaddr_in server_addr;
  fuzzer->socket = socket(AF_INET, SOCK_STREAM, 0);

  server_addr.sin_family = AF_INET;
  server_addr.sin_port = htons(fuzzer->port);
  server_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

  setsockopt(fuzzer->socket, SOL_SOCKET, SO_REUSEADDR, &(int){1}, sizeof(int));

  bind(fuzzer->socket, (struct sockaddr *)&server_addr, sizeof(server_addr));
  listen(fuzzer->socket, 1);
}

void *Server(void *args) {

  Fuzzer *fuzzer = (Fuzzer *)args;
  {
    int client;
    char clientData[10240];
    struct sockaddr_in clientAddr;
    uint32_t clientSZ = sizeof(clientAddr);

    client = accept(fuzzer->socket, (struct sockaddr *)&clientAddr, &clientSZ);

    send(client, fuzzer->buffer, fuzzer->size, 0);
    recv(client, clientData, sizeof(clientData), 0);

    shutdown(client, SHUT_RDWR);
    close(client);
  }
  pthread_exit(NULL);
}

void clean(Fuzzer *fuzzer) {
  { // Server
    shutdown(fuzzer->socket, SHUT_RDWR);
    close(fuzzer->socket);
  }
  free(fuzzer);
}

extern int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

  if (size < kMinInputLength || size > kMaxInputLength) {
    return 0;
  }

  Fuzzer *fuzzer = (Fuzzer *)malloc(sizeof(Fuzzer));
  fuzzer->port = PORT;
  fuzzer->size = size;
  fuzzer->buffer = data;

  fuzzinit(fuzzer);

  pthread_create(&fuzzer->thread, NULL, Server, fuzzer);
  client(fuzzer);
  pthread_join(fuzzer->thread, NULL); /* To Avoid UAF*/

  clean(fuzzer);
  return 0;
}

int client(Fuzzer *fuzzer) {

  uint16_t *tab_wr_registers = NULL;
  uint16_t *tab_rp_registers = NULL;
  modbus_t *ctx = NULL;
  int rc;
  const int write_nb = UT_REGISTERS_NB;
  const int read_nb = UT_REGISTERS_NB;

  ctx = modbus_new_tcp("127.0.0.1", fuzzer->port);

  if (ctx == NULL) {
    fprintf(stderr, "Unable to allocate libmodbus context\n");
    return -1;
  }

  if (modbus_connect(ctx) == -1) {
    fprintf(stderr, "Connection failed: %s\n", modbus_strerror(errno));
    modbus_free(ctx);
    return -1;
  }

  /* This is the code block before the function being tested */
  /* Allocate and initialize the memory to store the registers */
  tab_wr_registers = (uint16_t *)malloc(write_nb * sizeof(uint16_t));
  memset(tab_wr_registers, 0, write_nb * sizeof(uint16_t));
  tab_rp_registers = (uint16_t *)malloc(read_nb * sizeof(uint16_t));
  memset(tab_rp_registers, 0, read_nb * sizeof(uint16_t));

  /* Write and Read registers */
  rc = modbus_write_and_read_registers(ctx, UT_REGISTERS_ADDRESS, write_nb,
                                       tab_wr_registers, UT_REGISTERS_ADDRESS,
                                       read_nb, tab_rp_registers);

  /* Free the memory */
  free(tab_wr_registers);
  free(tab_rp_registers);

  /* Close the connection */
  modbus_close(ctx);
  modbus_free(ctx);

  return rc;
}
