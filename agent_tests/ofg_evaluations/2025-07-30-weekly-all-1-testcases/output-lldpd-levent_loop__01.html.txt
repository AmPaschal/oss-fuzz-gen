
	
<description>
The `levent_loop` function is the main event loop of the lldpd daemon. It initializes the event library, enters a loop to process events, and cleans up resources on exit. The function takes a pointer to a `struct lldpd` as an argument, which holds the global configuration and state of the daemon.
</description>
<requirements>
<requirement>
The `cfg` structure must be allocated and initialized. The `lldpd_main` function shows a complete initialization sequence, which should be followed.
</requirement>
<requirement>
The `g_base` member of the `cfg` structure must be initialized by calling `levent_init(cfg)` before calling `levent_loop`. This is because `levent_loop` uses the event base to check for loop exit conditions.
</requirement>
<requirement>
The `g_chassis` tail queue within `cfg` must contain at least one `lldpd_chassis` structure, representing the local chassis. In `lldpd_main`, this is allocated with `calloc` and inserted into the queue.
</requirement>
<requirement>
The `g_hardware` tail queue should be initialized, although it can be empty. The `lldpd_loop` function, called by `levent_loop`, will populate this list by calling `interfaces_update`.
</requirement>
<requirement>
The `g_protocols` member of `cfg` should point to the global `protos` array. This is done in `lldpd_main` before calling `levent_loop`.
</requirement>
<requirement>
The `g_config` member of `cfg` must be initialized with valid parameters. Key fields to initialize are `c_receiveonly`, `c_tx_interval`, `c_tx_hold`, and `c_ttl` as these control the behavior of the event loop.
</requirement>
</requirements>

#define _GNU_SOURCE
#include "../src/daemon/lldpd.h"
#include <stdlib.h>
#include <string.h>
#include <event2/event.h>

/* Use this callback to avoid some logs */
void donothing(int pri, const char *msg) {};

/*
 * We are overriding levent_loop() to avoid an infinite loop during fuzzing.
 * Because levent_loop() is in the same object file (event.o) as many other
 * functions, we must provide our own implementations for all functions from
 * that file that are called by the rest of the library. The build script
 * removes the original object file from the static library to prevent linker errors.
 *
 * The signatures of these stubs MUST exactly match the declarations in lldpd.h.
 */

// Custom implementation of levent_init() for the fuzzer.
void levent_init(struct lldpd *cfg) {
    // We only need the event base for the fuzzer.
    cfg->g_base = event_base_new();
}

// Custom implementation of levent_shutdown() for the fuzzer.
void levent_shutdown(struct lldpd *cfg) {
    if (cfg->g_base) {
        event_base_free(cfg->g_base);
        cfg->g_base = NULL;
    }
}

// Stubs for other functions from event.o needed by the library.
void levent_ctl_close_clients(void) {}
void levent_hardware_init(struct lldpd_hardware *h) {}
void levent_hardware_release(struct lldpd_hardware *h) {}
void levent_schedule_cleanup(struct lldpd *cfg) {}
void levent_ctl_notify(char *iface, int state, struct lldpd_port *port) {}
void levent_schedule_pdu(struct lldpd_hardware *h) {}
int levent_iface_subscribe(struct lldpd *cfg, int s) { return 0; }
void levent_recv_error(int s, const char* err) {}
void levent_hardware_add_fd(struct lldpd_hardware *h, int fd) {}


// Custom, non-blocking implementation of levent_loop() for the fuzzer.
void levent_loop(struct lldpd *cfg) {
	levent_init(cfg); // Calls our custom init
	if (!cfg->g_base) {
		return;
	}
	lldpd_loop(cfg);

	// Run the event loop for a few non-blocking iterations.
	for (int i = 0; i < 5; i++) {
		event_base_loop(cfg->g_base, EVLOOP_ONCE | EVLOOP_NONBLOCK);
	}

	if (cfg->g_iface_timer_event != NULL) {
		event_free(cfg->g_iface_timer_event);
		cfg->g_iface_timer_event = NULL;
	}
	levent_ctl_close_clients();
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
	if (size < 1) {
		return 0;
	}

	log_register(donothing);

	struct lldpd *cfg = (struct lldpd *)calloc(1, sizeof(struct lldpd));
	if (!cfg) {
		return 0;
	}

	/* Initialize parts of the configuration */
	cfg->g_config.c_tx_hold = LLDPD_TX_HOLD;
	cfg->g_config.c_tx_interval = LLDPD_TX_INTERVAL * 1000;
	cfg->g_config.c_paused = 0;
	cfg->g_config.c_smart = data[0] % 15;

	TAILQ_INIT(&cfg->g_hardware);
	TAILQ_INIT(&cfg->g_chassis);

	struct lldpd_port *port = (struct lldpd_port *)calloc(1, sizeof(struct lldpd_port));
	if (!port) {
		free(cfg);
		return 0;
	}
	cfg->g_default_local_port = port;

	struct lldpd_chassis *lchassis = (struct lldpd_chassis *)calloc(1, sizeof(struct lldpd_chassis));
	if (!lchassis) {
		free(port);
		free(cfg);
		return 0;
	}
	lchassis->c_name = strdup("fuzz_chassis");
	lchassis->c_id_subtype = LLDP_CHASSISID_SUBTYPE_LOCAL;
	lchassis->c_id = strdup("fuzz_id");
	if (lchassis->c_id) {
		lchassis->c_id_len = strlen(lchassis->c_id);
	} else {
		lchassis->c_id_len = 0;
	}

	TAILQ_INSERT_TAIL(&cfg->g_chassis, lchassis, c_entries);
	lchassis->c_refcount++;

	// Call our custom levent_loop.
	levent_loop(cfg);

	// Call our custom shutdown function to clean up.
	levent_shutdown(cfg);

	lldpd_port_cleanup(cfg->g_default_local_port, 1);
	free(cfg->g_default_local_port);

	lchassis->c_refcount--;
	lldpd_chassis_cleanup(lchassis, 1);

	free(cfg);

	return 0;
}
