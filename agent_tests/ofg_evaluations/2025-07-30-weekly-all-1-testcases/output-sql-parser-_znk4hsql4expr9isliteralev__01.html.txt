// Helper function to test a single expression
void testExpression(const Expr* expr) {
  if (expr) {
    // Call the function under test
    expr->isLiteral();

    // Recursively test sub-expressions
    testExpression(expr->expr);
    testExpression(expr->expr2);
    if (expr->exprList) {
      for (Expr* sub_expr : *expr->exprList) {
        testExpression(sub_expr);
      }
    }
  }
}

// Helper function to test a list of expressions
void testExpressionList(const std::vector<Expr*>* exprList) {
  if (exprList) {
    for (const Expr* expr : *exprList) {
      testExpression(expr);
    }
  }
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  std::string input(reinterpret_cast<const char *>(data), size);
  SQLParserResult result;
  SQLParser::parse(input, &result);

  if (result.isValid()) {
    for (size_t i = 0; i < result.size(); ++i) {
        const SQLStatement* statement = result.getStatement(i);
        if (statement) {
            switch (statement->type()) {
                case kStmtSelect: {
                const auto* select_stmt = static_cast<const SelectStatement*>(statement);
                testExpressionList(select_stmt->selectList);
                testExpression(select_stmt->whereClause);
                if (select_stmt->groupBy) {
                    testExpression(select_stmt->groupBy->having);
                }
                break;
                }
                case kStmtInsert: {
                const auto* insert_stmt = static_cast<const InsertStatement*>(statement);
                if (insert_stmt->values) {
                    testExpressionList(insert_stmt->values);
                }
                break;
                }
                case kStmtUpdate: {
                const auto* update_stmt = static_cast<const UpdateStatement*>(statement);
                if (update_stmt->updates) {
                    for (const UpdateClause* update : *update_stmt->updates) {
                    testExpression(update->value);
                    }
                }
                testExpression(update_stmt->where);
                break;
                }
                case kStmtDelete: {
                const auto* delete_stmt = static_cast<const DeleteStatement*>(statement);
                testExpression(delete_stmt->expr);
                break;
                }
                default:
                break;
            }
        }
    }
  }

  return 0;
}
