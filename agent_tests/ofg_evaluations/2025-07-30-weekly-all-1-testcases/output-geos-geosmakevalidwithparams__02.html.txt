
	
<description>
The GEOSMakeValidWithParams function takes a geometry object and a parameters object to make the geometry valid. The function uses one of two methods to make the geometry valid, `GEOS_MAKE_VALID_LINEWORK` or `GEOS_MAKE_VALID_STRUCTURE`, which is specified in the `method` field of the `GEOSMakeValidParams` object. The `keepCollapsed` field in the `GEOSMakeValidParams` object is a boolean flag that determines whether to keep components that have collapsed into a lower dimensionality. The function returns a new valid geometry object, or NULL on error.
</description>

</requirement>
<requirement>

</requirement>
<requirement>
The `keepCollapsed` field of the GEOSMakeValidParams object should be set to 0 or 1. This field is treated as a boolean value by the function.
</requirement>
<requirement>

</requirement>

FILE * flogOut;

void

    fprintf( flogOut, "NOTICE: ");

}

    va_list ap;
    fprintf( flogOut, "ERROR: ");
    va_start (ap, fmt);
    vfprintf( flogOut, fmt, ap);
    va_end(ap);
    fprintf( flogOut, "\n" );
}

int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) {
    if (initialized == 0) {
        flogOut = fopen("/dev/null", "wb");
        initGEOS(notice, log_and_exit);
        initialized = 1;
    }

    if (Size < 3) {
        return 0;
    }

    // Use first two bytes for parameters, the rest for the geometry
    const enum GEOSMakeValidMethods method = (Data[0] % 2 == 0) ? GEOS_MAKE_VALID_LINEWORK : GEOS_MAKE_VALID_STRUCTURE;
    const int keepCollapsed = Data[1];
    const unsigned char *wkb_data = Data + 2;
    const size_t wkb_size = Size - 2;

    GEOSGeometry *g1 = GEOSGeomFromWKB_buf(wkb_data, wkb_size);
    if (g1 == NULL) {
        return 0;
    }

    GEOSMakeValidParams *params = GEOSMakeValidParams_create();
    if (params == NULL) {
        GEOSGeom_destroy(g1);
        return 0;
    }

    GEOSMakeValidParams_setMethod(params, method);
    GEOSMakeValidParams_setKeepCollapsed(params, keepCollapsed);

    GEOSGeometry *g2 = GEOSMakeValidWithParams(g1, params);

    // Clean up
    if (g2 != NULL) {
        GEOSGeom_destroy(g2);
    }
    GEOSMakeValidParams_destroy(params);
    GEOSGeom_destroy(g1);

    return 0;
