
	
<description>
The nxt_main_process_start function is responsible for initializing the main process in the NGINX Unit application. It sets the process type to NXT_PROCESS_MAIN, creates a port for the main process, sets the process title, and finally starts the discovery process. This function is called by nxt_runtime_initial_start, which is the initial start function for the runtime.
</description>
<requirements>
<requirement>
The `task` argument must be a valid pointer to a `nxt_task_t` struct, and its `thread` member must be a valid pointer to a `nxt_thread_t` struct. This is because the `thr` and `rt` arguments are derived from `task` in the caller function `nxt_runtime_initial_start`.
</requirement>
<requirement>
The `rt` argument, which is `task->thread->runtime`, must be a valid pointer to an initialized `nxt_runtime_t` struct. The `nxt_runtime_create` function should be called to create and initialize the runtime, as it allocates memory for the runtime and sets up its initial state.
</requirement>
<requirement>
The `thr` argument must be a valid pointer to a `nxt_thread_t` struct and should be equal to `task->thread`. This is because the caller `nxt_runtime_initial_start` sets `thr` to `task->thread`.
</requirement>
<requirement>
The `rt` argument should be equal to `task->thread->runtime`. This is because the caller `nxt_runtime_initial_start` sets `rt` to `thr->runtime` which is `task->thread->runtime`.
</requirement>
</requirements>

#include <nxt_main.h>

extern int LLVMFuzzerInitialize(int *argc, char ***argv);

extern char **environ;

int
LLVMFuzzerInitialize(int *argc, char ***argv)
{
    if (nxt_lib_start("fuzzing", NULL, &environ) != NXT_OK) {
        return 1;
    }

    return 0;
}

int
LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    nxt_task_t     *task;
    nxt_thread_t   *thr;
    nxt_runtime_t  *rt;
    nxt_mp_t       *mp;

    task = &nxt_main_task;
    thr = task->thread;

    if (thr->runtime != NULL) {
        /*
         * A previous run might have been interrupted, leaving thr->runtime in an
         * inconsistent state. It is safer to stop here.
         */
        return 0;
    }

    if (nxt_runtime_create(task) != NXT_OK) {
        /*
         * nxt_runtime_create() is expected to clean up after itself on
         * failure.
         */
        thr->runtime = NULL;
        return 0;
    }

    rt = thr->runtime;
    if (rt == NULL) {
        return 0;
    }

    /*
     * Keep a pointer to the memory pool, because rt itself will be freed
     * when the pool is destroyed.
     */
    mp = rt->mem_pool;

    nxt_main_process_start(thr, task, rt);

    /*
     * The intended shutdown sequence via nxt_runtime_quit() is asynchronous
     * and not suitable for a synchronous fuzzer. We manually destroy the
     * memory pool created by nxt_runtime_create() to clean up resources
     * and allow the fuzzer to loop.
     */
    nxt_mp_destroy(mp);

    /*
     * Nullify the runtime pointer to prevent use-after-free on the next
     * iteration.
     */
    thr->runtime = NULL;

    return 0;
}
