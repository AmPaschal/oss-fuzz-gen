
	
<description>
The `lodepng::encode` function encodes a raw image buffer into a PNG image and saves it to a file. It is a wrapper around the core encoding function `lodepng_encode_memory`, which in turn calls `lodepng_encode`. The `lodepng_encode` function performs several checks on the input parameters, particularly on the color type and bit depth, before proceeding with the encoding process.
</description>
<requirements>

<requirement>
The `colortype` and `bitdepth` parameters must be a valid combination. The `checkColorValidity` function defines the valid combinations: `LCT_GREY` with `bitdepth` 1, 2, 4, 8, or 16; `LCT_RGB` with `bitdepth` 8 or 16; `LCT_PALETTE` with `bitdepth` 1, 2, 4, or 8; `LCT_GREY_ALPHA` with `bitdepth` 8 or 16; and `LCT_RGBA` with `bitdepth` 8 or 16.
</requirement>
<requirement>
The `in` buffer must have a size that is compatible with the image dimensions (`w` and `h`), `colortype`, and `bitdepth`. The required size can be calculated using the `lodepng_get_raw_size_lct` function.
</requirement>
<requirement>
The `filename` parameter should be a valid, null-terminated string representing a writable file path. This is to ensure that the encoded PNG data can be successfully saved to a file.
</requirement>

</requirements>

#include "lodepng.h"
#include <fuzzer/FuzzedDataProvider.h>
#include <cstdint>
#include <string>
#include <vector>

// Amount of valid colortype/bidthdepth combinations in the PNG file format.
const size_t num_combinations = 15;

LodePNGColorType colortypes[num_combinations] = {
    LCT_GREY,       LCT_GREY,       LCT_GREY,    LCT_GREY,    LCT_GREY,
    LCT_RGB,        LCT_RGB,
    LCT_PALETTE,    LCT_PALETTE,    LCT_PALETTE, LCT_PALETTE,
    LCT_GREY_ALPHA, LCT_GREY_ALPHA,
    LCT_RGBA,       LCT_RGBA,
};

unsigned bitdepths[num_combinations] = {
    1, 2,  4, 8, 16,
    8, 16,
    1, 2,  4, 8,
    8, 16,
    8, 16,
};

// Re-implementation of internal lodepng functions to calculate raw image size,
// avoiding a linker error due to the original functions being static.
namespace {
unsigned get_channels(LodePNGColorType colortype) {
  switch(colortype) {
    case LCT_GREY: return 1;
    case LCT_RGB: return 3;
    case LCT_PALETTE: return 1;
    case LCT_GREY_ALPHA: return 2;
    case LCT_RGBA: return 4;
    default: return 0;
  }
}

unsigned get_bpp(LodePNGColorType colortype, unsigned bitdepth) {
  return get_channels(colortype) * bitdepth;
}

size_t get_raw_image_size(unsigned w, unsigned h, LodePNGColorType colortype, unsigned bitdepth) {
  size_t bitsize = (size_t)get_bpp(colortype, bitdepth) * w * h;
  return bitsize / 8;
}
} // namespace

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  // Select a valid color type and bit depth combination.
  size_t combination_index = provider.ConsumeIntegralInRange<size_t>(0, num_combinations - 1);
  LodePNGColorType color_type = colortypes[combination_index];
  unsigned bit_depth = bitdepths[combination_index];

  // Generate image dimensions. Keep them small to avoid large allocations.
  unsigned w = provider.ConsumeIntegralInRange<unsigned>(1, 256);
  unsigned h = provider.ConsumeIntegralInRange<unsigned>(1, 256);

  // Calculate the required size for the raw image data.
  size_t raw_image_size = get_raw_image_size(w, h, color_type, bit_depth);
  if (raw_image_size == 0 || raw_image_size > provider.remaining_bytes()) {
    return 0;
  }

  // Consume the raw image data.
  std::vector<uint8_t> image_data = provider.ConsumeBytes<uint8_t>(raw_image_size);
  if (image_data.size() != raw_image_size) {
    return 0;
  }

  // Use /dev/null to avoid writing to disk.
  const std::string filename = "/dev/null";

  // Call the function under test.
  lodepng::encode(filename, image_data.data(), w, h, color_type, bit_depth);

  return 0;
}
