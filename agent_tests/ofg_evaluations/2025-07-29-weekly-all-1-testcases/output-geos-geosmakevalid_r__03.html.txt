
	
<description>
The GEOSMakeValid_r function takes a GEOS context handle and a Geometry object, and returns a valid representation of the geometry. It does this by calling the GEOSMakeValidWithParams_r function with a default set of parameters, which in turn uses the `MakeValid` or `GeometryFixer` classes to perform the operation.
</description>

</requirement>
<requirement>
The `g` argument must be a valid `Geometry` object. This can be created by calling the `GEOSGeomFromWKB_buf_r` function with a byte buffer.
</requirement>
<requirement>
The `GEOS_init_r` function must be called before calling `GEOSMakeValid_r`. This is to ensure that the context handle is properly initialized.
</requirement>
<requirement>
The geometry object `g` should be fuzzed. The purpose of the function is to validate and repair invalid geometries, so it is important to test it with a wide range of malformed inputs.
</requirement>
</requirements>

#include <stdarg.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#include "geos_c.h"

// Note: GEOS >= 3.12 defines Geometry as GEOSGeometry
// This ensures compatibility with the 'Geometry' type used in the function signature.
#ifndef GEOS_GEOMETRY_DEFINED
#define GEOS_GEOMETRY_DEFINED
typedef struct GEOSGeom_t GEOSGeometry;
typedef GEOSGeometry Geometry;
#endif


static GEOSContextHandle_t handle = NULL;

// Suppress GEOS messages by providing empty handler functions
void
log_notice_handler(const char *fmt, ...)
{
}

void
log_error_handler(const char *fmt, ...)
{
}

int
LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    // One-time initialization of the GEOS context
    if (handle == NULL) {
        handle = GEOS_init_r();
        // Set handlers to avoid printing to stdout/stderr, which slows down fuzzing
        GEOSContext_setNoticeHandler_r(handle, log_notice_handler);
        GEOSContext_setErrorHandler_r(handle, log_error_handler);
    }

    // GEOSGeomFromWKB_buf_r requires at least 1 byte of input
    if (size == 0) {
        return 0;
    }

    // Create an input geometry from the raw fuzzer data (as WKB)
    Geometry *g_in = GEOSGeomFromWKB_buf_r(handle, data, size);

    // If the input was valid enough to create a geometry
    if (g_in != NULL) {
        // Call the function under test
        Geometry *g_out = GEOSMakeValid_r(handle, g_in);

        // Clean up the geometry returned by the function
        if (g_out != NULL) {
            GEOSGeom_destroy_r(handle, g_out);
        }

        // Clean up the input geometry
        GEOSGeom_destroy_r(handle, g_in);
    }

    return 0;
}
