// The build script renames haproxy's main() to main2(). We must declare it
// as an external function to be able to call it.
extern int main2(int argc, char **argv);

// The prompt requires a function `init` to be called. Since the target

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size == 0) {
    return 0;
  }

  // Create a mutable, null-terminated copy of the input data to build argv.
  char *buffer = (char *)malloc(size + 1);
  if (!buffer) {
    return 0;
  }
  memcpy(buffer, data, size);
  buffer[size] = '\0';

  const int MAX_ARGS = 64; // Set a reasonable limit for arguments.
  char *argv[MAX_ARGS + 2]; // +2 for program name and NULL terminator.
  int argc = 0;

  // argv[0] must be the program name.
  argv[argc++] = "haproxy";

  // Split the input buffer into arguments based on null bytes. This gives the
  // fuzzer control over argument separation.
  char *current_arg = buffer;
  const char *end_of_buffer = buffer + size;
  while (current_arg < end_of_buffer && argc < MAX_ARGS) {
    argv[argc++] = current_arg;
    // Find the next null byte to determine the end of the current argument.
    while (current_arg < end_of_buffer && *current_arg != '\0') {
      current_arg++;
    }
    // Move past the null byte for the next argument.
    current_arg++;
  }

  // The argv array must be null-terminated.
  argv[argc] = NULL;

  init(argc, argv);

  free(buffer);
  return 0;
}
