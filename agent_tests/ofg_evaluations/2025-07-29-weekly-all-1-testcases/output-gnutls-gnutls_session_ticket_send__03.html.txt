
	
<description>
The `gnutls_session_ticket_send` function is used by a TLS 1.3 server to send one or more session tickets to a client after the main handshake has completed. This allows the client to use these tickets to resume future sessions, which is a performance optimization. The function can be called after the initial handshake or after a post-handshake authentication exchange. It requires a fully established TLS 1.3 session and is only valid on the server side.
</description>
<requirements>
<requirement>
The `session` object must be initialized as a server session using `gnutls_init` with the `GNUTLS_SERVER` flag. This is because the function is only intended for server-side use and explicitly checks for it.
</requirement>
<requirement>
The session must be configured for the TLS 1.3 protocol. This is because session tickets are handled differently in TLS 1.3 and the function contains checks to enforce this.
</requirement>
<requirement>
A session ticket key must be generated and enabled for the session using `gnutls_session_ticket_key_generate` and `gnutls_session_ticket_enable_server`. This is a prerequisite for the server to be able to create and send any session tickets.
</requirement>
<requirement>
A complete TLS handshake must be performed using `gnutls_handshake` before calling this function. This function sends post-handshake messages and assumes a secure connection is already established.
</requirement>
<requirement>
The `nr` argument, representing the number of tickets to send, must be greater than zero. The function will return an `GNUTLS_E_INVALID_REQUEST` error if `nr` is zero.
</requirement>
<requirement>
The `flags` argument must be set to zero. This is specified by the function's documentation and is a reserved field for future use.
</requirement>
</requirements>

int __attribute__((visibility("protected")))
gnutls_rnd(gnutls_rnd_level_t level, void *data, size_t len)
{
	memset(data, 0xff, len);

	/* Flip the first byte to avoid infinite loop in the RSA
	 * blinding code of Nettle */
	if (len > 0)
		memset(data, 0x0, 1);
	return 0;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
	int res;
	gnutls_datum_t rsa_cert, rsa_key;
	gnutls_datum_t ecdsa_cert, ecdsa_key;
	gnutls_datum_t ed25519_cert, ed25519_key;
	gnutls_datum_t ed448_cert, ed448_key;
	gnutls_session_t session;
	gnutls_certificate_credentials_t xcred;
	gnutls_datum_t skey;
	struct mem_st memdata;
	unsigned int retry;
	unsigned int nr_tickets;

	if (size < 1) {
		return 0;
	}

	/* Use the first byte for the number of tickets, and the rest for the handshake */
	nr_tickets = (data[0] % 4) + 1; /* A number from 1 to 4 */
	memdata.data = data + 1;
	memdata.size = size - 1;

	res = gnutls_init(&session, GNUTLS_SERVER);
	assert(res >= 0);

	res = gnutls_certificate_allocate_credentials(&xcred);
	assert(res >= 0);

	rsa_cert.data = (unsigned char *)kRSACertificateDER;
	rsa_cert.size = sizeof(kRSACertificateDER);
	rsa_key.data = (unsigned char *)kRSAPrivateKeyDER;
	rsa_key.size = sizeof(kRSAPrivateKeyDER);

	ecdsa_cert.data = (unsigned char *)kECDSACertificateDER;
	ecdsa_cert.size = sizeof(kECDSACertificateDER);
	ecdsa_key.data = (unsigned char *)kECDSAPrivateKeyDER;
	ecdsa_key.size = sizeof(kECDSAPrivateKeyDER);

	ed25519_cert.data = (unsigned char *)kEd25519CertificateDER;
	ed25519_cert.size = sizeof(kEd25519CertificateDER);
	ed25519_key.data = (unsigned char *)kEd25519PrivateKeyDER;
	ed25519_key.size = sizeof(kEd25519PrivateKeyDER);

	ed448_cert.data = (unsigned char *)kEd448CertificateDER;
	ed448_cert.size = sizeof(kEd448CertificateDER);
	ed448_key.data = (unsigned char *)kEd448PrivateKeyDER;
	ed448_key.size = sizeof(kEd448PrivateKeyDER);

	res = gnutls_certificate_set_x509_key_mem(xcred, &rsa_cert, &rsa_key,
						  GNUTLS_X509_FMT_DER);
	assert(res >= 0);

	res = gnutls_certificate_set_x509_key_mem(
		xcred, &ecdsa_cert, &ecdsa_key, GNUTLS_X509_FMT_DER);
	assert(res >= 0);

	res = gnutls_certificate_set_x509_key_mem(
		xcred, &ed25519_cert, &ed25519_key, GNUTLS_X509_FMT_DER);
	assert(res >= 0);

	res = gnutls_certificate_set_x509_key_mem(
		xcred, &ed448_cert, &ed448_key, GNUTLS_X509_FMT_DER);
	assert(res >= 0);

	gnutls_certificate_set_known_dh_params(xcred, GNUTLS_SEC_PARAM_MEDIUM);

	res = gnutls_credentials_set(session, GNUTLS_CRD_CERTIFICATE, xcred);
	assert(res >= 0);

	/* Enable session tickets */
	res = gnutls_session_ticket_key_generate(&skey);
	assert(res >= 0);
	res = gnutls_session_ticket_enable_server(session, &skey);
	assert(res >= 0);

	res = gnutls_priority_set_direct(
		session, "NORMAL:-VERS-TLS-ALL:+VERS-TLS1.3", NULL);
	assert(res >= 0);

	gnutls_transport_set_push_function(session, error_push);
	gnutls_transport_set_pull_function(session, error_pull);
	gnutls_handshake_set_read_function(session, handshake_discard);

	retry = 0;
	do {
		res = gnutls_handshake(session);
		if (res == GNUTLS_E_AGAIN) {
			if (handshake_pull(session, &memdata) < 0) {
				res = GNUTLS_E_INTERNAL_ERROR;
				break;
			}
			if (retry > HANDSHAKE_MAX_RETRY_COUNT) {
				break;
			}
			retry++;
		} else {
			retry = 0;
		}
	} while (res < 0 && gnutls_error_is_fatal(res) == 0);

	if (res >= 0) {
		/* Handshake successful, now call the target function */
		do {
			res = gnutls_session_ticket_send(session, nr_tickets, 0);
		} while (res == GNUTLS_E_AGAIN || res == GNUTLS_E_INTERRUPTED);
	}

	gnutls_deinit(session);
	gnutls_certificate_free_credentials(xcred);
	gnutls_free(skey.data);
	return 0;
}
