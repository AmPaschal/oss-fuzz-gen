
	
<description>
The `sf_command` function is a versatile function that allows getting and setting various parameters of a sound file. It takes a command and a data pointer with its size, and performs the action specified by the command. The function is essentially a large switch statement that handles a wide variety of commands, each with its own specific requirements for the data and datasize parameters. Many commands can be called with a NULL `sndfile` pointer. For the commands that require a valid `sndfile`, it must be obtained by calling `sf_open`, `sf_open_fd` or `sf_open_virtual`.

The `command` argument should be one of the `SFC_` values defined in the `sndfile.h` header file. The `data` and `datasize` arguments are highly dependent on the `command` and must be set correctly to avoid errors. For example, for the `SFC_GET_SIMPLE_FORMAT_COUNT` command, `data` must be a pointer to an integer and `datasize` must be the size of an integer. For other commands, `data` might be a pointer to a struct or can be NULL.

There are also commands that have side effects on the `SNDFILE` object, such as setting normalization, scaling, or adding metadata chunks. Some commands require the file to be opened in a specific mode (e.g., `SFM_WRITE` for `SFC_SET_ADD_PEAK_CHUNK`).
</description>

#include <fuzzer/FuzzedDataProvider.h>
#include <sndfile.h>
#include <stdlib.h>
#include <string.h>
#include <vector>
#include <inttypes.h>

#include "sndfile_fuzz_header.h"

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  SNDFILE *sndfile = NULL;
  SF_INFO sndfile_info;
  memset(&sndfile_info, 0, sizeof(sndfile_info));

  // Some commands work with a NULL sndfile.
  if (provider.ConsumeBool()) {
    // Create a virtual file from the fuzzer input.
    VIO_DATA vio_data;
    SF_VIRTUAL_IO vio;
    // Use the rest of the data for the virtual file.
    std::vector<uint8_t> file_data = provider.ConsumeRemainingBytes<uint8_t>();
    if (sf_init_file(file_data.data(), file_data.size(), &sndfile, &vio_data, &vio, &sndfile_info) != 0) {
      // Initialization failed, proceed with a NULL handle.
      sndfile = NULL;
    }
  }

  const int command = provider.PickValueInArray({
      SFC_GET_LIB_VERSION,
      SFC_GET_LOG_INFO,
      SFC_GET_CURRENT_SF_INFO,
      SFC_GET_NORM_DOUBLE,
      SFC_GET_NORM_FLOAT,
      SFC_SET_NORM_DOUBLE,
      SFC_SET_NORM_FLOAT,
      SFC_SET_SCALE_FLOAT_INT_READ,
      SFC_SET_SCALE_INT_FLOAT_WRITE,
      SFC_GET_SIMPLE_FORMAT_COUNT,
      SFC_GET_SIMPLE_FORMAT,
      SFC_GET_FORMAT_INFO,
      SFC_GET_FORMAT_MAJOR_COUNT,
      SFC_GET_FORMAT_MAJOR,
      SFC_GET_FORMAT_SUBTYPE_COUNT,
      SFC_GET_FORMAT_SUBTYPE,
      SFC_CALC_SIGNAL_MAX,
      SFC_CALC_NORM_SIGNAL_MAX,
      SFC_CALC_MAX_ALL_CHANNELS,
      SFC_CALC_NORM_MAX_ALL_CHANNELS,
      SFC_GET_SIGNAL_MAX,
      SFC_GET_MAX_ALL_CHANNELS,
      SFC_SET_ADD_PEAK_CHUNK,
      SFC_UPDATE_HEADER_NOW,
      SFC_SET_UPDATE_HEADER_AUTO,
      SFC_FILE_TRUNCATE,
      SFC_SET_RAW_START_OFFSET,
      SFC_SET_DITHER_ON_WRITE,
      SFC_SET_DITHER_ON_READ,
      SFC_GET_DITHER_INFO_COUNT,
      SFC_GET_DITHER_INFO,
      SFC_GET_EMBED_FILE_INFO,
      SFC_SET_CLIPPING,
      SFC_GET_CLIPPING,
      SFC_GET_CUE_COUNT,
      SFC_GET_CUE,
      SFC_SET_CUE,
      SFC_GET_INSTRUMENT,
      SFC_SET_INSTRUMENT,
      SFC_GET_LOOP_INFO,
      SFC_GET_BROADCAST_INFO,
      SFC_SET_BROADCAST_INFO,
      SFC_GET_CHANNEL_MAP_INFO,
      SFC_SET_CHANNEL_MAP_INFO,
      SFC_RAW_DATA_NEEDS_ENDSWAP,
      SFC_WAVEX_SET_AMBISONIC,
      SFC_WAVEX_GET_AMBISONIC,
      SFC_RF64_AUTO_DOWNGRADE,
      SFC_SET_VBR_ENCODING_QUALITY,
      SFC_SET_COMPRESSION_LEVEL,
      SFC_SET_CART_INFO,
      SFC_GET_CART_INFO,
      SFC_TEST_IEEE_FLOAT_REPLACE,
  });

  void *cmd_data = NULL;
  int datasize = 0;

  static union {
    char c[2048];
    int i[512];
    double d[256];
    sf_count_t count;
    SF_INFO sfinfo;
    SF_FORMAT_INFO sfformatinfo;
    SF_DITHER_INFO ditherinfo;
    SF_INSTRUMENT instrument;
    SF_BROADCAST_INFO broadcastinfo;
    SF_CART_INFO cartinfo;
    SF_LOOP_INFO loopinfo;
    SF_EMBED_FILE_INFO embedfileinfo;
  } static_data;

  switch (command) {
  case SFC_GET_LIB_VERSION:
  case SFC_GET_LOG_INFO:
    datasize = provider.ConsumeIntegralInRange<int>(0, sizeof(static_data.c));
    cmd_data = static_data.c;
    break;

  case SFC_GET_SIMPLE_FORMAT_COUNT:
  case SFC_GET_FORMAT_MAJOR_COUNT:
  case SFC_GET_FORMAT_SUBTYPE_COUNT:
  case SFC_GET_DITHER_INFO_COUNT:
  case SFC_GET_CUE_COUNT:
    cmd_data = &static_data.i[0];
    datasize = sizeof(int);
    break;

  case SFC_GET_SIMPLE_FORMAT:
  case SFC_GET_FORMAT_MAJOR:
  case SFC_GET_FORMAT_SUBTYPE:
  case SFC_GET_FORMAT_INFO:
    cmd_data = &static_data.sfformatinfo;
    datasize = sizeof(SF_FORMAT_INFO);
    provider.ConsumeData(cmd_data, datasize);
    break;

  case SFC_GET_CURRENT_SF_INFO:
    cmd_data = &static_data.sfinfo;
    datasize = sizeof(SF_INFO);
    break;

  case SFC_CALC_SIGNAL_MAX:
  case SFC_CALC_NORM_SIGNAL_MAX:
  case SFC_GET_SIGNAL_MAX:
  case SFC_SET_VBR_ENCODING_QUALITY:
  case SFC_SET_COMPRESSION_LEVEL:
    cmd_data = &static_data.d[0];
    datasize = sizeof(double);
    provider.ConsumeData(cmd_data, datasize);
    break;

  case SFC_CALC_MAX_ALL_CHANNELS:
  case SFC_CALC_NORM_MAX_ALL_CHANNELS:
  case SFC_GET_MAX_ALL_CHANNELS:
  case SFC_GET_CHANNEL_MAP_INFO:
  case SFC_SET_CHANNEL_MAP_INFO:
    if (sndfile != NULL && sndfile_info.channels > 0 && sndfile_info.channels < 256) {
      datasize = sndfile_info.channels * sizeof(int);
      if (command != SFC_SET_CHANNEL_MAP_INFO && command != SFC_GET_CHANNEL_MAP_INFO)
        datasize = sndfile_info.channels * sizeof(double);
      cmd_data = static_data.d;
      provider.ConsumeData(cmd_data, datasize);
    }
    break;

  case SFC_FILE_TRUNCATE:
  case SFC_SET_RAW_START_OFFSET:
    cmd_data = &static_data.count;
    datasize = sizeof(sf_count_t);
    provider.ConsumeData(cmd_data, datasize);
    break;

  case SFC_SET_DITHER_ON_WRITE:
  case SFC_SET_DITHER_ON_READ:
    cmd_data = &static_data.ditherinfo;
    datasize = sizeof(SF_DITHER_INFO);
    provider.ConsumeData(cmd_data, datasize);
    break;

  case SFC_SET_INSTRUMENT:
  case SFC_GET_INSTRUMENT:
    cmd_data = &static_data.instrument;
    datasize = sizeof(SF_INSTRUMENT);
    provider.ConsumeData(cmd_data, datasize);
    break;

  case SFC_SET_BROADCAST_INFO:
  case SFC_GET_BROADCAST_INFO:
    cmd_data = &static_data.broadcastinfo;
    datasize = sizeof(SF_BROADCAST_INFO);
    provider.ConsumeData(cmd_data, datasize);
    break;

  case SFC_SET_CART_INFO:
  case SFC_GET_CART_INFO:
    cmd_data = &static_data.cartinfo;
    datasize = sizeof(SF_CART_INFO);
    provider.ConsumeData(cmd_data, datasize);
    break;

  case SFC_GET_LOOP_INFO:
    cmd_data = &static_data.loopinfo;
    datasize = sizeof(SF_LOOP_INFO);
    break;

  case SFC_GET_EMBED_FILE_INFO:
    cmd_data = &static_data.embedfileinfo;
    datasize = sizeof(SF_EMBED_FILE_INFO);
    break;

  default:
    datasize = provider.ConsumeBool();
    break;
  }

  sf_command(sndfile, command, cmd_data, datasize);

  if (sndfile != NULL) {
    sf_close(sndfile);
  }

  return 0;
}
