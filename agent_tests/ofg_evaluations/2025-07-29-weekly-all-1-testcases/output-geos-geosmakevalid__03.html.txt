
	
<description>
The GEOSMakeValid function takes a Geometry object and attempts to create a valid representation of it. It is a wrapper around the GEOSMakeValid_r function, which in turn calls GEOSMakeValidWithParams_r. The default method used is GEOS_MAKE_VALID_LINEWORK. The function does not support curved geometries and will return a null pointer if one is provided.
</description>
<requirements>
<requirement>
The input Geometry object should be created using functions such as GEOSGeomFromWKT or GEOSGeomFromHEX_buf. This is because these functions are used in the test cases to create valid Geometry objects.</requirement>
<requirement>
The input Geometry should not be a curved geometry. The test suite explicitly checks for this and expects a null return, indicating that curved geometries are not supported.</requirement>
</requirements>

#include <stdarg.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

#include "/src/geos/build/capi/geos_c.h"

static int initialized = 0;
FILE *flogOut;

void notice(const char *fmt, ...) {
  va_list ap;
  fprintf(flogOut, "NOTICE: ");
  va_start(ap, fmt);
  vfprintf(flogOut, fmt, ap);
  va_end(ap);
  fprintf(flogOut, "\n");
}

void log_and_exit(const char *fmt, ...) {
  va_list ap;
  fprintf(flogOut, "ERROR: ");
  va_start(ap, fmt);
  vfprintf(flogOut, fmt, ap);
  va_end(ap);
  fprintf(flogOut, "\n");
}

int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) {
  if (initialized == 0) {
    flogOut = fopen("/dev/null", "wb");
    initGEOS(notice, log_and_exit);
    initialized = 1;
  }

  GEOSGeometry *geom = NULL;

  // GEOSGeomFromWKT requires a null-terminated string.
  // Create a null-terminated copy of the input data.
  char *wkt = (char *)malloc(Size + 1);
  if (!wkt) {
    return 0;
  }
  memcpy(wkt, Data, Size);
  wkt[Size] = '\0';

  geom = GEOSGeomFromWKT(wkt);
  free(wkt);

  // If WKT parsing fails, try to parse it as WKB
  if (geom == NULL) {
      geom = GEOSGeomFromWKB_buf(Data, Size);
  }

  if (geom != NULL) {
    // If a geometry was successfully created, fuzz GEOSMakeValid
    GEOSGeometry *valid_geom = GEOSMakeValid(geom);

    if (valid_geom != NULL) {
      GEOSGeom_destroy(valid_geom);
    }
    GEOSGeom_destroy(geom);
  }

  return 0;
}
