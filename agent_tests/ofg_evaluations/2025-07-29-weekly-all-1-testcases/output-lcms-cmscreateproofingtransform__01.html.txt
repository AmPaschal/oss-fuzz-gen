
	
<description>
The function `cmsCreateProofingTransform` creates a proofing transform. This transform is a color transform that simulates the output of one device on another, while also providing gamut checking capabilities. The function takes as input three profiles (input, output, and proofing), their corresponding formats, the rendering intent, the proofing intent, and a set of flags. The function returns a handle to the transform, or NULL if the transform could not be created.
</description>

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

    const cmsUInt32Number *params = (const cmsUInt32Number *)data;
    cmsUInt32Number InputFormat    = params[0];
    cmsUInt32Number OutputFormat   = params[1];
    cmsUInt32Number nIntent        = params[2];
    cmsUInt32Number ProofingIntent = params[3];
    cmsUInt32Number dwFlags        = params[4];

    // Create valid profiles. Using standard profiles is more effective
    // than trying to parse random data from the fuzzer.
    cmsHPROFILE hInProfile = cmsCreate_sRGBProfile();
    if (hInProfile == NULL) {
        return 0;
    }

    cmsHPROFILE hOutProfile = cmsCreate_sRGBProfile();
    if (hOutProfile == NULL) {
        cmsCloseProfile(hInProfile);
        return 0;
    }

    cmsHPROFILE hProofProfile = cmsCreateLab4Profile(NULL);
    if (hProofProfile == NULL) {
        cmsCloseProfile(hInProfile);
        cmsCloseProfile(hOutProfile);
        return 0;
    }

    // Call the function under test.
    cmsHTRANSFORM hTransform = cmsCreateProofingTransform(
        hInProfile,
        InputFormat,
        hOutProfile,
        OutputFormat,
        hProofProfile,
        nIntent,
        ProofingIntent,
        dwFlags
    );

    // Clean up allocated resources.
    if (hTransform != NULL) {
        cmsDeleteTransform(hTransform);
    }

    cmsCloseProfile(hInProfile);
    cmsCloseProfile(hOutProfile);
