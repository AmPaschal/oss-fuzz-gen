// Dummy write function to consume the data generated by the parser
static int parser_write_stub(void *priv, uint8_t *data, int count) {
    // We don't need to do anything with the data, just acknowledge it was "written"
    // to allow the parser's internal buffer to be cleared.
    return count;
}

// Custom deleter for the usbredirparser to ensure proper cleanup
struct ParserDeleter {
  void operator()(struct usbredirparser *p) {
    if (p) {
      usbredirparser_destroy(p);
    }
  }
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    // Create and initialize the parser
    std::unique_ptr<struct usbredirparser, ParserDeleter> parser(usbredirparser_create());
    if (!parser) {
        return 0;
    }

    // Set a minimal set of callbacks. The write_func is essential for this target.
    parser->write_func = parser_write_stub;

    // Initialize the parser. We can use default/empty capabilities.
    uint32_t caps[USB_REDIR_CAPS_SIZE] = {0};
    usbredirparser_init(parser.get(), "fuzzer", caps, USB_REDIR_CAPS_SIZE, 0);

    // Consume data for the function call
    uint64_t id = provider.ConsumeIntegral<uint64_t>();

    // Prepare the header with fuzzed data
    struct usb_redir_buffered_bulk_packet_header header;
    header.stream_id = provider.ConsumeIntegral<uint32_t>();
    header.endpoint = provider.ConsumeIntegral<uint8_t>();
    header.status = provider.ConsumeIntegral<uint8_t>();

    // Prepare the data buffer using the remaining fuzzer bytes
    std::vector<uint8_t> packet_data = provider.ConsumeRemainingBytes<uint8_t>();
    int data_len = packet_data.size();
    
    // Ensure length consistency as required by the protocol
    header.length = data_len;

    // Call the function under test
    usbredirparser_send_buffered_bulk_packet(parser.get(), id, &header, packet_data.data(), data_len);

    // Process the write queue to actually exercise the packet serialization code path
    while (usbredirparser_has_data_to_write(parser.get())) {
        if (usbredirparser_do_write(parser.get()) < 0) {
            // An error occurred during write, stop processing
            break;
        }
    }
    
    // Parser is destroyed automatically by unique_ptr's deleter at scope exit
    return 0;
}
