It operates under the rules defined by a decContext, which specifies precision and rounding. The function handles special number cases (NaNs, infinities) and ensures that the inputs are valid before performing the power calculation. If the base is negative, the exponent must be an integer.
</description>

// Define a maximum precision for our fuzzing to avoid huge allocations.
// The `DECIMAL128` format uses 34 digits, which is a reasonable upper bound.
#define FUZZ_MAX_DIGITS 34

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    // We need at least 2 bytes for context settings.
    if (size < 2) {
        return 0;
    }

    // --- Initialize decContext ---
    decContext set;
    // Start with a default context.
    decContextDefault(&set, DEC_INIT_BASE);

    // Fuzz context settings from the first 2 bytes of data.
    // Constrain digits to a safe maximum to avoid excessive memory allocation.
    set.digits = 1 + (data[0] % FUZZ_MAX_DIGITS);
    // Constrain rounding mode to valid enum values.
    set.round = (enum rounding)(data[1] % DEC_ROUND_MAX);

    // Advance the data pointer past the context settings.
    data += 2;
    size -= 2;

    // --- Allocate decNumber structs with enough space for the coefficient ---
    // The number of 'units' needed depends on the precision (set.digits).
    int32_t units = (set.digits + DECDPUN - 1) / DECDPUN;
    // The decNumber struct already includes one unit in its definition, so we allocate for the rest.
    size_t dec_number_size = sizeof(decNumber) + (units > 1 ? (units - 1) * sizeof(decNumberUnit) : 0);

    decNumber *res = (decNumber *)malloc(dec_number_size);
    decNumber *lhs = (decNumber *)malloc(dec_number_size);
    decNumber *rhs = (decNumber *)malloc(dec_number_size);

    // Check for allocation failure.
    if (!res || !lhs || !rhs) {
        if (res) free(res);
        if (lhs) free(lhs);
        if (rhs) free(rhs);
        return 0;
    }

    // --- Create string inputs for decNumberFromString ---
    // Split the remaining data into two null-terminated strings for lhs and rhs.
    size_t lhs_str_len = size / 2;
    size_t rhs_str_len = size - lhs_str_len;

    char *lhs_str = (char *)malloc(lhs_str_len + 1);
    char *rhs_str = (char *)malloc(rhs_str_len + 1);

    if (!lhs_str || !rhs_str) {
        if (lhs_str) free(lhs_str);
        if (rhs_str) free(rhs_str);
        free(res);
        free(lhs);
        free(rhs);
        return 0;
    }

    memcpy(lhs_str, data, lhs_str_len);
    lhs_str[lhs_str_len] = '\0';
    memcpy(rhs_str, data + lhs_str_len, rhs_str_len);
    rhs_str[rhs_str_len] = '\0';

    // --- Initialize decNumbers using the strings ---
    // decNumberFromString is a robust way to create decNumber instances from arbitrary string data.
    decNumberFromString(lhs, lhs_str, &set);
    decNumberFromString(rhs, rhs_str, &set);

    // --- Call the target function ---
    // All pointers are non-NULL, the context is initialized, and the decNumbers are initialized.
    decNumberPower(res, lhs, rhs, &set);

    // --- Cleanup ---
    free(lhs_str);
    free(rhs_str);
    free(res);
    free(lhs);
    free(rhs);

    return 0;
