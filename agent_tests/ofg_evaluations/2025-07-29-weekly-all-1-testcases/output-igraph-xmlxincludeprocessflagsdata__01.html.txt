The `flags` argument is used to control the XInclude processing behavior. The `data` argument is a user-defined data pointer that is passed to the underlying processing function. The function first validates the input document and retrieves the root element. It then calls `xmlXIncludeProcessTreeFlagsData` to perform the actual XInclude processing on the document tree. The caller `xmlXIncludeProcessFlags` provides a NULL value for the data argument, indicating that it is not always required.
</description>

#include <fuzzer/FuzzedDataProvider.h>
#include <cstddef>
#include <cstdint>
#include <string>

extern "C" {
#include <libxml/parser.h>
#include <libxml/xinclude.h>
#include <libxml/xmlerror.h>
}

// Suppress libxml2 error messages to avoid spamming the logs during fuzzing.
void ignore_xml_errors(void *ctx, const char *msg, ...) {
    // Do nothing.
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    // Set a generic error handler to suppress output.
    xmlSetGenericErrorFunc(NULL, ignore_xml_errors);

    FuzzedDataProvider provider(data, size);

    int flags = provider.ConsumeIntegral<int>();
    std::string xml_content = provider.ConsumeRemainingBytesAsString();

    if (xml_content.empty()) {
        return 0;
    }

    // Use XML_PARSE_RECOVER to allow the parser to continue on non-fatal errors,
    // which can lead to deeper code coverage.
    xmlDocPtr doc = xmlReadMemory(xml_content.c_str(), xml_content.length(), "fuzz.xml", NULL, XML_PARSE_RECOVER);
    if (doc == NULL) {
        return 0;
    }

    xmlXIncludeProcessFlagsData(doc, flags, NULL);

    xmlFreeDoc(doc);
    // xmlCleanupParser() is not recommended in fuzz targets as it can interfere
    // with the fuzzing engine's state and is not thread-safe.

    return 0;
}
