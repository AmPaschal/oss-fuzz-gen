
	
<description>
The function `MmsValue_createArray` creates a new `MmsValue` of type `MMS_ARRAY`. It allocates memory for the array structure and for each of its elements. The elements are initialized with default values based on the provided `elementType`. The function is called by `MmsValue_newDefaultValue` when it needs to create an array.
</description>
<requirements>
<requirement>
The `elementType` argument must be a valid pointer to an `MmsVariableSpecification` struct. This is because it is passed to `MmsValue_newDefaultValue` which dereferences it.
</requirement>
<requirement>
The `size` argument should be a non-negative integer. This is because it is used as the size for memory allocation and as a loop counter.
</requirement>
<requirement>
The `type` member of the `elementType` struct should be a valid `MmsType` enum value. This is because it is used in a switch statement in the called function `MmsValue_newDefaultValue`.
</requirement>
<requirement>
The `typeSpec` member of the `elementType` struct should be initialized according to the `type` member. This is to ensure that the correct default value is created for the elements of the array.
</requirement>
</requirements>

#include <stdint.h>
#include <stdlib.h>
#include <string.h>

#include "/src/libiec61850/src/mms/inc/mms_value.h"
#include "mms_types.h"
#include "mms_common.h"

MmsValue * MmsValue_createArray(const MmsVariableSpecification *, int);

int
LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    if (size < sizeof(int) + 1) {
        return 0;
    }

    MmsVariableSpecification elementSpec;
    memset(&elementSpec, 0, sizeof(MmsVariableSpecification));

    /* Use one byte from input to select the element type */
    MmsType elementType = data[sizeof(int)] % (MMS_DATA_ACCESS_ERROR + 1);
    elementSpec.type = elementType;

    /*
     * MmsValue_newDefaultValue is called by MmsValue_createArray. For complex
     * types like ARRAY or STRUCTURE, it expects nested type specifications
     * which are complex to fuzz. To avoid crashes from NULL pointer
     * dereferences, we default to a simple type if a complex one is chosen.
     */
    if (elementType == MMS_ARRAY || elementType == MMS_STRUCTURE) {
        elementSpec.type = MMS_BOOLEAN;
    }

    /* Provide some default values for typeSpec union members where required. */
    switch (elementSpec.type) {
        case MMS_INTEGER:
            elementSpec.typeSpec.integer = 32;
            break;
        case MMS_UNSIGNED:
            elementSpec.typeSpec.unsignedInteger = 32;
            break;
        case MMS_FLOAT:
            elementSpec.typeSpec.floatingpoint.formatWidth = 32;
            elementSpec.typeSpec.floatingpoint.exponentWidth = 8;
            break;
        case MMS_BIT_STRING:
            elementSpec.typeSpec.bitString = 32;
            break;
        case MMS_OCTET_STRING:
            elementSpec.typeSpec.octetString = 32;
            break;
        case MMS_VISIBLE_STRING:
            elementSpec.typeSpec.visibleString = 32;
            break;
        case MMS_STRING:
            elementSpec.typeSpec.mmsString = 32;
            break;
        case MMS_BINARY_TIME:
            elementSpec.typeSpec.binaryTime = 4;
            break;
        default:
            /* Other simple types like MMS_BOOLEAN don't need typeSpec. */
            break;
    }

    int arraySize;
    memcpy(&arraySize, data, sizeof(int));

    /* Sanitize size to be non-negative and within a reasonable limit */
    if (arraySize < 0) {
        arraySize = 0;
    }
    arraySize = arraySize % 1024;

    MmsValue* arrayValue = MmsValue_createArray(&elementSpec, arraySize);

    if (arrayValue != NULL) {
        MmsValue_delete(arrayValue);
    }

    return 0;
}
