// Helper function to safely consume data from the input buffer
static int consume_data(void* dest, size_t len, const uint8_t** data, size_t* size) {
    if (*size < len) {
        return 0; // Not enough data
    }
    memcpy(dest, *data, len);
    *data += len;
    *size -= len;
    return 1;
}

// Recursively frees the MmsVariableSpecification structure
static void free_spec(MmsVariableSpecification* spec) {
    if (spec == NULL) {
        return;
    }

    if (spec->type == MMS_ARRAY) {
        free_spec(spec->typeSpec.array.elementTypeSpec);
    } else if (spec->type == MMS_STRUCTURE) {
        if (spec->typeSpec.structure.elements != NULL) {
            for (int i = 0; i < spec->typeSpec.structure.elementCount; i++) {
                free_spec(spec->typeSpec.structure.elements[i]);
            }
            free(spec->typeSpec.structure.elements);
        }
    }

    free(spec);
}
