
	
<description>
The `Pistache::Rest::Private::RouterHandler::onRequest` function is responsible for dispatching incoming HTTP requests to the appropriate route handler. It achieves this by calling the `route` method on its internal `router` member, which is a `std::shared_ptr<Rest::Router>`. For the `onRequest` function to be tested effectively, the `router` object must be initialized and configured with at least one route. A route is defined by an HTTP method, a resource path, and a handler function. When `onRequest` is called, the `route` method of the `router` object searches for a handler that matches the method and resource of the incoming request. If a match is found, the corresponding handler is invoked. Therefore, to achieve high code coverage and to avoid trivial rejections of all requests, the fuzzer should be set up to create a `RouterHandler` with a pre-configured `Rest::Router` instance containing one or more routes. The fuzzer should then generate `Http::Request` objects that have a high probability of matching one of the configured routes.
</description>
<requirements>
<requirement>
The `router` member of the `RouterHandler` class must be initialized with a valid `Rest::Router` object. This is because the `onRequest` method directly calls the `route` method on this object, and a null pointer would lead to a crash.
</requirement>
<requirement>
The `Rest::Router` object used to initialize the `RouterHandler` must have at least one route configured. This is necessary to ensure that the `router->route` call can potentially find and execute a handler, allowing for deeper code coverage of the request-handling logic.
</requirement>
<requirement>
The `req` argument, which is a `const Http::Request&`, must be a valid object. The `method` and `resource` fields of the request should be fuzzed to match the routes configured in the `Rest::Router` instance to test the routing logic effectively.
</requirement>
</requirements>

template <class F> void ignoreExceptions(const F &func) {
  try {
    func();
  } catch (...) {
  }
}

template <class T> void parseHttpHeader(const std::string &input) {
  T header;
  ignoreExceptions([&] { header.parse(input); });

  std::stringstream oss;
  header.write(oss);
}

template <> void parseHttpHeader<Pistache::Http::Header::Authorization>(const std::string &input) {
  Pistache::Http::Header::Authorization header;
  ignoreExceptions([&] { header.parse(input); });
  ignoreExceptions([&] { header.getMethod(); });
  ignoreExceptions([&] {
    const auto user = header.getBasicUser();
    const auto password = header.getBasicPassword();
    header.setBasicUserPassword(user, password);
  });

  std::stringstream oss;
  header.write(oss);
}

void fuzz_request_parser(const std::string &input) {
  constexpr size_t maxDataSize = 4096;
  Pistache::Http::RequestParser rparser(maxDataSize);

  if (rparser.feed(input.data(), input.size())) {
    auto state = Pistache::Http::Private::State::Done;
    ignoreExceptions([&] { state = rparser.parse(); });

    if (state == Pistache::Http::Private::State::Again)
      ignoreExceptions([&] { rparser.parse(); });
  }
}

void fuzz_router(const std::string &input) {
  std::string path_input;
  std::stringstream input_stream(input);
  Pistache::Rest::SegmentTreeNode tree;

  while (std::getline(input_stream, path_input, '\n')) {
    if (path_input.size() < 2)
      continue;

    const int test_case = path_input.back();
    path_input.pop_back();

    const auto sanitized = Pistache::Rest::SegmentTreeNode::sanitizeResource(path_input);
    std::shared_ptr<char> ptr(new char[sanitized.length()], std::default_delete<char[]>());
    memcpy(ptr.get(), sanitized.data(), sanitized.length());
    const std::string_view path{ptr.get(), sanitized.length()};

    switch (test_case) {
    case 'A':
      ignoreExceptions([&] {
        Pistache::Rest::Route::Handler handler = [](auto...) { return Pistache::Rest::Route::Result::Ok; };
        std::shared_ptr<char> ptr(new char[sanitized.length()], std::default_delete<char[]>());
        memcpy(ptr.get(), sanitized.data(), sanitized.length());
        const std::string_view path{ptr.get(), sanitized.length()};
        tree.addRoute(path, handler, ptr);
      });
      break;
    case 'R':
      ignoreExceptions([&] { tree.removeRoute(path); });
      break;
    case 'F':
      ignoreExceptions([&] { tree.findRoute(path); });
      break;
    }
  }
}

void fuzz_router_handler(const std::string &input) {
  if (input.empty()) {
    return;
  }

  auto router = std::make_shared<Pistache::Rest::Router>();

  std::string path = input;
  if (path[0] != '/') {
    path.insert(0, 1, '/');
  }

  router->get(path, [](const Pistache::Rest::Request &, Pistache::Http::ResponseWriter) { return Pistache::Rest::Route::Result::Ok; });

  Pistache::Rest::Private::RouterHandler router_handler(router);

  std::string raw_request = "GET " + path + " HTTP/1.1\r\nHost: fuzzer\r\n\r\n";
  Pistache::Http::RequestParser parser(4096);
  parser.feed(raw_request.data(), raw_request.size());

  if (parser.parse() != Pistache::Http::Private::State::Done) {
    return;
  }
  const auto &req = parser.request;

  Pistache::Http::ResponseWriter writer(Pistache::Http::Version::Http11, nullptr, &router_handler, {});

  ignoreExceptions([&] { router_handler.onRequest(req, std::move(writer)); });
}
