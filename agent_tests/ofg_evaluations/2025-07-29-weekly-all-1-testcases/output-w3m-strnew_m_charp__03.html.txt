static void *die_oom(size_t bytes) {
    fprintf(stderr, "Out of memory: %lu bytes unavailable!\n", (unsigned long)bytes);
    exit(1);
}

// Helper to create a null-terminated string from a data chunk using the GC
static const char *make_zstring(const uint8_t *data, size_t size) {
    if (!data) return NULL;
    char *str = (char *)GC_MALLOC(size + 1);
    if (!str) die_oom(size + 1);
    memcpy(str, data, size);
    str[size] = '\0';
    return str;
}
