
	
<description>
The function `ogs_gtp2_send_error_message` is responsible for sending a GTPv2 error message. It constructs an error message of a specific type, sets the cause value, and then sends it using a GTP transaction context. The message type determines which field in the `errmsg` union will hold the cause TLV. The function uses a transaction object `xact` to send the message, and it will crash if an unsupported message type is provided.
</description>
<requirements>

<requirement>
The `xact` parameter must be a valid pointer to an `ogs_gtp_xact_t` struct, initialized by the GTP transaction management functions. This is because the function calls `ogs_gtp_xact_update_tx` and `ogs_gtp_xact_commit` which operate on this transaction object.
</requirement>
<requirement>
The `type` parameter must be one of the following values: `OGS_GTP2_CREATE_SESSION_RESPONSE_TYPE`, `OGS_GTP2_MODIFY_BEARER_RESPONSE_TYPE`, `OGS_GTP2_DELETE_SESSION_RESPONSE_TYPE`, `OGS_GTP2_RELEASE_ACCESS_BEARERS_RESPONSE_TYPE`, `OGS_GTP2_DOWNLINK_DATA_NOTIFICATION_ACKNOWLEDGE_TYPE`, `OGS_GTP2_CREATE_BEARER_RESPONSE_TYPE`, `OGS_GTP2_UPDATE_BEARER_RESPONSE_TYPE`, `OGS_GTP2_DELETE_BEARER_RESPONSE_TYPE`, `OGS_GTP2_CREATE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE`, `OGS_GTP2_DELETE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE`, `OGS_GTP2_MODIFY_BEARER_FAILURE_INDICATION_TYPE`, `OGS_GTP2_DELETE_BEARER_FAILURE_INDICATION_TYPE`, or `OGS_GTP2_BEARER_RESOURCE_FAILURE_INDICATION_TYPE`. Any other value will cause the program to terminate via `ogs_fatal`.
</requirement>
<requirement>
The `teid` parameter is a 32-bit integer representing a tunnel endpoint identifier. It can be fuzzed as there are no internal checks on its value.
</requirement>
<requirement>
The `cause_value` parameter is an 8-bit integer representing the cause of the error. It can be fuzzed as there are no internal checks on its value.
</requirement>

</requirements>

#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include <sys/socket.h>

#include "fuzzing.h"
#include "ogs-gtp.h"

// Structure to map fuzzer input
typedef struct {
    uint32_t teid;
    uint8_t type_index;
    uint8_t cause_value;
} FuzzInput;

#define kMinInputLength sizeof(FuzzInput)

// Array of valid message types for the target function
static const uint8_t valid_types[] = {
    OGS_GTP2_CREATE_SESSION_RESPONSE_TYPE,
    OGS_GTP2_MODIFY_BEARER_RESPONSE_TYPE,
    OGS_GTP2_DELETE_SESSION_RESPONSE_TYPE,
    OGS_GTP2_RELEASE_ACCESS_BEARERS_RESPONSE_TYPE,
    OGS_GTP2_DOWNLINK_DATA_NOTIFICATION_ACKNOWLEDGE_TYPE,
    OGS_GTP2_CREATE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_UPDATE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_DELETE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_CREATE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE,
    OGS_GTP2_DELETE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE,
    OGS_GTP2_MODIFY_BEARER_FAILURE_INDICATION_TYPE,
    OGS_GTP2_DELETE_BEARER_FAILURE_INDICATION_TYPE,
    OGS_GTP2_BEARER_RESOURCE_FAILURE_INDICATION_TYPE,
};
static const size_t num_valid_types = sizeof(valid_types) / sizeof(valid_types[0]);

extern int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) {
    if (Size < kMinInputLength) {
        return 0;
    }

    if (!initialized) {
        initialize();
        ogs_log_install_domain(&__ogs_gtp_domain, "gtp", OGS_LOG_NONE);
        ogs_log_install_domain(&__ogs_tlv_domain, "tlv", OGS_LOG_NONE);
        ogs_gtp_context_init(1); // Initialize GTP context
    }

    const FuzzInput *input = (const FuzzInput *)Data;

    // Prepare parameters for the function-under-test
    uint32_t teid = input->teid;
    uint8_t type = valid_types[input->type_index % num_valid_types];
    uint8_t cause_value = input->cause_value;

    // Setup required objects for ogs_gtp_xact_t
    ogs_sockaddr_t *addr = NULL;
    if (ogs_getaddrinfo(&addr, AF_INET, "127.0.0.1", 2123, 0) != OGS_OK) {
        return 0;
    }

    // Add node to the global context. This assigns a valid socket to the node.
    ogs_gtp_node_t *gnode = ogs_gtp_node_add_by_addr(&ogs_gtp_self()->gtpc_list, addr);
    if (!gnode) {
        // ogs_gtp_node_add_by_addr frees addr on failure
        return 0;
    }

    ogs_pkbuf_t *pkbuf = ogs_pkbuf_alloc(NULL, 1);
    if (!pkbuf) {
        // gnode is in a global list, don't free it manually.
        return 0;
    }

    ogs_gtp2_header_t h;
    memset(&h, 0, sizeof(h));
    h.type = OGS_GTP2_CREATE_SESSION_REQUEST_TYPE; // Dummy type for xact creation

    ogs_gtp_xact_t *xact = ogs_gtp_xact_local_create(gnode, &h, pkbuf, NULL, NULL);
    if (!xact) {
        ogs_pkbuf_free(pkbuf);
        // gnode is in a global list, don't free it manually.
        return 0;
    }
    
    xact->gtp_version = 2;

    // Call the target function
    ogs_gtp2_send_error_message(xact, teid, type, cause_value);

    // Do not free gnode here. It's managed by the global context and the
    // transaction may still be alive and holds a pointer to it.
    // Leaking is acceptable in the fuzz target.

    return 0;
}
