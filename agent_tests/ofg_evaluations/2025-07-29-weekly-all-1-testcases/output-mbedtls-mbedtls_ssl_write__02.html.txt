
	
<description>
The `mbedtls_ssl_write()` function is used to write application data to an SSL/TLS connection. It encrypts the provided data and sends it over the underlying transport. The function is stateful and requires the SSL context to be fully initialized and the handshake process to be completed before it can be called. The function handles the fragmentation of data for stream-based transports if the input length exceeds the maximum fragment length.
</description>
<requirements>

<requirement>
The `mbedtls_ssl_context` structure must be initialized by calling `mbedtls_ssl_init()` and `mbedtls_ssl_setup()` before use. This is to ensure that all the internal fields of the context are set to a known and valid state.
</requirement>
<requirement>
A valid and seeded random number generator (RNG) must be configured for the SSL context using `mbedtls_ssl_conf_rng()`. The SSL/TLS protocol relies on a strong RNG for generating keys and other security-critical values.
</requirement>
<requirement>
The BIO (Basic Input/Output) callbacks for sending and receiving data must be set for the SSL context using `mbedtls_ssl_set_bio()`. This is necessary for `mbedtls_ssl_write()` to send the encrypted data over the network.
</requirement>
<requirement>
The SSL/TLS handshake must be completed by calling `mbedtls_ssl_handshake()` until it returns 0. `mbedtls_ssl_write()` will not send application data until the handshake is successfully completed and the session keys are established.
</requirement>
<requirement>
The second argument, `buf`, should be a pointer to a valid buffer containing the data to be written. This buffer can be fuzzed as it represents user-controlled application data.
</requirement>
<requirement>
The third argument, `len`, must be the length of the data in `buf`. While it can be 0, it should not exceed the actual size of the buffer pointed to by `buf` to prevent out-of-bounds reads.
</requirement>

</requirements>

int dummy_random(void *p_rng, unsigned char *output, size_t output_len)
{
    int ret;
    size_t i;
    unsigned long val;

    if (p_rng != NULL) {
        return dummy_entropy(p_rng, output, output_len);
    }

    for (i = 0; i < output_len; i++) {
        val = rand();
        output[i] = (unsigned char)val;
    }

    ret = 0;
    return ret;
}

    (void) len;

int fuzz_recv(void *ctx, unsigned char *buf, size_t len)
{
    fuzzBufferOffset_t *biomemfuzz = (fuzzBufferOffset_t *) ctx;

    if (biomemfuzz->Offset >= biomemfuzz->Size) {
        return MBEDTLS_ERR_SSL_WANT_READ;
    }

    if (len > biomemfuzz->Size - biomemfuzz->Offset) {
        len = biomemfuzz->Size - biomemfuzz->Offset;
    }

    memcpy(buf, biomemfuzz->Data + biomemfuzz->Offset, len);
    biomemfuzz->Offset += len;

    return len;
}

int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size)
{
#if defined(MBEDTLS_SSL_CLI_C) && \
    defined(MBEDTLS_ENTROPY_C) && \
    defined(MBEDTLS_CTR_DRBG_C)
    int ret;
    mbedtls_ssl_context ssl;
    mbedtls_ssl_config conf;
    mbedtls_ctr_drbg_context ctr_drbg;
    mbedtls_entropy_context entropy;
    fuzzBufferOffset_t biomemfuzz;
    uint16_t options;
    const unsigned char *write_buf;
    size_t write_len;
    size_t bio_len;

    if (initialized == 0) {
#if defined(MBEDTLS_X509_CRT_PARSE_C) && defined(MBEDTLS_PEM_PARSE_C)
        mbedtls_x509_crt_init(&cacert);
        if (mbedtls_x509_crt_parse(&cacert, (const unsigned char *) mbedtls_test_cas_pem,
                                   mbedtls_test_cas_pem_len) != 0) {
            return 1;
        }
#endif

        alpn_list[0] = "HTTP";
        alpn_list[1] = "fuzzalpn";
        alpn_list[2] = NULL;

        dummy_init();

        initialized = 1;
    }

    // we need at least 2 bytes for options and 1 for bio, 1 for write
    if (Size < 4) {
        return 0;
    }
    options = (Data[Size - 2] << 8) | Data[Size - 1];
    //Avoid warnings if compile options imply no options
    (void) options;

    // Split data into two halves: one for handshake, one for writing
    bio_len = (Size - 2) / 2;
    write_buf = Data + bio_len;
    write_len = (Size - 2) - bio_len;

    mbedtls_ssl_init(&ssl);
    mbedtls_ssl_config_init(&conf);
    mbedtls_ctr_drbg_init(&ctr_drbg);
    mbedtls_entropy_init(&entropy);

#if defined(MBEDTLS_USE_PSA_CRYPTO)
    psa_status_t status = psa_crypto_init();
    if (status != PSA_SUCCESS) {
        goto exit;
    }
#endif /* MBEDTLS_USE_PSA_CRYPTO */

    if (mbedtls_ctr_drbg_seed(&ctr_drbg, dummy_entropy, &entropy,
                              (const unsigned char *) pers, strlen(pers)) != 0) {
        goto exit;
    }

    if (mbedtls_ssl_config_defaults(&conf,
                                    MBEDTLS_SSL_IS_CLIENT,
                                    MBEDTLS_SSL_TRANSPORT_STREAM,
                                    MBEDTLS_SSL_PRESET_DEFAULT) != 0) {
        goto exit;
    }

#if defined(MBEDTLS_KEY_EXCHANGE_SOME_PSK_ENABLED)
    if (options & 2) {
        mbedtls_ssl_conf_psk(&conf, psk, sizeof(psk),
                             (const unsigned char *) psk_id, sizeof(psk_id) - 1);
    }
#endif

#if defined(MBEDTLS_X509_CRT_PARSE_C) && defined(MBEDTLS_PEM_PARSE_C)
    if (options & 4) {
        mbedtls_ssl_conf_ca_chain(&conf, &cacert, NULL);
        mbedtls_ssl_conf_authmode(&conf, MBEDTLS_SSL_VERIFY_REQUIRED);
    } else
#endif
    {
        mbedtls_ssl_conf_authmode(&conf, MBEDTLS_SSL_VERIFY_NONE);
    }
#if defined(MBEDTLS_SSL_EXTENDED_MASTER_SECRET)
    mbedtls_ssl_conf_extended_master_secret(&conf,
                                            (options &
                                             0x10) ? MBEDTLS_SSL_EXTENDED_MS_DISABLED : MBEDTLS_SSL_EXTENDED_MS_ENABLED);
#endif
#if defined(MBEDTLS_SSL_ENCRYPT_THEN_MAC)
    mbedtls_ssl_conf_encrypt_then_mac(&conf,
                                      (options &
                                       0x20) ? MBEDTLS_SSL_ETM_DISABLED : MBEDTLS_SSL_ETM_ENABLED);
#endif
#if defined(MBEDTLS_SSL_RENEGOTIATION)
    mbedtls_ssl_conf_renegotiation(&conf,
                                   (options &
                                    0x80) ? MBEDTLS_SSL_RENEGOTIATION_ENABLED : MBEDTLS_SSL_RENEGOTIATION_DISABLED);
#endif
#if defined(MBEDTLS_SSL_SESSION_TICKETS)
    mbedtls_ssl_conf_session_tickets(&conf,
                                     (options &
                                      0x100) ? MBEDTLS_SSL_SESSION_TICKETS_DISABLED : MBEDTLS_SSL_SESSION_TICKETS_ENABLED);
#endif
#if defined(MBEDTLS_SSL_ALPN)
    if (options & 0x200) {
        mbedtls_ssl_conf_alpn_protocols(&conf, alpn_list);
    }
#endif

    srand(1);
    mbedtls_ssl_conf_rng(&conf, dummy_random, &ctr_drbg);

    if (mbedtls_ssl_setup(&ssl, &conf) != 0) {
        goto exit;
    }

#if defined(MBEDTLS_X509_CRT_PARSE_C) && defined(MBEDTLS_PEM_PARSE_C)
    if ((options & 1) == 0) {
        if (mbedtls_ssl_set_hostname(&ssl, "localhost") != 0) {
            goto exit;
        }
    }
#endif

    biomemfuzz.Data = Data;
    biomemfuzz.Size = bio_len;
    biomemfuzz.Offset = 0;
    mbedtls_ssl_set_bio(&ssl, &biomemfuzz, dummy_send, fuzz_recv, NULL);

    ret = mbedtls_ssl_handshake(&ssl);
    if (ret == 0) {
        // Handshake successful, now call mbedtls_ssl_write
        mbedtls_ssl_write(&ssl, write_buf, write_len);
    }

exit:
    mbedtls_entropy_free(&entropy);
    mbedtls_ctr_drbg_free(&ctr_drbg);
    mbedtls_ssl_config_free(&conf);
    mbedtls_ssl_free(&ssl);
#if defined(MBEDTLS_USE_PSA_CRYPTO)
    mbedtls_psa_crypto_free();
#endif /* MBEDTLS_USE_PSA_CRYPTO */

#else
    (void) Data;
    (void) Size;
#endif /* MBEDTLS_SSL_CLI_C && MBEDTLS_ENTROPY_C && MBEDTLS_CTR_DRBG_C */

    return 0;
}
