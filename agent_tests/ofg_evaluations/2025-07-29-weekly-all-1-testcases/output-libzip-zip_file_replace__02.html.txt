
	
<description>
The function `zip_file_replace` replaces a file in a zip archive. The file to be replaced is identified by its index `idx`. The new file content is provided by the `source` argument. The `flags` argument controls the behavior of the replacement.
</description>
<requirements>
<requirement>
The first argument, `za`, must be a valid pointer to a `zip_t` struct that was not opened in read-only mode. This is because the underlying function `_zip_file_replace` will fail if the archive is read-only.
</requirement>
<requirement>
The second argument, `idx`, must be a valid index of an existing file in the archive. The function checks if `idx` is greater than or equal to the number of entries in the archive and fails if it is.
</requirement>
<requirement>
The third argument, `source`, must be a non-NULL pointer to a `zip_source_t` struct. It can be created using `zip_source_buffer`.
</requirement>
<requirement>
The fourth argument, `flags`, can be used to control the replacement behavior. While callers in the codebase often pass 0, it can be fuzzed with flags like `ZIP_FL_NOCASE`, `ZIP_FL_NODIR`, `ZIP_FL_ENC_UTF_8`, etc.
</requirement>
</requirements>

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    const char *path = "fuzz_replace.zip";
    int err = 0;
    zip_t *za;
    zip_source_t *s_orig = NULL;
    zip_source_t *s_replace = NULL;
    zip_int64_t idx;

    if (size < 1) {
        return 0;
    }

    (void)remove(path);
    if ((za = zip_open(path, ZIP_CREATE, &err)) == NULL) {
        return 0;
    }

    const char *orig_content = "original content";
    s_orig = zip_source_buffer(za, orig_content, strlen(orig_content), 0);
    if (s_orig == NULL) {
        zip_discard(za);
        (void)remove(path);
        return 0;
    }

    idx = zip_file_add(za, "file_to_replace.txt", s_orig, ZIP_FL_OVERWRITE);
    if (idx < 0) {
        // s_orig is consumed by zip_file_add on both success and failure
        zip_discard(za);
        (void)remove(path);
        return 0;
    }

    zip_flags_t flags = data[0];
    const uint8_t *replace_data = data + 1;
    size_t replace_size = size - 1;

    s_replace = zip_source_buffer(za, replace_data, replace_size, 0);
    if (s_replace == NULL) {
        zip_discard(za);
        (void)remove(path);
        return 0;
    }

    if (zip_file_replace(za, (zip_uint64_t)idx, s_replace, flags) < 0) {
        // s_replace must be freed on failure
        zip_source_free(s_replace);
    }
    // On success, zip_file_replace consumes s_replace

    if (zip_close(za) < 0) {
        // zip_close frees za on error
    }

    (void)remove(path);

    return 0;
}
