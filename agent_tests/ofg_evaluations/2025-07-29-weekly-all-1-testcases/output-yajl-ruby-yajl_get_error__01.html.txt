
	
<description>
The `yajl_get_error` function returns a descriptive string for the last error that occurred during parsing. It is a wrapper around `yajl_render_error_string`. The function requires a `yajl_handle` that has encountered a parse or lexical error. The verbosity of the error message is controlled by the `verbose` parameter. The `jsonText` and `jsonTextLen` parameters are used to provide context for the error in the returned string when `verbose` is enabled.
</description>
<requirements>

<requirement>
The `yajl_handle` (first argument) must be a valid handle returned by `yajl_alloc`. This is because the function dereferences this handle to access internal state.
</requirement>
<requirement>
The `yajl_handle` must be in an error state. This can be achieved by calling a parsing function like `yajl_parse` or `yajl_do_parse` with invalid JSON input, which sets the internal state to `yajl_state_parse_error` or `yajl_state_lexical_error`.
</requirement>
<requirement>
The `jsonText` (third argument) must be a valid pointer to a buffer of size `jsonTextLen` (fourth argument). This is to prevent memory access errors when the function generates a verbose error message.
</requirement>
<requirement>
The `verbose` parameter (second argument) should be either 0 or 1. This parameter controls the level of detail in the error message.
</requirement>

</requirements>

#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

#include "/src/yajl-ruby/ext/yajl/api/yajl_parse.h"

typedef struct {
    int arrayLevel;
    int objectLevel;
} context;

static int yajl_found_null(void* ctx) { 
    return 1;
}

static int yajl_found_boolean(void* ctx, int boolean) {
    return 1;
};

static int yajl_found_number(void* ctx, const char* v, unsigned int l) {
    assert(l > 0);
    return 1;
}

static int yajl_found_string(void* ctx, const unsigned char* s, unsigned int l) {
    return 1;
}

static int yajl_found_object_key(void* ctx, const unsigned char* v, unsigned int l) {
    assert(((context*)ctx)->objectLevel > 0);
    return 1;
}

static int yajl_found_start_object(void* ctx) {
    ((context*)ctx)->objectLevel++;
    return 1;
}

static int yajl_found_end_object(void* ctx) {
    assert(((context*)ctx)->objectLevel > 0);
    ((context*)ctx)->objectLevel--;
    return 1;
}

static int yajl_found_start_array(void* ctx) {
    ((context*)ctx)->arrayLevel++;
    return 1;
}

static int yajl_found_end_array(void* ctx) {
    assert(((context*)ctx)->arrayLevel > 0);
    ((context*)ctx)->arrayLevel--;
    return 1;
}

static yajl_callbacks callbacks = {
    yajl_found_null,
    yajl_found_boolean,
    NULL,
    NULL,
    yajl_found_number,
    yajl_found_string,
    yajl_found_start_object,
    yajl_found_object_key,
    yajl_found_end_object,
    yajl_found_start_array,
    yajl_found_end_array
};

int LLVMFuzzerTestOneInput(const uint8_t* data, size_t size) {
    context ctx = {
        .arrayLevel = 0,
        .objectLevel = 0,
    };
    yajl_parser_config cfg = {
        .allowComments = 1,
        .checkUTF8 = 1,
    };
    yajl_handle parser = yajl_alloc(&callbacks, &cfg, NULL, (void*)&ctx);

    yajl_status stat = yajl_parse(parser, data, size);

    // After parsing, check for an error and get the error string.
    if (stat == yajl_status_error) {
        // Test non-verbose mode.
        unsigned char * error_str_non_verbose = yajl_get_error(parser, 0, data, (unsigned int)size);
        if (error_str_non_verbose) {
            yajl_free_error(parser, error_str_non_verbose);
        }

        // Test verbose mode.
        unsigned char * error_str_verbose = yajl_get_error(parser, 1, data, (unsigned int)size);
        if (error_str_verbose) {
            yajl_free_error(parser, error_str_verbose);
        }
    }

    // Also check for errors after completing the parse, for incomplete JSON.
    stat = yajl_parse_complete(parser);
    if (stat == yajl_status_error) {
        // Test non-verbose mode.
        unsigned char * error_str_non_verbose = yajl_get_error(parser, 0, data, (unsigned int)size);
        if (error_str_non_verbose) {
            yajl_free_error(parser, error_str_non_verbose);
        }

        // Test verbose mode.
        unsigned char * error_str_verbose = yajl_get_error(parser, 1, data, (unsigned int)size);
        if (error_str_verbose) {
            yajl_free_error(parser, error_str_verbose);
        }
    }

    yajl_free(parser);

    return 0;
}
