
	
<description>
The `gnutls_bye` function is used to close a TLS session. It takes a `gnutls_session_t` and a `gnutls_close_request_t` as input. The `gnutls_session_t` must be a valid session that has been initialized by `gnutls_init` and has successfully completed a handshake. The `gnutls_close_request_t` argument specifies whether to shut down the read channel, the write channel, or both.
</description>
<requirements>
<requirement>
The first argument, `session`, must be a valid `gnutls_session_t` initialized by `gnutls_init`. This is because `gnutls_bye` operates on an active TLS session.
</requirement>
<requirement>
A handshake must be successfully completed on the session before calling `gnutls_bye`. This is to ensure that the session is in a valid state for termination.
</requirement>
<requirement>
The second argument, `how`, must be one of `GNUTLS_SHUT_WR` or `GNUTLS_SHUT_RDWR`. These are the only values handled by the function to control session shutdown.
</requirement>
</requirements>

/* In-memory transport buffers using public gnutls_datum_t */
static gnutls_datum_t c2s_buffer = { NULL, 0 };
static gnutls_datum_t s2c_buffer = { NULL, 0 };
static size_t c2s_read_pos = 0;
static size_t s2c_read_pos = 0;

/* Custom push function for gnutls_datum_t */
static ssize_t push_data(gnutls_datum_t *buffer, const void *data, size_t len) {
    void *new_ptr = realloc(buffer->data, buffer->size + len);
    if (new_ptr == NULL) {
        /* Out of memory */
        return -1;
    }
    buffer->data = new_ptr;
    memcpy(buffer->data + buffer->size, data, len);
    buffer->size += len;
    return len;
}

/* Transport Callbacks */
ssize_t client_push(gnutls_transport_ptr_t tr, const void *data, size_t len) {
  return push_data(&c2s_buffer, data, len);
}

ssize_t client_pull(gnutls_transport_ptr_t tr, void *data, size_t len) {
  if (s2c_read_pos >= s2c_buffer.size) {
    return GNUTLS_E_AGAIN;
  }
  size_t to_read = len;
  if (s2c_read_pos + to_read > s2c_buffer.size) {
    to_read = s2c_buffer.size - s2c_read_pos;
  }
  memcpy(data, s2c_buffer.data + s2c_read_pos, to_read);
  s2c_read_pos += to_read;
  return to_read;
}

ssize_t server_push(gnutls_transport_ptr_t tr, const void *data, size_t len) {
  return push_data(&s2c_buffer, data, len);
}

ssize_t server_pull(gnutls_transport_ptr_t tr, void *data, size_t len) {
  if (c2s_read_pos >= c2s_buffer.size) {
    return GNUTLS_E_AGAIN;
  }
  size_t to_read = len;
  if (c2s_read_pos + to_read > c2s_buffer.size) {
    to_read = c2s_buffer.size - c2s_read_pos;
  }
  memcpy(data, c2s_buffer.data + c2s_read_pos, to_read);
  c2s_read_pos += to_read;
  return to_read;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  gnutls_session_t client_session = NULL, server_session = NULL;
  gnutls_certificate_credentials_t client_cred = NULL, server_cred = NULL;
  int ret;

  if (size < 1) {
    return 0;
  }

  /* Initialize transport buffers */
  c2s_buffer.data = NULL;
  c2s_buffer.size = 0;
  s2c_buffer.data = NULL;
  s2c_buffer.size = 0;
  c2s_read_pos = 0;
  s2c_read_pos = 0;

  /* Initialize server */
  ret = gnutls_init(&server_session, GNUTLS_SERVER);
  assert(ret >= 0);
  ret = gnutls_certificate_allocate_credentials(&server_cred);
  assert(ret >= 0);

  gnutls_datum_t rsa_cert, rsa_key;
  rsa_cert.data = (unsigned char *)kRSACertificateDER;
  rsa_cert.size = sizeof(kRSACertificateDER);
  rsa_key.data = (unsigned char *)kRSAPrivateKeyDER;
  rsa_key.size = sizeof(kRSAPrivateKeyDER);
  ret = gnutls_certificate_set_x509_key_mem(server_cred, &rsa_cert, &rsa_key, GNUTLS_X509_FMT_DER);
  assert(ret >= 0);

  ret = gnutls_credentials_set(server_session, GNUTLS_CRD_CERTIFICATE, server_cred);
  assert(ret >= 0);
  ret = gnutls_priority_set_direct(server_session, "NORMAL", NULL);
  assert(ret >= 0);

  gnutls_transport_set_push_function(server_session, server_push);
  gnutls_transport_set_pull_function(server_session, server_pull);
  gnutls_transport_set_ptr(server_session, NULL);

  /* Initialize client */
  ret = gnutls_init(&client_session, GNUTLS_CLIENT);
  assert(ret >= 0);
  ret = gnutls_certificate_allocate_credentials(&client_cred);
  assert(ret >= 0);
  ret = gnutls_credentials_set(client_session, GNUTLS_CRD_CERTIFICATE, client_cred);
  assert(ret >= 0);
  ret = gnutls_priority_set_direct(client_session, "NORMAL", NULL);
  assert(ret >= 0);

  gnutls_transport_set_push_function(client_session, client_push);
  gnutls_transport_set_pull_function(client_session, client_pull);
  gnutls_transport_set_ptr(client_session, NULL);

  /* Perform handshake */
  int client_ret = 0, server_ret = 0;
  int rounds = 0;
  do {
    client_ret = gnutls_handshake(client_session);
    server_ret = gnutls_handshake(server_session);
    rounds++;
  } while ((client_ret < 0 && gnutls_error_is_fatal(client_ret) == 0) && (server_ret < 0 && gnutls_error_is_fatal(server_ret) == 0) && rounds < 50);

  /* If handshake is complete, fuzz gnutls_bye */
  if (client_ret >= 0 && server_ret >= 0) {
    gnutls_close_request_t how = (data[0] % 2 == 0) ? GNUTLS_SHUT_WR : GNUTLS_SHUT_RDWR;
    gnutls_bye(client_session, how);
  }

  /* Cleanup */
  if (client_session)
    gnutls_deinit(client_session);
  if (server_session)
    gnutls_deinit(server_session);
  if (client_cred)
    gnutls_certificate_free_credentials(client_cred);
  if (server_cred)
    gnutls_certificate_free_credentials(server_cred);

  free(c2s_buffer.data);
  free(s2c_buffer.data);

  return 0;
}
