
	
<description>

The function returns `true` if the expression is a literal, and `false` otherwise. The function is a const member function, so it does not modify the state of the `Expr` object.
</description>

</requirement>

#include <fuzzer/FuzzedDataProvider.h>
#include "/src/sql-parser/src/sql/Expr.h"
#include "/src/sql-parser/src/sql/ColumnType.h"
#include <string>
#include <cstring>

using namespace hsql;

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider stream(data, size);

  // ConsumeIntegralInRange is used as the enums don't have kMaxValue.
  // The max values are derived from the enum definitions in Expr.h and ColumnType.h
  ExprType expr_type = static_cast<ExprType>(stream.ConsumeIntegralInRange<int>(0, 16));
  Expr* expr = nullptr;

  switch (expr_type) {
    case kExprLiteralFloat:
      expr = Expr::makeLiteral(stream.ConsumeFloatingPoint<double>());
      break;
    case kExprLiteralString: {
      std::string str = stream.ConsumeRandomLengthString();
      // The destructor uses free, so we must use malloc-based allocation.
      expr = Expr::makeLiteral(strdup(str.c_str()));
      break;
    }
    case kExprLiteralInt:
      expr = Expr::makeLiteral(stream.ConsumeIntegral<int64_t>());
      break;
    case kExprLiteralNull:
      expr = Expr::makeNullLiteral();
      break;
    case kExprLiteralDate: {
      std::string date_str = "2023-01-01";
      expr = Expr::makeDateLiteral(strdup(date_str.c_str()));
      break;
    }
    case kExprLiteralInterval: {
      DatetimeField field = static_cast<DatetimeField>(stream.ConsumeIntegralInRange<int>(0, 6));
      expr = Expr::makeIntervalLiteral(stream.ConsumeIntegral<int64_t>(), field);
      break;
    }
    case kExprStar:
      expr = Expr::makeStar();
      break;
    case kExprParameter:
      expr = Expr::makeParameter(stream.ConsumeIntegral<int>());
      break;
    case kExprColumnRef: {
      std::string col_name = stream.ConsumeRandomLengthString();
      expr = Expr::makeColumnRef(strdup(col_name.c_str()));
      break;
    }
    case kExprFunctionRef: {
      std::string func_name = stream.ConsumeRandomLengthString();
      expr = Expr::makeFunctionRef(strdup(func_name.c_str()), nullptr, false, nullptr);
      break;
    }
    case kExprOperator: {
      OperatorType op = static_cast<OperatorType>(stream.ConsumeIntegralInRange<int>(0, 23));
      expr = Expr::makeOpUnary(op, nullptr);
      break;
    }
    case kExprSelect:
      // Creating a SelectStatement is complex, so we skip it for now.
      break;
    case kExprHint:
      // Not a creatable expression type.
      break;
    case kExprArray:
      expr = Expr::makeArray(nullptr);
      break;
    case kExprArrayIndex:
      expr = Expr::makeArrayIndex(nullptr, stream.ConsumeIntegral<int64_t>());
      break;
    case kExprExtract: {
      DatetimeField field = static_cast<DatetimeField>(stream.ConsumeIntegralInRange<int>(0, 6));
      expr = Expr::makeExtract(field, nullptr);
      break;
    }
    case kExprCast: {
      DataType dt = static_cast<DataType>(stream.ConsumeIntegralInRange<int>(0, 15));
      ColumnType col_type(dt);
      expr = Expr::makeCast(nullptr, col_type);
      break;
    }
  }

  if (expr) {
    expr->isLiteral();
    delete expr;
  }

  return 0;
}
