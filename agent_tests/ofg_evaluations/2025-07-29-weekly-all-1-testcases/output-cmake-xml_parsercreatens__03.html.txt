
	
<description>
The function `XML_ParserCreateNS` is a wrapper around `XML_ParserCreate_MM`. It creates a namespace-aware parser. The first argument, `encodingName`, specifies the character encoding. The second argument, `nsSep`, is the namespace separator.
</description>
<requirements>
<requirement>
The `encodingName` argument can be a NULL pointer. If `encodingName` is not NULL, it must be a valid, null-terminated string representing a character encoding that the Expat library can handle.
</requirement>
<requirement>
The `nsSep` argument can be any valid `XML_Char` value. It is used as the namespace separator.
</requirement>
</requirements>

#include <fuzzer/FuzzedDataProvider.h>
#include <cstddef>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  const XML_Char nsSep = provider.ConsumeIntegral<char>();
  const XML_Char *encoding = nullptr;
  std::string encoding_str;

  if (provider.ConsumeBool()) {
    encoding_str = provider.ConsumeRandomLengthString();
    encoding = encoding_str.c_str();
  }

  XML_Parser parser = XML_ParserCreateNS(encoding, nsSep);

  if (parser) {
    std::vector<uint8_t> remaining_data =
        provider.ConsumeRemainingBytes<uint8_t>();
    XML_Parse(parser, reinterpret_cast<const char *>(remaining_data.data()),
              remaining_data.size(), 1);
    XML_ParserFree(parser);
  }

  return 0;
}
