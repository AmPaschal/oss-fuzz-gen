
	
<description>
The `kex_gen_client` function is responsible for generating the client-side key pair for various key exchange methods in an SSH connection. It is not called directly, but rather used as a function pointer within the `ssh` struct, which is the main data structure for an SSH connection. The function inspects the `kex_type` field of the `ssh->kex` struct to determine which key exchange algorithm to use, and then calls the appropriate function to generate the key pair. After key generation, it sends the client's public key to the server.
</description>

</requirement>
<requirement>

</requirement>
<requirement>
The `ssh->kex->kex_type` field must be a valid and supported key exchange type. This is because the function uses this field in a switch statement to determine which key generation function to call. Invalid values will lead to an error.
</requirement>
<requirement>
The `ssh->kex->client_pub` field must be a valid buffer. This is because after key generation, this buffer is used to send the public key to the server.
</requirement>

// Helper to split a comma-separated string into a vector of strings.
static std::vector<std::string> split_string(const std::string& s, char delimiter) {
   std::vector<std::string> tokens;
   std::string token;
   std::istringstream tokenStream(s);
   while (std::getline(tokenStream, token, delimiter)) {
      if (!token.empty()) {
        tokens.push_back(token);
      }
   }
   return tokens;
}

// Fuzz target for kex_gen_client
extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    static bool initialized = false;
    static std::vector<std::string> kex_algorithms;

    if (!initialized) {
        // Suppress logging to avoid spamming the output
        log_init("kex_fuzz", SYSLOG_LEVEL_QUIET, SYSLOG_FACILITY_AUTH, 1);
        // Populate the list of KEX algorithms from the project's defaults
        kex_algorithms = split_string(KEX_CLIENT_KEX, ',');
        initialized = true;
    }

    if (kex_algorithms.empty()) {
        return 0;
    }

    FuzzedDataProvider provider(data, size);

    // Pick a valid KEX algorithm to test
    const std::string& kex_alg_str = kex_algorithms[provider.ConsumeIntegralInRange<size_t>(0, kex_algorithms.size() - 1)];
    
    struct kex_params kex_params;
    struct ssh *ssh = NULL;

    // Manually populate the proposal to ensure all fields are set.
    // ssh_init will take ownership of the allocated memory.
    kex_params.proposal[PROPOSAL_KEX_ALGS] = xstrdup(kex_alg_str.c_str());
    kex_params.proposal[PROPOSAL_SERVER_HOST_KEY_ALGS] = xstrdup(KEX_DEFAULT_PK_ALG);
    kex_params.proposal[PROPOSAL_ENC_ALGS_CTOS] = xstrdup(KEX_CLIENT_ENCRYPT);
    kex_params.proposal[PROPOSAL_ENC_ALGS_STOC] = xstrdup(KEX_CLIENT_ENCRYPT);
    kex_params.proposal[PROPOSAL_MAC_ALGS_CTOS] = xstrdup(KEX_CLIENT_MAC);
    kex_params.proposal[PROPOSAL_MAC_ALGS_STOC] = xstrdup(KEX_CLIENT_MAC);
    kex_params.proposal[PROPOSAL_COMP_ALGS_CTOS] = xstrdup(KEX_DEFAULT_COMP);
    kex_params.proposal[PROPOSAL_COMP_ALGS_STOC] = xstrdup(KEX_DEFAULT_COMP);
    kex_params.proposal[PROPOSAL_LANG_CTOS] = xstrdup(KEX_DEFAULT_LANG);
    kex_params.proposal[PROPOSAL_LANG_STOC] = xstrdup(KEX_DEFAULT_LANG);

    // Initialize the ssh struct in client mode (is_server = 0).
    // ssh_init will take ownership of the proposal strings and free them later.
    if (ssh_init(&ssh, 0, &kex_params) != 0) {
        // ssh_init frees the proposal on failure, so we don't need to.
        return 0;
    }

    // Call the function under test.
    // This will exercise the key generation logic. The subsequent packet
    // sending will fail as there is no real connection, which is expected.
    kex_gen_client(ssh);

    // Clean up all allocated resources.
    ssh_free(ssh);

    return 0;
}
