
	
<description>
The GEOSMakeValidWithParams function takes a geometry `g` and a parameters object `params` and returns a new geometry that is topologically valid. It is a wrapper for the re-entrant function GEOSMakeValidWithParams_r. The `params` argument is a struct that controls the validation method. If `params->method` is `GEOS_MAKE_VALID_LINEWORK`, the function will use the `geos::operation::valid::MakeValid` class to fix the geometry. If `params->method` is `GEOS_MAKE_VALID_STRUCTURE`, it uses the `geos::geom::util::GeometryFixer` class. An invalid method will cause an error and return a NULL pointer.
</description>
<requirements>

<requirement>
The first argument, `g`, must be a valid pointer to a `Geometry` object. This is because it is dereferenced and passed to internal functions, and a NULL pointer would cause a crash.
</requirement>
<requirement>
The second argument, `params`, must be a valid pointer to a `GEOSMakeValidParams` struct, created by `GEOSMakeValidParams_create_r`. This ensures the struct is properly initialized with default values.
</requirement>
<requirement>
The `method` member of the `GEOSMakeValidParams` struct must be either `GEOS_MAKE_VALID_LINEWORK` or `GEOS_MAKE_VALID_STRUCTURE`. Any other value will result in an error and a NULL return, preventing deeper code coverage.
</requirement>
<requirement>
The `keepCollapsed` member of the `GEOSMakeValidParams` struct should be either 0 or 1. This member is used as a boolean flag when the `method` is `GEOS_MAKE_VALID_STRUCTURE`.
</requirement>

</requirements>

static int initialized = 0;
FILE *flogOut;

void notice(const char *fmt, ...) {
  va_list ap;
  fprintf(flogOut, "NOTICE: ");
  va_start(ap, fmt);
  vfprintf(flogOut, fmt, ap);
  va_end(ap);
  fprintf(flogOut, "\n");
}

void log_and_exit(const char *fmt, ...) {
  va_list ap;
  fprintf(flogOut, "ERROR: ");
  va_start(ap, fmt);
  vfprintf(flogOut, fmt, ap);
  va_end(ap);
  fprintf(flogOut, "\n");
}

int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) {
  if (initialized == 0) {
    flogOut = fopen("/dev/null", "wb");
    initGEOS(notice, log_and_exit);
    initialized = 1;
  }

  if (Size < 3) {
    return 0;
  }

  const uint8_t method_choice = Data[0];
  const uint8_t keep_collapsed_choice = Data[1];
  const uint8_t *geom_data = Data + 2;
  const size_t geom_size = Size - 2;

  GEOSGeometry *g = GEOSGeomFromWKB_buf(geom_data, geom_size);
  if (g == NULL) {
    return 0;
  }

  GEOSMakeValidParams *params = GEOSMakeValidParams_create();
  if (params == NULL) {
    GEOSGeom_destroy(g);
    return 0;
  }

  enum GEOSMakeValidMethods method = (method_choice % 2 == 0)
                                         ? GEOS_MAKE_VALID_LINEWORK
                                         : GEOS_MAKE_VALID_STRUCTURE;
  GEOSMakeValidParams_setMethod(params, method);

  int keep_collapsed = keep_collapsed_choice % 2;
  GEOSMakeValidParams_setKeepCollapsed(params, keep_collapsed);

  GEOSGeometry *g_valid = GEOSMakeValidWithParams(g, params);

  if (g_valid != NULL) {
    GEOSGeom_destroy(g_valid);
  }
  GEOSMakeValidParams_destroy(params);
  GEOSGeom_destroy(g);

  return 0;
}
