
	
<description>
The `lou_translatePrehyphenated` function translates a wide character string into braille, with added support for hyphenation. It is a wrapper around the `lou_translate` function. If hyphenation information is provided in `inputHyphens`, the function will use it in conjunction with the main translation to produce hyphenated braille in `outputHyphens`.
</description>
<requirements>
<requirement>
The `tableList` argument must be a comma-separated list of valid, compiled translation table files. It is recommended to call `lou_compileString` to compile the necessary tables beforehand and use the resulting table list.
</requirement>
<requirement>
The `inlen` and `outlen` pointers must not be NULL. `*inlen` must be the length of the input buffer `inbufx`, and `*outlen` must be the capacity of the output buffer `outbuf`.
</requirement>
<requirement>
If `inputHyphens` is not NULL, then `outputHyphens` must also not be NULL. This is a precondition for the hyphenation logic to be executed.
</requirement>
<requirement>
If `inputHyphens` is not NULL, the `outputHyphens` buffer must have a size of at least `*outlen` (the capacity of the output buffer). This is to prevent a buffer overflow when writing to `outputHyphens`.
</requirement>
<requirement>
If `inputHyphens` is not NULL, the `inputHyphens` buffer must be large enough to be indexed by the values returned in the `inputPos` buffer. The values in `inputPos` can go up to the original input length (`*inlen`), so `inputHyphens` should be at least that size.
</requirement>
<requirement>
The `mode` parameter can be set to a bitmask of flags to control the translation. While it can be fuzzed, providing no flags (0) is a safe default for general-purpose translation.
</requirement>
<requirement>
The `inbufx` parameter should be fuzzed as it represents the primary user-controlled input to the translation process.
</requirement>
</requirements>

// Suppress liblouis log output
static void silent_log_callback(logLevels level, const char *msg) {}

extern "C" int lou_translatePrehyphenated(const char *, const widechar *, int *, widechar *, int *, formtype *, char *, int *, int *, int *, char *, char *, int);

// Array of valid typeforms values
static const typeforms kTypeforms[] = {
    plain_text, emph_1,         emph_2,         emph_3,      emph_4,
    emph_5,     emph_6,         emph_7,         emph_8,      emph_9,
    emph_10,    computer_braille, no_translate, no_contract,
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  static bool initialized = false;
  if (!initialized) {
    lou_registerLogCallback(silent_log_callback);
    initialized = true;
  }

  FuzzedDataProvider provider(data, size);

  // Use part of the input to create a temporary table file.
  const size_t table_content_size = provider.ConsumeIntegralInRange<size_t>(0, 512);
  std::string table_content = provider.ConsumeBytesAsString(table_content_size);

  char table_filename[] = "/tmp/fuzz-table-XXXXXX";
  int fd = mkstemp(table_filename);
  if (fd == -1) {
    return 0;
  }
  FILE *fp = fdopen(fd, "wb");
  if (!fp) {
    close(fd);
    unlink(table_filename);
    return 0;
  }
  fwrite(table_content.c_str(), 1, table_content.length(), fp);
  fclose(fp);

  std::string compile_command = "include " + std::string(table_filename);
  const char *compiled_table_name = "fuzzer.ctb";

  if (!lou_compileString(compiled_table_name, compile_command.c_str())) {
    unlink(table_filename);
    return 0;
  }

  // Prepare parameters for the function call
  std::string input_string = provider.ConsumeRemainingBytesAsString();
  if (input_string.empty()) {
    lou_free();
    unlink(table_filename);
    return 0;
  }

  std::vector<widechar> inbuf(input_string.length() + 1);
  int inlen = _lou_extParseChars(input_string.c_str(), inbuf.data());
  if (inlen <= 0) {
    lou_free();
    unlink(table_filename);
    return 0;
  }

  int outlen = inlen * 5 + 100; // Provide ample space for the output
  std::vector<widechar> outbuf(outlen);
  std::vector<formtype> typeform(inlen);
  for (int i = 0; i < inlen; i++) {
    typeform[i] = provider.PickValueInArray(kTypeforms);
  }

  std::vector<char> spacing(outlen, 0);
  std::vector<int> outputPos(outlen, 0);
  std::vector<int> inputPos(outlen, 0);
  int cursor = provider.ConsumeIntegral<int>();
  int mode = provider.ConsumeIntegral<int>();

  char *inputHyphens = nullptr;
  char *outputHyphens = nullptr;
  std::vector<char> inputHyphens_vec;
  std::vector<char> outputHyphens_vec;

  if (provider.ConsumeBool()) {
    inputHyphens_vec.resize(inlen);
    for (int i = 0; i < inlen; i++) {
      inputHyphens_vec[i] = provider.ConsumeBool() ? '1' : '0';
    }
    outputHyphens_vec.resize(outlen);
    inputHyphens = inputHyphens_vec.data();
    outputHyphens = outputHyphens_vec.data();
  }

  lou_translatePrehyphenated(compiled_table_name, inbuf.data(), &inlen,
                             outbuf.data(), &outlen, typeform.data(),
                             spacing.data(), outputPos.data(), inputPos.data(),
                             &cursor, inputHyphens, outputHyphens, mode);

  // Cleanup
  lou_free();
  unlink(table_filename);

  return 0;
}
