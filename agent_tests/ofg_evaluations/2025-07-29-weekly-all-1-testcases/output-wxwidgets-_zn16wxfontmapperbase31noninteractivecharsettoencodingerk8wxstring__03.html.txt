
	
<description>
The function `wxFontMapperBase::NonInteractiveCharsetToEncoding` translates a character set name, provided as a `wxString`, into the corresponding `wxFontEncoding` enumeration value. The function is designed to be robust against unknown or malformed input. It first attempts to resolve the charset using the application's configuration settings. If that fails, it compares the input string against a comprehensive internal list of known encoding names and their aliases. As a final fallback, it employs pattern matching to identify and parse common encoding formats such as "ISO-8859-X" and "CPXXXX". If all attempts to identify the encoding fail, it returns a system-default value. This function is primarily used internally by other font management components within the wxWidgets library.
</description>

// A subset of known encoding names from gs_encodingNames in fmapbase.cpp
// to test the primary lookup mechanism.
static const char* const g_knownEncodings[] = {
    "ISO-8859-1", "ISO-8859-15", "UTF-8", "UTF-16", "WINDOWS-1250",
    "WINDOWS-1251", "WINDOWS-1252", "CP437", "CP850", "CP932",
    "SHIFT_JIS", "BIG5", "GB2312", "KOI8-R", "KOI8-U", "US-ASCII",
    "MacRoman", "MacCyrillic"
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    // Suppress wxWidgets logging to avoid spamming the output.
    wxLogNull noLog;

    // wxFontMapper requires a wxApp instance to be available, wxInitializer
    // handles the library initialization and cleanup.
    wxInitializer initializer;
    if (!initializer.IsOk()) {
        return 0;
    }

    wxFontMapperBase* mapper = wxFontMapperBase::Get();
    if (!mapper) {
        // Should not happen if wxInitializer was successful.
        return 0;
    }

    FuzzedDataProvider provider(data, size);
    std::string charset_std_str;

    // Decide which type of string to generate to cover different code paths.
    switch (provider.ConsumeIntegralInRange<int>(0, 3)) {
        case 0:
            // Test with a known, valid encoding name.
            charset_std_str = provider.PickValueInArray(g_knownEncodings);
            break;

        case 1: {
            // Test pattern matching for "ISO-8859-N", "WINDOWS-N", "CPN".
            int num = provider.ConsumeIntegral<uint16_t>();
            switch (provider.ConsumeIntegralInRange<int>(0, 2)) {
                case 0:
                    charset_std_str = "ISO-8859-" + std::to_string(num);
                    break;
                case 1:
                    charset_std_str = "WINDOWS-" + std::to_string(num);
                    break;
                case 2:
                    charset_std_str = "CP" + std::to_string(num);
                    break;
            }
            break;
        }

        case 2:
            // Test with a completely random string.
            charset_std_str = provider.ConsumeRemainingBytesAsString();
            break;

        case 3:
            // Test with a random string that may have a prefix of a valid pattern.
            charset_std_str = provider.PickValueInArray({"ISO-", "CP", "WINDOWS-"}) +
                              provider.ConsumeRemainingBytesAsString();
            break;
    }

    // Test whitespace trimming and quote handling.
    if (provider.ConsumeBool()) {
        charset_std_str = "  " + charset_std_str + "\t";
    }
    if (provider.ConsumeBool()) {
        charset_std_str = "\"" + charset_std_str + "\"";
    }

    // Convert the fuzzed string to wxString. wxConvAuto is a robust choice
    // for converting arbitrary byte sequences.
    wxString charset_wxstr(charset_std_str.c_str(), wxConvAuto());

    // Call the public function which in turn calls the protected one.
    // The second parameter is set to false to avoid any interactive dialogs.
    mapper->CharsetToEncoding(charset_wxstr, false);

    return 0;
}
