namespace ot {
namespace Nexus {

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    FuzzedDataProvider provider(data, size);

    Core nexus;
    Node &node = nexus.CreateNode();
    auto &manager = node.GetInstance().Get<BorderRouter::RoutingManager>();

    // Per requirement, Init() must be called.
    // We use fuzzed data to explore different initial states.
    manager.Init(provider.ConsumeIntegral<uint32_t>(), provider.ConsumeBool());

    // Per requirement, Start() (i.e., SetEnabled(true)) must be called.
    // We also enable other related components for a more realistic scenario,
    // mirroring the setup in other OpenThread fuzzers.
    manager.SetEnabled(true);
    node.GetInstance().Get<Srp::Server>().SetAutoEnableMode(true);
    manager.SetDhcp6PdEnabled(true);
#if OPENTHREAD_CONFIG_NAT64_BORDER_ROUTING_ENABLE
    manager.SetNat64PrefixManagerEnabled(true);
#endif
    node.GetInstance().Get<Nat64::Translator>().SetEnabled(true);

    // To get the RoutingManager into a "running" state, the device must be attached.
    // We form a network and advance time to simulate this.
    node.Form();
    nexus.AdvanceTime(provider.ConsumeIntegralInRange<uint32_t>(1000, 60 * 1000));

    // Now that the manager should be running, call the function under test.
    manager.Stop();

    return 0;
}
