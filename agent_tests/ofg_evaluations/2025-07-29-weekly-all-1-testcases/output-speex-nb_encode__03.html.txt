// Required for the general Speex API, including modes and bitstream handling.
#include "speex/speex.h"

// The frame size for narrowband mode is fixed at 160 samples.
#define NB_FRAME_SIZE 160

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    // The function requires an input frame of a fixed size.
    // spx_int16_t is 2 bytes.
    if (size < NB_FRAME_SIZE * sizeof(spx_int16_t)) {
        return 0;
    }

    // 1. Initialize the Speex encoder state for narrowband mode using the public API.
    const SpeexMode *mode = speex_lib_get_mode(SPEEX_MODEID_NB);
    if (!mode) {
        return 0;
    }

    void *encoder_state = speex_encoder_init(mode);
    if (!encoder_state) {
        return 0;
    }

    // 2. Initialize the bit-packing structure.
    SpeexBits bits;
    speex_bits_init(&bits);

    // 3. Create an input audio frame and populate it with fuzzer data.
    std::vector<spx_int16_t> input_frame(NB_FRAME_SIZE);
    provider.ConsumeData(input_frame.data(), input_frame.size() * sizeof(spx_int16_t));

    // 4. Call the target function to encode the frame using the public integer API.
    speex_encode_int(encoder_state, input_frame.data(), &bits);

    // 5. Clean up allocated resources to avoid memory leaks.
    speex_bits_destroy(&bits);
    speex_encoder_destroy(encoder_state);

    return 0;
}
