It is a wrapper for the re-entrant function GEOSMakeValid_r, which in turn calls GEOSMakeValidWithParams_r with default parameters. The function will return a new valid geometry, or `nullptr` if the input geometry is invalid or of a type that is not supported (e.g., curved geometries).
</description>

#include <stdio.h>

static int initialized = 0;
FILE * flogOut;

void
notice(const char *fmt, ...) {
    va_list ap;
    fprintf( flogOut, "NOTICE: ");
    va_start (ap, fmt);
    vfprintf( flogOut, fmt, ap);
    va_end(ap);
    fprintf( flogOut, "\n" );
}

void
log_and_exit(const char *fmt, ...) {
    va_list ap;
    fprintf( flogOut, "ERROR: ");
    va_start (ap, fmt);
    vfprintf( flogOut, fmt, ap);
    va_end(ap);
    fprintf( flogOut, "\n" );
}

int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size) {
    if (initialized == 0) {
        flogOut = fopen("/dev/null", "wb");
        initGEOS(notice, log_and_exit);
        initialized = 1;
    }

    // GEOSGeomFromWKT requires a null-terminated string.
    char *wkt_str = (char *)malloc(Size + 1);
    if (wkt_str == NULL) {
        return 0;
    }
    memcpy(wkt_str, Data, Size);
    wkt_str[Size] = '\0';

    GEOSGeometry *g1 = GEOSGeomFromWKT(wkt_str);
    free(wkt_str);

    if (g1 != NULL) {
        GEOSGeometry *g_valid = GEOSMakeValid(g1);

        if (g_valid != NULL) {
            GEOSGeom_destroy(g_valid);
        }
        GEOSGeom_destroy(g1);
    }
    return 0;
}
