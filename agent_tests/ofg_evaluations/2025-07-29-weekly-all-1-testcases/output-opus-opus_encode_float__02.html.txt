
	
<description>
The `opus_encode_float` function encodes a frame of Opus audio from floating-point PCM data. It takes an initialized `OpusEncoder` state, a buffer of PCM audio samples, the number of samples per channel, an output buffer for the compressed data, and the maximum size of the output buffer. The function relies on the `OpusEncoder` state being properly configured for aspects like sample rate, channels, and application type. The `analysis_frame_size` must be a valid frame size supported by the encoder, which is typically a duration of 2.5, 5, 10, 20, 40, or 60 ms of audio data. The function returns the number of bytes written to the output buffer on success, or an error code on failure.
</description>

#include <fuzzer/FuzzedDataProvider.h>
#include <vector>
#include <cstdint>
#include <cstdlib>

#include "/src/opus/include/opus.h"

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider fdp(data, size);

    const opus_int32 sample_rates[] = {8000, 12000, 16000, 24000, 48000};
    const int channels[] = {1, 2};
    const int applications[] = {OPUS_APPLICATION_VOIP, OPUS_APPLICATION_AUDIO, OPUS_APPLICATION_RESTRICTED_LOWDELAY};

    const opus_int32 Fs = fdp.PickValueInArray(sample_rates);
    const int num_channels = fdp.PickValueInArray(channels);
    const int application = fdp.PickValueInArray(applications);

    const double frame_durations_ms[] = {2.5, 5, 10, 20, 40, 60};
    const double frame_duration_ms = fdp.PickValueInArray(frame_durations_ms);
    const int frame_size = static_cast<int>(Fs / 1000.0 * frame_duration_ms);

    int err;
    OpusEncoder *enc = opus_encoder_create(Fs, num_channels, application, &err);
    if (err != OPUS_OK || !enc) {
        return 0;
    }

    const size_t pcm_size_floats = frame_size * num_channels;
    const size_t pcm_size_bytes = pcm_size_floats * sizeof(float);

    std::vector<uint8_t> pcm_raw = fdp.ConsumeRemainingBytes<uint8_t>();
    if (pcm_raw.size() < pcm_size_bytes) {
        opus_encoder_destroy(enc);
        return 0;
    }

    const float *pcm = reinterpret_cast<const float*>(pcm_raw.data());
    
    const opus_int32 max_data_bytes = 4000;
    unsigned char out_data[max_data_bytes];

    opus_encode_float(enc, pcm, frame_size, out_data, max_data_bytes);

    opus_encoder_destroy(enc);
    return 0;
}
