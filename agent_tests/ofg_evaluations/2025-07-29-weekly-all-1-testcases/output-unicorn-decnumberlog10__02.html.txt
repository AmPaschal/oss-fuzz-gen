// Helper function to allocate a decNumber structure with enough space
// for the coefficient, based on the number of digits in the context.
// The decNumber struct itself only has space for one unit, so for
// higher precision, more space must be allocated.
static decNumber *allocate_decNumber_from_context(const decContext *context) {
    // DECDPUN is the number of decimal digits per decNumberUnit. Its value is 3.
    // We need to calculate how many units are needed for the given precision.
    int32_t units = (context->digits + DECDPUN - 1) / DECDPUN;
    
    // The decNumber struct contains lsu[1]. We need to allocate space for
    // (units - 1) additional units.
    size_t extra_units = (units > 1) ? (units - 1) : 0;
    size_t size = sizeof(decNumber) + extra_units * sizeof(decNumberUnit);
    
    decNumber *dn = (decNumber *)malloc(size);
    return dn;
}
