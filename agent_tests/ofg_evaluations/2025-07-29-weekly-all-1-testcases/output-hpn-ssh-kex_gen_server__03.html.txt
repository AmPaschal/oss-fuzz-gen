
	
<description>
The `kex_gen_server` function is responsible for setting up the key exchange process on the server side of an SSH connection. It accomplishes this by registering the `input_kex_gen_init` function as the handler for the `SSH2_MSG_KEX_ECDH_INIT` message. The actual key exchange logic, including the generation of ephemeral keys, verification of client data, and derivation of session keys, is handled within `input_kex_gen_init`. The behavior of `input_kex_gen_init` is dependent on the key exchange algorithm specified in the `kex->kex_type` field of the `ssh` struct.
</description>
<requirements>

<requirement>
The `ssh` struct should be initialized by calling `ssh_new`. This is because `ssh_new` allocates and initializes the necessary memory for the struct, preventing null pointer dereferences.
</requirement>
<requirement>
The `ssh->kex` member must be a valid, initialized `kex` struct. This is because the function accesses `ssh->kex` to set up the key exchange handler, and a null or uninitialized pointer would lead to a crash.
</requirement>
<requirement>
The `ssh->kex->kex_type` field must be set to one of the supported key exchange algorithm constants (e.g., `KEX_DH_GRP1_SHA1`, `KEX_ECDH_SHA2`). This is because this field determines which key exchange algorithm is used in the subsequent `input_kex_gen_init` function.
</requirement>
<requirement>
The `ssh` struct must have a valid host key loaded before calling `kex_gen_server`. This is because `input_kex_gen_init` will call `kex_load_hostkey` which expects a host key to be available.
</requirement>
<requirement>
A packet handler should be configured for the `ssh` object to provide data for the `SSH2_MSG_KEX_ECDH_INIT` message. This is because `input_kex_gen_init` will attempt to read from the packet payload, and a lack of data will cause the key exchange to fail.
</requirement>

struct shared_state {
  size_t nkeys;
  struct sshkey **privkeys;
};

static void store_key(struct shared_state *st, struct sshkey *privkey) {
  if (st == NULL || privkey->type < 0 || privkey->type > INT_MAX)
    abort();
  if ((size_t)privkey->type >= st->nkeys) {
    st->privkeys = (struct sshkey **)xrecallocarray(st->privkeys, st->nkeys, privkey->type + 1, sizeof(*st->privkeys));
    st->nkeys = privkey->type + 1;
  }
  st->privkeys[privkey->type] = privkey;
}

static int prepare_key(struct shared_state *st, int kt) {
  const char *privstr = NULL;
  struct sshkey *privkey = NULL;
  struct sshbuf *b = NULL;
  int r;

  switch (kt) {
  case KEY_RSA:
    privstr = PRIV_RSA;
    break;
  case KEY_DSA:
    privstr = PRIV_DSA;
    break;
  case KEY_ECDSA:
    privstr = PRIV_ECDSA;
    break;
  case KEY_ED25519:
    privstr = PRIV_ED25519;
    break;
  default:
    abort();
  }
  if ((b = sshbuf_from(privstr, strlen(privstr))) == NULL)
    abort();
  if ((r = sshkey_parse_private_fileblob(b, "", &privkey, NULL)) != 0) {
    sshbuf_free(b);
    abort();
  }
  sshbuf_free(b);
  store_key(st, privkey);
  return 0;
}

static int prepare_keys(struct shared_state *st) {
  if (prepare_key(st, KEY_RSA) != 0 || prepare_key(st, KEY_DSA) != 0 || prepare_key(st, KEY_ECDSA) != 0 || prepare_key(st, KEY_ED25519) != 0) {
    return -1;
  }
  return 0;
}

static struct sshkey *get_privkey(struct shared_state *st, int keytype) {
  if (st == NULL || keytype < 0 || (size_t)keytype >= st->nkeys || st->privkeys == NULL || st->privkeys[keytype] == NULL)
    return NULL;
  return st->privkeys[keytype];
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  static struct shared_state *st;
  static int initialized = 0;
  struct ssh *server = NULL;
  struct sshbuf *packet = NULL;
  char *myproposal[PROPOSAL_MAX] = {0};
  int r;
  sig_atomic_t quit = 0;

  if (size < 2) {
    return 0;
  }

  if (!initialized) {
    log_init("kex_fuzz", SYSLOG_LEVEL_ERROR, SYSLOG_FACILITY_AUTH, 1);
    st = (struct shared_state *)xcalloc(1, sizeof(*st));
    if (prepare_keys(st) != 0) {
      return 0;
    }
    initialized = 1;
  }

  const char *kex_algs[] = {
      "ecdh-sha2-nistp256", "ecdh-sha2-nistp384", "ecdh-sha2-nistp521", "curve25519-sha256@libssh.org", "sntrup761x25519-sha512@openssh.com", "diffie-hellman-group-exchange-sha256", "diffie-hellman-group14-sha256", "diffie-hellman-group16-sha512", "diffie-hellman-group18-sha512",
  };
  const size_t kex_algs_len = sizeof(kex_algs) / sizeof(kex_algs[0]);
  const char *kex_alg = kex_algs[data[0] % kex_algs_len];

  const int key_types[] = {KEY_RSA, KEY_ECDSA, KEY_ED25519, KEY_DSA};
  const size_t key_types_len = sizeof(key_types) / sizeof(key_types[0]);
  int hostkey_type = key_types[data[1] % key_types_len];
  struct sshkey *hostkey = get_privkey(st, hostkey_type);

  const uint8_t *packet_payload = data + 2;
  size_t packet_payload_size = size - 2;

  if (hostkey == NULL) {
    return 0;
  }

  struct kex_params kex_params;
  const char *proposal_template[PROPOSAL_MAX] = {KEX_SERVER};

  for (int i = 0; i < PROPOSAL_MAX; i++) {
    const char *ccp = proposal_template[i];
    if (i == PROPOSAL_KEX_ALGS) {
      ccp = kex_alg;
    } else if (i == PROPOSAL_SERVER_HOST_KEY_ALGS) {
      ccp = sshkey_ssh_name(hostkey);
    }
    if ((myproposal[i] = strdup(ccp)) == NULL) {
      goto cleanup;
    }
  }
  memcpy(kex_params.proposal, myproposal, sizeof(myproposal));

  if ((r = ssh_init(&server, 1, &kex_params)) != 0) {
    goto cleanup;
  }
  if ((r = ssh_add_hostkey(server, hostkey)) != 0) {
    goto cleanup;
  }

  packet = sshbuf_new();
  if (packet == NULL) {
    goto cleanup;
  }

  if (sshbuf_put_u8(packet, SSH2_MSG_KEX_ECDH_INIT) != 0 || sshbuf_put_string(packet, packet_payload, packet_payload_size) != 0) {
    goto cleanup;
  }

  if (ssh_input_append(server, sshbuf_ptr(packet), sshbuf_len(packet)) != 0) {
    goto cleanup;
  }

  kex_gen_server(server);

  ssh_dispatch_run(server, DISPATCH_NONBLOCK, &quit);

cleanup:
  for (int i = 0; i < PROPOSAL_MAX; i++) {
    free(myproposal[i]);
  }
  sshbuf_free(packet);
  ssh_free(server);

  return 0;
}
