// A custom decoder class that inherits from FLAC::Decoder::File
// and implements the required pure virtual callback functions.
class FuzzerFileDecoder : public FLAC::Decoder::File {
public:
  FuzzerFileDecoder() = default;
  ~FuzzerFileDecoder() override = default;

protected:
  // Override for the pure virtual write_callback from the base Stream class.
  // This function is called when decoded audio data is available.
  ::FLAC__StreamDecoderWriteStatus
  write_callback(const ::FLAC__Frame * /*frame*/,
                 const FLAC__int32 *const /*buffer*/[]) override {
    // We don't need to process the audio data for this fuzzer.
    // Just tell the decoder to continue.
    return FLAC__STREAM_DECODER_WRITE_STATUS_CONTINUE;
  }

  // Override for the pure virtual error_callback from the base Stream class.
  // This function is called when a decoding error occurs.
  void error_callback(::FLAC__StreamDecoderErrorStatus /*status*/) override {
    // For fuzzing, we can simply ignore the errors.
  }
};
