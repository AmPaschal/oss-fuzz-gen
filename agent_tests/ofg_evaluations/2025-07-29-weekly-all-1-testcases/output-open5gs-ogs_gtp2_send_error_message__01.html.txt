
	
<description>
The function `ogs_gtp2_send_error_message` is responsible for sending GTPv2 error messages. It constructs an error message of a specific type and cause, and then sends it using the provided transaction context. The function uses a switch statement to handle different message types and will terminate the program if an invalid message type is provided.
</description>
<requirements>

<requirement>
The `xact` parameter must be a valid, non-NULL pointer to a `ogs_gtp_xact_t` struct. This is because the function dereferences this pointer to access its members and passes it to other functions. A NULL or invalid pointer will cause a crash.
</requirement>
<requirement>
The `gtp_version` member of the `xact` struct should be initialized to 2. This is because the calling function `ogs_gtp_send_error_message` will only call the target function if `xact->gtp_version` is 2.
</requirement>
<requirement>
The `type` parameter must be one of the following values: `OGS_GTP2_CREATE_SESSION_RESPONSE_TYPE` (33), `OGS_GTP2_MODIFY_BEARER_RESPONSE_TYPE` (35), `OGS_GTP2_DELETE_SESSION_RESPONSE_TYPE` (37), `OGS_GTP2_RELEASE_ACCESS_BEARERS_RESPONSE_TYPE` (171), `OGS_GTP2_DOWNLINK_DATA_NOTIFICATION_ACKNOWLEDGE_TYPE` (178), `OGS_GTP2_CREATE_BEARER_RESPONSE_TYPE` (96), `OGS_GTP2_UPDATE_BEARER_RESPONSE_TYPE` (98), `OGS_GTP2_DELETE_BEARER_RESPONSE_TYPE` (100), `OGS_GTP2_CREATE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE` (165), `OGS_GTP2_DELETE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE` (167), `OGS_GTP2_MODIFY_BEARER_FAILURE_INDICATION_TYPE` (70), `OGS_GTP2_DELETE_BEARER_FAILURE_INDICATION_TYPE` (72), or `OGS_GTP2_BEARER_RESOURCE_FAILURE_INDICATION_TYPE` (74). This is because the function uses a switch statement to handle different message types and will terminate the program if an invalid message type is provided.
</requirement>
<requirement>
The `cause_value` parameter can be fuzzed with any `uint8_t` value. This is because the function does not impose any constraints on this value.
</requirement>

</requirements>

#include <stdio.h>
#include <stdint.h>
#include <string.h>

#include "fuzzing.h"
#include "ogs-gtp.h"

#define kMinInputLength 6

static const uint8_t valid_types[] = {
    OGS_GTP2_CREATE_SESSION_RESPONSE_TYPE,
    OGS_GTP2_MODIFY_BEARER_RESPONSE_TYPE,
    OGS_GTP2_DELETE_SESSION_RESPONSE_TYPE,
    OGS_GTP2_RELEASE_ACCESS_BEARERS_RESPONSE_TYPE,
    OGS_GTP2_DOWNLINK_DATA_NOTIFICATION_ACKNOWLEDGE_TYPE,
    OGS_GTP2_CREATE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_UPDATE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_DELETE_BEARER_RESPONSE_TYPE,
    OGS_GTP2_CREATE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE,
    OGS_GTP2_DELETE_INDIRECT_DATA_FORWARDING_TUNNEL_RESPONSE_TYPE,
    OGS_GTP2_MODIFY_BEARER_FAILURE_INDICATION_TYPE,
    OGS_GTP2_DELETE_BEARER_FAILURE_INDICATION_TYPE,
    OGS_GTP2_BEARER_RESOURCE_FAILURE_INDICATION_TYPE,
};
static const size_t num_valid_types = sizeof(valid_types) / sizeof(valid_types[0]);

extern int LLVMFuzzerTestOneInput(const uint8_t *Data, size_t Size)
{
    if (Size < kMinInputLength) {
        return 0;
    }

    if (!initialized) {
        initialize();
        ogs_log_install_domain(&__ogs_gtp_domain, "gtp", OGS_LOG_NONE);
        ogs_log_install_domain(&__ogs_tlv_domain, "tlv", OGS_LOG_NONE);
    }

    ogs_gtp_xact_t xact;
    memset(&xact, 0, sizeof(ogs_gtp_xact_t));
    xact.gtp_version = 2;

    uint32_t teid;
    memcpy(&teid, Data, sizeof(uint32_t));

    uint8_t type_index = Data[sizeof(uint32_t)];
    uint8_t type = valid_types[type_index % num_valid_types];

    uint8_t cause_value = Data[sizeof(uint32_t) + 1];

    ogs_gtp2_send_error_message(&xact, teid, type, cause_value);

    return 0;
}
