
	
<description>
The function `cmsCreateProofingTransformTHR` creates a color transformation for proofing. It can operate in two modes depending on the `dwFlags` parameter. If `dwFlags` contains either `cmsFLAGS_SOFTPROOFING` or `cmsFLAGS_GAMUTCHECK`, it creates a complex proofing transform by calling `cmsCreateExtendedTransform`. Otherwise, it creates a simpler transform by calling `cmsCreateTransformTHR`. The function is a thread-safe version of `cmsCreateProofingTransform`, and the caller `cmsCreateProofingTransform` retrieves the context ID from the input profile.
</description>
<requirements>

<requirement>
The `ContextID` argument can be set to NULL. The lcms2 engine is designed to use a default global context if this argument is NULL.
</requirement>
<requirement>
The `InputProfile`, `OutputProfile`, and `ProofingProfile` arguments must be valid `cmsHPROFILE` handles. These can be created using functions such as `cmsCreateSRGBProfile()` or `cmsOpenProfileFromFile()` to ensure they point to valid profile structures.
</requirement>
<requirement>
The `InputFormat` and `OutputFormat` arguments must be valid format specifiers. These are defined as `TYPE_` macros in `lcms2.h`, for example `TYPE_RGB_8` or `TYPE_CMYK_16`.
</requirement>
<requirement>
The `nIntent` and `ProofingIntent` arguments must be valid rendering intents. The valid values are `INTENT_PERCEPTUAL` (0), `INTENT_RELATIVE_COLORIMETRIC` (1), `INTENT_SATURATION` (2), and `INTENT_ABSOLUTE_COLORIMETRIC` (3).
</requirement>
<requirement>
The `dwFlags` argument should be a combination of `cmsFLAGS_` macros to control the transform creation. To properly exercise the proofing transform logic, `dwFlags` must include `cmsFLAGS_SOFTPROOFING` (0x4000) or `cmsFLAGS_GAMUTCHECK` (0x1000).
</requirement>

</requirements>

#include <stdint.h>
#include "/src/lcms/include/lcms2.h"

cmsHTRANSFORM cmsCreateProofingTransformTHR(cmsContext, cmsHPROFILE, cmsUInt32Number, cmsHPROFILE, cmsUInt32Number, cmsHPROFILE, cmsUInt32Number, cmsUInt32Number, cmsUInt32Number);

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

    if (size < 20) {
        return 0;
    }

    uint32_t dwFlags = *(const uint32_t *)(data);
    uint32_t nIntent = *(const uint32_t *)(data + 4) % 4;
    uint32_t ProofingIntent = *(const uint32_t *)(data + 8) % 4;
    uint32_t format_choice = *(const uint32_t *)(data + 12);

    dwFlags |= cmsFLAGS_SOFTPROOFING;

    uint32_t InputFormat;
    uint32_t OutputFormat;

    if (format_choice % 2 == 0) {
        InputFormat = TYPE_RGB_8;
        OutputFormat = TYPE_CMYK_8;
    } else {
        InputFormat = TYPE_CMYK_16;
        OutputFormat = TYPE_RGB_16;
    }

    data += 16;
    size -= 16;

    if (size < 2) {
        return 0;
    }

    const size_t size1 = size / 2;
    const uint8_t *data1 = data;
    const size_t size2 = size - size1;
    const uint8_t *data2 = data + size1;

    cmsHPROFILE hInputProfile = cmsOpenProfileFromMem(data1, size1);
    if (hInputProfile == NULL) {
        return 0;
    }

    cmsHPROFILE hProofingProfile = cmsOpenProfileFromMem(data2, size2);
    if (hProofingProfile == NULL) {
        cmsCloseProfile(hInputProfile);
        return 0;
    }

    cmsHPROFILE hOutputProfile = cmsCreate_sRGBProfile();
    if (hOutputProfile == NULL) {
        cmsCloseProfile(hInputProfile);
        cmsCloseProfile(hProofingProfile);
        return 0;
    }

    cmsHTRANSFORM hTransform = cmsCreateProofingTransformTHR(NULL,
                                                            hInputProfile,
                                                            InputFormat,
                                                            hOutputProfile,
                                                            OutputFormat,
                                                            hProofingProfile,
                                                            nIntent,
                                                            ProofingIntent,
                                                            dwFlags);

    if (hTransform != NULL) {
        cmsDeleteTransform(hTransform);
    }

    cmsCloseProfile(hInputProfile);
    cmsCloseProfile(hOutputProfile);
    cmsCloseProfile(hProofingProfile);

    return 0;
}
