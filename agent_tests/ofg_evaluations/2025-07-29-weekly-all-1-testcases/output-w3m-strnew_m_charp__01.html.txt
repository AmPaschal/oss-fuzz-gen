int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

#if (GC_VERSION_MAJOR > 7) || ((GC_VERSION_MAJOR == 7) && (GC_VERSION_MINOR >= 2))

    GC_oom_fn = die_oom;

  // To fuzz the variadic function, we split the input into
  // several null-terminated strings.
#define MAX_ARGS 6
  const char *args[MAX_ARGS];
  int arg_count = 0;

  const char *p = (const char *)data;
  const char *end = p + size;

  while (p < end && arg_count < MAX_ARGS) {
    args[arg_count++] = p;
    const char *next_null = (const char *)memchr(p, '\0', end - p);

      // No more complete strings in the buffer.
      break;
    }
    p = next_null + 1;

  // how many strings we parsed. Each call must be terminated by NULL.

    break;

    s = Strnew_m_charp(args[0], NULL);
    break;
  case 2:

    break;

    break;

    s = Strnew_m_charp(args[0], args[1], args[2], args[3], args[4], NULL);
    break;
  case 6:
    s = Strnew_m_charp(args[0], args[1], args[2], args[3], args[4], args[5], NULL);
    break;
  }

  if (s) {
