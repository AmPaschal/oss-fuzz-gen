    if (size == 0) {
        return 0;

    // Create a temporary file to act as the configuration file.
    char filename[] = "/tmp/fuzz.cfg.XXXXXX";
    int fd = mkstemp(filename);
    if (fd < 0) {
        return 0;
    }

    // Write the fuzzing data to the config file and close it.
    ssize_t written = write(fd, data, size);
    close(fd);

    if (written != (ssize_t)size) {
        unlink(filename);
        return 0;
    }

    // Prepare argc and argv for haproxy's main function.
    // We use "-f" to specify our fuzz-generated config file and "-d" to
    // enable debug mode, which helps prevent daemonization.
    char *argv[] = {
        (char*)"haproxy",
        (char*)"-f",
        filename,
        (char*)"-d",
        NULL
    };
    int argc = 4;

    // Call the init function as required.
    init(argc, argv);

    // Clean up the temporary file.
    unlink(filename);

    return 0;
}
