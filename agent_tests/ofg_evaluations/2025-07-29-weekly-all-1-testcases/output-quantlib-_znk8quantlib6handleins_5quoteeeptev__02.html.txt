using namespace QuantLib;

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider fdp(data, size);
  // SavedSettings ensures that any changes to global settings are
  // reverted at the end of the scope, providing a clean slate for each run.
  SavedSettings saved_settings;

  // Create a SimpleQuote with a value from the fuzzer.
  Real quote_value = fdp.ConsumeFloatingPoint<Real>();
  ext::shared_ptr<SimpleQuote> simple_quote =
      ext::make_shared<SimpleQuote>(quote_value);

  // Create a Handle to the Quote. This handle will not be empty.
  Handle<Quote> quote_handle(simple_quote);

  // Call the function under test: operator->
  // This will access the underlying shared_ptr.
  // If the handle were empty, this would trigger an assertion.
  try {
    const ext::shared_ptr<Quote> &quote_ptr = quote_handle.operator->();
    // We can optionally use the returned pointer to do something,
    // but the call itself is the primary test.
    (void)quote_ptr->value();
  } catch (const std::exception &e) {
    // Catch potential exceptions from QuantLib's error handling.
  }

  return 0;
}
