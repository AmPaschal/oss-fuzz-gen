
	
<description>
The function `zip_source_window_create` creates a new zip source from a window (a sub-range) of another zip source. The new source reads data from the original source, starting at the `start` offset and reading `len` bytes. If `len` is -1, it reads until the end of the source.
</description>
<requirements>
<requirement>
The first argument `src` should be a valid `zip_source_t` pointer. This can be created using `zip_source_buffer` or `zip_source_file`.
</requirement>
<requirement>
The `start` argument can be any `zip_uint64_t` value. However, it should be a valid offset within the provided `src`.
</requirement>
<requirement>
The `len` argument should be a `zip_int64_t` value greater than or equal to -1. A value of -1 means the window extends to the end of the source. An integer overflow check is performed for `start + len`.
</requirement>
<requirement>
The `error` argument is a pointer to a `zip_error_t` structure and can be NULL. If it is not NULL, it will be used to store error information.
</requirement>
</requirements>

int
LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    const size_t start_size = sizeof(zip_uint64_t);
    const size_t len_size = sizeof(zip_int64_t);

    if (size < start_size + len_size) {
        return 0;
    }

    zip_uint64_t start;
    zip_int64_t len;

    memcpy(&start, data, start_size);
    memcpy(&len, data + start_size, len_size);

    const uint8_t *source_data = data + start_size + len_size;
    const size_t source_size = size - (start_size + len_size);

    zip_error_t error;
    zip_error_init(&error);

    zip_source_t *src = zip_source_buffer_create(source_data, source_size, 0, &error);
    if (src == NULL) {
        zip_error_fini(&error);
        return 0;
    }

    zip_source_t *window_src = zip_source_window_create(src, start, len, &error);

    if (window_src != NULL) {
        zip_source_free(window_src);
    } else {
        zip_source_free(src);
    }

    zip_error_fini(&error);

    return 0;
}
