
	
<description>
The function `avahi_string_list_add_pair` is a utility function that adds a key-value pair to an `AvahiStringList`. If the `value` is NULL, it adds only the `key`. Otherwise, it adds the pair in the format "key=value". The function returns the head of the modified string list.
</description>
<requirements>
<requirement>
The first argument, `l`, which is the string list, can be NULL. If it is NULL, a new list is created.
</requirement>
<requirement>
The `key` argument must not be NULL. The function has an assertion `assert(key)` that will cause a crash if `key` is NULL.
</requirement>
<requirement>
The `value` argument can be NULL. The function handles this by adding only the key to the list.
</requirement>
<requirement>
The `key` and `value` arguments should be null-terminated strings. This is because they are passed to `avahi_string_list_add_printf` which expects format strings.
</requirement>
</requirements>

#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

#include "avahi-common/malloc.h"
#include "avahi-core/log.h"
#include "/src/avahi/avahi-common/strlst.h"

// Dummy log function to prevent log output during fuzzing
void log_function(AvahiLogLevel level, const char *txt) {
    (void)level;
    (void)txt;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    avahi_set_log_function(log_function);

    // The key cannot be NULL, so we need at least one byte for it.
    if (size == 0) {
        return 0;
    }

    char *key = NULL;
    char *value = NULL;

    // Split the input data into key and value at the first NUL byte.
    const uint8_t *separator = memchr(data, '\0', size);

    if (separator != NULL) {
        size_t key_len = separator - data;

        // The key must not be empty to be meaningful, and the function asserts the pointer.
        if (key_len == 0) {
            return 0;
        }

        key = avahi_malloc(key_len + 1);
        memcpy(key, data, key_len);
        key[key_len] = '\0';

        size_t value_len = size - key_len - 1;
        if (value_len > 0) {
            value = avahi_malloc(value_len + 1);
            memcpy(value, separator + 1, value_len);
            value[value_len] = '\0';
        }
    } else {
        // No separator found, use the entire input as the key and value as NULL.
        key = avahi_malloc(size + 1);
        memcpy(key, data, size);
        key[size] = '\0';
    }

    AvahiStringList *slist = NULL;

    AvahiStringList * avahi_string_list_add_pair(AvahiStringList *, const char *, const char *);

    // Test with a NULL list, which should create a new one.
    slist = avahi_string_list_add_pair(slist, key, value);

    // Test adding to an existing list.
    if (slist) {
        slist = avahi_string_list_add_pair(slist, key, value);
    }

    // Free the allocated resources.
    if (slist) {
        avahi_string_list_free(slist);
    }
    avahi_free(key);
    avahi_free(value); // avahi_free(NULL) is safe.

    return 0;
}
