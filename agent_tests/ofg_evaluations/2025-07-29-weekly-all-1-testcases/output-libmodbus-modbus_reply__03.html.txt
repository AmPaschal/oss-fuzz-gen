
	
<description>
The `modbus_reply` function is responsible for handling Modbus requests and sending corresponding responses. It takes a Modbus context, a request packet, the request length, and a Modbus mapping as input. The function parses the request, performs the requested action based on the function code, and constructs a response packet. The response is then sent back to the Modbus master.
</description>
<requirements>
<requirement>
The `ctx` argument, a `modbus_t` pointer, must be initialized by `modbus_new_tcp` or `modbus_new_rtu`. This is to ensure that the Modbus context is properly configured for communication.
</requirement>
<requirement>
The `mb_mapping` argument, a `modbus_mapping_t` pointer, must be initialized by `modbus_mapping_new` or `modbus_mapping_new_start_address`. This is to allocate the necessary memory for the Modbus mappings.
</requirement>
<requirement>
The `req` argument should be a valid Modbus request. This is because the function expects a certain structure and content in the request to process it correctly.
</requirement>
<requirement>
The `req_length` argument must be a valid length for the request `req`. An incorrect length can lead to buffer overflows or other memory-related errors.
</requirement>
</requirements>

#define PORT 8080
#define kMinInputLength 9
#define kMaxInputLength MODBUS_RTU_MAX_ADU_LENGTH

struct Fuzzer{
    uint16_t    port;
    const uint8_t*    buffer;
    size_t      size;

    pthread_t   thread;
};
typedef struct Fuzzer Fuzzer;

int server(Fuzzer *fuzzer);

void *client(void *args){

    Fuzzer *fuzzer = (Fuzzer*)args;
    int sockfd;
    struct sockaddr_in serv_addr;

    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    serv_addr.sin_family = AF_INET;
    serv_addr.sin_port = htons(fuzzer->port);
    serv_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

    while(1){/* Try until connect*/
        if (connect(sockfd, (struct sockaddr *)&serv_addr, sizeof(serv_addr)) < 0){
            continue;
        }else{
            break;
        }
    }

    send(sockfd,fuzzer->buffer,fuzzer->size,0);

    close(sockfd);
    pthread_exit(NULL);
}

int modbus_reply(modbus_t *, const uint8_t *, int, modbus_mapping_t *);

extern int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

    if (size < kMinInputLength || size > kMaxInputLength){
        return 0;
    }

    Fuzzer *fuzzer = (Fuzzer*)malloc(sizeof(Fuzzer));
    fuzzer->port = PORT;
    fuzzer->buffer = data;
    fuzzer->size = size;

    pthread_create(&fuzzer->thread, NULL,client,fuzzer);
    server(fuzzer);
    pthread_join(fuzzer->thread, NULL); /* Avoid UAF*/

    free(fuzzer);
    return 0;
}

int server(Fuzzer *fuzzer)
{
    int s = -1;
    modbus_t *ctx;
    modbus_mapping_t *mb_mapping;
    int rc;
    uint8_t *query;

    ctx = modbus_new_tcp("127.0.0.1", fuzzer->port);
    if (ctx == NULL) {
        return -1;
    }
    query = malloc(MODBUS_TCP_MAX_ADU_LENGTH);
    if (query == NULL) {
        modbus_free(ctx);
        return -1;
    }

    mb_mapping = modbus_mapping_new_start_address(
        UT_BITS_ADDRESS, UT_BITS_NB,
        UT_INPUT_BITS_ADDRESS, UT_INPUT_BITS_NB,
        UT_REGISTERS_ADDRESS, UT_REGISTERS_NB_MAX,
        UT_INPUT_REGISTERS_ADDRESS, UT_INPUT_REGISTERS_NB);

    if (mb_mapping == NULL) {
        free(query);
        modbus_free(ctx);
        return -1;
    }

    s = modbus_tcp_listen(ctx, 1);
    if (s == -1) {
        modbus_mapping_free(mb_mapping);
        free(query);
        modbus_free(ctx);
        return -1;
    }
    modbus_tcp_accept(ctx, &s);

    rc = modbus_receive(ctx, query);
    if (rc > 0) {
        modbus_reply(ctx, query, rc, mb_mapping);
    }

    if (s != -1) {
        close(s);
    }

    modbus_mapping_free(mb_mapping);
    free(query);
    modbus_close(ctx);
    modbus_free(ctx);

    return 0;
}
