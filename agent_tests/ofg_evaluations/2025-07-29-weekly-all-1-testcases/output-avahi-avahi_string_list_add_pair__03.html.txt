
	
<description>
The function `avahi_string_list_add_pair` adds a key-value pair to an `AvahiStringList`. If the input string list `l` is NULL, it creates a new list. The `key` must be a non-NULL string, while the `value` can be either NULL or a non-NULL string. The function uses `avahi_string_list_add_printf` to add the pair when the value is not NULL, and `avahi_string_list_add` when the value is NULL.
</description>
<requirements>
<requirement>
The first argument, `l`, can be a NULL pointer. If `l` is NULL, the function will allocate a new `AvahiStringList`.
</requirement>
<requirement>
The second argument, `key`, must be a non-NULL, null-terminated string. This is because the function asserts that `key` is not NULL before using it.
</requirement>
<requirement>
The third argument, `value`, can be a NULL pointer or a null-terminated string. The function handles both cases by either adding a key-only entry or a key-value pair.
</requirement>
</requirements>

AvahiStringList * avahi_string_list_add_pair(AvahiStringList *, const char *, const char *);

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size == 0) {
        return 0;
    }

    // Split the input data to create a key and a value string.
    const size_t key_len = size / 2;
    const size_t value_len = size - key_len;
    const uint8_t *value_data = data + key_len;

    // The function asserts that key is not NULL. We allocate a buffer
    // and create a null-terminated string.
    char *key = (char *) malloc(key_len + 1);
    if (!key) {
        return 0;
    }
    memcpy(key, data, key_len);
    key[key_len] = '\0';

    // The value can be NULL. We'll create a string from the fuzzer input.
    char *value = (char *) malloc(value_len + 1);
    if (!value) {
        free(key);
        return 0;
    }
    memcpy(value, value_data, value_len);
    value[value_len] = '\0';

    // --- Test Case 1: Initial list is NULL ---
    AvahiStringList *l1 = NULL;
    l1 = avahi_string_list_add_pair(l1, key, value);
    if (l1) {
        avahi_string_list_free(l1);
    }

    // --- Test Case 2: Initial list is NULL and value is NULL ---
    AvahiStringList *l2 = NULL;
    l2 = avahi_string_list_add_pair(l2, key, NULL);
    if (l2) {
        avahi_string_list_free(l2);
    }

    // --- Test Case 3: Add to a pre-existing list ---
    AvahiStringList *l3 = avahi_string_list_add(NULL, "pre-existing-entry");
    l3 = avahi_string_list_add_pair(l3, key, value);
    if (l3) {
        avahi_string_list_free(l3);
    }

    free(key);
    free(value);

    return 0;
}
