#include <fuzzer/FuzzedDataProvider.h>
#include "glslang/Public/ResourceLimits.h"
#include "glslang/Public/ShaderLang.h"

// The function under test, glslang::HlslParseContext::parseShaderStrings,
// is called indirectly by glslang::TShader::parse() when the correct
// HLSL message flags are set.

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size == 0) {
        return 0;
    }
    FuzzedDataProvider provider(data, size);

    glslang::InitializeProcess();

    // Fuzz the shader stage
    const EShLanguage stage = provider.PickValueInArray<EShLanguage>({
        EShLangVertex,
        EShLangTessControl,
        EShLangTessEvaluation,
        EShLangGeometry,
        EShLangFragment,
        EShLangCompute,
    });

    glslang::TShader shader(stage);
    
    // Set HLSL-specific options.
    shader.setHlslIoMapping(true);
    shader.setEnvTargetHlslFunctionality1();

    // Provide the shader source from the fuzzer input
    std::string shader_source = provider.ConsumeRemainingBytesAsString();
    const char* shader_strings[] = {shader_source.c_str()};
    const int shader_lengths[] = {static_cast<int>(shader_source.length())};
    shader.setStringsWithLengths(shader_strings, shader_lengths, 1);

    shader.setEntryPoint("main");

    // Add EShMsgReadHlsl to use HLSL parsing rules and semantics.
    // This is the key to ensuring the target function is called.
    EShMessages messages = (EShMessages)(EShMsgSpvRules | EShMsgVulkanRules | EShMsgHlslLegalization | EShMsgReadHlsl);

    // This call will create an HlslParseContext and invoke the target function
    // `parseShaderStrings` internally because EShMsgReadHlsl is set.
    shader.parse(GetDefaultResources(), 100, false, messages);

    glslang::FinalizeProcess();
    return 0;
}
