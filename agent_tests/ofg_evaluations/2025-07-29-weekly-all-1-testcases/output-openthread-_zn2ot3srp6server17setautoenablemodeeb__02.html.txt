The function `SetAutoEnableMode` is used to enable or disable the auto-enable mode for the SRP server. When called, it first checks if the new auto-enable mode is different from the current mode. If so, it updates the internal state `mAutoEnable` and notifies the `RoutingManager` of the change by calling `HandleSrpServerAutoEnableMode()`. If the compile-time option `OPENTHREAD_CONFIG_SRP_SERVER_FAST_START_MODE_ENABLE` is enabled, it also calls `DisableFastStartMode()` to disable the fast start mode.
</description>

// This custom FuzzDataProvider is modeled after the one used in the project's
// existing fuzzers, like `radio-one-node.cpp`.
class FuzzDataProvider
{
public:
    FuzzDataProvider(const uint8_t *aData, size_t aSize)
        : mData(aData)
        , mSize(aSize)
    {
    }

    void ConsumeData(void *aBuf, size_t aLength)
    {
        size_t consume_len = (aLength > mSize) ? mSize : aLength;
        if (consume_len > 0)
        {
            memcpy(aBuf, mData, consume_len);
            mData += consume_len;
            mSize -= consume_len;
        }
    }

    uint8_t ConsumeIntegralInRange(uint8_t aMin, uint8_t aMax)
    {
        assert(aMin <= aMax);

        uint8_t  result = aMin;
        uint16_t range = aMax - aMin;

        if (mSize > 0)
        {
            uint8_t val;
            ConsumeData(&val, sizeof(val));
            if (range > 0)
            {
                result = aMin + (val % (range + 1));
            }
        }

        return result;
    }

private:
    const uint8_t *mData;
    size_t         mSize;
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    // The fuzzer needs at least one byte to generate a boolean value.
    if (size == 0)
    {
        return 0;
    }

    FuzzDataProvider provider(data, size);

    Core nexus;
    Node &node = nexus.CreateNode();

    node.GetInstance().Get<ot::BorderRouter::RoutingManager>().Init(/* aInfraIfIndex */ 1, /* aInfraIfIsRunning */ true);
    node.GetInstance().Get<ot::BorderRouter::RoutingManager>().SetEnabled(true);

    ot::Srp::Server &server = node.GetInstance().Get<ot::Srp::Server>();

    const bool auto_enable = (provider.ConsumeIntegralInRange(0, 1) == 1);

    server.SetAutoEnableMode(auto_enable);

    return 0;
}
