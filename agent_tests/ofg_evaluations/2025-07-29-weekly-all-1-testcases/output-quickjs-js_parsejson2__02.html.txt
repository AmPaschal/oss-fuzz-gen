
	
<description>
The function `JS_ParseJSON2` parses a string `buf` of length `buf_len` as a JSON text and returns a `JSValue` representing the parsed value. The `filename` argument is used in error messages. The `flags` argument can be used to enable extensions to the JSON format (e.g., `JS_PARSE_JSON_EXT`). The function initializes a parser state, parses the value, and returns the result. It handles errors by returning `JS_EXCEPTION`.
</description>

#include <stddef.h>

// Include the main quickjs header for the function-under-test and related types.
#include "/src/quickjs/quickjs.h"

#include "/src/quickjs/quickjs-libc.h"

// Interrupt handler to prevent the fuzzer from hanging on complex inputs.
static int nb_interrupts = 0;
static int interrupt_handler(JSRuntime *rt, void *opaque) {
    nb_interrupts++;
    return (nb_interrupts > 100);
}

// The fuzz target entry point.
int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    // We need at least one byte for the flags.
    if (size < 1) {
        return 0;
    }

    JSRuntime *rt = JS_NewRuntime();
    if (!rt) {
        // Cannot proceed if runtime creation fails.
        return 0;
    }

    JSContext *ctx = JS_NewContext(rt);
    if (!ctx) {
        JS_FreeRuntime(rt);
        // Cannot proceed if context creation fails.
        return 0;
    }

    // Set a memory limit to prevent out-of-memory issues.
    JS_SetMemoryLimit(rt, 1024 * 1024 * 64); // 64 MB
    // Set a stack size limit.
    JS_SetMaxStackSize(rt, 1024 * 64); // 64 KB
    // Register the interrupt handler.
    JS_SetInterruptHandler(rt, interrupt_handler, NULL);

    // Initialize the JSON object in the context. This is required for JS_ParseJSON2.
    JS_AddIntrinsicJSON(ctx);

    // Reset the interrupt counter for each new input.
    nb_interrupts = 0;

    // Use the first byte of the input data to determine the parsing flags.
    // This allows the fuzzer to explore both standard and extended JSON parsing.
    int flags = (data[0] & 1) ? JS_PARSE_JSON_EXT : 0;
    
    // The remainder of the data is the JSON string to be parsed.
    const char *buf = (const char *)(data + 1);
    size_t buf_len = size - 1;
    const char *filename = "fuzz_input"; // Filename is used for error reporting.

    // Call the function under test.
    JSValue val = JS_ParseJSON2(ctx, buf, buf_len, filename, flags);

    // If parsing is successful and does not return an exception,
    // free the created JSValue to avoid memory leaks.
    if (!JS_IsException(val)) {
        JS_FreeValue(ctx, val);
    }

    // Clean up the QuickJS context and runtime.
    JS_FreeContext(ctx);
    JS_FreeRuntime(rt);

    return 0; // Indicate successful execution of the fuzzer test case.
}
