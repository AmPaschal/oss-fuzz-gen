
	
<description>
The `sf_command` function serves as a versatile interface for interacting with the libsndfile library, enabling users to retrieve information and modify parameters of sound files. It operates based on a command integer, which dictates the function's behavior and the interpretation of the `data` and `datasize` parameters. Commands are categorized into two groups: those that function independently of a file handle for general library inquiries, and those that necessitate a valid `SNDFILE` handle for file-specific operations. The latter commands often have prerequisites regarding the file's access mode and its modification history.
</description>
<requirements>

<requirement>
The `sndfile` argument must be a valid pointer returned by `sf_open` for most commands. This is because the function relies on the `SNDFILE` structure to access file-specific data and metadata.
</requirement>
<requirement>
The `command` argument must be one of the `SFC_` command macros defined in `sndfile.h`. Using an undefined command will result in an error.
</requirement>
<requirement>
The `data` and `datasize` arguments must match the requirements of the specific `command` being used. A mismatch can lead to memory corruption or crashes.
</requirement>
<requirement>
For commands that modify file metadata, such as `SFC_SET_ADD_PEAK_CHUNK` or `SFC_SET_BROADCAST_INFO`, they must be called before any audio data is written to the file. This is to ensure that the header is written correctly.
</requirement>
<requirement>
When using commands that depend on the file's open mode, such as `SFC_FILE_TRUNCATE`, ensure that the file was opened in the correct mode (e.g., `SFM_WRITE` or `SFM_RDWR`). This prevents errors and ensures the command can be executed successfully.
</requirement>
<requirement>
For the `SFC_SET_CHANNEL_MAP_INFO` command, the integer array passed in the `data` argument must contain values between `SF_CHANNEL_MAP_INVALID` and `SF_CHANNEL_MAP_MAX`. This is to prevent out-of-bounds access and ensure the channel map is valid.
</requirement>

</requirements>

#include <fuzzer/FuzzedDataProvider.h>
#include <sndfile.h>
#include <stdlib.h>
#include <string.h>
#include <vector>
#include <inttypes.h>

#include "sndfile_fuzz_header.h"

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  // Take a portion of the data to initialize the sound file.
  const size_t init_data_size = provider.ConsumeIntegralInRange<size_t>(0, provider.remaining_bytes());
  auto init_data = provider.ConsumeBytes<uint8_t>(init_data_size);

  SF_INFO sndfile_info;
  VIO_DATA vio_data;
  SF_VIRTUAL_IO vio;
  SNDFILE *sndfile = NULL;

  int err = sf_init_file(init_data.data(), init_data.size(), &sndfile, &vio_data, &vio, &sndfile_info);
  if (err || sndfile == NULL) {
    if (sndfile)
      sf_close(sndfile);
    return 0;
  }

  const int commands[] = {
      SFC_GET_LIB_VERSION,          SFC_GET_LOG_INFO,
      SFC_GET_CURRENT_SF_INFO,      SFC_GET_NORM_DOUBLE,
      SFC_GET_NORM_FLOAT,           SFC_SET_NORM_DOUBLE,
      SFC_SET_NORM_FLOAT,           SFC_SET_SCALE_FLOAT_INT_READ,
      SFC_SET_SCALE_INT_FLOAT_WRITE,SFC_GET_SIMPLE_FORMAT_COUNT,
      SFC_GET_SIMPLE_FORMAT,        SFC_GET_FORMAT_INFO,
      SFC_GET_FORMAT_MAJOR_COUNT,   SFC_GET_FORMAT_MAJOR,
      SFC_GET_FORMAT_SUBTYPE_COUNT, SFC_GET_FORMAT_SUBTYPE,
      SFC_CALC_SIGNAL_MAX,          SFC_CALC_NORM_SIGNAL_MAX,
      SFC_CALC_MAX_ALL_CHANNELS,    SFC_CALC_NORM_MAX_ALL_CHANNELS,
      SFC_GET_SIGNAL_MAX,           SFC_GET_MAX_ALL_CHANNELS,
      SFC_SET_ADD_PEAK_CHUNK,       SFC_UPDATE_HEADER_NOW,
      SFC_SET_UPDATE_HEADER_AUTO,   SFC_FILE_TRUNCATE,
      SFC_SET_RAW_START_OFFSET,     SFC_SET_DITHER_ON_WRITE,
      SFC_SET_DITHER_ON_READ,       SFC_GET_DITHER_INFO_COUNT,
      SFC_GET_DITHER_INFO,          SFC_GET_EMBED_FILE_INFO,
      SFC_SET_CLIPPING,             SFC_GET_CLIPPING,
      SFC_GET_CUE_COUNT,            SFC_GET_CUE,
      SFC_SET_CUE,                  SFC_GET_INSTRUMENT,
      SFC_SET_INSTRUMENT,           SFC_GET_LOOP_INFO,
      SFC_GET_BROADCAST_INFO,       SFC_SET_BROADCAST_INFO,
      SFC_GET_CHANNEL_MAP_INFO,     SFC_SET_CHANNEL_MAP_INFO,
      SFC_RAW_DATA_NEEDS_ENDSWAP,   SFC_WAVEX_SET_AMBISONIC,
      SFC_WAVEX_GET_AMBISONIC,      SFC_RF64_AUTO_DOWNGRADE,
      SFC_SET_VBR_ENCODING_QUALITY, SFC_SET_COMPRESSION_LEVEL,
      SFC_SET_CART_INFO,            SFC_GET_CART_INFO};

  int command = provider.PickValueInArray(commands);

  switch (command) {
  case SFC_GET_LIB_VERSION:
  case SFC_GET_LOG_INFO: {
    char buffer[256];
    sf_command(sndfile, command, buffer, sizeof(buffer));
    break;
  }
  case SFC_GET_CURRENT_SF_INFO: {
    SF_INFO info;
    sf_command(sndfile, command, &info, sizeof(info));
    break;
  }
  case SFC_SET_NORM_DOUBLE:
  case SFC_SET_NORM_FLOAT:
  case SFC_SET_SCALE_FLOAT_INT_READ:
  case SFC_SET_SCALE_INT_FLOAT_WRITE:
  case SFC_SET_ADD_PEAK_CHUNK:
  case SFC_SET_UPDATE_HEADER_AUTO:
  case SFC_SET_CLIPPING:
  case SFC_WAVEX_SET_AMBISONIC:
  case SFC_RF64_AUTO_DOWNGRADE: {
    sf_command(sndfile, command, NULL, provider.ConsumeBool());
    break;
  }
  case SFC_GET_SIMPLE_FORMAT:
  case SFC_GET_FORMAT_INFO:
  case SFC_GET_FORMAT_MAJOR:
  case SFC_GET_FORMAT_SUBTYPE: {
    SF_FORMAT_INFO info;
    provider.ConsumeData(&info, sizeof(info));
    sf_command(sndfile, command, &info, sizeof(info));
    break;
  }
  case SFC_CALC_SIGNAL_MAX:
  case SFC_CALC_NORM_SIGNAL_MAX:
  case SFC_GET_SIGNAL_MAX:
  case SFC_SET_VBR_ENCODING_QUALITY:
  case SFC_SET_COMPRESSION_LEVEL: {
    double val = provider.ConsumeFloatingPoint<double>();
    sf_command(sndfile, command, &val, sizeof(val));
    break;
  }
  case SFC_CALC_MAX_ALL_CHANNELS:
  case SFC_CALC_NORM_MAX_ALL_CHANNELS:
  case SFC_GET_MAX_ALL_CHANNELS: {
    if (sndfile_info.channels > 0 && sndfile_info.channels < 1024) {
      std::vector<double> peaks(sndfile_info.channels);
      sf_command(sndfile, command, peaks.data(),
                 sizeof(double) * sndfile_info.channels);
    }
    break;
  }
  case SFC_FILE_TRUNCATE:
  case SFC_SET_RAW_START_OFFSET: {
    sf_count_t offset = provider.ConsumeIntegral<sf_count_t>();
    sf_command(sndfile, command, &offset, sizeof(offset));
    break;
  }
  case SFC_SET_DITHER_ON_WRITE:
  case SFC_SET_DITHER_ON_READ: {
    SF_DITHER_INFO dither_info;
    provider.ConsumeData(&dither_info, sizeof(dither_info));
    sf_command(sndfile, command, &dither_info, sizeof(dither_info));
    break;
  }
  case SFC_GET_EMBED_FILE_INFO: {
    SF_EMBED_FILE_INFO embed_info;
    sf_command(sndfile, command, &embed_info, sizeof(embed_info));
    break;
  }
  case SFC_GET_CUE:
  case SFC_SET_CUE: {
    size_t data_size = provider.ConsumeIntegralInRange<size_t>(0, 2048);
    auto buffer = provider.ConsumeBytes<uint8_t>(data_size);
    sf_command(sndfile, command, buffer.data(), buffer.size());
    break;
  }
  case SFC_GET_INSTRUMENT:
  case SFC_SET_INSTRUMENT: {
    SF_INSTRUMENT inst;
    provider.ConsumeData(&inst, sizeof(inst));
    sf_command(sndfile, command, &inst, sizeof(inst));
    break;
  }
  case SFC_GET_LOOP_INFO: {
    SF_LOOP_INFO loop_info;
    sf_command(sndfile, command, &loop_info, sizeof(loop_info));
    break;
  }
  case SFC_GET_BROADCAST_INFO:
  case SFC_SET_BROADCAST_INFO: {
    size_t data_size = provider.ConsumeIntegralInRange<size_t>(0, 2048);
    auto buffer = provider.ConsumeBytes<uint8_t>(data_size);
    sf_command(sndfile, command, buffer.data(), buffer.size());
    break;
  }
  case SFC_GET_CHANNEL_MAP_INFO:
  case SFC_SET_CHANNEL_MAP_INFO: {
    if (sndfile_info.channels > 0 && sndfile_info.channels < 1024) {
      std::vector<int> map(sndfile_info.channels);
      provider.ConsumeData(map.data(), sizeof(int) * sndfile_info.channels);
      sf_command(sndfile, command, map.data(),
                 sizeof(int) * sndfile_info.channels);
    }
    break;
  }
  case SFC_SET_CART_INFO:
  case SFC_GET_CART_INFO: {
    size_t data_size = provider.ConsumeIntegralInRange<size_t>(0, 2048);
    auto buffer = provider.ConsumeBytes<uint8_t>(data_size);
    sf_command(sndfile, command, buffer.data(), buffer.size());
    break;
  }
  default: {
    // For commands that take no data or are simple gets
    sf_command(sndfile, command, NULL, 0);
    break;
  }
  }

  sf_close(sndfile);
  return 0;
}
