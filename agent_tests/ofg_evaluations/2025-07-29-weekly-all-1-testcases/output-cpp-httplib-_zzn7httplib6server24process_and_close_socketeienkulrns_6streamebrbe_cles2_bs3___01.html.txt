class FuzzedStream : public httplib::Stream {
public:
  FuzzedStream(FuzzedDataProvider &provider)
      : data_(provider.ConsumeRemainingBytes<uint8_t>()), read_pos_(0) {}

  ~FuzzedStream() override = default;

  bool is_readable() const override { return true; }

  bool wait_readable() const override { return true; }

  bool wait_writable() const override { return true; }

  time_t duration() const override { return 0; }

  ssize_t read(char *ptr, size_t size) override {
    if (read_pos_ >= data_.size()) {
      return -1; // No more data to read
    }
    size_t remaining_bytes = data_.size() - read_pos_;
    size_t bytes_to_copy = size < remaining_bytes ? size : remaining_bytes;
    if (bytes_to_copy > 0) {
      memcpy(ptr, data_.data() + read_pos_, bytes_to_copy);
    }
    read_pos_ += bytes_to_copy;
    return static_cast<ssize_t>(bytes_to_copy);
  }

  ssize_t write(const char *ptr, size_t size) override {
    // Do nothing, just pretend to write.
    return static_cast<ssize_t>(size);
  }

  void get_remote_ip_and_port(std::string &ip, int &port) const override {
    ip = "127.0.0.1";
    port = 8080;
  }

  void get_local_ip_and_port(std::string &ip, int &port) const override {
    ip = "127.0.0.1";
    port = 8080;
  }

  socket_t socket() const override { return 0; }

private:
  std::vector<uint8_t> data_;
  size_t read_pos_;
};
