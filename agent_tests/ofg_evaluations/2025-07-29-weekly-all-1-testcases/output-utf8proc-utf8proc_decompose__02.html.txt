
	
<description>
The `utf8proc_decompose` function is a wrapper around `utf8proc_decompose_custom`, which performs Unicode normalization and decomposition. It takes a UTF-8 encoded string and decomposes it into a buffer of 32-bit integers, representing Unicode code points. The behavior of the function is controlled by a set of options, which can specify, for example, whether to perform composition or decomposition, or whether to handle null-terminated strings.
</description>
<requirements>

<requirement>
The `options` argument must not contain both `UTF8PROC_COMPOSE` and `UTF8PROC_DECOMPOSE` flags simultaneously. The function will return `UTF8PROC_ERROR_INVALIDOPTS` if both flags are set.</requirement>
<requirement>
If the `UTF8PROC_STRIPMARK` option is set, either `UTF8PROC_COMPOSE` or `UTF8PROC_DECOMPOSE` must also be set. Otherwise, the function will return `UTF8PROC_ERROR_INVALIDOPTS`.</requirement>
<requirement>
If the `UTF8PROC_NULLTERM` option is not set, the `strlen` argument must be the correct length of the `str` buffer. This is to prevent out-of-bounds reads when iterating over the string.</requirement>
<requirement>
The input string `str` must be a valid UTF-8 string. An invalid UTF-8 sequence will cause an error during processing.</requirement>
<requirement>
The `bufsize` argument must be large enough to store the decomposed string. If `bufsize` is too small, the output will be truncated, but it will not cause a crash.</requirement>

</requirements>

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size < 1) {
    return 0;
  }

  /* Avoid timeout with long inputs */
  if (size > (16 * 1024)) {
    return 0;
  }

  /*
   * The function to be fuzzed:
   * utf8proc_ssize_t utf8proc_decompose(
   *   const utf8proc_uint8_t *str, utf8proc_ssize_t strlen,
   *   utf8proc_int32_t *buffer, utf8proc_ssize_t bufsize,
   *   utf8proc_option_t options
   * );
   */

  /*
   * Allocate a buffer for the decomposed output.
   * A single UTF-8 character can decompose into multiple codepoints.
   * A buffer of 4 times the input length is guaranteed to be large enough.
   */
  utf8proc_ssize_t bufsize = size * 4;
  utf8proc_int32_t *buffer = (utf8proc_int32_t *)malloc(bufsize * sizeof(utf8proc_int32_t));
  if (buffer == NULL) {
    return 0;
  }

  /*
   * Call utf8proc_decompose with various valid option combinations
   * to exercise different code paths.
   */
  utf8proc_option_t options;

  /* Test default decomposition. */
  options = UTF8PROC_DECOMPOSE;
  utf8proc_decompose(data, size, buffer, bufsize, options);

  /* Test decomposition with stripping combining marks. */
  options = UTF8PROC_DECOMPOSE | UTF8PROC_STRIPMARK;
  utf8proc_decompose(data, size, buffer, bufsize, options);

  /* Test decomposition with lumping. */
  options = UTF8PROC_DECOMPOSE | UTF8PROC_LUMP;
  utf8proc_decompose(data, size, buffer, bufsize, options);

  /* Test decomposition with stripping control characters. */
  options = UTF8PROC_DECOMPOSE | UTF8PROC_STRIPCC;
  utf8proc_decompose(data, size, buffer, bufsize, options);

  /* Test buffer truncation by providing a small buffer. */
  if (size > 1) {
    utf8proc_decompose(data, size, buffer, 1, UTF8PROC_DECOMPOSE);
  }

  /* Test with NULLTERM option if input is null-terminated. */
  if (data[size - 1] == '\0') {
    options = UTF8PROC_DECOMPOSE | UTF8PROC_NULLTERM;
    /* When NULLTERM is used, strlen should be -1. */
    utf8proc_decompose(data, -1, buffer, bufsize, options);
  }

  free(buffer);

  /*
   * Keep other calls from the original fuzzer to exercise more of the library
   * with the same fuzzer input.
   */
  free(utf8proc_NFD(data));
  free(utf8proc_NFC(data));
  free(utf8proc_NFKD(data));
  free(utf8proc_NFKC(data));
  free(utf8proc_NFKC_Casefold(data));

  return 0;
}
