
	
<description>
The `smux_accept` function accepts a new SMUX peer connection. It takes a socket descriptor `sd` as input, which should be a listening socket. The function calls `accept` to establish a new connection, and then performs a series of validation checks on the received data. It also checks if the maximum number of peers has been reached before accepting a new connection.
</description>
<requirements>

<requirement>
The `smux_init` function must be called before `smux_accept`. This is because `smux_init` creates and initializes the listening socket descriptor that is passed as an argument to `smux_accept`.
</requirement>
<requirement>
The input integer `sd` must be a valid listening socket descriptor. This is because it is passed to the `accept` system call to accept new connections.
</requirement>
<requirement>
The global variable `npeers` must be less than `SMUXMAXPEERS - 1` (i.e., less than 9). This is to prevent the function from rejecting the new connection because the peer limit has been reached.
</requirement>

</requirements>

#include <net-snmp/net-snmp-config.h>
#include <net-snmp/net-snmp-includes.h>
#include <net-snmp/agent/net-snmp-agent-includes.h>
#include "../../agent/mibgroup/smux.h"

#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>

/* Forward declarations are in smux.h, but we need the global variable */
extern int smux_listen_sd;

/* Non-public function from smux.c, needed for configuration */
void smux_parse_peer_auth(const char *token, char *cptr);

static int initialized = 0;
static struct sockaddr_in smux_addr;

static void
initialize_smux_fuzzer(void) {
    if (initialized) {
        return;
    }

    /* Use a random port to avoid conflicts. */
    netsnmp_ds_set_string(NETSNMP_DS_APPLICATION_ID, NETSNMP_DS_SMUX_SOCKET,
                          "127.0.0.1:0");

    /* Initialize the SMUX listener */
    real_init_smux();
    if (smux_listen_sd < 0) {
        return;
    }

    /* Get the actual address we bound to */
    socklen_t addrlen = sizeof(smux_addr);
    if (getsockname(smux_listen_sd, (struct sockaddr *)&smux_addr, &addrlen) != 0) {
        close(smux_listen_sd);
        smux_listen_sd = -1;
        return;
    }

    /*
     * Configure a valid peer so authentication can pass.
     * The string must be writable because smux_parse_peer_auth modifies it.
     * The OID and password must be present in the fuzzed Open PDU.
     */
    char auth_str[] = ".1.3.6.1.4.1.8072.1.3.1 mypassword";
    smux_parse_peer_auth("smuxpeer", auth_str);

    initialized = 1;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    initialize_smux_fuzzer();

    if (smux_listen_sd < 0) {
        return 0; /* Initialization failed */
    }

    int client_sd = socket(AF_INET, SOCK_STREAM, 0);
    if (client_sd < 0) {
        return 0;
    }

    if (connect(client_sd, (struct sockaddr *)&smux_addr, sizeof(smux_addr)) < 0) {
        close(client_sd);
        return 0;
    }

    /* Send the fuzzer data to the listening socket */
    send(client_sd, data, size, 0);
    
    /*
     * Call smux_accept. This will accept the connection from client_sd,
     * receive the data, and process it.
     */
    int new_fd = smux_accept(smux_listen_sd);

    if (new_fd >= 0) {
        /*
         * A peer was successfully accepted. We must close the new descriptor.
         * Note: we cannot call the static smux_peer_cleanup() function,
         * so the internal peer counter 'npeers' will not be decremented.
         * This means the fuzzer will stop accepting new peers after
         * SMUXMAXPEERS (10) successful runs.
         */
        close(new_fd);
    }

    close(client_sd);

    return 0;
}
