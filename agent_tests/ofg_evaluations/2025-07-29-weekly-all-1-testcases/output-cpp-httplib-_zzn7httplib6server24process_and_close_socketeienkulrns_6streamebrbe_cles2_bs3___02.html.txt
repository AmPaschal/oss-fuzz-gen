static FuzzableServer g_server;

// Initialize server with handlers, which is necessary for process_request.
extern "C" int LLVMFuzzerInitialize(int * /*argc*/, char *** /*argv*/) {
  g_server.Get(R"(.*)",
               [&](const httplib::Request & /*req*/, httplib::Response &res) {
                 res.set_content("response content", "text/plain");
               });
  g_server.Post(R"(.*)",
                [&](const httplib::Request & /*req*/, httplib::Response &res) {
                  res.set_content("response content", "text/plain");
                });
  g_server.Put(R"(.*)",
               [&](const httplib::Request & /*req*/, httplib::Response &res) {
                 res.set_content("response content", "text/plain");
               });
  g_server.Patch(R"(.*)",
                 [&](const httplib::Request & /*req*/, httplib::Response &res) {
                   res.set_content("response content", "text/plain");
                 });
  g_server.Delete(
      R"(.*)", [&](const httplib::Request & /*req*/, httplib::Response &res) {
        res.set_content("response content", "text/plain");
      });
  g_server.Options(
      R"(.*)", [&](const httplib::Request & /*req*/, httplib::Response &res) {
        res.set_content("response content", "text/plain");
      });
  return 0;
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  // Requirement: Use httplib::detail::BufferStream and initialize with fuzzed
  // data.
  httplib::detail::BufferStream stream;
  std::string request_data = provider.ConsumeRemainingBytesAsString();
  stream.write(request_data.data(), request_data.size());

  // Requirement: Fuzz the 'close_connection' boolean.
  bool close_connection = provider.ConsumeBool();

  // Requirement: Initialize 'connection_closed' to false.
  bool connection_closed = false;

  // Call the target function as specified by the prompt's signature.
  g_server(nullptr, stream, close_connection, connection_closed);

  return 0;
}
