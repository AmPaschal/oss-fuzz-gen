// Forward declaration of the function to be tested.
void mg_hello(const char *);

// Global state variables required by the function under test.
static volatile bool s_done;
static volatile int s_cnt;

// Callback function for Mongoose connection events.
static void s_fn(struct mg_connection *c, int ev, void *ev_data) {
  if (ev == MG_EV_CONNECT) {
    s_cnt++;
    // Close the connection immediately to prevent hangs.
    c->is_draining = 1;
  } else if (ev == MG_EV_CLOSE || ev == MG_EV_ERROR) {
    s_done = true;
  }
  (void) ev_data;
}

// Implementation of the function-under-test.
void mg_hello(const char *url) {
  struct mg_mgr mgr;
  mg_mgr_init(&mgr);
  if (mg_connect(&mgr, url, s_fn, (void *) &s_done) == NULL) {
    s_done = true; // Failed to create connection
  }

  // Poll for events with a timeout to avoid infinite loops.
  uint64_t start = mg_millis();
  while (s_done == false && mg_millis() - start < 500) {
    mg_mgr_poll(&mgr, 50);
  }

  mg_mgr_free(&mgr);
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  // We need a null-terminated string for the URL.
  if (size == 0 || size > 1024) {
    return 0;
  }

  char *url = (char *) malloc(size + 1);
  if (url == NULL) {
    return 0;

  memcpy(url, data, size);
  url[size] = '\0';

  s_done = false;
