
	
<description>
The function `lodepng::encode` takes raw image data and encodes it into a PNG format, which is then saved to a file. It is a wrapper around two other functions. First, it calls another `lodepng::encode` overload that takes a `std::vector<unsigned char>` for the output. This function, in turn, calls `lodepng_encode_memory` to perform the actual PNG encoding in memory. If the in-memory encoding is successful, the target function then calls `lodepng::save_file` to write the contents of the output vector to the specified filename. The function returns an error code, with 0 indicating success.
</description>
<requirements>

<requirement>
The `in` pointer must point to a valid memory buffer of a size that is at least the value returned by `lodepng_get_raw_size_lct(w, h, colortype, bitdepth)`. This is to prevent out-of-bounds memory access during encoding.
</requirement>
<requirement>
The `colortype` and `bitdepth` parameters must be a valid combination as checked by `lodepng_check_color_settings`. Invalid combinations will cause an error to be returned.
</requirement>
<requirement>

</requirement>
<requirement>

</requirement>

</requirement>
<requirement>

</requirement>

#include "/src/lodepng/lodepng.h"
#include <fuzzer/FuzzedDataProvider.h>
#include <string>
#include <vector>

// Array of valid colortype/bitdepth combinations, based on lodepng_fuzzer.cpp
const size_t num_combinations = 15;
LodePNGColorType colortypes[num_combinations] = {
    LCT_GREY,       LCT_GREY,       LCT_GREY,    LCT_GREY,    LCT_GREY,
    LCT_RGB,        LCT_RGB,
    LCT_PALETTE,    LCT_PALETTE,    LCT_PALETTE, LCT_PALETTE,
    LCT_GREY_ALPHA, LCT_GREY_ALPHA,
    LCT_RGBA,       LCT_RGBA,
};
unsigned bitdepths[num_combinations] = {
    1, 2,  4, 8, 16, // gray
    8, 16,           // rgb
    1, 2,  4, 8,     // palette
    8, 16,           // gray+alpha
    8, 16,           // rgb+alpha
};

// Forward declaration for the function under test.
unsigned int lodepng::encode(const std::string &, const unsigned char *, unsigned int, unsigned int, LodePNGColorType, unsigned int);

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    // 1. Generate parameters for the image.
    // Keep dimensions small to avoid timeouts.
    unsigned int w = provider.ConsumeIntegralInRange<unsigned int>(1, 512);
    unsigned int h = provider.ConsumeIntegralInRange<unsigned int>(1, 512);

    // Pick a valid color type and bit depth combination.
    size_t combination_idx = provider.ConsumeIntegralInRange<size_t>(0, num_combinations - 1);
    LodePNGColorType colortype = colortypes[combination_idx];
    unsigned int bitdepth = bitdepths[combination_idx];

    // 2. Calculate the required raw image buffer size using the library's function.
    LodePNGColorMode color_mode;
    lodepng_color_mode_init(&color_mode);
    color_mode.colortype = colortype;
    color_mode.bitdepth = bitdepth;
    
    size_t raw_size = lodepng_get_raw_size(w, h, &color_mode);
    lodepng_color_mode_cleanup(&color_mode);

    // lodepng_get_raw_size returns 0 on integer overflow for w*h.
    // Also cap the allocation size to prevent OOM.
    if (raw_size == 0 || raw_size > 2000000) {
        return 0;
    }

    // 3. Consume bytes for the raw image data.
    std::vector<uint8_t> image_data = provider.ConsumeBytes<uint8_t>(raw_size);
    if (image_data.size() < raw_size) {
        // Not enough data from the provider, exit early.
        return 0;
    }

    // 4. Define a filename. Using a temporary file path is sufficient.
    const std::string filename = "/tmp/fuzz.png";

    // 5. Call the function under test.
    lodepng::encode(filename, image_data.data(), w, h, colortype, bitdepth);

    return 0;
}
