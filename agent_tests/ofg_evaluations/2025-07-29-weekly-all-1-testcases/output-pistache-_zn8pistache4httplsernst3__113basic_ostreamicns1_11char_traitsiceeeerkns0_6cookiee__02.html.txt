
	
<description>
The function `Pistache::Http::operator<<` is an overloaded operator that streams a `Pistache::Http::Cookie` object to an output stream. It does this by calling the private `write` method of the `Cookie` class, which formats the cookie's attributes (name, value, path, domain, expiration, etc.) into a string according to HTTP cookie standards and writes it to the provided `std::ostream`.
</description>

</requirement>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider stream(data, size);
  const std::string cookie_str = stream.ConsumeRemainingBytesAsString();

  try {
    Pistache::Http::Cookie cookie =
        Pistache::Http::Cookie::fromString(cookie_str);
    std::ostringstream oss;
    oss << cookie;
  } catch (const std::runtime_error &) {
    // Ignore parsing errors, as they are expected for invalid input.
  }

  return 0;
}
