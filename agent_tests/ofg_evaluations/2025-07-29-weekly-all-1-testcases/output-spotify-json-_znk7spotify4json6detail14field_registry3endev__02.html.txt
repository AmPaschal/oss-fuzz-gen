This function is typically used in conjunction with `begin()` to iterate over the fields that have been registered. The `field_registry` object must be properly initialized for this function to be used meaningfully.
</description>

</requirement>
<requirement>

Calling `save()` adds a field to the `_field_list`, ensuring that the container is not empty, which is essential for any loop that iterates through the registered fields.
</requirement>

// A dummy object for which the codec will be created.
struct MyFuzzObject {
  std::string field1;
  std::string field2;
  std::string field3;
};

// A custom codec class that inherits from spotify::json::codec::object_t.
// This is possible because the build script removes the 'final' specifier.
// This allows us to access the protected '_fields' member (a field_registry)
// from the base class.
template <typename T>
class FuzzCodec : public spotify::json::codec::object_t<T> {
public:
  // A public method to explicitly call the target function on the _fields member.
  void fuzz_the_end() const {
    // Access the protected _fields member and call begin() and end() on it.
    // The call to end() is now guaranteed to be a real function call because
    // the build script added __attribute__((noinline)) to its definition.
    auto it_begin = this->_fields.begin();
    auto it_end = this->_fields.end(); // <<< EXPLICIT CALL TO FUNCTION-UNDER-TEST

    // Iterate over the fields to ensure the iterators are used.
    for (; it_begin != it_end; ++it_begin) {
      // Use volatile to be extra sure the loop isn't optimized away.
      volatile size_t len = it_begin->first.length();
      (void)len;
    }
  }
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  // 1. Create an instance of our custom FuzzCodec.
  FuzzCodec<MyFuzzObject> codec;

  // 2. Add fields to the codec using fuzzed names. This populates the
  // internal field_registry (_fields) via the public API.
  codec.required(provider.ConsumeRandomLengthString(10).c_str(), &MyFuzzObject::field1);
  if (provider.ConsumeBool()) {
    codec.required(provider.ConsumeRandomLengthString(10).c_str(), &MyFuzzObject::field2);
  }
  if (provider.ConsumeBool()) {
    codec.required(provider.ConsumeRandomLengthString(10).c_str(), &MyFuzzObject::field3);
  }

  // 3. Explicitly call our new method which contains the call to the target function.
  codec.fuzz_the_end();

  return 0;
}
