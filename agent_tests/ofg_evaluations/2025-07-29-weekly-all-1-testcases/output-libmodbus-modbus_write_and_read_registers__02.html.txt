
	
<description>
The function `modbus_write_and_read_registers` writes a block of contiguous registers and reads another block of contiguous registers in a single atomic transaction. It takes a pointer to a `modbus_t` context, the write and read starting addresses, the number of registers to write and read, a pointer to the data to write, and a pointer to store the read data.
</description>
<requirements>

<requirement>
The first argument, `ctx`, must be a valid pointer to a `modbus_t` struct returned by `modbus_new_tcp` or a similar function, and `modbus_connect` must be called. This is because the function internally dereferences this pointer to access backend-specific functions and data.
</requirement>
<requirement>
The `write_nb` argument must be between 1 and `MODBUS_MAX_WR_WRITE_REGISTERS` (121). This is to prevent the function from returning an error and setting errno to `EMBMDATA`.
</requirement>
<requirement>
The `read_nb` argument must be between 1 and `MODBUS_MAX_WR_READ_REGISTERS` (125). This is to prevent the function from returning an error and setting errno to `EMBMDATA`.
</requirement>
<requirement>
The `src` argument must be a valid pointer to an array of `uint16_t` with a size of at least `write_nb`. This is because the function reads `write_nb` elements from this array.
</requirement>
<requirement>
The `dest` argument must be a valid pointer to an array of `uint16_t` with a size of at least `read_nb`. This is because the function writes `read_nb` elements to this array.
</requirement>
<requirement>
The `write_addr` and `read_addr` arguments should be less than the maximum number of registers. The sum of the address and the number of registers to access should not exceed 65535.
</requirement>

</requirements>

#define PORT 8080
#define kMinInputLength (4 * sizeof(int) + sizeof(uint16_t))
#define kMaxInputLength MODBUS_RTU_MAX_ADU_LENGTH

struct Fuzzer {
  uint16_t port;
  char *file;

  FILE *inFile;
  uint64_t size;
  const uint8_t *buffer;

  pthread_t thread;
  int socket;
};
typedef struct Fuzzer Fuzzer;

int client(Fuzzer *fuzzer);

void fuzzinit(Fuzzer *fuzzer) {
  struct sockaddr_in server_addr;
  fuzzer->socket = socket(AF_INET, SOCK_STREAM, 0);

  server_addr.sin_family = AF_INET;
  server_addr.sin_port = htons(fuzzer->port);
  server_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

  setsockopt(fuzzer->socket, SOL_SOCKET, SO_REUSEADDR, &(int){1}, sizeof(int));

  bind(fuzzer->socket, (struct sockaddr *)&server_addr, sizeof(server_addr));
  listen(fuzzer->socket, 1);
}

void *Server(void *args) {

  Fuzzer *fuzzer = (Fuzzer *)args;
  {
    int client_sock;
    char clientData[10240];
    struct sockaddr_in clientAddr;
    uint32_t clientSZ = sizeof(clientAddr);

    client_sock =
        accept(fuzzer->socket, (struct sockaddr *)&clientAddr, &clientSZ);

    send(client_sock, fuzzer->buffer, fuzzer->size, 0);
    recv(client_sock, clientData, sizeof(clientData), 0);

    send(client_sock, fuzzer->buffer, fuzzer->size, 0);
    recv(client_sock, clientData, sizeof(clientData), 0);

    shutdown(client_sock, SHUT_RDWR);
    close(client_sock);
  }
  pthread_exit(NULL);
}

void clean(Fuzzer *fuzzer) {
  { // Server
    shutdown(fuzzer->socket, SHUT_RDWR);
    close(fuzzer->socket);
  }
  free(fuzzer);
}

extern int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

  if (size < kMinInputLength || size > kMaxInputLength) {
    return 0;
  }

  Fuzzer *fuzzer = (Fuzzer *)malloc(sizeof(Fuzzer));
  if (fuzzer == NULL) {
    return 0;
  }
  fuzzer->port = PORT;
  fuzzer->size = size;
  fuzzer->buffer = data;

  fuzzinit(fuzzer);

  pthread_create(&fuzzer->thread, NULL, Server, fuzzer);
  client(fuzzer);
  pthread_join(fuzzer->thread, NULL); /* To Avoid UAF*/

  clean(fuzzer);
  return 0;
}

int client(Fuzzer *fuzzer) {
  modbus_t *ctx = NULL;
  int rc = 0;

  ctx = modbus_new_tcp("127.0.0.1", fuzzer->port);

  if (ctx == NULL) {
    return -1;
  }

  if (modbus_connect(ctx) == -1) {
    modbus_free(ctx);
    return -1;
  }

  const uint8_t *p = fuzzer->buffer;
  const uint8_t *end = fuzzer->buffer + fuzzer->size;

  int write_addr_in, write_nb_in, read_addr_in, read_nb_in;

  memcpy(&write_addr_in, p, sizeof(int));
  p += sizeof(int);
  memcpy(&write_nb_in, p, sizeof(int));
  p += sizeof(int);
  memcpy(&read_addr_in, p, sizeof(int));
  p += sizeof(int);
  memcpy(&read_nb_in, p, sizeof(int));
  p += sizeof(int);

  int write_nb = (abs(write_nb_in) % MODBUS_MAX_WR_WRITE_REGISTERS) + 1;

  if ((size_t)(end - p) < (size_t)write_nb * sizeof(uint16_t)) {
    modbus_close(ctx);
    modbus_free(ctx);
    return 0;
  }
  const uint16_t *src = (const uint16_t *)p;

  int read_nb = (abs(read_nb_in) % MODBUS_MAX_WR_READ_REGISTERS) + 1;

  uint16_t *dest = (uint16_t *)malloc((size_t)read_nb * sizeof(uint16_t));
  if (dest == NULL) {
    modbus_close(ctx);
    modbus_free(ctx);
    return -1;
  }

  int write_addr = abs(write_addr_in) % (0x10000 - write_nb);
  int read_addr = abs(read_addr_in) % (0x10000 - read_nb);

  rc = modbus_write_and_read_registers(ctx, write_addr, write_nb, src,
                                       read_addr, read_nb, dest);

  free(dest);

  /* Close the connection */
  modbus_close(ctx);
  modbus_free(ctx);

  return rc;
}
