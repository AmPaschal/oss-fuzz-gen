
	
<description>
The `sb_encode` function is the core of the Speex wideband and ultra-wideband encoding process. It takes the audio input, encodes it, and writes the resulting bits to a SpeexBits struct. The function is not called directly, but rather through a function pointer `enc` in a `SpeexMode` struct. The `SpeexMode` struct is initialized with either `speex_wb_mode` or `speex_uwb_mode`, which both set the `enc` pointer to `sb_encode`. The `state` argument must be initialized by `sb_encoder_init` before calling `sb_encode`.
</description>
<requirements>

<requirement>
The `state` argument must be initialized by calling `sb_encoder_init` with either `speex_wb_mode` or `speex_uwb_mode`. This is because `sb_encoder_init` allocates and initializes the `SBEncState` struct that `sb_encode` expects.
</requirement>
<requirement>
The `vin` argument should be a pointer to a block of audio data. The size of this block should be `2 * frameSize`, where `frameSize` is a member of the `SpeexSBMode` struct used to initialize the encoder.
</requirement>
<requirement>
The `bits` argument must be a pointer to a `SpeexBits` struct. This struct must be initialized before calling `sb_encode` so that the encoded bitstream can be written to it.
</requirement>

</requirements>

#include <fuzzer/FuzzedDataProvider.h>
#include <cstdint>
#include <vector>

#ifdef HAVE_CONFIG_H
# include "config.h"
#endif

#include "speex/speex.h"

// Explicitly declare the internal C functions we need with C linkage.
extern "C" {
    void *sb_encoder_init(const SpeexMode *m);
    void sb_encoder_destroy(void *state);
    int sb_encoder_ctl(void *state, int request, void *ptr);
    int sb_encode(void *state, void *in, SpeexBits *bits);
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    const SpeexMode *mode = provider.ConsumeBool() ? &speex_wb_mode : &speex_uwb_mode;

    void *state = sb_encoder_init(mode);
    if (!state) {
        return 0;
    }

    int frame_size = 0;
    sb_encoder_ctl(state, SPEEX_GET_FRAME_SIZE, &frame_size);
    if (frame_size <= 0) {
        sb_encoder_destroy(state);
        return 0;
    }

    const int input_samples = 2 * frame_size;
    std::vector<spx_int16_t> vin_data(input_samples);
    const size_t consumed_bytes = provider.ConsumeData(vin_data.data(), input_samples * sizeof(spx_int16_t));
    if (consumed_bytes < input_samples * sizeof(spx_int16_t)) {
        sb_encoder_destroy(state);
        return 0;
    }

    SpeexBits bits;
    speex_bits_init(&bits);

    sb_encode(state, (void*)vin_data.data(), &bits);

    speex_bits_destroy(&bits);
    sb_encoder_destroy(state);

    return 0;
}
