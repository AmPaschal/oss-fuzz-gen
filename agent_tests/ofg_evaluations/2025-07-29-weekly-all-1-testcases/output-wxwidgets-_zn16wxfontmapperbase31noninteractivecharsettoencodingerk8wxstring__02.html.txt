
	
<description>
The function `wxFontMapperBase::NonInteractiveCharsetToEncoding` maps a charset string to a `wxFontEncoding` enumeration value. It first attempts to find a user-defined mapping in the configuration. If no mapping is found, it tries to recognize the charset from a built-in list of known encodings. The function is designed to handle various string formats, including quoted strings and strings with leading/trailing whitespace. It also performs case-insensitive comparisons for some encodings.
</description>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
    // Suppress any wxWidgets logging, which is not useful for fuzzing.
    wxLogNull noLog;

    FuzzedDataProvider provider(data, size);
    const std::string charset_str = provider.ConsumeRemainingBytesAsString();

    // Get the singleton instance of the font mapper.
    wxFontMapperBase* mapper = wxFontMapperBase::Get();
    if (!mapper)
    {
        // This shouldn't be reachable as Get() provides a fallback.
        return 0;
    }

    // Create a wxString from the fuzzed data. wxConvLibc is a reasonable
    // choice as charset names are typically ASCII or compatible.
    const wxString charset(charset_str.c_str(), wxConvLibc);

    // Call the public function which in turn calls the function under test.
    mapper->CharsetToEncoding(charset);

    return 0;
}
