
	
<description>
The `File::operator=` function performs a member-wise copy of a `File` object. The most critical member is `hFile`, the file handle. The function copies the file handle from the source `File` object (`SrcFile`) to the destination `File` object. To prevent the file from being closed twice, the `SkipClose` member of `SrcFile` is set to `true`. This makes the destination object responsible for closing the file handle. The other members copied are `NewFile`, `LastWrite`, `HandleType`, `TruncatedAfterReadError`, and `FileName`. These members are fundamental data types and do not have any complex requirements.

The primary requirement for this function is that `SrcFile` must contain a valid, open file handle. If `hFile` is invalid, the function will not crash, but the resulting `File` object will be in an invalid state, which could lead to crashes in subsequent operations. The caller must also be aware that after the assignment, it should not close the file handle associated with `SrcFile` as the ownership of the handle is transferred to the destination object.
</description>

#include <stddef.h>
#include <stdint.h>

#include <filesystem>
#include <fstream>
#include <string>
#include <system_error>
#include <unistd.h>

#include "rar.hpp"

namespace fs = std::__fs::filesystem;

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  // unrar can create files, so let's work in a temporary directory.
  std::error_code code, ok;
  fs::path original_path = fs::current_path(code);
  if (code != ok)
    return 0;

  fs::path out_path = original_path / "o";
  fs::create_directory(out_path, code);
  if (code != ok && !fs::is_directory(out_path, code)) {
    return 0;
  }
  
  fs::current_path(out_path, code);
  if (code != ok) {
    fs::current_path(original_path, code);
    fs::remove_all(out_path, code);
    return 0;
  }

  const std::string filename = "fuzz_file.dat";
  std::ofstream file(filename, std::ios::binary | std::ios::out | std::ios::trunc);
  if (!file.is_open()) {
    fs::current_path(original_path, code);
    fs::remove_all(out_path, code);
    return 0;
  }
  file.write(reinterpret_cast<const char *>(data), size);
  file.close();

  std::wstring w_filename(filename.begin(), filename.end());

  File srcFile;
  // Create a file to get a valid file handle for the assignment.
  if (!srcFile.Create(w_filename)) {
    unlink(filename.c_str());
    fs::current_path(original_path, code);
    fs::remove_all(out_path, code);
    return 0;
  }

  File destFile;

  // Call the function under test.
  destFile = srcFile;

  // The file handle is now owned by destFile.
  // The destructors for destFile and srcFile will handle closing correctly.
  
  // Cleanup.
  unlink(filename.c_str());
  fs::current_path(original_path, code);
  fs::remove_all(out_path, code);

  return 0;
}
