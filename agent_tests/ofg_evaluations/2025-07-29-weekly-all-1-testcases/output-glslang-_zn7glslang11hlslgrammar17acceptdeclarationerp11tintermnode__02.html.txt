
	
<description>
The `acceptDeclaration` function is a core part of the HLSL parser. It is responsible for parsing a single declaration, which can be a variable, a function, a typedef, a namespace, or a constant buffer. It operates on a token stream provided by the `HlslGrammar` base class and builds an Abstract Syntax Tree (AST) using the `TIntermediate` object. The function is called recursively by other parsing functions like `acceptDeclarationList` to parse a list of declarations. The function heavily relies on the state of the `HlslParseContext` to manage symbols, namespaces, and to report errors.
</description>

#include "glslang/Public/ResourceLimits.h"
#include "glslang/Public/ShaderLang.h"
#include "/src/glslang/glslang/HLSL/hlslGrammar.h"
#include <fuzzer/FuzzedDataProvider.h>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider stream(data, size);

    glslang::InitializeProcess();

    // Pick a shader stage, as parsing rules can differ
    const EShLanguage stage = stream.PickValueInArray<EShLanguage>({
        EShLangVertex,
        EShLangTessControl,
        EShLangTessEvaluation,
        EShLangGeometry,
        EShLangFragment,
        EShLangCompute,
        EShLangRayGen,
        EShLangIntersect,
        EShLangAnyHit,
        EShLangClosestHit,
        EShLangMiss,
        EShLangCallable,
        EShLangTask,
        EShLangMesh,
    });

    glslang::TShader shader(stage);
    const std::string shader_string = stream.ConsumeRemainingBytesAsString();
    const char *shader_strings[] = {shader_string.c_str()};
    const int shader_lengths[] = {(int)shader_string.length()};
    shader.setStringsWithLengths(shader_strings, shader_lengths, 1);
    
    // Set entry points for the shader
    shader.setEntryPoint("main");
    shader.setSourceEntryPoint("main");

    // Set HLSL parsing mode and other options
    EShMessages messages = EShMsgReadHlsl;
    if (stream.ConsumeBool()) {
        messages = static_cast<EShMessages>(messages | EShMsgVulkanRules);
    }
    if (stream.ConsumeBool()) {
        messages = static_cast<EShMessages>(messages | EShMsgSpvRules);
    }

    // The call to parse will internally create the HlslGrammar and call
    // acceptDeclaration for each declaration in the shader source.
    shader.parse(GetDefaultResources(), 500, ECoreProfile, false, false, messages);

    glslang::FinalizeProcess();
    return 0;
}
