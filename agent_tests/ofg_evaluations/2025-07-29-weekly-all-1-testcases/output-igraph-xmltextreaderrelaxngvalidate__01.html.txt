extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    FuzzedDataProvider provider(data, size);

    // Split the input into a RelaxNG schema and an XML document.
    std::string rng_schema = provider.ConsumeRandomLengthString();
    std::string xml_document = provider.ConsumeRemainingBytesAsString();

    if (xml_document.empty()) {
        return 0;
    }

    // Create an xmlTextReader for the in-memory XML document.
    xmlTextReaderPtr reader = xmlReaderForMemory(
        xml_document.c_str(),
        static_cast<int>(xml_document.length()),
        nullptr, // base URI
        nullptr, // encoding
        0        // options
    );

    if (reader == nullptr) {
        xmlCleanupParser();
        return 0;
    }

    // Call the function under test.
    xmlTextReaderRelaxNGValidate(reader, rng_schema.c_str());

    // Clean up resources.
    xmlFreeTextReader(reader);
    xmlCleanupParser();

    return 0;
}
