#define MAX_BUFFER_SIZE 65536
#define HANDSHAKE_MAX_ITERATIONS 20

/* An in-memory buffer for our transport simulation. */
typedef struct {
    uint8_t *data;
    size_t size;
    size_t capacity;
    size_t read_pos;
} mem_buffer_st;

/* The state for our transport, pointing to the read and write buffers. */
typedef struct {
    mem_buffer_st *read_buf;
    mem_buffer_st *write_buf;
} transport_state_st;

static int buffer_init(mem_buffer_st *buf) {
    buf->capacity = 4096;
    buf->data = malloc(buf->capacity);
    if (!buf->data) return -1;
    buf->size = 0;
    buf->read_pos = 0;
    return 0;
}

static void buffer_deinit(mem_buffer_st *buf) {
    free(buf->data);
}

/* The push function writes data to the peer's buffer. */
static ssize_t push_func(gnutls_transport_ptr_t ptr, const void *data, size_t len) {
    transport_state_st *state = (transport_state_st *)ptr;
    mem_buffer_st *buf = state->write_buf;

    if (buf->read_pos > 0) {
        if (buf->size > buf->read_pos) {
            memmove(buf->data, buf->data + buf->read_pos, buf->size - buf->read_pos);
        }
        buf->size -= buf->read_pos;
        buf->read_pos = 0;
    }

    if (buf->size + len > buf->capacity) {
        if (buf->capacity >= MAX_BUFFER_SIZE) return -1;
        size_t new_capacity = buf->capacity * 2;
        if (new_capacity > MAX_BUFFER_SIZE) new_capacity = MAX_BUFFER_SIZE;
        if (buf->size + len > new_capacity) return -1;

        uint8_t *new_data = realloc(buf->data, new_capacity);
        if (!new_data) return -1;
        buf->data = new_data;
        buf->capacity = new_capacity;
    }

    memcpy(buf->data + buf->size, data, len);
    buf->size += len;
    return len;
}

/* The pull function reads data from our own buffer. */
static ssize_t pull_func(gnutls_transport_ptr_t ptr, void *data, size_t len) {
    transport_state_st *state = (transport_state_st *)ptr;
    mem_buffer_st *buf = state->read_buf;

    size_t available = buf->size - buf->read_pos;
    if (available == 0) {
        errno = EAGAIN;
        return -1;
    }

    size_t to_read = (len > available) ? available : len;
    memcpy(data, buf->data + buf->read_pos, to_read);
    buf->read_pos += to_read;

    return to_read;
}
