// VIO callbacks and data structure from the original fuzzer
typedef struct {
  sf_count_t offset;
  sf_count_t len;
  std::vector<uint8_t> *data;
} VIO_DATA_WRITE;

static sf_count_t vio_get_filelen_write(void *user_data) {
  VIO_DATA_WRITE *vf = (VIO_DATA_WRITE *)user_data;
  return vf->data->size();
}

static sf_count_t vio_seek_write(sf_count_t offset, int whence, void *user_data) {
  VIO_DATA_WRITE *vf = (VIO_DATA_WRITE *)user_data;
  sf_count_t new_offset;
  switch (whence) {
  case SEEK_SET:
    new_offset = offset;
    break;
  case SEEK_CUR:
    new_offset = vf->offset + offset;
    break;
  case SEEK_END:
    new_offset = vf->data->size() + offset;
    break;
  default:
    return vf->offset;
  };

  if (new_offset < 0)
    new_offset = 0;

  vf->offset = new_offset;
  return vf->offset;
}

static sf_count_t vio_read_write(void *ptr, sf_count_t count, void *user_data) {
  VIO_DATA_WRITE *vf = (VIO_DATA_WRITE *)user_data;
  if (vf->offset + count > vf->data->size())
    count = vf->data->size() - vf->offset;

  if (count > 0)
    memcpy(ptr, vf->data->data() + vf->offset, count);
  vf->offset += count;
  return count;
}

static sf_count_t vio_write_write(const void *ptr, sf_count_t count,
                                  void *user_data) {
  VIO_DATA_WRITE *vf = (VIO_DATA_WRITE *)user_data;
  if (vf->offset + count > vf->data->capacity()) {
      vf->data->resize(vf->offset + count);
  }
  if (vf->offset + count > vf->data->size()) {
    vf->data->resize(vf->offset + count);
  }

  memcpy(vf->data->data() + vf->offset, ptr, count);
  vf->offset += count;
  return count;
}

static sf_count_t vio_tell_write(void *user_data) {
  VIO_DATA_WRITE *vf = (VIO_DATA_WRITE *)user_data;
  return vf->offset;
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  // Setup for sf_open_virtual
  SF_INFO sfinfo;
  memset(&sfinfo, 0, sizeof(sfinfo));

  // Fuzz SF_INFO parameters
  sfinfo.format = provider.ConsumeIntegralInRange<int>(0x010000, 0x08FFFF);
  sfinfo.channels = provider.ConsumeIntegralInRange<int>(1, 16);
  sfinfo.samplerate = provider.ConsumeIntegralInRange<int>(8000, 96000);

  // We need a writeable buffer for the virtual IO
  std::vector<uint8_t> buffer(4096);

  VIO_DATA_WRITE vio_data;
  vio_data.offset = 0;
  vio_data.len = buffer.size();
  vio_data.data = &buffer;

  SF_VIRTUAL_IO vio;
  vio.get_filelen = vio_get_filelen_write;
  vio.seek = vio_seek_write;
  vio.read = vio_read_write;
  vio.write = vio_write_write;
  vio.tell = vio_tell_write;

  SNDFILE *sndfile = sf_open_virtual(&vio, SFM_WRITE, &sfinfo, &vio_data);
  if (sndfile == NULL) {
    return 0;
  }

  // Fuzz parameters for sf_set_string
  const int str_type = provider.PickValueInArray({
      SF_STR_TITLE,       SF_STR_COPYRIGHT, SF_STR_SOFTWARE, SF_STR_ARTIST,
      SF_STR_COMMENT,     SF_STR_DATE,      SF_STR_ALBUM,    SF_STR_LICENSE,
      SF_STR_TRACKNUMBER, SF_STR_GENRE,
  });

  std::string str = provider.ConsumeRemainingBytesAsString();

  // Call the target function
  sf_set_string(sndfile, str_type, str.c_str());

  sf_close(sndfile);

  return 0;
}
