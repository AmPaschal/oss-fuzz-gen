
	
<description>
The function `usbredirparser_send_interrupt_packet` sends an interrupt packet. It is a wrapper around the `usbredirparser_queue` function, which queues the packet to be sent.
</description>
<requirements>
<requirement>
The `parser` argument must be a valid pointer to a `usbredirparser` struct, initialized by `usbredirparser_create`. This is because the function internally uses the parser to queue the packet.
</requirement>
<requirement>
The `interrupt_header` argument must be a valid pointer to a `usb_redir_interrupt_packet_header` struct. This is because the function dereferences this pointer to read header information.
</requirement>
<requirement>
The `data_len` argument must be equal to the `length` field of the `interrupt_header` struct. This consistency is maintained by all callers of the function in the codebase.
</requirement>
<requirement>
If `data_len` is greater than 0, the `data` argument must be a valid pointer to a buffer of at least `data_len` bytes. This is because the function will copy `data_len` bytes from this buffer.
</requirement>
<requirement>
If `data_len` is 0, the `data` argument can be NULL. This is a valid use case as seen in the `usbredirhost_send_interrupt_status` function.
</requirement>
</requirements>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider fdp(data, size);

  if (!parser) {

  parser->get_alt_setting_func = dummy_get_alt_setting;
  parser->alt_setting_status_func = dummy_alt_setting_status;
  parser->start_iso_stream_func = dummy_start_iso_stream;
  parser->stop_iso_stream_func = dummy_stop_iso_stream;
  parser->iso_stream_status_func = dummy_iso_stream_status;
  parser->start_interrupt_receiving_func = dummy_start_interrupt_receiving;
  parser->stop_interrupt_receiving_func = dummy_stop_interrupt_receiving;
  parser->interrupt_receiving_status_func = dummy_interrupt_receiving_status;
  parser->alloc_bulk_streams_func = dummy_alloc_bulk_streams;
  parser->free_bulk_streams_func = dummy_free_bulk_streams;
  parser->bulk_streams_status_func = dummy_bulk_streams_status;
  parser->cancel_data_packet_func = dummy_cancel_data_packet;
  parser->control_packet_func = dummy_control_packet;
  parser->bulk_packet_func = dummy_bulk_packet;
  parser->iso_packet_func = dummy_iso_packet;
  parser->interrupt_packet_func = dummy_interrupt_packet;
  parser->alloc_lock_func = dummy_alloc_lock;
  parser->lock_func = dummy_lock;
  parser->unlock_func = dummy_unlock;
  parser->free_lock_func = dummy_free_lock;
  parser->hello_func = dummy_hello;

  usbredirparser_init(parser, "fuzzer", nullptr, 0, 0);

  const uint64_t id = fdp.ConsumeIntegral<uint64_t>();
  
  usb_redir_interrupt_packet_header header;
  const size_t header_bytes = fdp.ConsumeData(&header, sizeof(header));
  if (header_bytes != sizeof(header)) {
      usbredirparser_destroy(parser);
      return 0;
  }

  // Cap length to avoid excessive memory allocation
  const int data_len = header.length > 0 ? (header.length % 4096) : 0;
  header.length = data_len;

  std::vector<uint8_t> packet_data;
  uint8_t* data_ptr = nullptr;
  if (data_len > 0) {
      packet_data = fdp.ConsumeBytes<uint8_t>(data_len);
      // Ensure we have enough data for the specified length
      if (packet_data.size() < data_len) {
          usbredirparser_destroy(parser);
          return 0;
      }
      data_ptr = packet_data.data();
  }

  usbredirparser_send_interrupt_packet(parser, id, &header, data_ptr, data_len);

  // The packet is queued, we need to write it out to exercise the code
  while (usbredirparser_has_data_to_write(parser)) {
      usbredirparser_do_write(parser);
  }

  usbredirparser_destroy(parser);

  return 0;
}
