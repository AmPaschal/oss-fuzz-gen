
	
<description>
The function `zip_file_replace` replaces the content of a file in a zip archive. The file is identified by its index. The new content is provided as a `zip_source_t`.
</description>
<requirements>
<requirement>
The first argument, `za`, must be a valid pointer to a `zip_t` struct created by `zip_open` or a similar function. This is because the function dereferences this pointer to access archive information.
</requirement>
<requirement>
The archive `za` must not be read-only. This is because the underlying function `_zip_file_replace` checks for this condition and will fail if the archive is read-only.
</requirement>
<requirement>
The `idx` argument must be a valid index in the zip archive, i.e., `idx < za->nentry`. This is because the function checks this condition and returns an error if it is not met.
</requirement>
<requirement>
The `source` argument must not be NULL. This is because the function checks for a NULL source and returns an error.
</requirement>
<requirement>
The `source` argument should be created using a `zip_source_*` function like `zip_source_buffer`. This ensures that the source is correctly initialized for the replace operation, as seen in the callers of `zip_file_replace`.
</requirement>
</requirements>

/**
This fuzzing target tests the zip_file_replace function.
It creates a new ZIP archive, adds an initial file, and then
replaces that file's content using zip_file_replace with
data provided by the fuzzer.
**/

#ifdef __cplusplus
extern "C"
#endif
int
LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    if (size < 1) {
        return 0;
    }

    const char *path = "fuzz_replace.zip";
    zip_t *za;
    int err;

    (void)remove(path);
    if ((za = zip_open(path, ZIP_CREATE, &err)) == NULL) {
        return 0;
    }

    const char *initial_content = "initial";
    zip_source_t *s_initial = zip_source_buffer(za, initial_content, strlen(initial_content), 0);
    if (s_initial == NULL) {
        zip_discard(za);
        return 0;
    }

    if (zip_file_add(za, "initial_file.txt", s_initial, ZIP_FL_OVERWRITE) < 0) {
        zip_source_free(s_initial);
        zip_discard(za);
        return 0;
    }

    zip_flags_t flags = data[0];
    const uint8_t *replace_content = data + 1;
    size_t replace_size = size - 1;

    zip_source_t *s_replace = zip_source_buffer(za, replace_content, replace_size, 0);
    if (s_replace == NULL) {
        zip_discard(za);
        return 0;
    }

    zip_uint64_t idx_to_replace = 0;

    if (zip_file_replace(za, idx_to_replace, s_replace, flags) < 0) {
        zip_source_free(s_replace);
    }

    if (zip_close(za) < 0) {
        // zip_close frees the archive even on error.
    }
    (void)remove(path);

    return 0;
}
