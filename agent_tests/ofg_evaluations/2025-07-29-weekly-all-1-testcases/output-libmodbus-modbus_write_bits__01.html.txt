
	
<description>
The `modbus_write_bits` function is used to write multiple bits (coils) to a remote Modbus slave device. It takes a pointer to a `modbus_t` context, a starting address `addr`, the number of bits to write `nb`, and a pointer to a buffer `src` containing the values of the bits to be written. The function first performs checks on the validity of its arguments, such as ensuring the context is not NULL and the number of bits does not exceed the maximum limit. It then constructs a raw request packet, sends it to the slave, and waits for a confirmation. The `modbus_t` context must be properly initialized (eg. using `modbus_new_tcp` or `modbus_new_rtu`) and connected to a Modbus server before this function is called.
</description>
<requirements>

<requirement>
The first argument, `ctx`, must be a valid pointer to a `modbus_t` structure. This structure should be initialized by calling `modbus_new_tcp`, `modbus_new_tcp_pi`, or `modbus_new_rtu`.
</requirement>
<requirement>
A connection must be established by calling `modbus_connect` on the `ctx` object before calling `modbus_write_bits`. This is necessary for the function to be able to send the request to the remote device.
</requirement>
<requirement>
The third argument, `nb`, which represents the number of bits to write, must be greater than 0 and less than or equal to `MODBUS_MAX_WRITE_BITS` (1968). If `nb` is larger than this value, the function will return an error immediately.
</requirement>
<requirement>
The fourth argument, `src`, must be a valid pointer to an array of `uint8_t` with a size of at least `nb`. The function reads `nb` bits from this array to construct the request.
</requirement>
<requirement>
The second argument, `addr`, represents the starting address for writing the bits. While it can be fuzzed, for successful execution, it should correspond to a valid address range on the Modbus slave device. Invalid addresses will result in exceptions from the slave but will not crash the client.
</requirement>

struct Fuzzer {
  uint16_t port;
  size_t size;
  const uint8_t *buffer;

  pthread_t thread;
  int socket;
};
typedef struct Fuzzer Fuzzer;

void fuzzinit(Fuzzer *fuzzer) {
  struct sockaddr_in server_addr;
  fuzzer->socket = socket(AF_INET, SOCK_STREAM, 0);

  server_addr.sin_family = AF_INET;
  server_addr.sin_port = htons(fuzzer->port);
  server_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

  setsockopt(fuzzer->socket, SOL_SOCKET, SO_REUSEADDR, &(int){1}, sizeof(int));

  bind(fuzzer->socket, (struct sockaddr *)&server_addr, sizeof(server_addr));
  listen(fuzzer->socket, 1);
}

void *Server(void *args) {

  Fuzzer *fuzzer = (Fuzzer *)args;
  {
    int client_socket;
    char clientData[MODBUS_RTU_MAX_ADU_LENGTH];
    struct sockaddr_in clientAddr;
    uint32_t clientSZ = sizeof(clientAddr);

    client_socket = accept(fuzzer->socket, (struct sockaddr *)&clientAddr, &clientSZ);
    if (client_socket < 0) {
        pthread_exit(NULL);
    }

    // For write operations, the server first receives the request
    recv(client_socket, clientData, sizeof(clientData), 0);
    // Then sends a confirmation response based on fuzzer data
    send(client_socket, fuzzer->buffer, fuzzer->size, 0);

    shutdown(client_socket, SHUT_RDWR);
    close(client_socket);
  }
  pthread_exit(NULL);
}

void clean(Fuzzer *fuzzer) {
  { // Server
    shutdown(fuzzer->socket, SHUT_RDWR);
    close(fuzzer->socket);
  }
}

int client(modbus_t *ctx, int addr, int nb, const uint8_t *src_data) {
  if (modbus_connect(ctx) == -1) {
    return -1;
  }

  modbus_write_bits(ctx, addr, nb, src_data);

  modbus_close(ctx);
  return 0;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

  if (size < kMinInputLength) {
    return 0;
  }

  Fuzzer fuzzer_instance;
  Fuzzer *fuzzer = &fuzzer_instance;
  fuzzer->port = PORT;

  // The first part of the data is for the server's response
  const size_t server_response_size = data[0];
  if (size < 1 + server_response_size + 4) {
    return 0;
  }
  fuzzer->buffer = data + 1;
  fuzzer->size = server_response_size;

  const uint8_t *client_data = data + 1 + server_response_size;
  const size_t client_data_size = size - 1 - server_response_size;

  if (client_data_size < 4) {
      return 0;
  }

  int addr = (client_data[0] << 8) | client_data[1];
  int nb = (client_data[2] << 8) | client_data[3];

  // Apply constraints on nb
  nb = (nb % (MODBUS_MAX_WRITE_BITS + 1));
  if (nb == 0) {
    nb = 1;
  }

  const uint8_t *src = client_data + 4;
  const size_t src_size = client_data_size - 4;

  if (src_size < (size_t)nb) {
    return 0;
  }

  fuzzinit(fuzzer);

  modbus_t *ctx = modbus_new_tcp("127.0.0.1", fuzzer->port);
  if (ctx == NULL) {
    clean(fuzzer);
    return 0;
  }
  // Set a short timeout to avoid long hangs
  modbus_set_response_timeout(ctx, 0, 200000);

  pthread_create(&fuzzer->thread, NULL, Server, fuzzer);
  client(ctx, addr, nb, src);
  pthread_join(fuzzer->thread, NULL);

  modbus_free(ctx);
  clean(fuzzer);
  return 0;
}
