
	
<description>
The `crxLoadDecodeLoop` function decodes image data from a CRX file. It iterates through the image planes, decoding each one using the `crxDecodePlane` function. The first argument is a pointer to a `CrxImage` struct, which contains information about the image and its encoding. The second argument specifies the number of planes to decode.
</description>
<requirements>

<requirement>
The first argument, a void pointer, must be a pointer to a `CrxImage` struct. This is because the function internally casts the pointer to `CrxImage*` and accesses its members.
</requirement>
<requirement>
The `CrxImage` struct must be initialized by calling `crxSetupImageData`. This is because `crxSetupImageData` populates the fields of the struct that are necessary for `crxLoadDecodeLoop` to function correctly.
</requirement>
<requirement>
The `nPlanes` argument must be less than or equal to 4. This is to prevent an out-of-bounds write in the `results` array, which has a size of 4.
</requirement>
<requirement>
If the `encType` member of the `CrxImage` struct is 3, the `nPlanes` member must be 4, and the `nBits` member must be greater than 8. This ensures that the `planeBuf` member of the `CrxImage` struct is allocated, which is required for decoding.
</requirement>

</requirements>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  if (size < 1024)
    return 0;

  FuzzedDataProvider fdp(data, size);
  FuzzLibRaw lib_raw;

  std::vector<uint8_t> stream_buffer = fdp.ConsumeRemainingBytes<uint8_t>();
  if (lib_raw.open_buffer(stream_buffer.data(), stream_buffer.size()) !=
      LIBRAW_SUCCESS) {
    return 0;
  }

  crx_data_header_t hdr;
  memset(&hdr, 0, sizeof(hdr));

  hdr.f_width = fdp.ConsumeIntegralInRange<int32_t>(32, 1024);
  hdr.f_height = fdp.ConsumeIntegralInRange<int32_t>(32, 1024);
  hdr.tileWidth = fdp.ConsumeIntegralInRange<int32_t>(16, hdr.f_width);
  hdr.tileHeight = fdp.ConsumeIntegralInRange<int32_t>(16, hdr.f_height);
  hdr.nBits = fdp.ConsumeIntegralInRange<int32_t>(8, 16);
  hdr.nPlanes = fdp.ConsumeIntegralInRange<int32_t>(1, 4);
  hdr.encType = fdp.PickValueInArray<int32_t>({0, 1, 3});

  if (hdr.encType == 3) {
    hdr.nPlanes = 4;
    if (hdr.nBits <= 8) {
      hdr.nBits = 9;
    }
  }

  hdr.cfaLayout = fdp.ConsumeIntegralInRange<int32_t>(0, 3);
  hdr.imageLevels = fdp.ConsumeIntegralInRange<int32_t>(0, 3);
  hdr.medianBits = hdr.nBits;
  hdr.version = fdp.PickValueInArray<int32_t>({0x100, 0x200});

  size_t mdatHdrSize = fdp.ConsumeIntegralInRange<size_t>(0, 512);
  std::vector<uint8_t> mdatHdrBuf = fdp.ConsumeBytes<uint8_t>(mdatHdrSize);
  hdr.mdatHdrSize = mdatHdrBuf.size();

  CrxImage img;
  memset(&img, 0, sizeof(img));
  img.input = lib_raw.get_internal_data().internal_data.input;

  size_t raw_alloc_size =
      (size_t)hdr.f_width * hdr.f_height * 2 * sizeof(ushort);
  if (raw_alloc_size > 20000000)
    raw_alloc_size = 20000000;

  ushort *raw_image = (ushort *)malloc(raw_alloc_size);
  if (!raw_image)
    return 0;

  if (crxSetupImageData(&hdr, &img, (int16_t *)raw_image, 0,
                        stream_buffer.size(), mdatHdrBuf.data(),
                        hdr.mdatHdrSize) == 0) {
    lib_raw.call_crxLoadDecodeLoop(&img, hdr.nPlanes);
  }

  crxFreeImageData(&img);
  free(raw_image);

  return 0;
}
