extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  kj::runCatchingExceptions([&]() {
    // Requirement 1: Create a StructSchema.
    // We use TestAllTypes as it has a rich set of fields.
    auto schema = capnp::Schema::from<capnproto_test::capnp::test::TestAllTypes>();
    auto fields = schema.getFields();
    if (fields.size() == 0) {
      return;
    }

    // --- Create the orphan to be adopted ---

    // Requirement 5: Create a valid Orphan<DynamicValue>.
    // To do this, we create a separate message, initialize a field, and disown it.
    capnp::MallocMessageBuilder orphanMessage;
    capnp::DynamicStruct::Builder orphanStruct = orphanMessage.initRoot<capnp::DynamicStruct>(schema);

    // Pick a field to initialize and then disown.
    size_t disownIndex = provider.ConsumeIntegralInRange<size_t>(0, fields.size() - 1);
    auto fieldToDisown = fields[disownIndex];

    // Initialize the field with some fuzzed data. We'll set a simple int field
    // to ensure we can create a valid orphan value.
    orphanStruct.set(fieldToDisown, provider.ConsumeIntegral<int32_t>());
    capnp::Orphan<capnp::DynamicValue> orphan = orphanStruct.disown(fieldToDisown);

    // --- Adopt the orphan into a new struct ---

    // Requirement 2: Create a MallocMessageBuilder.
    capnp::MallocMessageBuilder message;
    // Requirement 3: Create a DynamicStruct::Builder.
    capnp::DynamicStruct::Builder root = message.initRoot<capnp::DynamicStruct>(schema);

    // Requirement 4: The `name` argument must be a valid field name.
    size_t adoptIndex = provider.ConsumeIntegralInRange<size_t>(0, fields.size() - 1);
    auto fieldToAdoptInto = fields[adoptIndex];
    kj::StringPtr adoptFieldName = fieldToAdoptInto.getProto().getName();

    // Call the function under test.
    // This will throw an exception if the orphan's type is not compatible
    // with the target field's type. kj::runCatchingExceptions will handle this.
    root.adopt(adoptFieldName, kj::mv(orphan));
  });

  return 0;
}
