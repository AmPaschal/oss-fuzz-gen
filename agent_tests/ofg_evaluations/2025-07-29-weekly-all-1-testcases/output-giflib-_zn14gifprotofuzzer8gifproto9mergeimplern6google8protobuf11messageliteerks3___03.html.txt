#include <fuzzer/FuzzedDataProvider.h>
#include "genfiles/gif_fuzz_proto.pb.h"
#include <google/protobuf/message_lite.h>
#include <vector>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t* data, size_t size) {
  FuzzedDataProvider provider(data, size);

  // Split the input into two parts for two separate proto messages.
  size_t to_size = provider.ConsumeIntegralInRange<size_t>(0, provider.remaining_bytes());
  std::vector<uint8_t> to_data = provider.ConsumeBytes<uint8_t>(to_size);
  std::vector<uint8_t> from_data = provider.ConsumeRemainingBytes<uint8_t>();

  gifProtoFuzzer::GifProto to_proto;
  to_proto.ParseFromArray(to_data.data(), to_data.size());

  gifProtoFuzzer::GifProto from_proto;
  from_proto.ParseFromArray(from_data.data(), from_data.size());

  // Call the public MergeFrom, which in turn calls the target MergeImpl.
  to_proto.MergeFrom(from_proto);

  return 0;
}
