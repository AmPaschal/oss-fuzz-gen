
	
<description>
The `gnutls_bye` function is used to terminate a TLS session. It sends a `close_notify` alert to the peer and, depending on the `how` parameter, may wait for the peer's `close_notify` alert. This function is typically called after the application has finished sending and receiving data over the TLS connection.
</description>
<requirements>
<requirement>
The `gnutls_init` function must be called to initialize the `gnutls_session_t` object before it is passed to `gnutls_bye`. This is because `gnutls_bye` operates on an initialized session object.
</requirement>
<requirement>
The `session` object should not be `NULL`. The function performs a `NULL` check on the `session` object and returns an error if it is `NULL`.
</requirement>
<requirement>
A TLS handshake should be successfully completed before calling `gnutls_bye`. The function operates on an established TLS session, which is created by the handshake process.
</requirement>
<requirement>
The `how` parameter must be one of `GNUTLS_SHUT_RDWR` or `GNUTLS_SHUT_WR`. These values control the termination behavior of the function, and other values may lead to unexpected behavior.
</requirement>
</requirements>

/* This is a dummy random function to make fuzzing deterministic.
 * It is based on the other fuzzers in the gnutls suite.
 */
int __attribute__((visibility("protected")))
gnutls_rnd(gnutls_rnd_level_t level, void *data, size_t len)
{
	memset(data, 0xff, len);

	/* Flip the first byte to avoid infinite loop in the RSA
	 * blinding code of Nettle */
	if (len > 0)
		memset(data, 0x0, 1);
	return 0;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size)
{
	int res;
	gnutls_session_t session;
	gnutls_certificate_credentials_t xcred;
	struct mem_st memdata;
	unsigned int retry;
    gnutls_close_request_t how;

    /* We need at least 1 byte to select the 'how' parameter. */
    if (size < 1) {
        return 0;
    }

    /* Use the first byte of the input to select the 'how' parameter. */
    how = (data[0] % 2 == 0) ? GNUTLS_SHUT_RDWR : GNUTLS_SHUT_WR;

    /* The rest of the data will be used for the handshake. */
    const uint8_t *handshake_data = data + 1;
    size_t handshake_size = size - 1;

	res = gnutls_init(&session, GNUTLS_CLIENT);
	assert(res >= 0);

	res = gnutls_certificate_allocate_credentials(&xcred);
	assert(res >= 0);
	res = gnutls_credentials_set(session, GNUTLS_CRD_CERTIFICATE, xcred);
	assert(res >= 0);

	res = gnutls_priority_set_direct(
		session, "NORMAL", NULL);
	assert(res >= 0);

	memdata.data = handshake_data;
	memdata.size = handshake_size;

	gnutls_transport_set_push_function(session, error_push);
	gnutls_transport_set_pull_function(session, error_pull);
	gnutls_handshake_set_read_function(session, handshake_discard);

	retry = 0;
	do {
		res = gnutls_handshake(session);
		if (res == GNUTLS_E_AGAIN) {
			if (handshake_pull(session, &memdata) < 0) {
				res = GNUTLS_E_INTERNAL_ERROR;
				break;
			}
			if (retry > HANDSHAKE_MAX_RETRY_COUNT) {
				break;
			}
			retry++;
		} else {
			retry = 0;
		}
	} while (res < 0 && gnutls_error_is_fatal(res) == 0);

    /*
     * After the handshake attempt, call the function under test.
     * This is done even if the handshake failed, to fuzz cleanup paths.
     */
    gnutls_bye(session, how);

	gnutls_deinit(session);
	gnutls_certificate_free_credentials(xcred);
	return 0;
}
