extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  struct libtpms_callbacks cbs = {
      .sizeOfStruct = sizeof(struct libtpms_callbacks),
      .tpm_nvram_init = NULL,
      .tpm_nvram_loaddata = NULL,
      .tpm_nvram_storedata = NULL,
      .tpm_nvram_deletename = NULL,
      .tpm_io_init = mytpm_io_init,
      .tpm_io_getlocality = mytpm_io_getlocality,
  };

  if (TPMLIB_RegisterCallbacks(&cbs) != TPM_SUCCESS) {
    return 0;
  }

  if (TPMLIB_ChooseTPMVersion(TPMLIB_TPM_VERSION_1_2) != TPM_SUCCESS) {
    return 0;
  }

  if (TPMLIB_MainInit() != TPM_SUCCESS) {
    return 0;
  }

  // Construct the GetCapability command payload
  uint32_t capArea = provider.ConsumeIntegral<uint32_t>();
  std::vector<uint8_t> subCap = provider.ConsumeRemainingBytes<uint8_t>();
  
  // Manually construct the full TPM command buffer
  // Header: tag (2), size (4), ordinal (4) = 10 bytes
  // Payload: capArea (4), subCapSize (4), subCap (variable)
  size_t payload_size = sizeof(capArea) + sizeof(uint32_t) + subCap.size();
  size_t full_command_size = 10 + payload_size;

  std::vector<unsigned char> command(full_command_size);
  unsigned char *buffer_ptr = command.data();

  // Header
  *(uint16_t*)buffer_ptr = htons(TPM_TAG_RQU_COMMAND);
  buffer_ptr += 2;
  *(uint32_t*)buffer_ptr = htonl(full_command_size);
  buffer_ptr += 4;
  *(uint32_t*)buffer_ptr = htonl(TPM_ORD_GetCapability);
  buffer_ptr += 4;

  // Payload
  *(uint32_t*)buffer_ptr = htonl(capArea);
  buffer_ptr += 4;
  *(uint32_t*)buffer_ptr = htonl(subCap.size());
  buffer_ptr += 4;
  if (!subCap.empty()) {
    memcpy(buffer_ptr, subCap.data(), subCap.size());
  }

  // Process the command
  unsigned char *rbuffer = NULL;
  uint32_t rlength;
  uint32_t rtotal = 0;
  TPMLIB_Process(&rbuffer, &rlength, &rtotal, command.data(), command.size());

  // Cleanup
  TPM_Free(rbuffer);
  TPMLIB_Terminate();

  return 0;
}
