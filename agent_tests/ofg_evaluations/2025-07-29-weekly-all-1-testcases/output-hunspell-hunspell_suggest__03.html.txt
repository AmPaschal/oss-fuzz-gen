#include "/src/hunspell/src/hunspell/hunspell.h"

#include <cstddef>

const char* kAffPath = "/tmp/fuzz.aff";
const char* kDicPath = "/tmp/fuzz.dic";

extern "C" int LLVMFuzzerTestOneInput(const uint8_t* data, size_t size) {
    FuzzedDataProvider provider(data, size);

    // Split the input data into three parts: affix data, dictionary data, and a word.
    // This method provides more variability in the size of each part.
    std::string aff_data = provider.ConsumeRandomLengthString(provider.remaining_bytes() / 2);
    std::string dic_data = provider.ConsumeRandomLengthString(provider.remaining_bytes());
    std::string word = provider.ConsumeRemainingBytesAsString();

    if (word.empty()) {
        return 0;
    }

    // Write the affix data to a temporary file.
    std::ofstream aff_stream(kAffPath, std::ios::binary | std::ios::trunc);
    aff_stream.write(aff_data.data(), aff_data.size());
    aff_stream.close();

    // Write the dictionary data to a temporary file.
    std::ofstream dic_stream(kDicPath, std::ios::binary | std::ios::trunc);
    dic_stream.write(dic_data.data(), dic_data.size());
    dic_stream.close();

    // Create a Hunspell handle using the temporary files.
    Hunhandle* handle = Hunspell_create(kAffPath, kDicPath);
    if (!handle) {
        return 0;
    }

    char** slst = nullptr;
    // Call the function under test to get suggestions.
    int num_sugg = Hunspell_suggest(handle, &slst, word.c_str());

    // Clean up the allocated suggestion list.
    if (num_sugg > 0 && slst != nullptr) {
        Hunspell_free_list(handle, &slst, num_sugg);
    }

    // Destroy the Hunspell handle to free resources.
    Hunspell_destroy(handle);

    return 0;
}
