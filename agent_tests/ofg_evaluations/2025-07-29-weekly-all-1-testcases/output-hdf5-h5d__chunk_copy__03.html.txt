
	
<description>
This function copies chunked dataset data from a source to a destination file. It handles data type conversions, including for variable-length (VLEN) and reference types, and manages memory for the copy process. The function iterates through the chunks in the source, copying each one to the destination. It also handles special cases, such as chunks stored in the cache.

The function is called by `H5O__layout_copy_file` when copying a dataset with a chunked layout. The caller prepares the necessary structures, including the source and destination file pointers, layout and storage information for the chunks, and the dataspace and datatype of the source dataset. The caller also provides copy options through the `cpy_info` parameter.
</description>

</requirement>

</requirement>

#include "hdf5.h"
#include <stdlib.h>

#include <unistd.h>

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
    // Need at least 1 byte for rank, and 2 per rank for dims/cdims.
    // Let's set a reasonable minimum.
    if (size < 3) {
        return 0;
    }

    // Suppress error messages.
    H5Eset_auto(H5E_DEFAULT, NULL, NULL);

    char src_filename[256];
    char dst_filename[256];
    sprintf(src_filename, "/tmp/fuzz_src.%d", getpid());
    sprintf(dst_filename, "/tmp/fuzz_dst.%d", getpid());

    // Create source and destination files.
    hid_t src_fid = H5Fcreate(src_filename, H5F_ACC_TRUNC, H5P_DEFAULT, H5P_DEFAULT);
    if (src_fid < 0) {
        remove(src_filename);
        return 0;
    }

    hid_t dst_fid = H5Fcreate(dst_filename, H5F_ACC_TRUNC, H5P_DEFAULT, H5P_DEFAULT);
    if (dst_fid < 0) {
        H5Fclose(src_fid);
        remove(src_filename);
        remove(dst_filename);
        return 0;
    }

    // Use fuzzer data to define dataset and chunk properties.
    int rank = (data[0] % H5S_MAX_RANK);
    if (rank == 0) rank = 1;

    if (size < (1 + (size_t)rank * 2)) {
        H5Fclose(src_fid);
        H5Fclose(dst_fid);
        remove(src_filename);
        remove(dst_filename);
        return 0;
    }

    hsize_t dims[H5S_MAX_RANK];
    hsize_t cdims[H5S_MAX_RANK];
    const uint8_t *params = data + 1;
    for (int i = 0; i < rank; i++) {
        // Keep dimensions small to avoid excessive memory usage.
        dims[i] = (params[i] % 128) + 1;
        cdims[i] = (params[i + rank] % 128) + 1;
        if (cdims[i] > dims[i]) {
            cdims[i] = dims[i];
        }
    }

    // Create a dataspace.
    hid_t sid = H5Screate_simple(rank, dims, NULL);
    if (sid >= 0) {
        // Create a dataset creation property list for chunking.
        hid_t dcpl = H5Pcreate(H5P_DATASET_CREATE);
        if (dcpl >= 0) {
            if (H5Pset_chunk(dcpl, rank, cdims) >= 0) {
                // Create the chunked dataset in the source file.
                hid_t did = H5Dcreate2(src_fid, "dset", H5T_NATIVE_INT, sid, H5P_DEFAULT, dcpl, H5P_DEFAULT);
                if (did >= 0) {
                    // Write some data to ensure chunks are allocated.
                    hssize_t n_points = H5Sget_simple_extent_npoints(sid);
                    if (n_points > 0) {
                        size_t buf_size = (size_t)n_points * sizeof(int);
                        // Limit allocation size.
                        if (buf_size > 0 && buf_size < 500000) {
                            int *buf = (int *)malloc(buf_size);
                            if (buf) {
                                for (size_t i = 0; i < (size_t)n_points; i++) {
                                    buf[i] = (int)i;
                                }
                                H5Dwrite(did, H5T_NATIVE_INT, H5S_ALL, H5S_ALL, H5P_DEFAULT, buf);
                                free(buf);
                            }
                        }
                    }

                    // Copy the dataset, which should trigger H5D__chunk_copy.
                    H5Ocopy(src_fid, "dset", dst_fid, "dset_copy", H5P_DEFAULT, H5P_DEFAULT);
                    H5Dclose(did);
                }
            }
            H5Pclose(dcpl);
        }
        H5Sclose(sid);
    }

    // Clean up.
    H5Fclose(src_fid);
    H5Fclose(dst_fid);
    remove(src_filename);
    remove(dst_filename);

    return 0;
}
