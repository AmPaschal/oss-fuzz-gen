
	
<description>
The `Cookie::write` function serializes a `Cookie` object to an output stream. The function iterates through the cookie's attributes, including its name, value, path, domain, expiration date, and other properties, and writes them to the provided `ostream` in the standard cookie string format. The function handles optional attributes gracefully, only writing them if they have a value. It also correctly formats the extensions map. Therefore, to fuzz this function, we need to create a `Cookie` object and populate its fields with fuzzed data. The `Cookie` object can be initialized using its constructor, which takes a name and a value. The other attributes can then be set on the object. The `FullDate` object, if used, can be initialized with a `time_point` created from a fuzzed duration. The output stream can be a `std::stringstream` to capture the output for verification.
</description>
<requirements>

<requirement>
The `Cookie` object should be initialized. The constructor `Cookie(std::string name, std::string value)` can be used for this purpose.
</requirement>
<requirement>
The `name` and `value` members of the `Cookie` object should be fuzzed. These are fundamental to the cookie and are always written to the output stream.
</requirement>
<requirement>
The `path`, `domain`, and `maxAge` optional members of the `Cookie` object should be fuzzed, including the case where they do not have a value. This is to test the conditional logic in the `write` function.
</requirement>
<requirement>
The `expires` optional member of the `Cookie` object should be fuzzed. This involves either not setting a value, or initializing a `FullDate` object with a fuzzed `time_point`.
</requirement>
<requirement>
The `secure` and `httpOnly` boolean members of the `Cookie` object should be fuzzed. This is to test the output of the 'Secure' and 'HttpOnly' flags.
</requirement>
<requirement>
The `ext` map member of the `Cookie` object should be fuzzed. This involves populating the map with a variable number of fuzzed key-value pairs to test the loop that writes the extensions.
</requirement>

</requirements>

#include <pistache/http.h>
#include <pistache/router.h>
#include <fuzzer/FuzzedDataProvider.h>
#include <chrono>

using namespace Pistache;

template <class F>
void ignoreExceptions(const F& func)
{
    try
    {
        func();
    }
    catch (...)
    {
    }
}

template <class T>
void parseHttpHeader(const std::string& input)
{
    T header;
    ignoreExceptions([&] { header.parse(input); });

    std::stringstream oss;
    header.write(oss);
}

template <>
void parseHttpHeader<Pistache::Http::Header::Authorization>(const std::string& input)
{
    Pistache::Http::Header::Authorization header;
    ignoreExceptions([&] { header.parse(input); });
    ignoreExceptions([&] { header.getMethod(); });
    ignoreExceptions(
        [&] {
            const auto user     = header.getBasicUser();
            const auto password = header.getBasicPassword();
            header.setBasicUserPassword(user, password);
        });

    std::stringstream oss;
    header.write(oss);
}

void fuzz_headers(const std::string& input)
{
    parseHttpHeader<Pistache::Http::Header::Accept>(input);
    parseHttpHeader<Pistache::Http::Header::CacheControl>(input);
    parseHttpHeader<Pistache::Http::Header::Connection>(input);
    parseHttpHeader<Pistache::Http::Header::AcceptEncoding>(input);
    parseHttpHeader<Pistache::Http::Header::ContentEncoding>(input);
    parseHttpHeader<Pistache::Http::Header::ContentLength>(input);
    parseHttpHeader<Pistache::Http::Header::ContentType>(input);
    parseHttpHeader<Pistache::Http::Header::Authorization>(input);
    parseHttpHeader<Pistache::Http::Header::Date>(input);
    parseHttpHeader<Pistache::Http::Header::Expect>(input);
    parseHttpHeader<Pistache::Http::Header::Host>(input);
    parseHttpHeader<Pistache::Http::Header::Server>(input);
}

void fuzz_cookies(const std::string& input)
{
    Pistache::Http::CookieJar cookie_jar;
    ignoreExceptions([&] { Pistache::Http::Cookie::fromString(input); });
    ignoreExceptions([&] { cookie_jar.addFromRaw(input.data(), input.size()); });
}

void fuzz_cookie_write(const std::string &input) {
    FuzzedDataProvider provider(reinterpret_cast<const uint8_t*>(input.data()), input.size());

    ignoreExceptions([&] {
        std::string name = provider.ConsumeRandomLengthString(100);
        std::string value = provider.ConsumeRandomLengthString(100);

        if (name.empty()) {
            return;
        }

        Pistache::Http::Cookie cookie(name, value);

        if (provider.ConsumeBool()) {
            cookie.path = provider.ConsumeRandomLengthString(50);
        }
        if (provider.ConsumeBool()) {
            cookie.domain = provider.ConsumeRandomLengthString(50);
        }
        if (provider.ConsumeBool()) {
            cookie.maxAge = provider.ConsumeIntegral<int>();
        }
        if (provider.ConsumeBool()) {
            auto time_since_epoch = std::chrono::seconds(provider.ConsumeIntegral<uint32_t>());
            Pistache::Http::FullDate::time_point tp(time_since_epoch);
            cookie.expires = Pistache::Http::FullDate(tp);
        }
        cookie.secure = provider.ConsumeBool();
        cookie.httpOnly = provider.ConsumeBool();

        uint8_t ext_count = provider.ConsumeIntegralInRange<uint8_t>(0, 5);
        for (uint8_t i = 0; i < ext_count; ++i) {
            std::string key = provider.ConsumeRandomLengthString(20);
            std::string val = provider.ConsumeRandomLengthString(20);
            if (!key.empty()) {
                cookie.ext[key] = val;
            }
        }
        std::stringstream ss;
        ss << cookie;
    });
}

void fuzz_request_parser(const std::string& input)
{
    constexpr size_t maxDataSize = 4096;
    Pistache::Http::RequestParser rparser(maxDataSize);

    if (rparser.feed(input.data(), input.size()))
    {
        auto state = Pistache::Http::Private::State::Done;
        ignoreExceptions([&] { state = rparser.parse(); });

        if (state == Pistache::Http::Private::State::Again)
            ignoreExceptions([&] { rparser.parse(); });
    }
}

void fuzz_response_parser(const std::string& input)
{
    constexpr size_t maxDataSize = 4096;
    Pistache::Http::ResponseParser rparser(maxDataSize);

    if (rparser.feed(input.data(), input.size()))
    {
        auto state = Pistache::Http::Private::State::Done;
        ignoreExceptions([&] { state = rparser.parse(); });

        if (state == Pistache::Http::Private::State::Again)
            ignoreExceptions([&] { rparser.parse(); });
    }
}

void fuzz_router(const std::string& input)
{
    std::string path_input;
    std::stringstream input_stream(input);
    Pistache::Rest::SegmentTreeNode tree;

    while (std::getline(input_stream, path_input, '\n'))
    {
        if (path_input.size() < 2)
            continue;

        const int test_case = path_input.back();
        path_input.pop_back();

        const auto sanitized = Pistache::Rest::SegmentTreeNode::sanitizeResource(path_input);
        std::shared_ptr<char> ptr(new char[sanitized.length()], std::default_delete<char[]>());
        std::memcpy(ptr.get(), sanitized.data(), sanitized.length());
        const std::string_view path { ptr.get(), sanitized.length() };

        switch (test_case)
        {
        case 'A':
            ignoreExceptions([&] {
                Pistache::Rest::Route::Handler handler = [](auto...) { return Pistache::Rest::Route::Result::Ok; };
                std::shared_ptr<char> ptr(new char[sanitized.length()], std::default_delete<char[]>());
                std::memcpy(ptr.get(), sanitized.data(), sanitized.length());
                const std::string_view path { ptr.get(), sanitized.length() };
                tree.addRoute(path, handler, ptr);
            });
            break;
        case 'R':
            ignoreExceptions([&] { tree.removeRoute(path); });
            break;
        case 'F':
            ignoreExceptions([&] { tree.findRoute(path); });
            break;
        }
    }
}

void fuzz_other(const std::string& input)
{
    // URI parsing
    Http::Uri::Query query1;
    query1.add(input, input);
}

extern "C" int LLVMFuzzerTestOneInput(const uint8_t* data, size_t size)
{
    if (size < 1)
        return -1;

    const uint8_t type = data[0];
    const std::string input(reinterpret_cast<const char*>(data + 1), size - 1);

    switch (type)
    {
    case 'H':
        fuzz_headers(input);
        break;
    case 'C':
        fuzz_cookies(input);
        break;
    case 'R':
        fuzz_request_parser(input);
        break;
    case 'A':
        fuzz_response_parser(input);
        break;
    case 'S':
        fuzz_router(input);
        break;
    case 'O':
        fuzz_other(input);
        break;
    case 'W':
        fuzz_cookie_write(input);
        break;
    }

    return 0;
}
