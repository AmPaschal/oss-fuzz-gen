The `_field_list` is a `std::vector` of `std::pair<std::string, std::shared_ptr<const field>>`. The function is an inline getter and does not perform any checks. The main purpose of the function is to provide an iterator for iterating over the fields registered in the `field_registry` object.
</description>

</requirement>

// A minimal, self-contained implementation of the abstract base class
// spotify::json::detail::field. This allows us to create field objects
// without depending on the library's internal, non-public concrete classes.
class FuzzField final : public spotify::json::detail::field {
 public:
  FuzzField(bool required, size_t required_field_idx)
      : spotify::json::detail::field(required, required_field_idx) {}

  void decode(spotify::json::decode_context &context,
              void * /*object*/) const override {
    // Correctly call skip_value from its 'detail' namespace.
    spotify::json::detail::skip_value(context);
  }

  void encode(spotify::json::encode_context & /*context*/,
              const std::string & /*escaped_key*/,
              const void * /*object*/) const override {
    // Do nothing for encode.
  }
};

// Place a volatile array of function pointers at global scope.
// The volatile keyword prevents the compiler from optimizing away the indirect
// call, as it must assume the array contents can change at any time.

    &spotify::json::detail::field_registry::end

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);
  spotify::json::detail::field_registry registry;

  // Populate the registry with a fuzzed number of our custom field objects.
  const auto fields_to_add = provider.ConsumeIntegralInRange<size_t>(0, 20);
  for (size_t i = 0; i < fields_to_add; ++i) {
    const auto field_name = provider.ConsumeRandomLengthString(32);
    const bool is_required = provider.ConsumeBool();
    
    const auto field_ptr = std::make_shared<FuzzField>(
        is_required, is_required ? registry.num_required_fields() : 0);
    
    registry.save(field_name, is_required, field_ptr);
  }

  // Select one of the function pointers from the global volatile array based on fuzzer input.
  const auto end_func_ptr = func_array[provider.ConsumeBool()];

  // Call the function under test via the data-dependent, volatile-guarded pointer.
  auto end_iterator = (registry.*end_func_ptr)();

  // Use the result in a loop to prevent the call from being optimized away.
  for (auto it = registry.begin(); it != end_iterator; ++it) {
    volatile const auto &name = it->first;
    (void)name;
  }

  return 0;
}
