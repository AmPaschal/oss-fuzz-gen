
	
<description>
`crxDecodePlane` is a function within the LibRaw library, responsible for decoding a single plane of a CRX image. It iterates through the tiles of the image, and for each tile, it decodes the subbands that make up the plane. The decoded data is then converted to the correct format and stored in the output buffer. The function is called by `crxLoadDecodeLoop`, which in turn is called by `crxLoadRaw`. The `crxLoadRaw` function is responsible for reading the CRX file, parsing the headers, and setting up the `CrxImage` structure that is passed to `crxDecodePlane`.
</description>
<requirements>

<requirement>
The first argument, a void pointer, should be a pointer to a `CrxImage` struct initialized by the `crxSetupImageData` function. This is because the function casts the void pointer to a `CrxImage` pointer and accesses its members.
</requirement>
<requirement>
The `CrxImage` struct's `nPlanes` member must be initialized to either 1 or 4. This is because the caller, `crxLoadRaw`, enforces this constraint.
</requirement>
<requirement>
The second argument, `planeNumber`, must be less than the `nPlanes` member of the `CrxImage` struct. This is to prevent out-of-bounds access to the `tile->comps` array.
</requirement>
<requirement>
The `CrxImage` struct's `input` member must be a valid pointer to a `LibRaw_abstract_datastream` object. This is because the function reads data from this stream.
</requirement>

</requirements>

#include <fuzzer/FuzzedDataProvider.h>
#include <libraw.h>
#include "libraw_internal.h"
#include <cstdint>
#include <vector>

// Re-declaration of internal LibRaw structures from src/decoders/crx.cpp
// This is necessary because they are not exposed in public headers.

struct CrxSubband {
  uint8_t *entropyBuf;
  uint32_t entropySize;
  int16_t *quantTbl;
  int32_t *idwtTbl;
  int16_t *median;
  uint16_t width;
  uint16_t height;
  uint8_t subbandNum;
  uint8_t levelNum;
  uint8_t quantIdx;
  uint8_t medianNum;
  int64_t dataSize;
  int64_t dataOffset;
  short rowStartAddOn;
  short rowEndAddOn;
  short colStartAddOn;
  short colEndAddOn;
  short levelShift;
};

struct CrxWaveletTransform; // Forward declaration

struct CrxPlaneComp {
  uint8_t *compBuf;
  CrxSubband *subBands;
  CrxWaveletTransform *wvltTransform;
  int8_t compNumber;
  int64_t dataOffset;
  int32_t compSize;
  bool supportsPartial;
  int32_t roundedBitsMask;
  int8_t tileFlag;
};

struct CrxQStep {
  uint32_t *qStepTbl;
  int width;
  int height;
};

struct CrxTile {
  CrxPlaneComp *comps;
  int8_t tileFlag;
  int8_t tileNumber;
  int64_t dataOffset;
  int32_t tileSize;
  uint16_t width;
  uint16_t height;
  bool hasQPData;
  CrxQStep *qStep;
  uint32_t mdatQPDataSize;
  uint16_t mdatExtraSize;
};

struct CrxImage {
  uint8_t nPlanes;
  uint16_t planeWidth;
  uint16_t planeHeight;
  uint8_t samplePrecision;
  uint8_t medianBits;
  uint8_t subbandCount;
  uint8_t levels;
  uint8_t nBits;
  uint8_t encType;
  uint8_t tileCols;
  uint8_t tileRows;
  CrxTile *tiles;
  uint64_t mdatOffset;
  uint64_t mdatSize;
  int16_t *outBufs[4]; // one per plane
  int16_t *planeBuf;
  LibRaw_abstract_datastream *input;
};

// Derived class to access protected members of LibRaw
class LibRawFuzzer : public LibRaw {
public:
  int public_crxDecodePlane(void *p, uint32_t planeNumber) {
    return crxDecodePlane(p, planeNumber);
  }
  LibRaw_abstract_datastream *get_datastream() {
    return libraw_internal_data.internal_data.input;
  }
};

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);
  LibRawFuzzer lib_raw;

  // open_buffer is needed to initialize the datastream
  std::vector<char> buffer = provider.ConsumeRemainingBytes<char>();
  if (lib_raw.open_buffer(buffer.data(), buffer.size()) != LIBRAW_SUCCESS) {
    return 0;
  }

  CrxImage img = {0};

  // Populate CrxImage with fuzzed data
  img.nPlanes = provider.PickValueInArray<uint8_t>({1, 4});
  img.planeWidth = provider.ConsumeIntegral<uint16_t>();
  img.planeHeight = provider.ConsumeIntegral<uint16_t>();
  img.samplePrecision = provider.ConsumeIntegral<uint8_t>();
  img.medianBits = provider.ConsumeIntegral<uint8_t>();
  img.subbandCount = provider.ConsumeIntegralInRange<uint8_t>(1, 31);
  img.levels = provider.ConsumeIntegral<uint8_t>();
  img.nBits = provider.ConsumeIntegral<uint8_t>();
  img.encType = provider.ConsumeIntegral<uint8_t>();
  img.tileCols = provider.ConsumeIntegralInRange<uint8_t>(1, 16);
  img.tileRows = provider.ConsumeIntegralInRange<uint8_t>(1, 16);
  img.mdatOffset = provider.ConsumeIntegral<uint64_t>();
  img.mdatSize = provider.ConsumeIntegral<uint64_t>();
  img.input = lib_raw.get_datastream();

  if (!img.input) {
    return 0;
  }

  uint32_t planeNumber = provider.ConsumeIntegralInRange<uint32_t>(0, img.nPlanes > 0 ? img.nPlanes - 1 : 0);

  size_t num_tiles = static_cast<size_t>(img.tileCols) * img.tileRows;
  if (num_tiles == 0) {
    return 0;
  }
  img.tiles = new CrxTile[num_tiles]();
  for (size_t i = 0; i < num_tiles; ++i) {
    img.tiles[i].comps = new CrxPlaneComp[img.nPlanes]();
    for (uint8_t j = 0; j < img.nPlanes; ++j) {
      if (img.subbandCount > 0) {
        img.tiles[i].comps[j].subBands = new CrxSubband[img.subbandCount]();
      }
    }
  }

  // Call the target function
  lib_raw.public_crxDecodePlane(&img, planeNumber);

  // Cleanup
  for (size_t i = 0; i < num_tiles; ++i) {
    for (uint8_t j = 0; j < img.nPlanes; ++j) {
      if (img.tiles[i].comps[j].subBands) {
        delete[] img.tiles[i].comps[j].subBands;
      }
    }
    delete[] img.tiles[i].comps;
  }
  delete[] img.tiles;

  return 0;
}
