
	
<description>
The function `sepol_policydb_to_image` converts a policy database (`sepol_policydb_t`) into a binary image. It is a wrapper around the `policydb_to_image` function. The callers of this function, such as `load_policy.c`, first create a policy database using `sepol_policydb_create`, then read the policy data from a file into the database using `sepol_policydb_read`, and also set the policy version using `sepol_policydb_set_vers`. The resulting database is then passed to `sepol_policydb_to_image` to generate the binary image.
</description>
<requirements>

<requirement>
The `sepol_handle_t` pointer can be NULL. In `load_policy.c`, this function is called with a NULL handle.
</requirement>
<requirement>
The `sepol_policydb_t` pointer must be a valid, initialized structure. The caller in `load_policy.c` initializes this structure by calling `sepol_policydb_create`, `sepol_policydb_read`, and `sepol_policydb_set_vers` in sequence.
</requirement>
<requirement>
The `newdata` and `newlen` arguments must be valid pointers. The function writes the address and size of the newly created policy image to these pointers.
</requirement>

</requirements>

#include <sepol/policydb.h>
#include <sepol/sepol.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <setjmp.h>

/*
 * This symbol is required by the checkpolicy object files that are linked
 * with this fuzzer, even though the fuzzer itself does not use them.
 * Defining it here resolves the linker error.
 */
jmp_buf fuzzing_pre_parse_stack_state;

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
	sepol_policydb_t *policydb = NULL;
	sepol_policy_file_t *pf = NULL;
	void *image = NULL;
	size_t image_len;

	if (size < 1) {
		return 0;
	}

	if (sepol_policy_file_create(&pf) != 0) {
		goto cleanup;
	}

	/* The fuzzer input is used as the policy file content */
	sepol_policy_file_set_mem(pf, (char *)data, size);

	if (sepol_policydb_create(&policydb) != 0) {
		goto cleanup;
	}

	/* Read the policy from the in-memory file */
	if (sepol_policydb_read(policydb, pf) != 0) {
		goto cleanup;
	}

	/*
	 * Set a policy version. Without this, to_image might fail.
	 * We can derive this from the input data to increase coverage.
	 */
	unsigned int policy_vers_max = sepol_policy_kern_vers_max();
	unsigned int policy_vers_min = sepol_policy_kern_vers_min();
	unsigned int policy_vers = policy_vers_max;
	if (size > 4) {
		/* Use a portion of the input to determine the version */
		policy_vers = (*(const unsigned int *)data) %
			      (policy_vers_max - policy_vers_min + 1) + policy_vers_min;
	}
	sepol_policydb_set_vers(policydb, policy_vers);

	/* Call the function under test */
	sepol_policydb_to_image(NULL, policydb, &image, &image_len);

cleanup:
	if (image) {
		free(image);
	}
	if (policydb) {
		sepol_policydb_free(policydb);
	}
	if (pf) {
		sepol_policy_file_free(pf);
	}

	return 0;
}
