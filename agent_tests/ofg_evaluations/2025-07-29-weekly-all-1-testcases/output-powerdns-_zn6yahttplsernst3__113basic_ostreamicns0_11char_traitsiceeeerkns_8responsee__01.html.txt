#include <fuzzer/FuzzedDataProvider.h>
#include "ext/yahttp/yahttp/yahttp.hpp"
#include <sstream>
#include <string>
#include <vector>
#include <stdexcept>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  YaHTTP::Response res;

  // Revert to the member names that the compiler seems to expect.
  res.version = provider.PickValueInArray<int>({9, 10, 11});
  res.status = provider.ConsumeIntegral<int>();
  res.statusText = provider.ConsumeRandomLengthString(128);

  // Fuzz headers
  int num_headers = provider.ConsumeIntegralInRange<uint8_t>(0, 20);
  for (int i = 0; i < num_headers; ++i) {
    std::string key = provider.ConsumeRandomLengthString(64);
    std::string value = provider.ConsumeRandomLengthString(256);
    if (provider.remaining_bytes() == 0) {
      break;
    }
    res.headers[key] = value;
  }

  // Fuzz content body with its correct name
  res.body = provider.ConsumeRemainingBytesAsString();

  // Create a stream and call the target operator
  std::ostringstream oss;
  try {
    oss << res;
  } catch (const std::exception &e) {
    // Catch a generic exception since the specific type is not found.
  }

  return 0;
}
