static void *die_oom(size_t bytes) {
  fprintf(stderr, "Out of memory: %lu bytes unavailable!\n", (unsigned long)bytes);
  exit(1);
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  static int init_done = 0;

  if (!init_done) {
    setenv("GC_LARGE_ALLOC_WARN_INTERVAL", "30000", 1);
    GC_INIT();
#if (GC_VERSION_MAJOR > 7) || ((GC_VERSION_MAJOR == 7) && (GC_VERSION_MINOR >= 2))
    GC_set_oom_fn(die_oom);
#else
    GC_oom_fn = die_oom;
#endif
#ifdef USE_M17N
#ifdef USE_UNICODE
    wtf_init(WC_CES_UTF_8, WC_CES_UTF_8);
#else
    wtf_init(WC_CES_EUC_JP, WC_CES_EUC_JP);
#endif
#endif
    init_done = 1;
  }

  const char *s1, *s2, *s3;
  const uint8_t *p, *next;
  Str s = NULL;

  // The fuzzer is expected to provide null-separated strings.
  // We will parse up to 3 strings.

  // Find the first string
  p = (const uint8_t *)memchr(data, '\0', size);
  if (p == NULL) {
    return 0; // Require at least one null-terminated string.
  }
  s1 = (const char *)data;
  next = p + 1;

  // If there's no more data, call with one string.
  if (next >= data + size) {
    s = Strnew_m_charp(s1, NULL);
    if (s) Strfree(s);
    return 0;
  }

  // Find the second string
  p = (const uint8_t *)memchr(next, '\0', data + size - next);
  if (p == NULL) {
    return 0; // For simplicity, require subsequent strings to be null-terminated.
  }
  s2 = (const char *)next;
  next = p + 1;

  // If there's no more data, call with two strings.
  if (next >= data + size) {
    s = Strnew_m_charp(s1, s2, NULL);
    if (s) Strfree(s);
    return 0;
  }

  // Find the third string. The rest of the data is ignored.
  p = (const uint8_t *)memchr(next, '\0', data + size - next);
  if (p == NULL) {
    return 0;
  }
  s3 = (const char *)next;

  // Call with three strings.
  s = Strnew_m_charp(s1, s2, s3, NULL);
  if (s) Strfree(s);

  return 0;
