#include "capnp/dynamic.h"
#include "capnp/message.h"
#include "capnp/schema.h"
#include "capnp/test.capnp.h"
#include "kj/main.h"

#include <iostream>

extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  kj::runCatchingExceptions([&]() {
    capnp::MallocMessageBuilder message;
    capnp::StructSchema schema = capnp::Schema::from<capnp::_::TestAllTypes>();
    capnp::DynamicStruct::Builder builder = message.initRoot<capnp::DynamicStruct>(schema);

    auto fields = schema.getFields();
    if (fields.size() == 0) {
      return;
    }

    // Pick a random valid field to set.
    uint16_t field_idx = provider.ConsumeIntegralInRange<uint16_t>(0, fields.size() - 1);
    auto field = fields[field_idx];
    auto name = field.getProto().getName();
    auto type = field.getType();

    // Create a value of the appropriate type.
    switch (type.which()) {
    case capnp::schema::Type::VOID:
      builder.set(name, capnp::DynamicValue::Reader(capnp::Void()));
      break;
    case capnp::schema::Type::BOOL:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeBool()));
      break;
    case capnp::schema::Type::INT8:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeIntegral<int8_t>()));
      break;
    case capnp::schema::Type::INT16:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeIntegral<int16_t>()));
      break;
    case capnp::schema::Type::INT32:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeIntegral<int32_t>()));
      break;
    case capnp::schema::Type::INT64:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeIntegral<int64_t>()));
      break;
    case capnp::schema::Type::UINT8:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeIntegral<uint8_t>()));
      break;
    case capnp::schema::Type::UINT16:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeIntegral<uint16_t>()));
      break;
    case capnp::schema::Type::UINT32:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeIntegral<uint32_t>()));
      break;
    case capnp::schema::Type::UINT64:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeIntegral<uint64_t>()));
      break;
    case capnp::schema::Type::FLOAT32:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeFloatingPoint<float>()));
      break;
    case capnp::schema::Type::FLOAT64:
      builder.set(name, capnp::DynamicValue::Reader(provider.ConsumeFloatingPoint<double>()));
      break;
    case capnp::schema::Type::TEXT: {
      std::string value = provider.ConsumeRandomLengthString();
      builder.set(name, capnp::DynamicValue::Reader(capnp::Text::Reader(value)));
      break;
    }
    case capnp::schema::Type::DATA: {
      std::vector<uint8_t> value = provider.ConsumeRemainingBytes<uint8_t>();
      builder.set(name, capnp::DynamicValue::Reader(capnp::Data::Reader(value.data(), value.size())));
      break;
    }
    case capnp::schema::Type::ENUM: {
      auto enumSchema = type.asEnum();
      auto enumerants = enumSchema.getEnumerants();
      if (enumerants.size() > 0) {
        uint16_t idx = provider.ConsumeIntegralInRange<uint16_t>(0, enumerants.size() - 1);
        auto enumerant = enumerants[idx];
        capnp::DynamicEnum dynamicEnum(enumerant);
        builder.set(name, capnp::DynamicValue::Reader(dynamicEnum));
      }
      break;
    }
    // Other types (LIST, STRUCT, INTERFACE, ANY_POINTER) are more complex to
    // instantiate and are skipped for now.
    default:
      break;
    }
  });

  return 0;
}
