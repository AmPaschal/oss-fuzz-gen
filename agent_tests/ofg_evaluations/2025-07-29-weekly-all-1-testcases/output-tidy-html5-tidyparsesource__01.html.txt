
	
<description>
The `tidyParseSource` function is a wrapper around the `tidyDocParseSource` function. It takes a `TidyDoc` and a `TidyInputSource` as input, and then calls `tidyDocParseSource` to parse the input source. The `TidyInputSource` struct contains function pointers for getting the next byte of input, ungetting a byte, and checking for the end of the file. The actual parsing is handled by the `TY_(DocParseStream)` function, which is called by `tidyDocParseSource`. The `TY_(UserInput)` function is used to create a `StreamIn` object from the `TidyInputSource`.
</description>

</requirement>
<requirement>
The `ungetByte` member of the `TidyInputSource` struct pointed to by the `source` argument should be a valid function pointer. This is because it is used to unget bytes from the input source.
</requirement>
<requirement>
The `eof` member of the `TidyInputSource` struct pointed to by the `source` argument should be a valid function pointer. This is because it is used to check for the end of the input source.
</requirement>
<requirement>
The `tdoc` argument must be a valid `TidyDoc` object. This is because it is dereferenced in the `tidyDocToImpl` function.
</requirement>
</requirements>

// A struct to hold the state for our custom input source.
// This will be passed as the `sourceData` member of TidyInputSource.
typedef struct {
  const uint8_t *data;
  size_t size;
  size_t offset;
} FuzzerInputSource;

// Callback function to get the next byte from the fuzzer's input buffer.
// This function's signature matches TidyGetByteFunc.
int get_byte_callback(void *sourceData) {
  FuzzerInputSource *source = (FuzzerInputSource *)sourceData;
  if (source->offset < source->size) {
    return source->data[source->offset++];
  }
  return EOF;
}

// Callback function to "unget" a byte (move the offset back).
// This function's signature matches TidyUngetByteFunc.
void unget_byte_callback(void *sourceData, byte bt) {
  (void)bt; // The byte is unused, we just move the offset.
  FuzzerInputSource *source = (FuzzerInputSource *)sourceData;
  if (source->offset > 0) {
    source->offset--;
  }
}

// Callback function to check for the end of the input.
// This function's signature matches TidyEOFFunc.
Bool eof_callback(void *sourceData) {
  FuzzerInputSource *source = (FuzzerInputSource *)sourceData;
  return (source->offset >= source->size) ? yes : no;
}

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  // Create a Tidy document.
  TidyDoc tdoc = tidyCreate();
  if (!tdoc) {
    return 0;
  }

  // It's good practice to set an error buffer to capture diagnostics.
  TidyBuffer errbuf = {0};
  tidySetErrorBuffer(tdoc, &errbuf);

  // Initialize our custom input source with the fuzzer data.
  FuzzerInputSource fuzz_source;
  fuzz_source.data = data;
  fuzz_source.size = size;
  fuzz_source.offset = 0;

  // Set up the TidyInputSource struct with our callbacks.
  TidyInputSource input_source;
  input_source.sourceData = &fuzz_source;
  input_source.getByte = get_byte_callback;
  input_source.ungetByte = unget_byte_callback;
  input_source.eof = eof_callback;

  // Call the function under test.
  tidyParseSource(tdoc, &input_source);

  // Clean up resources.
  tidyBufFree(&errbuf);
  tidyRelease(tdoc);

  return 0;
}
