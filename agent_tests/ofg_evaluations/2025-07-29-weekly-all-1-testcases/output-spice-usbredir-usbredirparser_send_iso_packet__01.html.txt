
	
<description>
The function `usbredirparser_send_iso_packet` is a wrapper around `usbredirparser_queue`, and is used to send an ISO packet. It sets the packet type to `usb_redir_iso_packet` and queues it for writing. The callers of this function in `usbredirhost.c` use it for sending ISO data as well as sending empty ISO packets to signal the end of a transfer.
</description>

int wobbly_read_write_count(int count) {
  if (count > (1024 * 1024)) {
    return count;
  }

  return std::min(count, fdp->ConsumeIntegralInRange(1, 4 * count));
}

// Read over complete input buffer to detect buffer overflows
void read_all(const void *data, size_t count) {
#ifdef __cpp_lib_smart_ptr_for_overwrite
  const auto buf = std::make_unique_for_overwrite<uint8_t[]>(count);
#else
  const auto buf = std::make_unique<uint8_t[]>(count);
#endif

  memcpy(buf.get(), data, count);
}

template <typename T, typename = std::enable_if_t<std::is_class<T>::value>> void read_all(const T *ptr) { read_all(ptr, sizeof(T)); }

int parser_write(void *priv, uint8_t *data, int count) {
  // Simulate short writes
  count = wobbly_read_write_count(count);

  read_all(data, count);

  return count;
}

  parser->buffered_bulk_packet_func = parser_buffered_bulk_packet;

  for (uint32_t &cap : caps) {
    cap = fdp->ConsumeIntegral<decltype(caps)::value_type>();
  }

  const int init_flags = fdp->ConsumeIntegral<uint8_t>() & (usbredirparser_fl_usb_host | usbredirparser_fl_no_hello);

  usbredirparser_init(parser.get(), "fuzzer", caps.data(), caps.size(), init_flags);

  uint64_t id = fdp->ConsumeIntegral<uint64_t>();

  struct usb_redir_iso_packet_header iso_header;
  iso_header.endpoint = fdp->ConsumeIntegral<uint8_t>();
  iso_header.status = fdp->ConsumeIntegral<uint8_t>();

  std::vector<uint8_t> data_vec = fdp->ConsumeRemainingBytes<uint8_t>();
  if (data_vec.size() > std::numeric_limits<uint16_t>::max()) {
    data_vec.resize(std::numeric_limits<uint16_t>::max());
  }
  int data_len = data_vec.size();
  iso_header.length = data_len;

  uint8_t *payload_data = (data_len > 0) ? data_vec.data() : nullptr;

  usbredirparser_send_iso_packet(parser.get(), id, &iso_header, payload_data, data_len);

  while (usbredirparser_has_data_to_write(parser.get())) {
    if (usbredirparser_do_write(parser.get()) < 0) {
      break;
    }
  }

  parser.reset();
  fdp.reset();

  return 0;
