
	
<description>
The function `modbus_read_input_registers` reads `nb` input registers from the remote device starting from the address `addr`, and stores the result in the `dest` array. It is a wrapper around the `read_registers` function, and it is used for reading the input registers of a modbus device. Before calling this function, a `modbus_t` context must be created using `modbus_new_tcp` or `modbus_new_rtu` and a connection must be established using `modbus_connect`.
</description>
<requirements>
<requirement>
The first argument, `ctx`, must be a valid pointer to a `modbus_t` struct initialized by `modbus_new_tcp` or `modbus_new_rtu` and connected using `modbus_connect`. This is because the function needs a valid and connected context to communicate with the modbus device.
</requirement>
<requirement>
The third argument, `nb`, must be an integer between 1 and 125 (inclusive). This is to prevent a buffer overflow, as the internal buffer for receiving the response is limited by `MODBUS_MAX_READ_REGISTERS`.
</requirement>
<requirement>
The fourth argument, `dest`, must be a pointer to a `uint16_t` array that is large enough to store `nb` values. This is to avoid a buffer overflow when writing the read data into the `dest` array.
</requirement>
</requirements>

#define PORT 8080
#define kMinInputLength 9
#define kMaxInputLength MODBUS_RTU_MAX_ADU_LENGTH

struct Fuzzer {
  uint16_t port;
  char *file;

  FILE *inFile;
  uint64_t size;
  const uint8_t *buffer;

  pthread_t thread;
  int socket;
};
typedef struct Fuzzer Fuzzer;

int client(Fuzzer *fuzzer);

void fuzzinit(Fuzzer *fuzzer) {
  struct sockaddr_in server_addr;
  fuzzer->socket = socket(AF_INET, SOCK_STREAM, 0);

  server_addr.sin_family = AF_INET;
  server_addr.sin_port = htons(fuzzer->port);
  server_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

  setsockopt(fuzzer->socket, SOL_SOCKET, SO_REUSEADDR, &(int){1}, sizeof(int));

  bind(fuzzer->socket, (struct sockaddr *)&server_addr, sizeof(server_addr));
  listen(fuzzer->socket, 1);
}

void *Server(void *args) {

  Fuzzer *fuzzer = (Fuzzer *)args;
  {
    int client_sock;
    char clientData[10240];
    struct sockaddr_in clientAddr;
    socklen_t clientSZ = sizeof(clientAddr);

    client_sock = accept(fuzzer->socket, (struct sockaddr *)&clientAddr, &clientSZ);

    send(client_sock, fuzzer->buffer, fuzzer->size, 0);
    recv(client_sock, clientData, sizeof(clientData), 0);

    send(client_sock, fuzzer->buffer, fuzzer->size, 0);
    recv(client_sock, clientData, sizeof(clientData), 0);

    shutdown(client_sock, SHUT_RDWR);
    close(client_sock);
  }
  pthread_exit(NULL);
}

void clean(Fuzzer *fuzzer) {
  { // Server
    shutdown(fuzzer->socket, SHUT_RDWR);
    close(fuzzer->socket);
  }
  free(fuzzer);
}

extern int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {

  if (size < kMinInputLength || size > kMaxInputLength) {
    return 0;
  }

  Fuzzer *fuzzer = (Fuzzer *)malloc(sizeof(Fuzzer));
  fuzzer->port = PORT;
  fuzzer->size = size;
  fuzzer->buffer = data;

  fuzzinit(fuzzer);

  pthread_create(&fuzzer->thread, NULL, Server, fuzzer);
  client(fuzzer);
  pthread_join(fuzzer->thread, NULL); /* To Avoid UAF*/

  clean(fuzzer);
  return 0;
}

int client(Fuzzer *fuzzer) {

  uint16_t *tab_rp_registers = NULL;
  modbus_t *ctx = NULL;
  int nb_points;
  int rc;

  ctx = modbus_new_tcp("127.0.0.1", fuzzer->port);

  if (ctx == NULL) {
    fprintf(stderr, "Unable to allocate libmodbus context\n");
    return -1;
  }

  if (modbus_connect(ctx) == -1) {
    fprintf(stderr, "Connection failed: %s\n", modbus_strerror(errno));
    modbus_free(ctx);
    return -1;
  }

  /* Allocate and initialize the memory to store the registers */
  nb_points = (UT_REGISTERS_NB > UT_INPUT_REGISTERS_NB) ? UT_REGISTERS_NB
                                                       : UT_INPUT_REGISTERS_NB;
  tab_rp_registers = (uint16_t *)malloc(nb_points * sizeof(uint16_t));
  memset(tab_rp_registers, 0, nb_points * sizeof(uint16_t));

  int modbus_read_input_registers(modbus_t *, int, int, uint16_t *);
  rc = modbus_read_input_registers(ctx, UT_INPUT_REGISTERS_ADDRESS,
                                   UT_INPUT_REGISTERS_NB, tab_rp_registers);

  /* Free the memory */
  free(tab_rp_registers);

  /* Close the connection */
  modbus_close(ctx);
  modbus_free(ctx);

  return rc;
}
