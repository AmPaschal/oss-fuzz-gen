The `_field_list` is a `std::vector` of pairs, where each pair contains a field name (as a string) and a shared pointer to a `field` object. This function is typically used to iterate over the fields that have been registered in the `field_registry`.
</description>

// A dummy implementation of the abstract base class spotify::json::detail::field.
// This is necessary because the concrete implementations in the library are
// private, but the field_registry::save method requires a std::shared_ptr<field>.
struct MyFuzzField : public spotify::json::detail::field {
  MyFuzzField(bool required, size_t required_field_idx)
      : spotify::json::detail::field(required, required_field_idx) {}

  void decode(spotify::json::decode_context &context,
              void *object) const override {}

  void encode(spotify::json::encode_context &context,
              const std::string &escaped_key,
              const void *object) const override {}
};

// Define a type for a pointer to the member function we want to call.
using begin_fn_ptr = spotify::json::detail::field_registry::const_iterator (spotify::json::detail::field_registry::*)() const;

// This function returns a pointer to the function-under-test.
// By marking it `noinline`, we prevent the compiler from knowing the
// return value at the call site, which forces a true indirect function call later.
__attribute__((noinline))
begin_fn_ptr get_begin_ptr() {
    return &spotify::json::detail::field_registry::begin;
}

// The main fuzzing function.
extern "C" int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  FuzzedDataProvider provider(data, size);

  // Instantiate field_registry directly.
  spotify::json::detail::field_registry registry;

  size_t required_field_count = 0;
  while (provider.remaining_bytes() > 0) {
    const auto field_name = provider.ConsumeRandomLengthString(32);
    if (field_name.empty()) {
      continue;
    }
    const bool is_required = provider.ConsumeBool();
    if (is_required) {
      required_field_count++;
    }

    // Create a shared_ptr to our dummy field implementation.
    auto field_ptr =
        std::make_shared<MyFuzzField>(is_required, required_field_count);

    // Populate the registry using the public save() method.
    registry.save(field_name, is_required, field_ptr);
  }

  // Get the function pointer from our non-inlined helper function.
  begin_fn_ptr ptr = get_begin_ptr();

  // Call the function under test via the opaque function pointer.
  // This should result in a non-optimizable, indirect call.
  spotify::json::detail::field_registry::const_iterator it = (registry.*ptr)();

  // To ensure the call and the loop are not optimized away, we use the
  // returned iterator and read from it with a volatile pointer.
  for (; it != registry.end(); ++it) {
    volatile const char *c = it->first.c_str();
    (void)c;
  }

  return 0;
}
