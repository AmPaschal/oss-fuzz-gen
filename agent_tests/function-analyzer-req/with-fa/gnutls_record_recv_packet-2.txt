<description>
The gnutls_record_recv_packet function is used to receive a TLS packet from a session. It reads data from the underlying transport, decrypts it, and returns the packet data in the gnutls_packet_t struct. Callers typically initialize a session, perform a TLS handshake, and then call this function in a loop to process incoming data. After processing the packet, gnutls_packet_deinit should be called to free its resources.
</description>

</requirement>
<requirement>
The second argument, `packet`, must be a valid pointer to a `gnutls_packet_t` structure. This structure will be populated with the received packet's data and must not be NULL to avoid a crash.
</requirement>
<requirement>
After successfully receiving a packet and processing its contents, `gnutls_packet_deinit` must be called on the `packet` variable. This is necessary to free the internal memory allocated for the packet data.
</requirement>

#include <assert.h>
#include <stdint.h>
#include <string.h>
#include <stdlib.h>
#include <stdbool.h>

#include <gnutls/gnutls.h>
#include "mem.h"
#include "fuzzer.h"

int LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) {
  int res;
  gnutls_session_t session;
  gnutls_certificate_credentials_t xcred;
  struct mem_st memdata;

  res = gnutls_init(&session, GNUTLS_CLIENT);
  assert(res >= 0);

  res = gnutls_certificate_allocate_credentials(&xcred);
  assert(res >= 0);
  res = gnutls_credentials_set(session, GNUTLS_CRD_CERTIFICATE, xcred);
  assert(res >= 0);

  res = gnutls_priority_set_direct(session, "NORMAL", NULL);
  assert(res >= 0);

  memdata.data = data;
  memdata.size = size;

  gnutls_transport_set_push_function(session, mem_push);
  gnutls_transport_set_pull_function(session, mem_pull);
  gnutls_transport_set_pull_timeout_function(session, mem_pull_timeout);
  gnutls_transport_set_ptr(session, &memdata);

  do {
    res = gnutls_handshake(session);
  } while (res < 0 && gnutls_error_is_fatal(res) == 0);

  if (res >= 0) {
    while (true) {
      gnutls_packet_t packet = NULL;
      ssize_t ret = gnutls_record_recv_packet(session, &packet);

      if (ret > 0) {
        if (packet != NULL) {
          gnutls_packet_deinit(packet);
        }
      } else {
        if (packet != NULL) {
          gnutls_packet_deinit(packet);
        }
        break;
      }
    }
  }

  gnutls_deinit(session);
  gnutls_certificate_free_credentials(xcred);
  return 0;
}
